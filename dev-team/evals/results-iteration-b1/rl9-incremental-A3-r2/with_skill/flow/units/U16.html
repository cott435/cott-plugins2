GRID_CODES = {0, 1, 2, 3}  # empty, head, body, food or goal


@dataclass(frozen=True)
class EnvRecording:
    """One recorded game: its teams, the reset, then each step's actions, result and frame."""

    teams: dict[str, TeamSpec]
    board: Literal["grid", "tank"]
    seed: int
    obs: dict[str, np.ndarray]
    info: dict[str, float]
    start_frame: np.ndarray
    actions: tuple[dict[str, np.ndarray], ...]
    results: tuple[StepResult, ...]
    frames: tuple[np.ndarray, ...]


class ReplayEnv(Env):
    """Serves a recording: `reset` rewinds to its start, `step` returns the next result.

    `step` validates the actions as the contract says (`ValueError`) and raises `RuntimeError`
    once the recorded game is over or before the first `reset`.
    """

    def __init__(self, recording: EnvRecording) -> None:
        self.teams = recording.teams
        self.board = recording.board
        self.recording = recording
        self._t = -1

    def reset(self, seed: int) -> tuple[dict[str, np.ndarray], dict[str, float]]:
        self._t = 0
        return {k: v.copy() for k, v in self.recording.obs.items()}, dict(self.recording.info)

    def step(self, actions: Mapping[str, np.ndarray]) -> StepResult:
        if not 0 <= self._t < len(self.recording.results):
            raise RuntimeError("step called while no game is running")
        check_actions(self.teams, actions)
        result = self.recording.results[self._t]
        self._t += 1
        return result

    def frame(self) -> np.ndarray:
        if self._t <= 0:
            return self.recording.start_frame.copy()
        return self.recording.frames[self._t - 1].copy()


def check_actions(teams: Mapping[str, TeamSpec], actions: Mapping[str, np.ndarray]) -> None:
    """`ValueError` for a missing team, a wrong shape, an out-of-range or non-finite action."""
    for team, spec in teams.items():
        if team not in actions:
            raise ValueError(f"no actions for team {team!r}")
        action = np.asarray(actions[team])
        space = spec.actions
        shape = (spec.n_agents,) if space.kind == "discrete" else (spec.n_agents, space.dim)
        if action.shape != shape:
            raise ValueError(f"team {team!r}: actions shaped {action.shape}, expected {shape}")
        if not np.all(np.isfinite(action)):
            raise ValueError(f"team {team!r}: non-finite action")
        if space.kind == "discrete" and not np.all((action >= 0) & (action < space.n)):
            raise ValueError(f"team {team!r}: action outside 0..{space.n - 1}")

    _assert_obs(env.teams, obs)
    _assert_info(env.teams, info)
    _assert_frame(env.board, env.frame())
    over = False
    for step_actions in actions:
        result = env.step(step_actions)
        _assert_result(env.teams, result)
        _assert_frame(env.board, env.frame())
        over = result.terminated or result.truncated
        if over:
            break
    if over:
        with pytest.raises(RuntimeError):
            env.step(actions[-1])
    env.reset(seed)
    with pytest.raises(ValueError, match="team"):
        env.step({})

@dataclass(frozen=True)
class AgentRecording:
    """The calls one agent received, acting for `team`."""

    team: TeamSpec
    calls: tuple[AgentCall, ...]

    def acts(self) -> list[AgentCall]:

class ReplayAgent(Agent):
    """Returns the recorded actions in order, starting again after the last.

    Keeps what it was shown (`observed`) and taught (`learned`) for the consumer's test;
    `act` raises `ValueError` for a batch not shaped like the recording's team.
    """

    def __init__(self, recording: AgentRecording) -> None:
        self.recording = recording
        self.observed: list[tuple[np.ndarray, bool]] = []
        self.learned: list[Transition] = []
        self._actions = [call.action for call in recording.acts() if call.action is not None]
        self._next = 0

    def act(self, obs: np.ndarray, *, explore: bool) -> np.ndarray:
        team = self.recording.team
        if obs.shape != (team.n_agents, *team.obs_shape):
            raise ValueError(f"observation shaped {obs.shape} for team {team}")
        self.observed.append((obs.copy(), explore))
        action = self._actions[self._next % len(self._actions)]
        self._next += 1
        return action.copy()

    def learn(self, t: Transition) -> None:
        self.learned.append(t)

    def save(self, path: Path) -> None:
        path.write_text(json.dumps({"next": self._next}), encoding="utf-8")

    def load(self, path: Path) -> None:
        self._next = json.loads(path.read_text(encoding="utf-8"))["next"]

            status=status,
            checkpoints=dict(self.checkpoints),
            generations={team: tuple(paths) for team, paths in self.generations.items()},
            games=self.games,

def schema_errors(rule: dict[str, Any], value: Any, where: str = "$") -> list[str]:
    """Every way `value` breaks `rule`, as `path: message` lines (empty when it conforms)."""
    if "anyOf" in rule:
        if any(not schema_errors(option, value, where) for option in rule["anyOf"]):
            return []
        return [f"{where}: matches no anyOf option"]
    errors = _scalar_errors(rule, value, where)
    if errors:
        return errors
    if isinstance(value, dict):
        errors += _object_errors(rule, value, where)
    if isinstance(value, list):
        errors += _array_errors(rule, value, where)
    return errors


def _type_ok(name: str, value: Any) -> bool:
    if isinstance(value, bool) and name in {"integer", "number"}:
        return False
    return isinstance(value, _JSON_TYPES[name])


def _scalar_errors(rule: dict[str, Any], value: Any, where: str) -> list[str]:
    kinds = rule.get("type")
    names = [kinds] if isinstance(kinds, str) else kinds
    errors = []
    if names is not None and not any(_type_ok(n, value) for n in names):
        errors.append(f"{where}: expected {kinds}, got {type(value).__name__}")
    elif "const" in rule and value != rule["const"]:
        errors.append(f"{where}: expected {rule['const']!r}, got {value!r}")
    elif "enum" in rule and value not in rule["enum"]:
        errors.append(f"{where}: {value!r} not one of {rule['enum']}")
    elif isinstance(value, (int, float)) and not isinstance(value, bool):
        errors += _range_errors(rule, value, where)
    return errors


def _range_errors(rule: dict[str, Any], value: float, where: str) -> list[str]:

def build(hint: Any, data: Any) -> Any:
    """`data` (decoded JSON) as a value of type `hint`."""
    if is_dataclass(hint) and isinstance(hint, type):
        return _build_dataclass(hint, data)
    if hint is np.ndarray:
        return _build_array(data)
    if get_origin(hint) is not None:
        return _build_generic(hint, data)
    return _build_scalar(hint, data)


def _build_generic(hint: Any, data: Any) -> Any:
    origin = get_origin(hint)
    if origin is Literal:
        if data not in get_args(hint):
            raise ValueError(f"{data!r} is not one of {get_args(hint)}")
        return data
    if origin in {Union, types.UnionType}:
        return _build_union(hint, data)
    if origin is tuple:
        return _build_tuple(hint, data)
    if origin in {dict, collections.abc.Mapping}:
        value_hint = get_args(hint)[1]
        return {key: build(value_hint, item) for key, item in data.items()}
    raise TypeError(f"cannot build {hint}")


def _build_dataclass(cls: type, data: dict[str, Any]) -> Any:
    hints = get_type_hints(cls)
    names = {f.name for f in fields(cls)}
    unknown = set(data) - names
    if unknown:
        raise ValueError(f"{cls.__name__} has no fields {sorted(unknown)}")
    return cls(**{name: build(hints[name], data[name]) for name in names if name in data})


def _build_array(data: dict[str, Any]) -> np.ndarray:
    array = np.asarray(data["data"], dtype=data["dtype"])
    if array.size == 0:
        array = array.reshape(data["shape"])
    if list(array.shape) != list(data["shape"]):
        raise ValueError(f"array shape {array.shape} is not {data['shape']}")
    return array


def _build_union(hint: Any, data: Any) -> Any:
    options = [a for a in get_args(hint) if a is not type(None)]
    if data is None and len(options) < len(get_args(hint)):
        return None
    if len(options) != 1:
        raise TypeError(f"cannot build the union {hint}")
    return build(options[0], data)


def _build_tuple(hint: Any, data: list[Any]) -> tuple[Any, ...]:
    args = get_args(hint)
    if len(args) == 2 and args[1] is Ellipsis:
        return tuple(build(args[0], item) for item in data)
    if len(args) != len(data):
        raise ValueError(f"expected {len(args)} items, got {len(data)}")
    return tuple(build(a, item) for a, item in zip(args, data, strict=True))


def _build_scalar(hint: Any, data: Any) -> Any:
    if hint is Any:
        return data
    if hint is Path:
        return Path(data)
    if hint is float and isinstance(data, (int, float)) and not isinstance(data, bool):
        return float(data)
    if hint in {int, str, bool} and type(data) is hint:
        return data
    raise TypeError(f"cannot build {hint} from {data!r}")

    """A typed value as JSON: the inverse of `build`."""
    if is_dataclass(value) and not isinstance(value, type):

        return {str(key): encode(item) for key, item in value.items()}
    if isinstance(value, (tuple, list)):
        return [encode(item) for item in value]
    return _encode_scalar(value)

import numpy as np

from rllab.core import SOLO, Agent, Env, RunSettings, StepResult, Transition
from rllab.evaluate import EvalSpec, evaluate
from rllab.runs import CurvePoint, RunRecord, RunWriter, create_run

RECORDED_GAMES = 5
PROGRESS_EVERY_S = 30.0
TRAINING_SEEDS = 1_000_000  # game seeds stay below the evaluation seeds (DATA-01)


def train(
    make_env: Callable[[], Env],
    agents: Mapping[str, Agent],
    settings: RunSettings,
    *,
    root: Path = Path("runs"),
) -> RunRecord:
    """Step for `settings.steps` steps, learning after each, with a curve point every `eval_every`."""
    writer = create_run(settings, root=root)
    try:
        _run_steps(make_env, agents, settings, writer)
        _save_results(make_env, agents, settings, writer)
    except KeyboardInterrupt:
        writer.finish("incomplete")
        raise
    return writer.finish("complete")


def learning_teams(settings: RunSettings) -> list[str]:
    """The teams that learn: `SOLO` in a single-team run, else those `opponents` marks `learn`."""
    if not settings.opponents:
        return [SOLO]
    return [team for team, role in settings.opponents.items() if role == "learn"]


def _run_steps(
    make_env: Callable[[], Env],
    agents: Mapping[str, Agent],
    settings: RunSettings,
    writer: RunWriter,
) -> None:
    learners = learning_teams(settings)
    env = make_env()
    rng = np.random.default_rng(settings.seed)
    start = last_print = monotonic()
    games = 0
    obs, _ = env.reset(int(rng.integers(TRAINING_SEEDS)))
    active = _all_active(obs)
    for step in range(settings.steps):
        if step % settings.eval_every == 0:
            writer.add_point(_curve_point(make_env, agents, settings, step, games, start))
        actions = {team: agents[team].act(obs[team], explore=True) for team in env.teams}
        result = env.step(actions)
        for team in learners:
            agents[team].learn(_transition(team, obs, actions, result, active[team]))
        if result.terminated or result.truncated:
            games += 1
            obs, _ = env.reset(int(rng.integers(TRAINING_SEEDS)))
            active = _all_active(obs)
        else:
            obs = result.obs
            active = {team: ~done for team, done in result.done.items()}
        if monotonic() - last_print >= PROGRESS_EVERY_S:
            last_print = monotonic()
            print(_progress(settings, step + 1, games, last_print - start), flush=True)
    writer.add_point(_curve_point(make_env, agents, settings, settings.steps, games, start))


def _all_active(obs: Mapping[str, np.ndarray]) -> dict[str, np.ndarray]:
    return {team: np.ones(len(batch), dtype=np.bool_) for team, batch in obs.items()}


def _transition(
    team: str,
    obs: Mapping[str, np.ndarray],
    actions: Mapping[str, np.ndarray],
    result: StepResult,
    active: np.ndarray,
) -> Transition:
    """One team's batch-first step; agents done only by truncation are not terminated."""
    return Transition(
        obs=obs[team],
        actions=actions[team],
        reward=result.reward[team],
        next_obs=result.obs[team],
        terminated=result.done[team] & active & (not result.truncated),
        active=active,
        truncated=result.truncated,
    )


def _curve_point(
    make_env: Callable[[], Env],
    agents: Mapping[str, Agent],
    settings: RunSettings,
    step: int,
    games: int,
    start: float,
) -> CurvePoint:

    spec = EvalSpec(
        player=settings.name,
        train_seed=settings.seed,
        team=learning_teams(settings)[0],
        n_games=settings.eval_games,
    )
    scores = evaluate(make_env, agents, spec).scores

        step=step, games=games, mean_score=float(np.mean(scores)), elapsed_s=monotonic() - start

from collections.abc import Callable, Mapping
from pathlib import Path
from time import monotonic

import numpy as np

from rllab.core import SOLO, Agent, Env, RunSettings, StepResult, Transition
from rllab.evaluate import EvalSpec, evaluate
from rllab.runs import CurvePoint, RunRecord, RunWriter, create_run

RECORDED_GAMES = 5
PROGRESS_EVERY_S = 30.0
TRAINING_SEEDS = 1_000_000  # game seeds stay below every evaluation seed (DATA-01)


def train(
    make_env: Callable[[], Env],
    agents: Mapping[str, Agent],
    settings: RunSettings,
    *,
    root: Path = Path("runs"),
) -> RunRecord:
    """Step for `settings.steps` steps, learning after each, with a curve point every `eval_every`."""
    writer = create_run(settings, root=root)
    try:
        _run_steps(make_env, agents, settings, writer)
        _save_results(make_env, agents, settings, writer)
    except KeyboardInterrupt:
        writer.finish("incomplete")
        raise
    return writer.finish("complete")


def learning_teams(settings: RunSettings) -> list[str]:
    """The teams that learn: `SOLO` in a single-team run, else those `opponents` marks `learn`."""
    if not settings.opponents:
        return [SOLO]
    return [team for team, role in settings.opponents.items() if role == "learn"]


def _run_steps(
    make_env: Callable[[], Env],
    agents: Mapping[str, Agent],
    settings: RunSettings,
    writer: RunWriter,
) -> None:
    """The step loop: act for every team, then each learning team learns from its Transition."""
    learners = learning_teams(settings)
    spec = EvalSpec(settings.name, settings.seed, team=learners[0], n_games=settings.eval_games)
    env = make_env()
    rng = np.random.default_rng(settings.seed)
    start = last_print = monotonic()
    games = 0
    obs, _ = env.reset(int(rng.integers(TRAINING_SEEDS)))
    active = _all_active(obs)
    for step in range(settings.steps):
        if step % settings.eval_every == 0:
            score = _mean_score(make_env, agents, spec)
            writer.add_point(CurvePoint(step, games, score, monotonic() - start))
        actions = {team: agents[team].act(obs[team], explore=True) for team in env.teams}
        result = env.step(actions)
        for team in learners:
            agents[team].learn(_transition(team, obs, actions, result, active=active[team]))
        if result.terminated or result.truncated:
            games += 1
            obs, _ = env.reset(int(rng.integers(TRAINING_SEEDS)))
            active = _all_active(obs)
        else:
            obs = result.obs
            active = {team: ~done for team, done in result.done.items()}
        if monotonic() - last_print >= PROGRESS_EVERY_S:
            last_print = monotonic()
            print(_progress(settings, step + 1, games, last_print - start), flush=True)
    score = _mean_score(make_env, agents, spec)
    writer.add_point(CurvePoint(settings.steps, games, score, monotonic() - start))


def _all_active(obs: Mapping[str, np.ndarray]) -> dict[str, np.ndarray]:
    return {team: np.ones(len(batch), dtype=np.bool_) for team, batch in obs.items()}


def _transition(
    team: str,
    obs: Mapping[str, np.ndarray],
    actions: Mapping[str, np.ndarray],
    result: StepResult,
    *,
    active: np.ndarray,
) -> Transition:
    """One team's batch-first step; agents done only by truncation are not terminated."""
    return Transition(
        obs=obs[team],
        actions=actions[team],
        reward=result.reward[team],
        next_obs=result.obs[team],
        terminated=result.done[team] & active & (not result.truncated),
        active=active,
        truncated=result.truncated,
    )


def _mean_score(make_env: Callable[[], Env], agents: Mapping[str, Agent], spec: EvalSpec) -> float:
    """A greedy evaluation of the first learning team against the run's opposition."""

def _two_team_recording():
    discrete = ActionSpace(kind="discrete", n=2)
    teams = {"predator": TeamSpec(1, (3,), discrete), "prey": TeamSpec(2, (3,), discrete)}

    def obs(value):
        return {
            "predator": np.full((1, 3), value, dtype=np.float32),
            "prey": np.full((2, 3), value, dtype=np.float32),
        }

    def result(value, prey_done, terminated):
        return StepResult(
            obs=obs(value),
            reward={
                "predator": np.zeros(1, dtype=np.float32),
                "prey": np.ones(2, dtype=np.float32),
            },
            done={"predator": np.array([terminated]), "prey": np.array(prey_done)},
            terminated=terminated,
            truncated=False,
            info={"steps": value, "score": 0.0, "score:predator": 0.0, "score:prey": 0.0},
        )

    frame = np.zeros((1, 4), dtype=np.float32)
    return envs.EnvRecording(
        teams=teams,
        board="tank",
        seed=0,
        obs=obs(0.0),
        info={"steps": 0, "score": 0.0, "score:predator": 0.0, "score:prey": 0.0},
        start_frame=frame,
        actions=(),
        results=(
            result(1.0, [False, True], terminated=False),
            result(2.0, [False, True], terminated=False),
            result(3.0, [True, True], terminated=True),
        ),
        frames=(frame, frame, frame),
    )
