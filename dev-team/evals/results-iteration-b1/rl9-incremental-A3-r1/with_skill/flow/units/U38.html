230	class EnvReplay:
231	    """A core.Env replaying one recorded game: `reset` returns the recorded start, and each
232	    `step` must get the recorded actions (`recorded_actions`), returning the recorded result."""
233	
234	    def __init__(self, recording: dict[str, Any]) -> None:
235	        self.teams: dict[str, TeamSpec] = decode(dict[str, TeamSpec], recording["teams"])
236	        self.board: Literal["grid", "tank"] = recording["board"]
237	        self.seed: int = recording["seed"]
238	        self._start = decode(dict[str, np.ndarray], recording["reset"]["obs"])
239	        self._info = decode(dict[str, float], recording["reset"]["info"])
240	        self._frames = [decode(np.ndarray, f) for f in recording["frames"]]
241	        self.recorded_actions: list[dict[str, np.ndarray]] = [
242	            decode(dict[str, np.ndarray], s["actions"]) for s in recording["steps"]
243	        ]
244	        self._results: list[StepResult] = [
245	            decode(StepResult, s["result"]) for s in recording["steps"]
246	        ]
247	        self.reset_seeds: list[int] = []
248	        self._t = -1

250	    def reset(self, seed: int) -> tuple[dict[str, np.ndarray], dict[str, float]]:
251	        self.reset_seeds.append(seed)
252	        self._t = 0
253	        return dict(self._start), dict(self._info)
254	
255	    def step(self, actions: Mapping[str, np.ndarray]) -> StepResult:
256	        if self._t < 0 or self._t >= len(self._results):
257	            raise RuntimeError("step called before reset or after the game is over")
258	        expected = self.recorded_actions[self._t]
259	        if set(actions) != set(expected):
260	            raise ValueError(
261	                f"teams {sorted(actions)} differ from the recording's {sorted(expected)}"
262	            )
263	        for team, action in expected.items():
264	            if not np.array_equal(np.asarray(actions[team]), action):
265	                raise ValueError(f"step {self._t}: {team} actions differ from the recording")
266	        result = self._results[self._t]
267	        self._t += 1
268	        return result
269	
270	    def frame(self) -> np.ndarray:
271	        return self._frames[max(self._t, 0)]

275	    """A core.Agent replaying recorded calls in order: each `act` must get the recorded

286	        if self._next >= len(self._calls):

def _load_kit() -> ModuleType:
    spec = importlib.util.spec_from_file_location("contract_kit", TESTS / "contracts" / "kit.py")
    assert spec is not None
    assert spec.loader is not None
    module = importlib.util.module_from_spec(spec)
    sys.modules["contract_kit"] = module
    spec.loader.exec_module(module)
    return module


kit = _load_kit()


def _load(name: str) -> Any:
    return json.loads((FIXTURE / name).read_text(encoding="utf-8"))


def _records(name: str, settings_case: str, run_set: dict[str, Any]) -> dict[str, RunRecord]:
    """One complete RunRecord per seed of `run_set`, keyed by its run folder."""
    base: RunRecord = kit.sample("runs.RunRecord", "complete_corridor", RunRecord)
    settings: RunSettings = kit.sample("core.RunSettings", settings_case, RunSettings)
    out: dict[str, RunRecord] = {}
    for seed in run_set["seeds"]:
        folder = f"runs/{name}-s{seed}-20261018-000000"
        run_settings = dataclasses.replace(
            settings,
            name=name,
            agent=run_set["agent"],
            ablation=run_set["ablation"],
            steps=run_set["steps"],
            seed=seed,
        )
        out[folder] = dataclasses.replace(base, path=Path(folder), settings=run_settings)
    return out


def _targets(full: dict[str, RunRecord], ablation: dict[str, RunRecord]) -> Any:
    def player(folders: dict[str, RunRecord]) -> dict[str, Any]:
        return {"runs": list(folders), "baseline": None, "team": "agents", "vs": []}

    return kit.decode(
        ReportTargets,
        {
            "players": {FULL: player(full), ABLATION: player(ablation)},
            "baselines": [],
            "ladder": [FULL],
            "ablations": [ABLATION],
            "crossplay": [],
        },
    )


def _serve(monkeypatch: pytest.MonkeyPatch, records: dict[str, RunRecord]) -> list[str]:
    """Patch `runs.open_run` to serve `records` by folder name; returns the folders opened."""
    by_name = {Path(folder).name: record for folder, record in records.items()}
    opened: list[str] = []

    def fake_open_run(path: Path) -> RunRecord:
        name = Path(path).name
        if name not in by_name:
            raise FileNotFoundError(f"run {path}: no settings.json")
        opened.append(name)
        return by_name[name]

    monkeypatch.setattr(runs, "open_run", fake_open_run)
    monkeypatch.setattr(players, "open_run", fake_open_run, raising=False)
    return opened


def _case(monkeypatch: pytest.MonkeyPatch, case: str) -> tuple[Any, list[str], list[str]]:
    data = _load("input.json")
    full = _records(FULL, "dqn_compact", data["full"])
    ablation = _records(ABLATION, "no_target", data["ablations"][case])
    opened = _serve(monkeypatch, {**full, **ablation})
    folders = [Path(f).name for f in [*full, *ablation]]
    return _targets(full, ablation), opened, folders
