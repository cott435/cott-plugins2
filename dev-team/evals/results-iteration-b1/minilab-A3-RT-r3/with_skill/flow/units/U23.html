+import json
+from pathlib import Path
+
+from risk.annualize import annualize
+from risk.types import Returns, VolSeries
+
+SAMPLE = Path(__file__).parents[2] / "contracts" / "risk.VolSeries" / "sample.json"
+
+
+def _daily_sample() -> VolSeries:
+    data = json.loads(SAMPLE.read_text())
+    returns = data["returns"]
+    return VolSeries(
+        returns=Returns(
+            base_date=returns["base_date"],
+            dates=tuple(returns["dates"]),
+            tickers=tuple(returns["tickers"]),
+            values={t: tuple(v) for t, v in returns["values"].items()},
+        ),
+        window=data["window"],
+        annualized=False,
+        values={t: tuple(v) for t, v in data["values"].items()},
+    )
+
+
+def test_output_is_marked_annualized() -> None:
+    assert annualize(_daily_sample()).annualized is True
+
+
+def test_daily_values_pass_through_unchanged() -> None:
+    out = annualize(_daily_sample())
+    assert out.values == {
+        "AAPL": (None, None, 0.372194, 0.059487, 0.252872),
+        "MSFT": (None, None, 0.041322, 0.055861, 0.047833),
+    }
+
+
+def test_returns_and_window_are_kept() -> None:
+    vol = _daily_sample()
+    out = annualize(vol, periods_per_year=252)
+    assert out.returns == vol.returns
+    assert out.window == 2
+    assert len(out.values["AAPL"]) == len(out.returns.dates)
+
+
+def test_input_is_not_modified() -> None:
+    vol = _daily_sample()
+    annualize(vol)
+    assert vol.annualized is False
