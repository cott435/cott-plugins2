from statistics import mean, stdev

def summary_stats(vol: VolSeries, periods_per_year: int = 252) -> tuple[TickerSummary, ...]:
    """Annualized return, volatility, Sharpe and the last rolling volatility (RISK-04).

    Spine version: the daily mean and sample stdev of the returns, not annualized.

    if len(vol.returns.dates) < 2:

    return tuple(_ticker_summary(vol, ticker) for ticker in vol.returns.tickers)

    series = vol.returns.values[ticker]
