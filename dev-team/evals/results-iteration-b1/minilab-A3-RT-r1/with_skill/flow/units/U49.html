"""risk_report on the real PL-02 and PL-01: inputs pass through, and their errors surface."""

from __future__ import annotations

import pytest

from mdata import GapError
from risk.report_pipeline import risk_report

DATES = ("2024-01-02", "2024-01-03", "2024-01-04", "2024-01-05")


def _csv(tmp_path, rows):
    path = tmp_path / "prices.csv"
    path.write_text("date,ticker,close\n" + "\n".join(rows) + "\n", encoding="utf-8")
    return path


def _rows(ticker, closes, dates=DATES):
    return [f"{day},{ticker},{close}" for day, close in zip(dates, closes, strict=True)]


def test_requested_tickers_limit_the_report(tmp_path):
    path = _csv(tmp_path, _rows("A", (1, 2, 3, 4)) + _rows("B", (4, 3, 2, 1)))

    report = risk_report(path, tmp_path / "out.json", window=2, tickers=["B"])

    assert [row["ticker"] for row in report.rows] == ["B"]
    assert report.rows[0]["max_drawdown"] == pytest.approx(-0.75, abs=1e-12)


def test_window_sets_when_rolling_vol_starts(tmp_path):
    path = _csv(tmp_path, _rows("A", (1, 2, 3, 4)))

    report = risk_report(path, tmp_path / "out.json", window=3)

    assert report.rows[0]["last_vol"] is None


def test_unknown_ticker_raises_key_error(tmp_path):
    path = _csv(tmp_path, _rows("A", (1, 2, 3, 4)))

    with pytest.raises(KeyError):
        risk_report(path, tmp_path / "out.json", window=2, tickers=["Z"])


def test_long_gap_raises_gap_error_and_writes_nothing(tmp_path):
    dates = tuple(f"2024-01-{day:02d}" for day in range(2, 10))
    rows = _rows("A", range(1, 9), dates) + _rows("B", (5, 6), (dates[0], dates[-1]))
    out = tmp_path / "out.json"

    with pytest.raises(GapError, match="B"):
        risk_report(_csv(tmp_path, rows), out, window=2)
    assert not out.exists()


def test_one_return_is_too_few_for_a_summary(tmp_path):
    path = _csv(tmp_path, _rows("A", (1, 2), DATES[:2]))

    with pytest.raises(ValueError, match="two returns"):
        risk_report(path, tmp_path / "out.json", window=2)
