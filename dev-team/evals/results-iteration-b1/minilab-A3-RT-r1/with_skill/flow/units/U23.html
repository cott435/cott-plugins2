+

+def _sample() -> VolSeries:
+    raw = json.loads(SAMPLE.read_text())
+    r = raw["returns"]
+    returns = Returns(
+        base_date=r["base_date"],
+        dates=tuple(r["dates"]),
+        tickers=tuple(r["tickers"]),
+        values={t: tuple(v) for t, v in r["values"].items()},
+    )
+    return VolSeries(
+        returns=returns,
+        window=raw["window"],
+        annualized=raw["annualized"],
+        values={t: tuple(v) for t, v in raw["values"].items()},
+    )
+
+
+def _daily() -> VolSeries:
+    return replace(_sample(), annualized=False)
+
+
+def test_output_is_marked_annualized() -> None:
+    assert annualize(_daily()).annualized is True
+
+
+def test_values_returns_and_window_pass_through() -> None:
+    daily = _daily()
+    out = annualize(daily)
+    assert out.values == daily.values
+    assert out.values["BBB"][:3] == (None, None, 0.0)
+    assert out.returns is daily.returns
