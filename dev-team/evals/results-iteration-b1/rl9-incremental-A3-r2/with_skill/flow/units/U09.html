TESTS = Path(__file__).resolve().parents[2]
TC20 = TESTS / "fixtures" / "rllab" / "TC-20"
SAMPLES = TESTS / "contracts" / "baselines.RandomPlayer" / "samples.json"


def _load(path: Path) -> dict:
    return json.loads(path.read_text())


def _sample_obs() -> list[np.ndarray]:
    calls = _load(SAMPLES)["seeded_10"]["calls"]
    return [np.asarray(c["obs"]["data"], dtype=c["obs"]["dtype"]) for c in calls]


def _shares(explore: bool) -> np.ndarray:
    case = _load(TC20["input.json"] if False else TC20 / "input.json")
    n_actions = case["n_actions"]
    player = RandomPlayer(n_actions, seed=case["seed"])
    obs = _sample_obs()
    counts = np.zeros(n_actions, dtype=np.int64)
    for i in range(case["calls"]):
        action = np.asarray(player.act(obs[i % len(obs)], explore=explore))
        assert action.shape == (1,)
        a = int(action[0])
        assert 0 <= a < n_actions
        counts[a] += 1
    return counts / case["calls"]


@pytest.mark.intent
@pytest.mark.parametrize("explore", [True, False])
def test_tc20_each_action_share_is_in_the_band(explore):
    expected = _load(TC20 / "expected.json")
    shares = _shares(explore)
    assert np.all(shares >= expected["low"]), shares
    assert np.all(shares <= expected["high"]), shares


@pytest.mark.intent
def test_same_seed_gives_the_same_actions():
    obs = _sample_obs()
    a = RandomPlayer(3, seed=0)
    b = RandomPlayer(3, seed=0)
    first = [int(np.asarray(a.act(o, explore=True))[0]) for o in obs * 10]
    second = [int(np.asarray(b.act(o, explore=True))[0]) for o in obs * 10]
    assert first == second
    assert len(set(first)) == 3
