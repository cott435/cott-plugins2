"""PL-02 step 2: daily returns."""

from __future__ import annotations

import math
from itertools import pairwise

from mdata.types import Prices
from risk.types import Returns


def log_returns(prices: Prices) -> Returns:
    """r_t = ln(close_t / close_t-1), dated t (RISK-01).

    Raises ValueError with fewer than two dates.
    """
    if len(prices.dates) < 2:
        raise ValueError(f"log_returns needs at least two dates, got {len(prices.dates)}")
    values = {
        ticker: tuple(
            math.log(today / yesterday) for yesterday, today in pairwise(prices.closes[ticker])
        )
        for ticker in prices.tickers
    }
    return Returns(
        base_date=prices.dates[0],
        dates=prices.dates[1:],
        tickers=prices.tickers,
        values=values,
    )
