ISO_DATE = re.compile(r"^\d{4}-\d{2}-\d{2}$")
SAMPLE_TOL = 1e-5  # the samples' derived values are rounded to about six digits


def _returns(d: dict[str, Any]) -> Returns:
    return Returns(
        base_date=d["base_date"],
        dates=tuple(d["dates"]),
        tickers=tuple(d["tickers"]),
        values={t: tuple(v) for t, v in d["values"].items()},
    )


def _vol(d: dict[str, Any]) -> VolSeries:
    return VolSeries(
        returns=_returns(d["returns"]),
        window=d["window"],
        annualized=d["annualized"],
        values={t: tuple(v) for t, v in d["values"].items()},
    )


def _prices(d: dict[str, Any]) -> Prices:
    return Prices(
        dates=tuple(d["dates"]),
        tickers=tuple(d["tickers"]),
        closes={t: tuple(v) for t, v in d["closes"].items()},
    )

def test_vol_series_is_the_annualized_window_2_vol_of_the_returns(contract_sample):
    vol = _vol(contract_sample("risk.VolSeries"))
    assert vol.returns == _returns(contract_sample("risk.Returns"))
    for ticker, rets in vol.returns.values.items():
        for i in range(vol.window, len(rets)):
            expected = statistics.stdev(rets[i - vol.window : i]) * math.sqrt(252)
            assert vol.values[ticker][i] == pytest.approx(expected, abs=SAMPLE_TOL)


def test_report_rows_summarize_the_vol_series(contract_sample):
    vol = _vol(contract_sample("risk.VolSeries"))
    for row in contract_sample("risk.Report")["rows"]:
        rets = vol.returns.values[row["ticker"]]
        ann_return = 252 * statistics.mean(rets)
        ann_vol = math.sqrt(252) * statistics.stdev(rets)
        assert row["ann_return"] == pytest.approx(ann_return, abs=SAMPLE_TOL)
        assert row["ann_vol"] == pytest.approx(ann_vol, abs=SAMPLE_TOL)
        assert row["sharpe"] == pytest.approx(row["ann_return"] / row["ann_vol"], abs=SAMPLE_TOL)
        assert row["last_vol"] == vol.values[row["ticker"]][-1]

    assert returns.dates == tuple(expected["dates"])
    assert returns.tickers == tuple(expected["tickers"])
    assert set(returns.values) == set(expected["values"])
    for ticker, want in expected["values"].items():
        assert len(returns.values[ticker]) == len(want)
        assert returns.values[ticker] == pytest.approx(want, abs=expected["tol"])
