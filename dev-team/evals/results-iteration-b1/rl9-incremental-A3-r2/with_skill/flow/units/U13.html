def load_settings(path: Path) -> RunSettings:
    """Read a TOML settings file into fully resolved `RunSettings`."""
    raw = tomllib.loads(path.read_text(encoding="utf-8"))
    _check_names(RunSettings, {**_TOP_DEFAULTS, "opponents": {}, "generations": 0, **raw}, "")

def _check_names(cls: type, table: dict[str, Any], prefix: str) -> None:
    """Raise `ValueError` naming the first unknown, missing or unknown-valued key of `table`."""
    known = {f.name: f for f in fields(cls)}
    for key in table:
        if key not in known:
            raise ValueError(f"{prefix}{key}: unknown setting")
    for name, field in known.items():
        if name not in table:
            if field.default is MISSING:
                raise ValueError(f"{prefix}{name}: missing setting")
            continue
        if get_origin(field.type) is Literal and table[name] not in get_args(field.type):
            raise ValueError(f"{prefix}{name}: unknown value {table[name]!r}")


def _check_opponents(opponents: dict[str, Any]) -> None:
    for team, kind in opponents.items():
        if kind not in _OPPONENT_KINDS:
            raise ValueError(f"opponents.{team}: unknown value {kind!r}")


def _resolve(raw: dict[str, Any]) -> RunSettings:
    """`raw` with every unspecified field at its default."""
    env = raw["env"]
    opponents = dict(raw.get("opponents", {}))
    rewards = {**(_TANK_REWARDS if env == "tank" else {}), **raw.get("rewards", {})}

            "rewards": Rewards(**rewards),
            "exploration": Exploration(**raw.get("exploration", {})),
            "params": AgentParams(**raw["params"]),
        }
    )

from rllab.core import AgentParams, Exploration, Rewards, RunSettings
from rllab.core.settings import load_settings, with_seed

def _toml(data: dict) -> str:
    """A RunSettings sample as a settings file: scalars first, then one table per section."""
    lines = [f"{k} = {json.dumps(v)}" for k, v in data.items() if not isinstance(v, dict)]
    for section, table in data.items():
        if isinstance(table, dict):
            lines.append(f"\n[{section}]")
            lines += [f"{k} = {json.dumps(v)}" for k, v in table.items()]
    return "\n".join(lines) + "\n"


def _write(tmp_path: Path, text: str) -> Path:
    path = tmp_path / "settings.toml"
    path.write_text(text, encoding="utf-8")
    return path


def _ladder_with(tmp_path: Path, old: str, new: str) -> Path:
    text = FILE_SAMPLES["dqn_compact"]
    assert old in text
    return _write(tmp_path, text.replace(old, new, 1))


def test_ladder_file_loads_to_the_run_settings_sample(tmp_path):
    loaded = load_settings(_write(tmp_path, FILE_SAMPLES["dqn_compact"]))
    assert loaded == codec.sample("core.RunSettings", "dqn_compact", RunSettings)


@pytest.mark.parametrize("case", sorted(SETTINGS_SAMPLES))
def test_every_run_settings_sample_round_trips(tmp_path, case):
    text = _toml(SETTINGS_SAMPLES[case])
    assert load_settings(_write(tmp_path, text)) == codec.sample(

def test_unspecified_fields_take_the_defaults(tmp_path):
    text = (
        'name = "bare"\nenv = "corridor"\nview = "index"\nagent = "tabular"\n'
        "steps = 100\nseed = 3\neval_every = 10\neval_games = 2\n\n"
        "[params]\nlr = 0.1\ngamma = 0.9\n"
    )
    loaded = load_settings(_write(tmp_path, text))
    assert loaded.n_agents == 1
    assert loaded.opponents == {}
    assert loaded.self_play == "none"
    assert loaded.pool_latest == 0.5
    assert loaded.generations == 0
    assert loaded.ablation == "none"
    assert loaded.rewards == Rewards(food=1.0, death=-1.0, step=0.0)
    assert loaded.exploration == Exploration(start=1.0, end=0.05, decay_steps=50_000)
    assert loaded.params == AgentParams(lr=0.1, gamma=0.9)

    assert load_settings(_write(tmp_path, _toml(data))).generations == 10

    settings = codec.sample("core.RunSettings", "dqn_compact", RunSettings)
    seeded = with_seed(settings, 4)
    assert seeded.seed == 4
    assert settings.seed == 0
    assert seeded == replace(settings, seed=4)
    others = [f.name for f in fields(RunSettings) if f.name != "seed"]
    assert all(getattr(seeded, n) == getattr(settings, n) for n in others)
