"""PL-02 step 4: annualize the volatility."""

from __future__ import annotations

import math
from dataclasses import replace

from risk.types import VolSeries


def annualize(vol: VolSeries, periods_per_year: int = 252) -> VolSeries:
    """Each daily volatility times sqrt(periods_per_year), gaps kept (RISK-03).

    Raises ValueError if `vol` is already annualized.
    """
    if vol.annualized:
        raise ValueError("vol is already annualized")
    scale = math.sqrt(periods_per_year)
    values = {
        ticker: tuple(None if v is None else v * scale for v in series)
        for ticker, series in vol.values.items()
    }
    return replace(vol, annualized=True, values=values)

def _daily(values: dict[str, tuple[float | None, ...]]) -> VolSeries:
    n = len(next(iter(values.values())))
    returns = Returns(
        base_date="2024-01-01",
        dates=tuple(f"2024-01-{d:02d}" for d in range(2, 2 + n)),
        tickers=tuple(values),
        values={t: (0.0,) * n for t in values},
    )
    return VolSeries(returns=returns, window=2, annualized=False, values=values)

    # sqrt(252) = 15.874507866387544
    out = annualize(_daily({"AAA": (None, None, 0.01, 0.02)}))
