+
+from collections.abc import Mapping
+from typing import Any
+
+import gymnasium as gym
+import numpy as np
+
+from rllab.core import SOLO, ActionSpace, Agent, Env
+
+TRAINING_SEEDS = 1_000_000
+"""Game seeds stay below every evaluation seed (DATA-01)."""
+
+
+class GymAdapter(gym.Env[np.ndarray, np.ndarray | int]):
+    """`team`'s one agent as a `gymnasium.Env`: unbatched observations, one action.
+
+    Every other team is played by its frozen agent in `others`, greedily (AD-15). Each reset
+    starts a game on the next seed drawn from a generator seeded by the latest
+    `reset(seed=...)`, so the same seed gives the same games.
+    """
+
+    def __init__(
+        self, env: Env, *, team: str = SOLO, others: Mapping[str, Agent] | None = None
+    ) -> None:
+        others = dict(others or {})
+        if team not in env.teams:
+            raise ValueError(f"team {team!r} is not one of {sorted(env.teams)}")
+        spec = env.teams[team]
+        if spec.n_agents != 1:
+            raise ValueError(f"the adapter needs team {team!r} to have one agent, got {spec}")
+        missing = sorted(set(env.teams) - {team} - set(others))
+        if missing:
+            raise ValueError(f"no frozen agent in `others` for teams {missing}")
+        self.env = env
+        self.team = team
+        self.others = others
+        self.observation_space = gym.spaces.Box(0.0, 1.0, shape=spec.obs_shape, dtype=np.float32)
+        self.action_space = _action_space(spec.actions)
+        self._games = np.random.default_rng(0)
+        self._obs: dict[str, np.ndarray] = {}
+
+    def reset(
+        self, *, seed: int | None = None, options: dict[str, object] | None = None
+    ) -> tuple[np.ndarray, dict[str, float]]:
+        super().reset(seed=seed)
+        if seed is not None:
+            self._games = np.random.default_rng(seed)
+        self._obs, info = self.env.reset(int(self._games.integers(TRAINING_SEEDS)))
+        return self._obs[self.team][0], dict(info)
+
+    def step(
+        self, action: np.ndarray | int
+    ) -> tuple[np.ndarray, float, bool, bool, dict[str, float]]:
+        actions = {
+            name: agent.act(self._obs[name], explore=False) for name, agent in self.others.items()
+        }
+        actions[self.team] = self._batched(action)
+        result = self.env.step(actions)
+        self._obs = result.obs
+        reward = float(result.reward[self.team][0])
+        return (
+            self._obs[self.team][0],
+            reward,
+            result.terminated,
+            result.truncated,
+            dict(result.info),
+        )
+

+        if isinstance(self.action_space, gym.spaces.Discrete):

+        return np.asarray(action, dtype=np.float32).reshape(1, -1)

+    """One game as the adapter shows it: spaces, the reset, then each step's action and result."""

+    obs_shape: tuple[int, ...]
+    n_actions: int

+class ReplayGym(gym.Env[np.ndarray, Any]):
+    """Serves a recording: `reset` rewinds to its start, `step` returns the next result.
+
+    `step` raises `RuntimeError` once the recorded game is over or before the first `reset`.
+    """
+
+    def __init__(self, recording: GymRecording) -> None:
+        self.recording = recording
+        self.observation_space = gym.spaces.Box(
+            0.0, 1.0, shape=recording.obs_shape, dtype=np.float32
+        )
+        self.action_space = gym.spaces.Discrete(recording.n_actions)
+        self._t = -1
+
+    def reset(
+        self, *, seed: int | None = None, options: dict[str, object] | None = None
+    ) -> tuple[np.ndarray, dict[str, float]]:
+        super().reset(seed=seed)
+        self._t = 0
+        return self.recording.obs.copy(), dict(self.recording.info)
+
+    def step(
+        self, action: np.ndarray | int
+    ) -> tuple[np.ndarray, float, bool, bool, dict[str, float]]:
+        r, t = self.recording, self._t
+        if not 0 <= t < len(r.actions):
+            raise RuntimeError("step called while no game is running")
+        self._t += 1
+        return (
+            r.observations[t].copy(),
+            r.rewards[t],
+            r.terminated[t],
+            r.truncated[t],
+            dict(r.infos[t]),
+        )
