"""PL-02 step 3: rolling volatility from earlier returns."""

from __future__ import annotations

from statistics import stdev

from risk.types import Returns, VolSeries


def rolling_vol(returns: Returns, window: int) -> VolSeries:
    """Sample stdev of the `window` returns before each date, never its own (RISK-02).

    Raises ValueError if `window` < 2.
    """
    if window < 2:
        raise ValueError(f"window must be at least 2, got {window}")
    values = {
        ticker: tuple(
            stdev(series[:i]) if i >= window else None for i in range(len(series))
        )
        for ticker, series in returns.values.items()
    }
    return VolSeries(returns=returns, window=window, annualized=False, values=values)

def _returns(values: dict[str, tuple[float, ...]], n: int) -> Returns:
    dates = tuple(f"2024-01-{d:02d}" for d in range(2, 2 + n))
    return Returns(base_date="2024-01-01", dates=dates, tickers=tuple(values), values=values)


def test_sample_returns_give_none_until_window_then_earlier_stdev(contract_sample):
    data = contract_sample("risk.Returns")
    returns = Returns(
        base_date=data["base_date"],
        dates=tuple(data["dates"]),
        tickers=tuple(data["tickers"]),
        values={t: tuple(v) for t, v in data["values"].items()},
    )
    vol = rolling_vol(returns, 2)
    assert vol.returns is returns
    assert vol.window == 2
    assert vol.annualized is False
    assert set(vol.values) == {"A", "B"}
    assert all(len(v) == len(returns.dates) for v in vol.values.values())
    assert vol.values["A"][:2] == (None, None)
    # two returns 0.014 and -0.007: sample stdev |a - b| / sqrt(2)
    assert vol.values["A"][2] == pytest.approx(0.021 / math.sqrt(2), abs=1e-12)
    # B's first two returns are equal
    assert vol.values["B"][2] == pytest.approx(0.0, abs=1e-12)


def test_uses_every_earlier_return_never_its_own():
    returns = _returns({"X": (0.0, 2.0, 4.0, 100.0)}, 4)
    vol = rolling_vol(returns, 2)
    assert vol.values["X"][0] is None
    assert vol.values["X"][1] is None
    assert vol.values["X"][2] == pytest.approx(math.sqrt(2.0))
    # mean 2 of (0, 2, 4): squares 4 + 0 + 4 over 2 -> 4; the 100 is not used
    assert vol.values["X"][3] == pytest.approx(2.0)


def test_window_three_waits_for_three_earlier_returns():
    returns = _returns({"X": (1.0, 2.0, 3.0, 4.0)}, 4)
    assert rolling_vol(returns, 3).values["X"] == (None, None, None, pytest.approx(1.0))
