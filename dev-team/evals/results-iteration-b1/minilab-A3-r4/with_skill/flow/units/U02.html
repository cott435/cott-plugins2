base_commit() {

# A boundary contract is frozen once its folder exists at the base (CR-09): each folder under
# tests/contracts/ at the base, and any file directly in it. A folder new since the base is the
# interface task freezing it, so that task may write it; the next task's base freezes it. A frozen
# folder the task's .claude/.task_writes names (from its Writes:, by dt claim) is a re-freeze
# interface task's, so it is open to that task; dt land checks the Writes: against the plan.
CONTRACTS_GLOB="tests/contracts/**"
claimed_contract() {
  local line
  [ -s .claude/.task_writes ] || return 1
  while IFS= read -r line; do
    line="${line%%[[:space:]]}"
    line="${line%/}"; line="${line%/\*\*}"
    [ "$line" = "$1" ] && return 0
  done <.claude/.task_writes
  return 1
}
frozen_contracts() {
  local base meta path
  base=$(base_commit) || return 0
  git ls-tree "$base" -- tests/contracts/ 2>/dev/null | while IFS=$'\t' read -r meta path; do
    case "$meta" in
      *" tree "*) claimed_contract "$path" || echo "$path/**" ;;
      *) echo "$path" ;;
    esac
  done
}

# Fail if anything matching the given pathspecs changed since the base, untracked files included.
unchanged_since_base() {
  local label="$1" base new
  shift
  echo "-> $label unchanged since $(git rev-parse -q --verify "$BASE_REF" >/dev/null 2>&1 && echo "$BASE_REF" || echo HEAD)" >&2
  if ! base=$(base_commit); then
    echo "no commit to compare against: commit the tester's work first" >&2
    rc=1; failed+=("$label: no commit"); return
  fi
  git --no-pager diff --exit-code --stat "$base" -- "$@" >&2 || { rc=1; failed+=("$label changed"); }
  new=$(git ls-files --others --exclude-standard -- "$@")
  if [ -n "$new" ]; then
    printf 'new untracked files in %s:\n%s\n' "$label" "$new" >&2
    rc=1; failed+=("$label changed")
  fi
}

def parse_diff(text: str) -> Diff:
    """Added lines (with new line numbers), removed lines and deleted files, per file."""
    diff = Diff()
    old = new = ""
    line_no = 0
    in_header = False
    for line in text.splitlines():
        if line.startswith("diff --git "):
            old, new, in_header = "", "", True
        elif in_header and line.startswith("--- "):
            old = _diff_path(line[4:])
        elif in_header and line.startswith("+++ "):
            new = _diff_path(line[4:])
            if new == "/dev/null":
                diff.deleted.add(old)
        elif match := HUNK.match(line):
            in_header, line_no = False, int(match[1])
        elif in_header:
            continue
        elif line.startswith("+"):
            diff.added.setdefault(new, []).append((line_no, line[1:]))
            line_no += 1
        elif line.startswith("-"):
            diff.removed.setdefault(old, []).append(line[1:])
    return diff

    if isinstance(node, ast.Assign):
        return any(_is_sys_path(target) for target in node.targets)
    if isinstance(node, ast.AugAssign):
        return _is_sys_path(node.target)
    return False


def _outside_finding(path: str, node: ast.AST) -> Finding | None:
    """CR-08: an import from references/ or probes/, or either folder added to sys.path."""
    line = getattr(node, "lineno", 0)
    for folder, word in OUTSIDE.items():
        if any(_in_folder(name, folder) for name in _imported_modules(node)):
            return Finding(f"{folder}-import", path, line, f"import from {folder}/")
        if _changes_sys_path(node) and _mentions(node, word):
            return Finding(f"{folder}-path", path, line, f"{folder}/ added to sys.path")
    return None


def _asserts(func: ast.AST) -> bool:
    for node in ast.walk(func):
        if isinstance(node, ast.Assert):
            return True
        name = _name(node) if isinstance(node, ast.Call) else None
        if name and (
            name.startswith("assert") or name in {"raises", "warns", "deprecated_call", "fail"}
        ):
            return True
    return False


def _stub_pinned(path: str, node: ast.AST, test: bool) -> list[Finding]:
    """A test naming NotImplementedError: it passes only while a stub stands (CR-02)."""
    if not test or isinstance(node, ast.Call) or _name(node) != "NotImplementedError":
        return []
    message = "a test expects NotImplementedError: check the signature, not the stub body"
    return [Finding("stub-pinned", path, getattr(node, "lineno", 0), message)]


def ast_findings(path: str, lines: set[int]) -> list[Finding]:
    try:
        tree = ast.parse(Path(path).read_text(encoding="utf-8"))
    except (SyntaxError, UnicodeDecodeError, OSError):
        return []  # ruff reports syntax errors; a deleted file has nothing to check
    exempt, test = _exempt_functions(tree), is_test_file(path)
    found: list[Finding] = []
    for node in ast.walk(tree):
        if not _touches(node, lines):
            continue
        if outside := _outside_finding(path, node):
            found.append(outside)
        found += _stub_pinned(path, node, test)
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            if (
                node not in exempt
                and not path.endswith(".pyi")
                and _is_trivial(node.body)
                and not _task_stub(node.body)
            ):
                found.append(
                    Finding("stub", path, node.lineno, f"`{node.name}` has a placeholder body")
                )
            if test and node.name.startswith("test") and not _asserts(node):
                found.append(
                    Finding(
                        "test-without-assertion",
                        path,
                        node.lineno,
                        f"`{node.name}` asserts nothing",
                    )
                )
        elif isinstance(node, ast.ExceptHandler) and _is_trivial(node.body):
            if (None if node.type is None else _name(node.type)) in TRIVIAL_EXCEPT:
                found.append(
                    Finding("swallowed-exception", path, node.lineno, "exception swallowed")
                )
    return found

def _loosened(was: Any, now: Any, direction: int) -> bool:
    if not isinstance(was, (int, float)):
        return False
    if not isinstance(now, (int, float)):
        return True  # a limit that disappeared
    return (now - was) * direction > 0


def _list_findings(old: dict[str, Any], new: dict[str, Any]) -> list[str]:
    out = [f"{k} lost {sorted(_as_set(_get(old, k)) - _as_set(_get(new, k)))}" for k in NO_SHRINK]
    for k in NO_GROW:
        gained = _as_set(_get(new, k)) - _as_set(_get(old, k))
        out.append(f"{k} gained {sorted(g for g in gained if not OUTSIDE_EXCLUDE.fullmatch(g))}")
    for table in NO_GROW_TABLES:
        was, now = _get(old, table) or {}, _get(new, table) or {}
        out += [
            f"{table}.{key} gained {sorted(_as_set(codes) - _as_set(was.get(key)))}"
            for key, codes in now.items()
            if _as_set(codes) - _as_set(was.get(key))
        ]
    flags = _as_set((_get(new, "tool.pytest.ini_options.addopts") or "").split())
    out += [

        for f in STRICT_FLAGS

    out += [
        f"{k} changed (boundaries and type config change only through build-setup)"
        for k in MUST_MATCH
        if _get(old, k) != _get(new, k)

def _section(lines: list[str], title: str) -> tuple[int, int]:
    """1-based line range of a `## title` section, or (0, -1)."""
    start = next((i for i, t in enumerate(lines, 1) if t.strip().lower() == f"## {title}"), 0)
    if not start:
        return 0, -1
    end = next(
        (i - 1 for i, t in enumerate(lines, 1) if i > start and t.startswith("## ")), len(lines)
    )
    return start, end


def _table_rows(lines: list[str], start: int, end: int) -> list[tuple[int, list[str]]]:
    rows = []
    for n in range(start + 1, end + 1):
        text = lines[n - 1].strip()
        if not text.startswith("|"):
            continue
        cells = [c.strip() for c in text.strip("|").split("|")]
        if cells and cells[0] not in {"", "ID"} and not set(cells[0]) <= set("-: "):
            rows.append((n, cells))
    return rows


def constraints_findings(diff: Diff) -> list[Finding]:

def load_exceptions(today: date) -> list[tuple[str, str]]:
    """(rule cell, path glob) for each unexpired row in CONSTRAINTS.md's Exceptions table."""
    path = Path("CONSTRAINTS.md")

        return []
    lines = path.read_text(encoding="utf-8").splitlines()
    allowed = []

    return allowed

# Minilab — Spec

**Status:** Approved · **Last updated:** 2026-10-17 · **Brief:** [brief.md](brief.md)

## Features

### Prices (`DATA`)
**Serves:** Goal 1

**Requirements**
- **DATA-01** A `date,ticker,close` CSV is read row by row.
  - Given a CSV, when it is read, then each row comes back in file order, and a blank close is missing, not zero.
- **DATA-02** Unusable rows are dropped and duplicates resolved.
  - Given rows with blank, zero or negative closes and a repeated (date, ticker), when validated, then those rows are gone, the later of the duplicates wins, and rows are sorted by date, then ticker.
  - Given a requested ticker with no usable row, when validated, then a `KeyError` names it.
- **DATA-03** Every ticker sits on one calendar.
  - Given tickers that start on different dates and miss some dates, when aligned, then the calendar is the dates any ticker trades from the first date on which every ticker has a close, and a missing date takes that ticker's previous close.
- **DATA-04** A long gap is an error, not a fill.
  - Given a ticker missing more than 5 calendar dates in a row, when aligned, then a `GapError` names the ticker.

### Risk (`RISK`)
**Serves:** Goal 2

**Requirements**
- **RISK-01** Daily returns are log returns.
  - Given closes, when returns are computed, then r_t = ln(close_t / close_t−1), dated t.
- **RISK-02** Rolling volatility uses only earlier returns.
  - Given a window of w, when the volatility for date t is computed, then it is the sample standard deviation (n − 1) of the w returns before t, never t's own return, and there is no value until w earlier returns exist.
- **RISK-03** Volatility is annualized over trading days.
  - Given a daily volatility σ, when annualized, then the value is σ·√252.
- **RISK-04** Each ticker gets a whole-period summary.
  - Given returns, when summarized, then the annualized return is 252 × the mean, the annualized volatility is √252 × the sample standard deviation, the Sharpe ratio is their quotient (0 when the volatility is 0; no risk-free rate), and the last rolling volatility is reported.
- **RISK-05** Maximum drawdown comes with its dates.
  - Given closes, when the drawdown is computed, then wealth starts at 1 on the first date, and the result is the deepest fall below the running peak (trough / peak − 1, at most 0) with the date of that peak and the date of the trough; the first if tied. With no fall, both dates are the first date.

### Report (`REP`)
**Serves:** Goal 3

    cls: type, fields: dict[str, object]
) -> None:
    assert cls.__module__ == "risk.types"
    assert cls.__dataclass_params__.frozen
    assert [f.name for f in dataclasses.fields(cls)] == list(fields)
    assert typing.get_type_hints(cls) == fields

def assert_returns_valid(returns: Returns) -> None:
    """risk.Returns's invariants: dated after `base_date`; one value per date and ticker."""
    assert ISO_DATE.fullmatch(returns.base_date)
    assert list(returns.dates) == sorted(set(returns.dates))
    assert returns.base_date < returns.dates[0]
    assert list(returns.tickers) == sorted(set(returns.tickers))
    assert set(returns.values) == set(returns.tickers)
    for values in returns.values.values():
        assert len(values) == len(returns.dates)
        assert all(isinstance(v, float) and math.isfinite(v) for v in values)


def assert_vol_valid(vol: VolSeries) -> None:
    """risk.VolSeries's invariants: aligned to the returns; None until `window` earlier returns."""
    assert_returns_valid(vol.returns)
    assert set(vol.values) == set(vol.returns.tickers)
    for values in vol.values.values():
        assert len(values) == len(vol.returns.dates)
        assert all(v is None for v in values[: vol.window])
        assert all(isinstance(v, float) and v >= 0 for v in values[vol.window :])

def test_drawdown_sample_fits_the_type(contract_sample: Sample) -> None:
    fall, no_fall = (Drawdown(**case) for case in contract_sample("risk.Drawdown")["cases"])
    assert fall.depth < 0
    assert ISO_DATE.fullmatch(fall.peak_date)
    assert fall.peak_date < fall.trough_date
    assert no_fall.depth == 0
    assert ISO_DATE.fullmatch(no_fall.peak_date)
    assert no_fall.peak_date == no_fall.trough_date


def test_report_sample_fits_the_type(contract_sample: Sample) -> None:
    data = contract_sample("risk.Report")
    report = Report(path=data["path"], rows=tuple(data["rows"]))
    tickers = [row["ticker"] for row in report.rows]
    assert tickers == sorted(set(tickers))
    for row in report.rows:
        assert set(row) == REPORT_KEYS
        assert row["max_drawdown"] <= 0
        assert row["peak_date"] <= row["trough_date"]


def assert_vol_close(actual: VolSeries, expected: VolSeries) -> None:
    assert actual.returns.base_date == expected.returns.base_date
    assert actual.returns.dates == expected.returns.dates
    assert actual.returns.tickers == expected.returns.tickers
    assert (actual.window, actual.annualized) == (expected.window, expected.annualized)
    assert set(actual.values) == set(expected.values)
    for ticker, values in expected.values.items():
        assert actual.returns.values[ticker] == pytest.approx(expected.returns.values[ticker])
        assert actual.values[ticker] == pytest.approx(values)


def check_vol_series(vol_series: Callable[..., VolSeries], csv: Path, expected: VolSeries) -> None:
    """The risk.vol_series cases, for any provider: the fake now, the real one once it is built."""
    vol = vol_series(csv, window=2)
    assert_vol_close(vol, expected)
    assert_vol_valid(vol)
    last = expected.returns.tickers[-1]
    one = vol_series(csv, window=2, tickers=[last])
    assert one.returns.tickers == (last,)
    assert set(one.values) == {last}
    assert one.values[last] == pytest.approx(expected.values[last])
    with pytest.raises(ValueError, match="window"):
        vol_series(csv, window=1)
    with pytest.raises(KeyError, match="NOPE"):
        vol_series(csv, window=2, tickers=["NOPE"])
