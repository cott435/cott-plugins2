+"""Unit tests for the write_report spine: JSON rows, one per ticker, sorted."""
+
+from __future__ import annotations
+
+import json
+from pathlib import Path
+
+import pytest
+
+from risk.report import write_report
+from risk.types import Drawdown, TickerSummary
+
+CONTRACTS = Path(__file__).parents[2] / "contracts"
+
+
+def _summary_sample() -> TickerSummary:
+    return TickerSummary(
+        **json.loads((CONTRACTS / "risk.TickerSummary" / "sample.json").read_text())
+    )
+
+
+def _drawdown_samples() -> list[Drawdown]:
+    data = json.loads((CONTRACTS / "risk.Drawdown" / "sample.json").read_text())
+    return [Drawdown(**d) for d in data["drawdowns"]]
+
+
+def _report_rows() -> list[dict[str, object]]:
+    rows: list[dict[str, object]] = json.loads(
+        (CONTRACTS / "risk.Report" / "sample.json").read_text()
+    )["rows"]
+    return rows
+
+
+def _bbb_summary() -> TickerSummary:
+    return TickerSummary(
+        ticker="BBB",
+        ann_return=1.512,
+        ann_vol=0.0354964786985977,
+        sharpe=42.59577443831724,
+        last_vol=0.05612486080160912,
+    )
+
+
+def test_summary_sample_gives_one_json_row(tmp_path: Path) -> None:
+    out = tmp_path / "report.json"
+    aaa_drawdown = _drawdown_samples()[0]
+    report = write_report([_summary_sample()], [aaa_drawdown], out)
+    expected = _report_rows()[0]
+    assert report.path == str(out)
+    assert report.rows == (expected,)
+    assert json.loads(out.read_text()) == {"tickers": [expected]}
+
+
+def test_rows_are_sorted_by_ticker_and_match_the_file(tmp_path: Path) -> None:
+    out = tmp_path / "report.json"
+    report = write_report([_bbb_summary(), _summary_sample()], _drawdown_samples(), out)
+    assert [r["ticker"] for r in report.rows] == ["AAA", "BBB"]
+    assert list(report.rows) == _report_rows()
+    assert json.loads(out.read_text())["tickers"] == _report_rows()
