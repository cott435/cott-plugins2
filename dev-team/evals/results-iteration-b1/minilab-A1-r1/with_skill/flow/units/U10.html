"""Intent tests for risk.max_drawdown: the deepest fall with its peak and trough dates (RISK-05, TC-08)."""

from __future__ import annotations

import pytest
from hypothesis import given
from hypothesis import strategies as st

from risk import Drawdown, Returns, max_drawdown


def _returns(payload: dict) -> Returns:
    return Returns(
        base_date=payload["base_date"],
        dates=tuple(payload["dates"]),
        tickers=tuple(payload["tickers"]),
        values={t: tuple(v) for t, v in payload["values"].items()},
    )


def _tc08(golden) -> tuple[tuple[Drawdown, ...], dict]:
    data, expected = golden("risk", "TC-08")
    return max_drawdown(_returns(data["returns"])), expected


@pytest.mark.intent
def test_tc08_one_drawdown_per_ticker_in_order(golden):
    result, expected = _tc08(golden)
    assert [d.ticker for d in result] == [row["ticker"] for row in expected["drawdowns"]]
    assert all(isinstance(d, Drawdown) for d in result)


@pytest.mark.intent
@pytest.mark.parametrize("index", [0, 1, 2], ids=["X-fall", "Y-rising", "Z-first-day-fall"])
def test_tc08_depth_within_tolerance(golden, index):
    result, expected = _tc08(golden)
    row = expected["drawdowns"][index]
    got = {d.ticker: d for d in result}[row["ticker"]]
    assert got.depth == pytest.approx(row["depth"], abs=expected["tol"])


@pytest.mark.intent
@pytest.mark.parametrize("index", [0, 1, 2], ids=["X-fall", "Y-rising", "Z-first-day-fall"])
def test_tc08_peak_and_trough_dates_exact(golden, index):
    result, expected = _tc08(golden)
    row = expected["drawdowns"][index]
    got = {d.ticker: d for d in result}[row["ticker"]]
    assert (got.peak_date, got.trough_date) == (row["peak_date"], row["trough_date"])


_RETURN = st.floats(min_value=-0.5, max_value=0.5, allow_nan=False, allow_infinity=False)


@st.composite
def _random_returns(draw) -> Returns:
    n = draw(st.integers(min_value=1, max_value=12))
    tickers = draw(st.lists(st.sampled_from(["AAA", "BBB", "CCC"]), min_size=1, max_size=3, unique=True))
    dates = tuple(f"2025-01-{day:02d}" for day in range(2, 2 + n))
    values = {t: tuple(draw(st.lists(_RETURN, min_size=n, max_size=n))) for t in tickers}
    return Returns(base_date="2025-01-01", dates=dates, tickers=tuple(tickers), values=values)


@pytest.mark.intent
@given(_random_returns())
def test_drawdown_invariants_hold(returns):
    result = max_drawdown(returns)
    assert [d.ticker for d in result] == list(returns.tickers)
    all_dates = (returns.base_date, *returns.dates)
    for d in result:
        assert -1.0 < d.depth <= 0.0
        assert d.peak_date in all_dates
        assert d.trough_date in all_dates
        assert d.peak_date <= d.trough_date
        if d.depth == 0.0:
            assert (d.peak_date, d.trough_date) == (returns.base_date, returns.base_date)


@pytest.mark.intent
@given(_random_returns())
def test_no_fall_gives_zero_depth_at_first_date(returns):
    rising = Returns(
        base_date=returns.base_date,
        dates=returns.dates,
        tickers=returns.tickers,
        values={t: tuple(abs(r) for r in v) for t, v in returns.values.items()},
    )
    for d in max_drawdown(rising):
        assert d.depth == 0.0
        assert (d.peak_date, d.trough_date) == (returns.base_date, returns.base_date)
