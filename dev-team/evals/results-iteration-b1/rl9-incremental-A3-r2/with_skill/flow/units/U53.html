def train_reference(settings: RunSettings, *, root: Path = Path("runs")) -> RunRecord:
    """Train the library agent on the settings' environment, with the same curve points."""
    make_env = _env_factory(settings)
    writer = create_run(settings, root=root)
    kwargs: dict[str, Any] = dqn_kwargs(settings)  # each value typed by the library's DQN
    model = DQN("MlpPolicy", GymAdapter(make_env()), **kwargs)
    player = _TrainingPlayer(model)
    try:
        model.learn(settings.steps, callback=_Curve(make_env, player, settings, writer))
        writer.save_checkpoint(player)
        writer.save_games(_recorded_games(make_env, player, settings))
    except KeyboardInterrupt:
        writer.finish("incomplete")
        raise
    return writer.finish("complete")


def _env_factory(settings: RunSettings) -> Callable[[], Env]:
    """Snake with the compact view: the one reference this milestone trains (AD-04)."""
    if settings.agent != "reference":
        raise ValueError(f"agent {settings.agent!r}: train_reference needs agent 'reference'")
    if settings.env != "snake" or settings.view != "compact":
        raise ValueError(f"the reference cannot train {settings.env}/{settings.view}")
    rewards = settings.rewards

        return Snake(view="compact", rewards=rewards)

    return make_env


class _TrainingPlayer(ReferencePlayer):
    """The model being trained, played greedily as an Agent."""

    def __init__(self, model: DQN) -> None:
        self.model = model


class _Curve(BaseCallback):
    """A curve point at step 0, every `eval_every` steps and at the last step."""

    def __init__(
        self,
        make_env: Callable[[], Env],
        player: _TrainingPlayer,
        settings: RunSettings,
        writer: RunWriter,
    ) -> None:
        super().__init__()
        self.make_env = make_env
        self.player = player
        self.settings = settings
        self.writer = writer
        self.spec = EvalSpec(settings.name, settings.seed, n_games=settings.eval_games)
        self.games = 0
        self.start = monotonic()

    def _on_training_start(self) -> None:
        self.start = monotonic()
        self._add_point(0)

    def _on_step(self) -> bool:
        self.games += int(np.sum(self.locals["dones"]))
        step = self.num_timesteps
        if step % self.settings.eval_every == 0 or step == self.settings.steps:
            self._add_point(step)
        return True

    def _add_point(self, step: int) -> None:
        scores = evaluate(self.make_env, {SOLO: self.player}, self.spec).scores
        elapsed = monotonic() - self.start
        self.writer.add_point(CurvePoint(step, self.games, float(np.mean(scores)), elapsed))


def _recorded_games(

from pathlib import Path

import numpy as np
from stable_baselines3 import DQN

from rllab.core import Agent, Transition


class ReferencePlayer(Agent):
    """The library model's deterministic action, batch-first with one agent."""

    def __init__(self, checkpoint: Path) -> None:
        self.model = DQN.load(checkpoint, device="cpu")

    def act(self, obs: np.ndarray, *, explore: bool) -> np.ndarray:
        """The greedy action for each row of `obs`; the reference never explores."""
        action, _ = self.model.predict(obs, deterministic=True)
        return np.asarray(action, dtype=np.int64)

    def learn(self, t: Transition) -> None:
        """A trained reference does not learn while it plays: the transition is dropped."""
        del t

    def save(self, path: Path) -> None:
        self.model.save(path)

    def load(self, path: Path) -> None:
        self.model = DQN.load(path, device="cpu")
