@pytest.mark.intent
def test_tc04_geometric_closes_give_closed_form_log_returns(golden):
    inp, expected = golden("risk", "TC-04")
    tol = expected["tol"]

    returns = log_returns(_prices(inp["prices"]))

    assert isinstance(returns, Returns)
    assert returns.base_date == expected["base_date"]
    assert returns.dates == tuple(expected["dates"])
    assert returns.tickers == tuple(expected["tickers"])
    for ticker in expected["tickers"]:
        assert len(returns.values[ticker]) == len(expected["values"][ticker])
        assert list(returns.values[ticker]) == pytest.approx(
            expected["values"][ticker], rel=0, abs=tol
        )


@pytest.mark.intent
