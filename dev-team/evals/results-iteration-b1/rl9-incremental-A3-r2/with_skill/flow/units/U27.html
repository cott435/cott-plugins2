import math
from collections.abc import Mapping, Sequence

import numpy as np

from rllab.compare.types import Claim, ClaimSpec
from rllab.evaluate import ScoreSet
from rllab.viz import Chart

PERMUTATIONS = 9999
PERMUTATION_BATCH = 1000
PERMUTATION_SEED = 0


def run_claims(
    scores: Mapping[str, Sequence[ScoreSet]],
    specs: Sequence[ClaimSpec],
    *,
    alpha: float = 0.05,
    charts: Sequence[Chart] = (),
    out: Path,
) -> list[Claim]:
    """Test every claim with one correction over all of them, and write `out/report.md`."""
    pooled = pool_scores(scores)
    check_players(specs, pooled)
    p_values = [permutation_p_value(*tested_pair(spec, pooled)) for spec in specs]
    corrected = holm(p_values)
    method = (
        "statistic: difference in pooled mean score (one-sided); "
        f"null: permutation of player labels over pooled games ({PERMUTATIONS} shuffles); "
        f"correction: Holm over {len(specs)} claim(s)"
    )
    claims = [
        Claim(
            spec=spec,
            mean_a=float(pooled[spec.a].mean()),
            mean_b=float(pooled[spec.b].mean()),
            p_value=p,
            p_corrected=p_corr,
            passed=claim_passed(spec, pooled, p_corr, alpha),
            method=method,
        )
        for spec, p, p_corr in zip(specs, p_values, corrected, strict=True)
    ]
    write_report(out / "report.md", pooled, claims, charts)
    return claims


def pool_scores(scores: Mapping[str, Sequence[ScoreSet]]) -> dict[str, np.ndarray]:
    """Each player's games pooled across its training seeds."""
    return {
        player: np.concatenate([s.scores for s in sets]).astype(np.float64)
        for player, sets in scores.items()
    }


def check_players(specs: Sequence[ClaimSpec], pooled: Mapping[str, np.ndarray]) -> None:
    """Raise `ValueError` naming any player a spec uses that has no scores."""
    for spec in specs:
        missing = [p for p in (spec.a, spec.b) if p not in pooled]
        if missing:
            raise ValueError(f"claim {spec.name!r} names players without scores: {missing}")


def tested_pair(
    spec: ClaimSpec, pooled: Mapping[str, np.ndarray]
) -> tuple[np.ndarray, np.ndarray]:
    """The (x, y) whose H1 is mean(x) > mean(y): a over b, or b over a for "not worse"."""
    if spec.direction == "greater":
        return pooled[spec.a], pooled[spec.b]
    return pooled[spec.b], pooled[spec.a]


def permutation_p_value(x: np.ndarray, y: np.ndarray) -> float:
    """One-sided p-value of mean(x) - mean(y) under shuffled labels, (hits + 1) / (n + 1)."""
    rng = np.random.default_rng(PERMUTATION_SEED)
    both = np.concatenate([x, y])
    observed = x.mean() - y.mean()
    hits = 0
    for start in range(0, PERMUTATIONS, PERMUTATION_BATCH):
        n = min(PERMUTATION_BATCH, PERMUTATIONS - start)
        shuffled = rng.permuted(np.tile(both, (n, 1)), axis=1)
        diffs = shuffled[:, : len(x)].mean(axis=1) - shuffled[:, len(x) :].mean(axis=1)
        hits += int(np.count_nonzero(diffs >= observed))
    return (hits + 1) / (PERMUTATIONS + 1)


def holm(p_values: Sequence[float]) -> list[float]:
    """Holm step-down adjusted p-values, in input order."""
    m = len(p_values)
    adjusted = [0.0] * m
    running = 0.0
    for rank, i in enumerate(sorted(range(m), key=lambda k: p_values[k])):
        running = max(running, min(1.0, (m - rank) * p_values[i]))
        adjusted[i] = running
    return adjusted


def claim_passed(
    spec: ClaimSpec, pooled: Mapping[str, np.ndarray], p_corrected: float, alpha: float
) -> bool:
    """`greater`: significant; `not_worse`: "a worse" not significant and the ratio bar holds."""
    significant = p_corrected <= alpha
    if spec.direction == "greater":
        return significant
    if spec.min_ratio is None:
        return not significant
    return not significant and ratio_holds(
        float(pooled[spec.a].mean()), float(pooled[spec.b].mean()), spec.min_ratio
    )


def ratio_holds(mean_a: float, mean_b: float, min_ratio: float) -> bool:
    """`mean_a >= min_ratio * mean_b`, inclusive up to float rounding."""
    bar = min_ratio * mean_b
    return mean_a >= bar or math.isclose(mean_a, bar)


def write_report(
    path: Path, pooled: Mapping[str, np.ndarray], claims: Sequence[Claim], charts: Sequence[Chart]
) -> None:
    """Write each player's mean and spread, each claim, and a link to each cross-play chart."""
    lines = ["# Report", "", "## Players", ""]
    lines += [
        f"- {player}: mean {s.mean():.3f}, std {s.std():.3f}, "
        f"min {s.min():.3f}, max {s.max():.3f} over {s.size} games"
        for player, s in pooled.items()
    ]
    lines += ["", "## Claims", ""]
    lines += [claim_line(claim) for claim in claims]
    if claims:
        lines += ["", f"Method: {claims[0].method}"]
    if charts:
        lines += ["", "## Cross-play", ""]
        lines += [f"- [{chart.path.name}]({chart.path})" for chart in charts]
    path.write_text("\n".join(lines) + "\n")
