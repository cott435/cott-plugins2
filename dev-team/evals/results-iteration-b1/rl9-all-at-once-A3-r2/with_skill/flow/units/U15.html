def make_learners(config: RunConfig, spec: WorldSpec) -> dict[str, Learner]:
    """One learner per team of `spec`, each built from its team's method config.

    children = np.random.SeedSequence(config.seed).spawn(len(spec.teams))

        )
        for team, child in zip(spec.teams, children, strict=True)
    }
