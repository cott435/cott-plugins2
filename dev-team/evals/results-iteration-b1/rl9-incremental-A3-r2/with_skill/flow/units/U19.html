class LoadingAgent:
    """A player that only remembers the checkpoints it was loaded from."""

    def __init__(self) -> None:
        self.loaded: list[Path] = []

    def act(self, obs: np.ndarray, *, explore: bool) -> np.ndarray:
        return np.zeros(len(obs), dtype=np.int64)

    def learn(self, t: Transition) -> None:
        self.loaded = list(self.loaded)

    def save(self, path: Path) -> None:
        path.write_text("", encoding="utf-8")

    def load(self, path: Path) -> None:
        self.loaded.append(path)
