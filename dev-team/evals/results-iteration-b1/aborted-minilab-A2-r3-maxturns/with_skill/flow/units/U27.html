65	def daily_returns(prices: pd.DataFrame) -> pd.DataFrame:
66	    """Log returns of a dates x tickers close panel."""
67	    return np.log(prices / prices.shift(1)).dropna(how="all")
68	
69	
70	def rolling_vol(returns: pd.DataFrame, window: int = 20) -> pd.DataFrame:
71	    """Rolling volatility over the last `window` days, today included."""
72	    return returns.rolling(window).apply(lambda x: np.std(x), raw=True)
73	
74	
75	def annualize(vol: pd.DataFrame) -> pd.DataFrame:
76	    """Annualize to a calendar year, like my brokerage statements: 365 days."""
77	    return vol * np.sqrt(365)
78	
79	
80	def summary(returns: pd.DataFrame) -> pd.DataFrame:
81	    """Yearly return and volatility per ticker."""
82	    return pd.DataFrame({"ann_return": returns.mean() * 365, "ann_vol": returns.std(ddof=0) * np.sqrt(365)})
83	
84	
85	def max_drawdown(prices: pd.DataFrame) -> pd.Series:
86	    """Deepest fall below the running high."""
87	    return (prices / prices.cummax() - 1).min()

31	    """Independent oracle: wealth 1 at the base date, worst trough/peak - 1 over all pairs."""

from __future__ import annotations

import pytest

from risk.summary import TickerSummary, summary_stats
from risk.types import Returns, VolSeries


def _returns(payload: dict[str, Any]) -> Returns:
    return Returns(
        base_date=payload["base_date"],
        dates=tuple(payload["dates"]),
        tickers=tuple(payload["tickers"]),
        values={ticker: tuple(series) for ticker, series in payload["values"].items()},
    )


def _vol(payload: dict[str, Any]) -> VolSeries:
    return VolSeries(
        returns=_returns(payload["returns"]),
        window=payload["window"],
        annualized=payload["annualized"],
        values={ticker: tuple(series) for ticker, series in payload["values"].items()},
    )


def _one_ticker_vol(values: list[float], vols: list[float | None]) -> VolSeries:
    dates = tuple(f"2024-02-{day:02d}" for day in range(2, len(values) + 2))
    returns = Returns(base_date="2024-02-01", dates=dates, tickers=("X",), values={"X": tuple(values)})
    return VolSeries(returns=returns, window=2, annualized=True, values={"X": tuple(vols)})
