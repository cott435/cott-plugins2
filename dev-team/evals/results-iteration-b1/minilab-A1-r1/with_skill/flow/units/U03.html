+from __future__ import annotations
+
+import json
+from collections.abc import Sequence
+from pathlib import Path
+
+from risk import Returns, VolSeries
+
+SAMPLE = Path(__file__).resolve().parent.parent / "risk.VolSeries" / "sample.json"
+
+
+def vol_series(
+    path: str | Path, *, window: int = 20, tickers: Sequence[str] | None = None
+) -> VolSeries:
+    """The sample's series, whatever `path` names, narrowed to `tickers` (sorted).
+
+    The fake serves only its fixture's window; any other window is a `ValueError`.
+    """
+    payload = json.loads(SAMPLE.read_text())
+    if window != payload["window"]:
+        raise ValueError(f"the fake serves window {payload['window']}, not {window}")
+    raw = payload["returns"]
+    chosen = sorted(set(tickers)) if tickers is not None else sorted(raw["values"])
+    for ticker in chosen:
+        if ticker not in raw["values"]:
+            raise KeyError(ticker)
+    returns = Returns(
+        base_date=raw["base_date"],
+        dates=tuple(raw["dates"]),
+        tickers=tuple(chosen),
+        values={t: tuple(raw["values"][t]) for t in chosen},
+    )
+    return VolSeries(
+        returns=returns,
+        window=payload["window"],
+        annualized=payload["annualized"],
+        values={t: tuple(payload["values"][t]) for t in chosen},
+    )
