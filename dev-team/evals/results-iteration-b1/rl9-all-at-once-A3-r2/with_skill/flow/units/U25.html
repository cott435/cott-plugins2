from dataclasses import dataclass
from typing import Any

from rllab.core import EpisodeResult
from rllab.runlog import Checkpoint, EvalPoint, RunRecord


@dataclass(frozen=True)
class ReplayRun:
    record: RunRecord
    recorded_episodes: tuple[EpisodeResult, ...]
    recorded_evals: tuple[EvalPoint, ...]
    recorded_checkpoints: tuple[Checkpoint, ...]

    def episodes(self) -> list[EpisodeResult]:
        return list(self.recorded_episodes)

    def evals(self) -> list[EvalPoint]:
        return list(self.recorded_evals)

    def checkpoints(self) -> list[Checkpoint]:
        return list(self.recorded_checkpoints)


def build(kit: Any, recording: dict[str, Any]) -> ReplayRun:
    return ReplayRun(
        kit.decode(RunRecord, recording["record"]),
        kit.decode(tuple[EpisodeResult, ...], recording["episodes"]),
        kit.decode(tuple[EvalPoint, ...], recording["evals"]),
        kit.decode(tuple[Checkpoint, ...], recording["checkpoints"]),
    )

import json
from pathlib import Path
from typing import Any

import numpy as np
import pytest

from rllab.core import ActionError, Discrete
from rllab.worlds.checks import CheckDiscrete

TC03 = Path(__file__).parents[2] / "fixtures" / "rllab" / "TC-03"

# Action encoding of the corridor's Discrete(2): the contract recording
# (tests/contracts/worlds.CheckWorld/samples/corridor.json) moves right on action 1.
ACTIONS = {"left": 0, "right": 1}

# AD-09: the exit left of state 0 pays 0.8, the exit right of state 5 pays 1.0.
EXIT_REWARD = {"left": 0.8, "right": 1.0}


def expected() -> dict[str, Any]:
    data: dict[str, Any] = json.loads((TC03 / "expected.json").read_text())
    return data


def play_until_done(world: Any, action: int, seed: int) -> float:
    step = world.reset(seed)
    for _ in range(world.spec.max_steps):
        step = world.step(np.asarray([action]))
        if step.done:
            return float(step.rewards[0])
    raise AssertionError(f"action {action} never reached an exit")


@pytest.mark.intent
def test_checkdiscrete_has_the_fixture_number_of_states() -> None:
    world = CheckDiscrete()
    states = world.states()
    assert states.shape == (expected()["check-discrete"]["states"], world.spec.obs_len)
    assert len({tuple(row) for row in np.asarray(states).tolist()}) == states.shape[0]


@pytest.mark.intent
def test_checkdiscrete_best_actions_match_tc03() -> None:
    case = expected()
    want = [ACTIONS[name] for name in case["check-discrete"]["best_actions"]]
    best = np.asarray(CheckDiscrete().best_actions()).reshape(-1)
    assert best.tolist() == pytest.approx(want, abs=case["tol"])


@pytest.mark.intent
def test_checkdiscrete_action_space_is_discrete_two() -> None:
    world = CheckDiscrete()
    assert isinstance(world.spec.action, Discrete)
    assert world.spec.action.n == len(ACTIONS)
    assert world.spec.world == "check-discrete"


@pytest.mark.intent
@pytest.mark.parametrize("side", ["left", "right"])
def test_checkdiscrete_each_exit_pays_its_reward(side: str) -> None:
    reward = play_until_done(CheckDiscrete(), ACTIONS[side], seed=0)
    assert reward == pytest.approx(EXIT_REWARD[side], abs=1e-6)
