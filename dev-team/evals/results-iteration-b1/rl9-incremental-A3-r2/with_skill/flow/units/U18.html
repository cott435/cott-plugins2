"""core.Agent: a replay double serving recorded actions, and the check any Agent must pass.

`check_agent(agent, recording, tmp)` takes the provider, so the real agents and baselines run
the same check as the double.
"""

import json
from dataclasses import dataclass
from pathlib import Path
from typing import Literal

import numpy as np

from rllab.core import Agent, TeamSpec, Transition


@dataclass(frozen=True)
class AgentCall:
    """One recorded call: `act` (obs, explore, the action returned) or `learn` (a transition)."""

    kind: Literal["act", "learn"]
    obs: np.ndarray | None = None
    explore: bool | None = None
    action: np.ndarray | None = None
    transition: Transition | None = None


@dataclass(frozen=True)
class AgentRecording:
    """The calls one agent received, acting for `team`."""

    team: TeamSpec
    calls: tuple[AgentCall, ...]

    def acts(self) -> list[AgentCall]:
        return [call for call in self.calls if call.kind == "act"]


class ReplayAgent(Agent):
    """Returns the recorded actions in order, starting again after the last.

    Keeps what it was shown (`observed`) and taught (`learned`) for the consumer's test;
    `act` raises `ValueError` for a batch not shaped like the recording's team.
    """

    def __init__(self, recording: AgentRecording) -> None:
        self.recording = recording
        self.observed: list[tuple[np.ndarray, bool]] = []
        self.learned: list[Transition] = []
        self._actions = [call.action for call in recording.acts() if call.action is not None]
        self._next = 0

    def act(self, obs: np.ndarray, *, explore: bool) -> np.ndarray:
        team = self.recording.team
        if obs.shape != (team.n_agents, *team.obs_shape):
            raise ValueError(f"observation shaped {obs.shape} for team {team}")
        self.observed.append((obs.copy(), explore))
        action = self._actions[self._next % len(self._actions)]
        self._next += 1
        return action.copy()

    def learn(self, t: Transition) -> None:
        self.learned.append(t)

    def save(self, path: Path) -> None:
        path.write_text(json.dumps({"next": self._next}), encoding="utf-8")

    def load(self, path: Path) -> None:
        self._next = json.loads(path.read_text(encoding="utf-8"))["next"]


def check_agent(agent: Agent, recording: AgentRecording, tmp: Path) -> None:
    """The core.Agent contract on the recorded calls: action shapes, `learn`, save then load."""
    for call in recording.calls:
        if call.kind == "act":
            assert call.obs is not None
            assert call.explore is not None
            assert_actions(recording.team, agent.act(call.obs, explore=call.explore))
        else:
            assert call.transition is not None
            agent.learn(call.transition)
    first = recording.acts()[0].obs
    assert first is not None
    checkpoint = tmp / "checkpoint"
    agent.save(checkpoint)
    before = agent.act(first, explore=False)
    agent.load(checkpoint)
    np.testing.assert_array_equal(agent.act(first, explore=False), before)

    """One action per agent, shaped and ranged by the team's action space."""

        assert np.all((actions >= 0) & (actions < space.n))

        assert np.issubdtype(actions.dtype, np.floating)
