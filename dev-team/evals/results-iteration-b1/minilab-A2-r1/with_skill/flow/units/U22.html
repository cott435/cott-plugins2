+
+from __future__ import annotations
+
+from collections.abc import Sequence
+
+from mdata.types import Prices, RawRow
+
+
+def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
+    """Prices on the dates every ticker has a close (spine: no fill yet)."""
+    closes = {(row.date, row.ticker): row.close for row in rows if row.close is not None}
+    tickers = tuple(sorted({ticker for _, ticker in closes}))
+    dates = tuple(
+        sorted(
+            date
+            for date in {date for date, _ in closes}
+            if all((date, ticker) in closes for ticker in tickers)
+        )
+    )
+    return Prices(
+        dates=dates,
+        tickers=tickers,
+        closes={ticker: tuple(closes[date, ticker] for date in dates) for ticker in tickers},
+    )

+"""PL-01 step 1: read the `date,ticker,close` rows of a CSV."""

+
+import csv
+import datetime as dt
+from pathlib import Path

+COLUMNS = ("date", "ticker", "close")
+
+
+def read_csv(path: str | Path) -> list[RawRow]:
+    """Every row in file order; a blank close is None, not zero."""
+    with Path(path).open(newline="", encoding="utf-8") as handle:
+        reader = csv.DictReader(handle)
+        missing = [column for column in COLUMNS if column not in (reader.fieldnames or ())]
+        if missing:
+            raise ValueError(f"{path}: missing column(s) {', '.join(missing)}")
+        return [_row(record, reader.line_num) for record in reader]
+
+
+def _row(record: dict[str, str], line: int) -> RawRow:
+    date = record["date"].strip()
+    if not _is_iso_date(date):
+        raise ValueError(f"line {line}: malformed date {date!r}")
+    close = record["close"].strip()
+    return RawRow(date=date, ticker=record["ticker"].strip(), close=float(close) if close else None)
+
+
+def _is_iso_date(text: str) -> bool:
+    try:
+        return dt.date.fromisoformat(text).isoformat() == text
+    except ValueError:
+        return False
