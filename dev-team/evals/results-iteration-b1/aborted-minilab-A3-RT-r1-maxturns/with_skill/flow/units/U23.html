"""PL-01 step 1: read the rows of a `date,ticker,close` CSV."""

import csv
from datetime import date
from pathlib import Path

from mdata.types import RawRow

_COLUMNS = ("date", "ticker", "close")


def read_csv(path: str | Path) -> list[RawRow]:
    """Every row in file order; a blank close is None. ValueError for a missing column or a malformed date."""
    with Path(path).open(newline="") as handle:
        reader = csv.DictReader(handle)
        missing = [column for column in _COLUMNS if column not in (reader.fieldnames or ())]
        if missing:
            raise ValueError(f"{path}: missing column(s) {', '.join(missing)}")
        return [_to_row(record) for record in reader]


def _to_row(record: dict[str, str]) -> RawRow:
    """One CSV record as a RawRow, checking the date is ISO `yyyy-mm-dd`."""
    day = record["date"].strip()
    if not _is_iso_date(day):
        raise ValueError(f"malformed date {day!r}, expected yyyy-mm-dd")
    close = record["close"].strip()
    return RawRow(date=day, ticker=record["ticker"].strip(), close=float(close) if close else None)


def _is_iso_date(text: str) -> bool:
    """True when `text` is a real calendar date written exactly as `yyyy-mm-dd`."""
    try:
        return date.fromisoformat(text).isoformat() == text
    except ValueError:
        return False
"""PL-01 step 2: keep the usable rows, sorted."""

from collections.abc import Sequence

from mdata.types import RawRow


def validate_rows(rows: Sequence[RawRow], tickers: Sequence[str] | None = None) -> list[RawRow]:
    """Positive closes only, the later of duplicate (date, ticker) rows, sorted by (date, ticker).

    KeyError for a requested ticker with no usable row.
    """
    kept = [row for row in rows if row.close is not None]
    return sorted(kept, key=lambda row: (row.date, row.ticker))

    """Start on the first date every ticker has a close; gaps take the previous close.

    GapError naming the ticker when more than `max_gap` dates in a row are missing.
    Spine version: keep only the dates every ticker has a close on, with no fill.
    """
    by_ticker: dict[str, dict[str, float]] = {}
    for row in rows:
        closes = by_ticker.setdefault(row.ticker, {})
        if row.close is not None:
            closes[row.date] = row.close
    tickers = tuple(sorted(by_ticker))
    shared = (
        set.intersection(*(set(closes) for closes in by_ticker.values())) if by_ticker else set()
    )
    dates = tuple(sorted(shared))
    return Prices(
        dates=dates,
        tickers=tickers,
        closes={ticker: tuple(by_ticker[ticker][date] for date in dates) for ticker in tickers},
    )
