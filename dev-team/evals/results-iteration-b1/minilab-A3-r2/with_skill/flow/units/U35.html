+    """Every date any ticker trades, from the first date on which every ticker has a close."""
+    all_dates = sorted({date for closes in by_ticker.values() for date in closes})

+from __future__ import annotations
+
+from statistics import stdev
+
+from risk.types import Returns, VolSeries
+
+
+def rolling_vol(returns: Returns, window: int) -> VolSeries:
+    """Volatility of the `window` returns before each date; `ValueError` if `window` < 2.
+
+    The value at date i is the sample stdev (n - 1) of returns i - window .. i - 1, never
+    date i's own return; None until `window` earlier returns exist.
+    """
+    if window < 2:
+        raise ValueError(f"window must be at least 2, got {window}")
+    values = {
+        ticker: tuple(
+            stdev(series[i - window : i]) if i >= window else None for i in range(len(series))
+        )
+        for ticker, series in returns.values.items()
+    }
+    return VolSeries(returns=returns, window=window, annualized=False, values=values)
