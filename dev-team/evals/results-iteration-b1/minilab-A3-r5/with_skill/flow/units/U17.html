"""Whole-period summary per ticker (RISK-04)."""

from statistics import fmean, stdev

from risk.types import TickerSummary, VolSeries


def summary_stats(vol: VolSeries, periods_per_year: int = 252) -> tuple[TickerSummary, ...]:
    """Annualized return, volatility, Sharpe ratio and last rolling volatility per ticker.

    Spine version: daily mean and sample stdev, not yet annualized by `periods_per_year`.
    """
    if len(vol.returns.dates) < 2:
        raise ValueError(f"summary_stats needs at least two returns, got {len(vol.returns.dates)}")
    return tuple(_ticker_summary(vol, ticker) for ticker in vol.returns.tickers)


def _ticker_summary(vol: VolSeries, ticker: str) -> TickerSummary:
    returns = vol.returns.values[ticker]
    mean = fmean(returns)
    sd = stdev(returns)
    vols = vol.values[ticker]
    return TickerSummary(
        ticker=ticker,
        ann_return=mean,
        ann_vol=sd,
        sharpe=mean / sd if sd else 0.0,
        last_vol=vols[-1] if vols else None,
    )

from risk.types import Returns, VolSeries

def _sample_vol() -> VolSeries:
    data = json.loads(SAMPLE.read_text())
    returns = data["returns"]
    return VolSeries(
        returns=Returns(
            base_date=returns["base_date"],
            dates=tuple(returns["dates"]),
            tickers=tuple(returns["tickers"]),
            values={t: tuple(v) for t, v in returns["values"].items()},
        ),
        window=data["window"],
        annualized=data["annualized"],
        values={t: tuple(v) for t, v in data["values"].items()},
    )


def test_sample_gives_one_summary_per_ticker_in_order() -> None:
    result = summary_stats(_sample_vol())

    assert [s.ticker for s in result] == ["ACME", "BOLT"]


def test_sample_daily_mean_and_stdev() -> None:
    acme, bolt = summary_stats(_sample_vol())

    # ACME: mean 0.015 / 5; squared deviations sum to 854e-6 over n - 1 = 4.
    assert acme.ann_return == pytest.approx(0.003)
    assert acme.ann_vol == pytest.approx(math.sqrt(213.5e-6))
    assert acme.sharpe == pytest.approx(0.003 / math.sqrt(213.5e-6))
    # BOLT: mean 0.006 / 5; squared deviations sum to 842.8e-6 over 4.
    assert bolt.ann_return == pytest.approx(0.0012)
    assert bolt.ann_vol == pytest.approx(math.sqrt(210.7e-6))


def test_last_vol_is_the_final_rolling_value() -> None:
    acme, bolt = summary_stats(_sample_vol())

    assert acme.last_vol == pytest.approx(0.17959955456514918)
    assert bolt.last_vol == pytest.approx(0.39287402561126384)


def test_flat_returns_give_zero_sharpe() -> None:
    returns = Returns(
        base_date="2024-01-01",
        dates=("2024-01-02", "2024-01-03"),
        tickers=("ACME",),
        values={"ACME": (0.01, 0.01)},
    )
    vol = VolSeries(returns=returns, window=2, annualized=False, values={"ACME": (None, None)})

    (summary,) = summary_stats(vol)

    assert summary.ann_vol == 0.0
    assert summary.sharpe == 0.0
    assert summary.last_vol is None
