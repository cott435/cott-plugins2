"""The scripted baseline on the compact view (BASE-02)."""

from pathlib import Path

import numpy as np

from rllab.core import Agent, Transition
from rllab.envs import COMPACT_FIELDS

# Relative moves in tie order: action 0 straight, 1 turn left, 2 turn right.
_MOVES = (0, 1, 2)
# Compass directions clockwise, so a left turn is -1 and a right turn +1.
_HEADING = [COMPACT_FIELDS.index(f"head_{d}") for d in ("up", "right", "down", "left")]
_FOOD = [COMPACT_FIELDS.index(f"food_{d}") for d in ("up", "right", "down", "left")]
_DANGER = [COMPACT_FIELDS.index(f"danger_{m}") for m in ("straight", "left", "right")]
_TURN = (0, -1, 1)


class ScriptedPlayer(Agent):
    """Toward food if safe, else any safe move, else straight; ties straight, left, right."""

    def __init__(self) -> None:
        """The scripted player holds no state: its choice depends only on the observation."""

    def act(self, obs: np.ndarray, *, explore: bool) -> np.ndarray:
        """One move per agent from its compact view; `explore` changes nothing."""
        if obs.ndim != 2 or obs.shape[1:] != (len(COMPACT_FIELDS),):
            raise ValueError(f"each observation must be shape (11,), got {obs.shape[1:]}")
        return np.asarray([_choose(row) for row in obs], dtype=np.int64)

    def learn(self, t: Transition) -> None:
        """A scripted player does not learn: the transition is dropped."""
        del t

    def save(self, path: Path) -> None:
        """Nothing to save: the script is the whole player."""
        del path

    def load(self, path: Path) -> None:
        """Nothing to load: the script is the whole player."""
        del path


def _choose(view: np.ndarray) -> int:
    """The first safe move toward food, else the first safe move, else straight."""
    heading = int(np.argmax(view[_HEADING]))
    safe = [m for m in _MOVES if view[_DANGER[m]] == 0]
    toward = [m for m in safe if view[_FOOD[(heading + _TURN[m]) % 4]] == 1]
    return (toward or safe or [0])[0]
