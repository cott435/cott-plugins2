11	def log_returns(prices: Prices) -> Returns:
12	    """r_t = ln(close_t / close_t-1), dated t, for every ticker.
13	
14	    Spine version: simple returns, close_t / close_t-1 - 1.
15	
16	    Raises ValueError with fewer than two dates.

20	    values = {
21	        ticker: tuple(
22	            today / yesterday - 1.0 for yesterday, today in pairwise(prices.closes[ticker])
23	        )
24	        for ticker in prices.tickers
25	    }
26	    return Returns(
27	        base_date=prices.dates[0],
28	        dates=prices.dates[1:],
29	        tickers=prices.tickers,
30	        values=values,

"""Daily log returns (RISK-01)."""

from __future__ import annotations

import math
from itertools import pairwise

from mdata.types import Prices
from risk.types import Returns


def log_returns(prices: Prices) -> Returns:
    """r_t = ln(close_t / close_t-1), dated t, for every ticker.

    Raises ValueError with fewer than two dates.
    """
    if len(prices.dates) < 2:
        raise ValueError(f"log_returns needs at least two dates, got {len(prices.dates)}")
    values = {
        ticker: tuple(
            math.log(today / yesterday) for yesterday, today in pairwise(prices.closes[ticker])
        )
        for ticker in prices.tickers
    }
    return Returns(
        base_date=prices.dates[0],
        dates=prices.dates[1:],
        tickers=prices.tickers,
        values=values,
    )
