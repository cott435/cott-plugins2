def _greedy_agent(recording: dict[str, Any], team: str, n_games: int) -> Any:
    """An Agent replay double that expects, for each of `n_games` games, the recorded game's
    observations with `explore=False`, returns the recorded actions, and has no `learn` call."""
    observations = [recording["reset"]["obs"][team]]
    observations += [s["result"]["obs"][team] for s in recording["steps"][:-1]]
    actions = [s["actions"][team] for s in recording["steps"]]
    game = [
        {"method": "act", "obs": obs, "explore": False, "returns": act}
        for obs, act in zip(observations, actions, strict=True)
    ]
    return kit.AgentReplay({"team": recording["teams"][team], "calls": game * n_games})

def train(
    make_env: Callable[[], Env],
    agents: Mapping[str, Agent],
    settings: RunSettings,
    *,
    root: Path = Path("runs"),
) -> RunRecord:
    """Train for `settings.steps` steps, evaluating every `eval_every`, into a run folder."""
    env = make_env()
    learners = _learning_teams(env, agents, settings)
    writer = create_run(settings, root=root)
    loop = _Loop(
        make_env=make_env, agents=agents, settings=settings, writer=writer, learners=learners
    )
    try:
        games = loop.run(env)
    except KeyboardInterrupt:
        writer.finish("incomplete")
        raise

        writer.save_checkpoint(agents[team], team=team)
    writer.save_games(games)
    return writer.finish("complete")


def _learning_teams(
    env: Env, agents: Mapping[str, Agent], settings: RunSettings
) -> tuple[str, ...]:
    """The teams that learn, in the environment's order: those `opponents` marks `learn`, or
    the one team of a single-team run. ValueError unless `agents` has one agent per team."""
    if set(agents) != set(env.teams):
        raise ValueError(f"agents {sorted(agents)} must name the teams {sorted(env.teams)}")
    learners = tuple(t for t in env.teams if settings.opponents.get(t, "learn") == "learn")
    if not learners:
        raise ValueError("no team learns in this run")
    return learners


@dataclass
class _Loop:
    """One run's step loop: training games played, the clock, and the training-seed stream."""

    make_env: Callable[[], Env]
    agents: Mapping[str, Agent]
    settings: RunSettings
    writer: RunWriter
    learners: tuple[str, ...]
    games: int = 0
    last_score: float = 0.0
    start: float = field(init=False)
    last_print: float = field(init=False)
    seeds: np.random.Generator = field(init=False)

    def __post_init__(self) -> None:
        self.start = self.last_print = monotonic()
        self.seeds = np.random.default_rng(self.settings.seed)

    def run(self, env: Env) -> tuple[GameRecording, ...]:
        """Step `settings.steps` times; returns the games the final evaluation recorded."""
        steps = self.settings.steps
        self._evaluate(0)
        obs, active = self._reset(env)
        for step in range(1, steps + 1):
            obs, active = self._step(env, obs, active)
            if step % self.settings.eval_every == 0 and step < steps:
                self._evaluate(step)
            self._save_generations(step)
            self._report(step)
        return self._evaluate(steps).games

    def _reset(self, env: Env) -> tuple[Batch, Batch]:
        obs, _ = env.reset(int(self.seeds.integers(_TRAIN_SEEDS)))
        return obs, {team: np.ones(len(o), dtype=bool) for team, o in obs.items()}

    def _step(self, env: Env, obs: Batch, active: Batch) -> tuple[Batch, Batch]:
        actions = {team: agent.act(obs[team], explore=True) for team, agent in self.agents.items()}
        result = env.step(actions)
        for team in self.learners:
            terminated = np.logical_and(result.done[team], not result.truncated)
            self.agents[team].learn(
                Transition(
                    obs=obs[team],
                    actions=np.asarray(actions[team]),
                    reward=result.reward[team],
                    next_obs=result.obs[team],
                    terminated=terminated,
                    active=active[team],
                    truncated=result.truncated,
                )
            )
        if result.terminated or result.truncated:
            self.games += 1
            return self._reset(env)
        return result.obs, {team: active[team] & ~result.done[team] for team in active}

    def _evaluate(self, step: int) -> ScoreSet:
        n_games = self.settings.eval_games
        record = min(RECORDED_GAMES, n_games) if step == self.settings.steps else 0
        scores = evaluate(
            self.make_env,
            self.agents,
            player=self.settings.name,
            train_seed=self.settings.seed,
            options=EvalOptions(team=self.learners[0], n_games=n_games, record=record),
        )
        self.last_score = float(np.mean(scores.scores))
        point = CurvePoint(
            step=step,
            games=self.games,
            mean_score=self.last_score,
            elapsed_s=monotonic() - self.start,
        )
        self.writer.add_point(point)
        return scores

    def _save_generations(self, step: int) -> None:
        """Each learner's generation each time the run crosses another 1/`generations` of it."""
        n, steps = self.settings.generations, self.settings.steps
        index = step * n // steps - 1
        if index < 0 or index == (step - 1) * n // steps - 1:  # none yet, or no new one
            return
        for team in self.learners:
            self.writer.save_generation(self.agents[team], team=team, index=index)

    def _report(self, step: int) -> None:
        now = monotonic()
        if now - self.last_print < PROGRESS_EVERY_S:
            return
        self.last_print = now
        print(
            f"{self.settings.name} seed {self.settings.seed}: step {step}/{self.settings.steps}, "
            f"{self.games} games, last score {self.last_score:.3f}, {now - self.start:.0f}s",
            flush=True,
        )

def eval_seeds(train_seed: int, n: int) -> list[int]:
    """`1_000_000 + 1000 * train_seed + i` for `i < n`: never a training seed."""
    if n > _SEEDS_PER_TRAIN_SEED:
        raise ValueError(f"n must be at most {_SEEDS_PER_TRAIN_SEED}, got {n}")

    return [base + i for i in range(n)]


def evaluate(
    make_env: Callable[[], Env],
    agents: Mapping[str, Agent],
    *,
    player: str,
    train_seed: int,
    options: EvalOptions = _DEFAULT_OPTIONS,
) -> ScoreSet:

    env = make_env()

        raise ValueError(f"agents {sorted(agents)} must name the teams {sorted(env.teams)}")

class _CountdownEnv:
    """A game lasting `seed % 3 + 1` steps; each step scores 1 and rewards 0.5."""

    teams: Mapping[str, TeamSpec] = {SOLO: _SPEC}
    board: Literal["grid", "tank"] = "grid"

    def __init__(self) -> None:
        self.seeds: list[int] = []
        self.t = 0
        self.length = 0

    def reset(self, seed: int) -> tuple[dict[str, np.ndarray], dict[str, float]]:
        self.seeds.append(seed)
        self.t, self.length = 0, seed % 3 + 1
        return {SOLO: np.zeros((1, 1), dtype=np.float32)}, {"steps": 0.0, f"score:{SOLO}": 0.0}

    def step(self, actions: Mapping[str, np.ndarray]) -> StepResult:
        self.t += 1
        over = self.t == self.length
        return StepResult(
            obs={SOLO: np.full((1, 1), self.t, dtype=np.float32)},
            reward={SOLO: np.array([0.5])},
            done={SOLO: np.array([over])},
            terminated=over,
            truncated=False,
            info={"steps": float(self.t), f"score:{SOLO}": float(self.t)},
        )

    def frame(self) -> np.ndarray:
        return np.full((2, 2), self.t, dtype=np.int8)


class _SpyAgent:
    """Always action 0; records each `explore` flag and refuses to learn."""

    def __init__(self) -> None:
        self.explores: list[bool] = []

    def act(self, obs: np.ndarray, *, explore: bool) -> np.ndarray:
        self.explores.append(explore)
        return np.zeros(len(obs), dtype=np.int64)

    def learn(self, t: Transition) -> None:
        raise AssertionError("evaluate must never learn")

    def save(self, path: Path) -> None:
        raise AssertionError("unused")

    def load(self, path: Path) -> None:
        raise AssertionError("unused")
