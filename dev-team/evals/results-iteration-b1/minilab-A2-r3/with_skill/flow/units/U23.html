75	    """Annualize to a calendar year, like my brokerage statements: 365 days."""

from __future__ import annotations

import math
from dataclasses import dataclass
from itertools import accumulate

from risk.types import Returns


@dataclass(frozen=True)
class Drawdown:
    """The deepest fall below the running peak (`depth` <= 0) and the dates of that peak and trough."""

    ticker: str
    depth: float
    peak_date: str
    trough_date: str


def max_drawdown(returns: Returns) -> tuple[Drawdown, ...]:
    """One drawdown per ticker, in `returns.tickers` order; wealth starts at 1 on `base_date`."""
    dates = (returns.base_date, *returns.dates)
    return tuple(
        _ticker_drawdown(ticker, dates, returns.values[ticker]) for ticker in returns.tickers
    )


def _ticker_drawdown(ticker: str, dates: tuple[str, ...], values: tuple[float, ...]) -> Drawdown:
    """The deepest fall of one ticker's wealth, dated at its peak and trough; the first if tied."""
    wealth = [1.0, *(math.exp(r) for r in accumulate(values))]
    peak = trough = running_peak = 0
    depth = 0.0
    for day, value in enumerate(wealth):
        if value > wealth[running_peak]:
            running_peak = day
        fall = value / wealth[running_peak] - 1
        if fall < depth:
            depth, peak, trough = fall, running_peak, day
    return Drawdown(ticker=ticker, depth=depth, peak_date=dates[peak], trough_date=dates[trough])
