    summary: Sequence[TickerSummary], drawdowns: Sequence[Drawdown], out_path: str | Path
) -> Report:
    """Write `{"tickers": [...]}` to `out_path`: each ticker's summary and drawdown, by ticker."""
    by_ticker = {d.ticker: d for d in drawdowns}
    if sorted(s.ticker for s in summary) != sorted(by_ticker):
        raise ValueError("summary and drawdowns cover different tickers")
    rows = tuple(

            "ann_vol": s.ann_vol,
            "sharpe": s.sharpe,
            "last_vol": s.last_vol,

            "peak_date": by_ticker[s.ticker].peak_date,
            "trough_date": by_ticker[s.ticker].trough_date,
        }
        for s in sorted(summary, key=lambda s: s.ticker)
    )
    path = Path(out_path)
    path.write_text(json.dumps({"tickers": list(rows)}, indent=2) + "\n", encoding="utf-8")

def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
    """Put every ticker on the dates any ticker trades, from the first date all have a close.

    A missing date takes the ticker's previous close; more than `max_gap` missing dates in a
    row raise `GapError` naming the ticker.
    """

    for row in rows:
        if row.close is not None:
            closes.setdefault(row.ticker, {})[row.date] = row.close
    tickers = tuple(sorted(closes))
    start = max((min(by_date) for by_date in closes.values()), default="")
    dates = tuple(sorted({day for by_date in closes.values() for day in by_date if day >= start}))
    return Prices(
        dates=dates,
        tickers=tickers,
        closes={ticker: _fill(ticker, closes[ticker], dates, max_gap) for ticker in tickers},

def _fill(
    ticker: str, by_date: Mapping[str, float], dates: Sequence[str], max_gap: int
) -> tuple[float, ...]:
    """One ticker's closes on `dates`, each missing date taking the previous close."""
    filled: list[float] = []
    missing = 0

        if day in by_date:
            filled.append(by_date[day])
            missing = 0

        if missing > max_gap:
            raise GapError(f"{ticker} misses more than {max_gap} dates in a row up to {day}")
        filled.append(filled[-1])
    return tuple(filled)

def test_risk_report_raises_gap_error_for_a_long_gap(tmp_path: Path) -> None:
    days = [f"2024-01-{d:02d}" for d in range(1, 10)]
    rows = [f"{day},AAA,100" for day in days] + [f"{days[0]},BBB,50", f"{days[-1]},BBB,51"]
    path = tmp_path / "prices.csv"
    path.write_text("date,ticker,close\n" + "\n".join(rows) + "\n", encoding="utf-8")
    with pytest.raises(GapError, match="BBB"):
        risk.risk_report(path, tmp_path / "r.json", window=2)
