def prices_of(raw: dict[str, Any]) -> Prices:
    return Prices(
        dates=tuple(raw["dates"]),
        tickers=tuple(raw["tickers"]),
        closes={t: tuple(v) for t, v in raw["closes"].items()},
    )


@pytest.mark.intent
def test_tc04_geometric_closes_match_closed_form(golden) -> None:
    raw, expected = golden("risk", "TC-04")
    tol = expected["tol"]

    returns = log_returns(prices_of(raw["prices"]))

    assert isinstance(returns, Returns)
    assert returns.base_date == expected["base_date"]
    assert returns.dates == tuple(expected["dates"])
    assert returns.tickers == tuple(expected["tickers"])
    assert set(returns.values) == set(expected["values"])
    for ticker, want in expected["values"].items():
        assert tuple(returns.values[ticker]) == pytest.approx(tuple(want), abs=tol, rel=0)


@pytest.mark.intent
def test_returns_are_dated_after_the_base_date(contract_sample) -> None:
    raw = contract_sample("mdata.Prices")

    returns = log_returns(prices_of(raw))

    assert returns.base_date == raw["dates"][0]
    assert returns.dates == tuple(raw["dates"][1:])
    assert returns.tickers == tuple(raw["tickers"])
    for ticker in raw["tickers"]:
        assert len(returns.values[ticker]) == len(raw["dates"]) - 1
