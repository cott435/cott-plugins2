"""The snake rule baseline (SNAKE-07)."""

import numpy as np

from rllab.baselines.types import RulePolicy
from rllab.core import Array

# The snake's observation (SNAKE-06): danger after each action (0 straight, 1 turn left,
# 2 turn right), then its heading and the food's direction, each as up/right/down/left flags.
DANGER = slice(0, 3)
HEADING = slice(3, 7)
FOOD = slice(7, 11)

# Actions in the order the rule tries them, each with its quarter turns clockwise. Turning first
# moves the snake onto the other axis, closing the food's offset in a staircase.
MOVES = ((1, -1), (2, 1), (0, 0))


def choose(row: Array) -> int:
    """The first safe action toward the food; else the first safe action; else straight."""
    heading = int(np.argmax(row[HEADING]))
    food = row[FOOD]
    safe = [action for action, _ in MOVES if row[DANGER][action] == 0]
    toward = [action for action, turn in MOVES if action in safe and food[(heading + turn) % 4]]
    return (toward or safe or [0])[0]


class SnakeRule(RulePolicy):
    """Toward the food on the shorter axis, unless that move is fatal next step."""

    def __init__(self) -> None:
        self.description = (
            "Turns toward the food along the shorter axis, "
            "unless that move would end the episode on the next step."
        )

    def act(self, obs: Array, rng: np.random.Generator) -> Array:
        return np.array([choose(row) for row in obs], dtype=np.int64)
