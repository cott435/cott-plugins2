78	def daily_returns(prices: pd.DataFrame) -> pd.DataFrame:
79	    """Log returns of a dates x tickers close panel."""
80	    return np.log(prices / prices.shift(1)).dropna(how="all")
81	
82	
83	def rolling_vol(returns: pd.DataFrame, window: int = 20) -> pd.DataFrame:
84	    """Rolling volatility over the last `window` days, today included."""
85	    return returns.rolling(window).apply(lambda x: np.std(x), raw=True)
86	
87	
88	def annualize(vol: pd.DataFrame) -> pd.DataFrame:
89	    """Annualize to a calendar year, like my brokerage statements: 365 days."""
90	    return vol * np.sqrt(365)
91	
92	
93	def summary(returns: pd.DataFrame) -> pd.DataFrame:
94	    """Yearly return and volatility per ticker."""
95	    return pd.DataFrame({"ann_return": returns.mean() * 365, "ann_vol": returns.std(ddof=0) * np.sqrt(365)})
96	
97	
98	def max_drawdown(prices: pd.DataFrame) -> pd.Series:
99	    """Deepest fall below the running high."""
100	    return (prices / prices.cummax() - 1).min()

from __future__ import annotations

import math

import pytest

from risk import Returns, TickerSummary, VolSeries, summary_stats


def _returns(data: dict) -> Returns:
    return Returns(
        base_date=data["base_date"],
        dates=tuple(data["dates"]),
        tickers=tuple(data["tickers"]),
        values={t: tuple(v) for t, v in data["values"].items()},
    )


def _vol(data: dict) -> VolSeries:
    return VolSeries(
        returns=_returns(data["returns"]),
        window=data["window"],
        annualized=data["annualized"],
        values={t: tuple(v) for t, v in data["values"].items()},
    )


def _by_ticker(summaries: tuple[TickerSummary, ...]) -> dict[str, TickerSummary]:
    return {s.ticker: s for s in summaries}


@pytest.mark.intent
def test_tc07_one_summary_per_ticker(golden):
    data, expected = golden("risk", "TC-07")
    result = summary_stats(_vol(data["vol"]))
    assert all(isinstance(s, TickerSummary) for s in result)
    assert [s.ticker for s in result] == [row["ticker"] for row in expected["summary"]]


@pytest.mark.intent
def test_tc07_annualized_return_vol_sharpe_and_last_vol(golden):
    data, expected = golden("risk", "TC-07")
    tol = expected["tol"]
    got = _by_ticker(summary_stats(_vol(data["vol"])))
    for row in expected["summary"]:
        s = got[row["ticker"]]
        assert s.ann_return == pytest.approx(row["ann_return"], abs=tol)
        assert s.ann_vol == pytest.approx(row["ann_vol"], abs=tol)
        assert s.sharpe == pytest.approx(row["sharpe"], abs=tol)
        assert s.last_vol == pytest.approx(row["last_vol"], abs=tol)


@pytest.mark.intent
def test_tc07_zero_volatility_gives_zero_sharpe(golden):
    data, expected = golden("risk", "TC-07")
    tol = expected["tol"]
    got = _by_ticker(summary_stats(_vol(data["vol"])))
    for row in expected["summary"]:
        if row["ann_vol"] == 0.0:
            s = got[row["ticker"]]
            assert s.ann_vol == pytest.approx(0.0, abs=tol)
            assert s.sharpe == 0.0
            assert math.isfinite(s.sharpe)


@pytest.mark.intent
def test_tc07_periods_per_year_scales_the_annualization(golden):
    data, expected = golden("risk", "TC-07")
    tol = expected["tol"]
    got = _by_ticker(summary_stats(_vol(data["vol"]), periods_per_year=12))

        s = got[row["ticker"]]

        assert s.ann_vol == pytest.approx(row["ann_vol"] * math.sqrt(12 / 252), abs=tol)
