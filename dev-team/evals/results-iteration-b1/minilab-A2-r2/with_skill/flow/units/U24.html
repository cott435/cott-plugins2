+def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
+    """Prices on the dates every ticker has a close; no date is filled yet (spine)."""
+    by_ticker: dict[str, dict[str, float]] = {}
+    for row in rows:
+        if row.close is not None:
+            by_ticker.setdefault(row.ticker, {})[row.date] = row.close
+    tickers = tuple(sorted(by_ticker))
+    shared = set.intersection(*(set(closes) for closes in by_ticker.values())) if tickers else set()
+    dates = tuple(sorted(shared))
+    return Prices(
+        dates=dates,
+        tickers=tickers,
+        closes={ticker: tuple(by_ticker[ticker][date] for date in dates) for ticker in tickers},
+    )

+import datetime as dt
+import re
+from pathlib import Path
+

+COLUMNS = ("date", "ticker", "close")
+ISO_DATE = re.compile(r"\d{4}-\d{2}-\d{2}")
+
+
+def read_csv(path: str | Path) -> list[RawRow]:
+    """Every row of the CSV at `path`, in file order; a blank close is None, not zero."""
+    with Path(path).open(newline="", encoding="utf-8") as handle:
+        reader = csv.DictReader(handle)
+        missing = [column for column in COLUMNS if column not in (reader.fieldnames or ())]
+        if missing:
+            raise ValueError(f"{path}: missing column(s) {', '.join(missing)}")
+        return [_row(record, reader.line_num) for record in reader]
+
+
+def _row(record: dict[str, str], line: int) -> RawRow:
+    """One CSV record as a RawRow; `ValueError` for a malformed date or close."""
+    date = (record["date"] or "").strip()
+    if not _is_iso_date(date):
+        raise ValueError(f"line {line}: malformed date {date!r}")
+    close = (record["close"] or "").strip()
+    return RawRow(
+        date=date,
+        ticker=(record["ticker"] or "").strip(),
+        close=float(close) if close else None,
+    )
+
+
+def _is_iso_date(text: str) -> bool:
+    """Whether `text` is a real calendar date written as yyyy-mm-dd."""
+    if not ISO_DATE.fullmatch(text):
+        return False
+    try:
+        dt.date.fromisoformat(text)
+    except ValueError:
+        return False
+    return True
