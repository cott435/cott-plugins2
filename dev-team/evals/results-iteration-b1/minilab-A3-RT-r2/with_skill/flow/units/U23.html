+"""Unit tests for risk.annualize (spine: daily values pass through)."""
+
+from __future__ import annotations
+
+import json
+from dataclasses import replace
+from pathlib import Path
+
+from risk import Returns, VolSeries, annualize
+
+SAMPLE = Path(__file__).parents[2] / "contracts" / "risk.VolSeries" / "sample.json"
+
+
+def _sample_vol() -> VolSeries:
+    data = json.loads(SAMPLE.read_text())
+    ret = data["returns"]
+    returns = Returns(
+        base_date=ret["base_date"],
+        dates=tuple(ret["dates"]),
+        tickers=tuple(ret["tickers"]),
+        values={k: tuple(v) for k, v in ret["values"].items()},
+    )
+    return VolSeries(
+        returns=returns,
+        window=data["window"],
+        annualized=data["annualized"],
+        values={k: tuple(v) for k, v in data["values"].items()},
+    )
+
+
+def test_sample_values_pass_through_annualized() -> None:
+    vol = _sample_vol()
+
+    out = annualize(vol)
+
+    assert out.annualized is True
+    assert out.values == {
+        "AAA": (None, None, 0.3350986802582836, 0.44461147241128745, 0.3880623746755),
+        "BBB": (None, None, 0.110591753288, 0.166298925491, 0.0557071722029),
+    }
+    assert out.returns == vol.returns
+    assert out.window == 2
+
+
+def test_daily_series_comes_out_marked_annualized() -> None:
+    daily = replace(_sample_vol(), annualized=False)
+
+    out = annualize(daily, periods_per_year=12)
+
+    assert out.annualized is True
+    assert out.values == daily.values
+    assert out.returns is daily.returns
