75	import numpy as np
76	import pandas as pd
77	
78	
79	def daily_returns(prices: pd.DataFrame) -> pd.DataFrame:
80	    """Log returns of a dates x tickers close panel."""
81	    return np.log(prices / prices.shift(1)).dropna(how="all")
82	
83	
84	def rolling_vol(returns: pd.DataFrame, window: int = 20) -> pd.DataFrame:
85	    """Rolling volatility over the last `window` days, today included."""
86	    return returns.rolling(window).apply(lambda x: np.std(x), raw=True)
87	
88	
89	def annualize(vol: pd.DataFrame) -> pd.DataFrame:
90	    """Annualize to a calendar year, like my brokerage statements: 365 days."""
91	    return vol * np.sqrt(365)
92	
93	
94	def summary(returns: pd.DataFrame) -> pd.DataFrame:
95	    """Yearly return and volatility per ticker."""
96	    return pd.DataFrame({"ann_return": returns.mean() * 365, "ann_vol": returns.std(ddof=0) * np.sqrt(365)})
97	
98	
99	def max_drawdown(prices: pd.DataFrame) -> pd.Series:
100	    """Deepest fall below the running high."""
101	    return (prices / prices.cummax() - 1).min()
102	```

@pytest.mark.intent
def test_tc05_rolling_vol_from_earlier_returns(golden):
    data, expected = golden("risk", "TC-05")
    returns = _returns(data["returns"])
    vol = rolling_vol(returns, data["window"])
    assert isinstance(vol, VolSeries)
    assert vol.returns == returns
    assert vol.window == expected["window"]
    assert vol.annualized is expected["annualized"]
    assert set(vol.values) == set(expected["values"])
    tol = expected["tol"]
    for ticker, want in expected["values"].items():
        got = vol.values[ticker]
        assert len(got) == len(want)
        for g, w in zip(got, want, strict=True):
            if w is None:
                assert g is None
            else:
                assert g is not None
                assert g == pytest.approx(w, abs=tol)


@pytest.mark.intent
def test_tc05_value_ignores_the_dates_own_return(golden):
    data, expected = golden("risk", "TC-05")
    window = data["window"]
    changed = dict(data["returns"])
    values = {k: list(v) for k, v in data["returns"]["values"].items()}
    values["X"][window] = 1.0
    changed["values"] = values
    vol = rolling_vol(_returns(changed), window)
    got = vol.values["X"][window]
    assert got is not None
    assert got == pytest.approx(expected["values"]["X"][window], abs=expected["tol"])
