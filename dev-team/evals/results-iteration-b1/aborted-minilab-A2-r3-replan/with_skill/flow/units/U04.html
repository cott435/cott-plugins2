from __future__ import annotations

import csv

from pathlib import Path

COLUMNS = ("date", "ticker", "close")


def read_csv(path: str | Path) -> list[RawRow]:
    """Every row of `path` in file order; a blank close is None, not zero."""
    with Path(path).open(newline="") as handle:
        reader = csv.DictReader(handle)
        missing = [column for column in COLUMNS if column not in (reader.fieldnames or ())]
        if missing:
            raise ValueError(f"{path}: missing column(s) {', '.join(missing)}")
        return [_parse_row(record, line) for line, record in enumerate(reader, start=2)]


def _parse_row(record: dict[str, str], line: int) -> RawRow:
    """One CSV record as a RawRow; `ValueError` naming the line for a malformed date or close."""
    date = (record["date"] or "").strip()
    close = (record["close"] or "").strip()
    try:
        parsed = dt.date.fromisoformat(date)
    except ValueError as error:
        raise ValueError(f"line {line}: malformed date {date!r}") from error
    if parsed.isoformat() != date:
        raise ValueError(f"line {line}: date {date!r} is not yyyy-mm-dd")
    try:
        value = float(close) if close else None
    except ValueError as error:
        raise ValueError(f"line {line}: malformed close {close!r}") from error
    return RawRow(date=date, ticker=(record["ticker"] or "").strip(), close=value)
