"""Intent tests for risk.log_returns (RISK-01, TC-04)."""

import json
import math
from pathlib import Path

import pytest
from mdata import Prices
from risk import Returns, log_returns

ROOT = Path(__file__).resolve().parents[3]
TC04 = ROOT / "tests" / "fixtures" / "risk" / "TC-04"
PRICES_SAMPLE = ROOT / "tests" / "contracts" / "mdata.Prices" / "sample.json"


def _load(path: Path) -> dict:
    return json.loads(path.read_text())


def _prices(raw: dict) -> Prices:
    return Prices(
        dates=tuple(raw["dates"]),
        tickers=tuple(raw["tickers"]),
        closes={t: tuple(v) for t, v in raw["closes"].items()},
    )


@pytest.fixture
def tc04() -> tuple[Prices, dict]:
    inp = _load(TC04 / "input.json")
    expected = _load(TC04 / "expected.json")
    return _prices(inp["prices"]), expected


@pytest.mark.intent
def test_tc04_returns_match_closed_form(tc04: tuple[Prices, dict]) -> None:
    prices, expected = tc04
    result = log_returns(prices)
    assert isinstance(result, Returns)
    for ticker in expected["tickers"]:
        assert result.values[ticker] == pytest.approx(
            expected["values"][ticker], abs=expected["tol"], rel=0
        )


@pytest.mark.intent
def test_tc04_dates_and_base_date(tc04: tuple[Prices, dict]) -> None:
    prices, expected = tc04
    result = log_returns(prices)
    assert result.base_date == expected["base_date"]
    assert tuple(result.dates) == tuple(expected["dates"])
    assert tuple(result.tickers) == tuple(expected["tickers"])


@pytest.mark.intent
def test_one_value_per_date_and_ticker() -> None:
    prices = _prices(_load(PRICES_SAMPLE))
    result = log_returns(prices)
    assert result.base_date == prices.dates[0]
    assert tuple(result.dates) == prices.dates[1:]
    assert tuple(result.tickers) == prices.tickers
    for ticker in prices.tickers:
        assert len(result.values[ticker]) == len(prices.dates) - 1


@pytest.mark.intent
def test_return_is_log_of_close_ratio_dated_t() -> None:
    # RISK-01: r_t = ln(close_t / close_{t-1}), dated t.
    prices = _prices(_load(PRICES_SAMPLE))
    result = log_returns(prices)
    for ticker in prices.tickers:
        closes = prices.closes[ticker]
        for i, date in enumerate(result.dates):
            t = prices.dates.index(date)
            want = math.log(closes[t] / closes[t - 1])
            assert result.values[ticker][i] == pytest.approx(want, abs=1e-12, rel=0)


@pytest.mark.intent
def test_fewer_than_two_dates_raises() -> None:
    raw = _load(PRICES_SAMPLE)
    one = Prices(
        dates=(raw["dates"][0],),
        tickers=tuple(raw["tickers"]),
        closes={t: (v[0],) for t, v in raw["closes"].items()},
    )
    with pytest.raises(ValueError, match=r"(?i)date"):
        log_returns(one)
