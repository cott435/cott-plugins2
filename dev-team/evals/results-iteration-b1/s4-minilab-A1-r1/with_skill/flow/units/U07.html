from __future__ import annotations

import pytest

from mdata import Prices
from risk import Returns, log_returns


def _prices(payload: dict) -> Prices:
    return Prices(
        dates=tuple(payload["dates"]),
        tickers=tuple(payload["tickers"]),
        closes={t: tuple(v) for t, v in payload["closes"].items()},
    )


@pytest.mark.intent
def test_tc04_geometric_closes_give_the_closed_form(golden):
    given, expected = golden("risk", "TC-04")
    returns = log_returns(_prices(given["prices"]))
    assert isinstance(returns, Returns)
    assert returns.base_date == expected["base_date"]
    assert returns.dates == tuple(expected["dates"])
    assert returns.tickers == tuple(expected["tickers"])
    assert set(returns.values) == set(expected["values"])
    for ticker, values in expected["values"].items():
        assert len(returns.values[ticker]) == len(values)
        assert returns.values[ticker] == pytest.approx(values, abs=expected["tol"])


@pytest.mark.intent
def test_returns_are_dated_at_the_later_close(contract_sample):
    prices = _prices(contract_sample("mdata.Prices"))
    returns = log_returns(prices)
    assert returns.base_date == prices.dates[0]
    assert returns.dates == prices.dates[1:]
    assert returns.tickers == prices.tickers


@pytest.mark.intent
def test_contract_sample_prices_give_the_contract_sample_returns(contract_sample):
    sample = contract_sample("risk.Returns")
    returns = log_returns(_prices(contract_sample("mdata.Prices")))
    assert returns.base_date == sample["base_date"]
    assert returns.dates == tuple(sample["dates"])
    for ticker, values in sample["values"].items():
        # the sample is rounded to 12 decimals
        assert returns.values[ticker] == pytest.approx(values, abs=1e-11)
