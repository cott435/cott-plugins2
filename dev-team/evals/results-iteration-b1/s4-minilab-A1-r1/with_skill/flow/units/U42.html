+def max_drawdown(returns: Returns) -> tuple[Drawdown, ...]:
+    """Each ticker's worst fall from a peak, with its dates (RISK-05)."""
+    return tuple(_ticker_drawdown(returns, ticker) for ticker in returns.tickers)

+def _ticker_drawdown(returns: Returns, ticker: str) -> Drawdown:
+    """Worst fall of the ticker's wealth path, which starts at 1.0 on `base_date`."""
+    wealth = 0.0  # log wealth
+    peak, peak_date = 0.0, returns.base_date
+    worst = Drawdown(ticker, 0.0, returns.base_date, returns.base_date)
+    for date, value in zip(returns.dates, returns.values[ticker], strict=True):

+    return worst
