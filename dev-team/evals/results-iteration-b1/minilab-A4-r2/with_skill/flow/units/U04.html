from __future__ import annotations

from pathlib import Path

from mdata.io import read_csv
from mdata.types import RawRow


def _write(tmp_path: Path, text: str) -> Path:
    path = tmp_path / "prices.csv"
    path.write_text(text)
    return path


def test_reads_every_row_in_file_order(tmp_path: Path) -> None:
    path = _write(
        tmp_path,
        "date,ticker,close\n"
        "2024-01-03,BBB,50.5\n"
        "2024-01-02,AAA,100.0\n"
        "2024-01-02,BBB,50\n",
    )
    assert read_csv(path) == [
        RawRow(date="2024-01-03", ticker="BBB", close=50.5),
        RawRow(date="2024-01-02", ticker="AAA", close=100.0),
        RawRow(date="2024-01-02", ticker="BBB", close=50.0),
    ]


def test_blank_close_is_none_not_zero(tmp_path: Path) -> None:
    path = _write(tmp_path, "date,ticker,close\n2024-01-04,AAA,\n2024-01-05,AAA,0\n")
    rows = read_csv(path)
    assert rows[0].close is None
    assert rows[1].close == 0.0


def test_accepts_str_path_and_header_only_file(tmp_path: Path) -> None:
    path = _write(tmp_path, "date,ticker,close\n")
    assert read_csv(str(path)) == []
