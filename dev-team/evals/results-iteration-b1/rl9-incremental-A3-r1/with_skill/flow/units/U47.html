def _group(
    name: str, case: str, seeds: tuple[int, ...], steps: int, **changes: object
) -> dict[Path, RunRecord]:
    base: RunRecord = kit.sample("runs.RunRecord", "complete_corridor", RunRecord)
    settings: RunSettings = kit.sample("core.RunSettings", case, RunSettings)
    return {
        Path(f"runs/{name}-s{seed}"): dataclasses.replace(
            base,
            path=Path(f"runs/{name}-s{seed}"),
            settings=dataclasses.replace(settings, steps=steps, seed=seed, **changes),
        )
        for seed in seeds
    }


def _open(
    monkeypatch: pytest.MonkeyPatch,
    ladder: dict[str, dict[Path, RunRecord]],
    ablations: dict[str, dict[Path, RunRecord]],
) -> players.CurveRuns:
    groups = {**ladder, **ablations}
    records = {path: r for group in groups.values() for path, r in group.items()}
    monkeypatch.setattr(players, "open_run", records.__getitem__)
    targets = ReportTargets(
        players={name: PlayerTarget(runs=tuple(group)) for name, group in groups.items()},
        baselines=(),
        ladder=tuple(ladder),
        ablations=tuple(ablations),
    )
    return players.open_curve_runs(targets)
