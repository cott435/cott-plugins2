+    ticker: str
+    depth: float
+    peak_date: str
+    trough_date: str

+def max_drawdown(returns: Returns) -> tuple[Drawdown, ...]:
+    """Return each ticker's deepest fall of wealth below its running peak, in ticker order.
+
+    Wealth is 1 on `returns.base_date` and compounds the log returns after it.
+    Ties keep the first peak and the first trough; with no fall, both dates are the base date.
+    """
+    dates = (returns.base_date, *returns.dates)
+    return tuple(_drawdown(ticker, dates, returns.values[ticker]) for ticker in returns.tickers)
+
+
+def _drawdown(ticker: str, dates: tuple[str, ...], log_returns: tuple[float, ...]) -> Drawdown:
+    wealth = 1.0
+    peak, peak_date = 1.0, dates[0]
+    worst = Drawdown(ticker=ticker, depth=0.0, peak_date=dates[0], trough_date=dates[0])
+    for date, r in zip(dates[1:], log_returns, strict=True):
+        wealth *= math.exp(r)
+        if wealth > peak:
+            peak, peak_date = wealth, date
+        depth = wealth / peak - 1.0
+        if depth < worst.depth:
+            worst = Drawdown(ticker=ticker, depth=depth, peak_date=peak_date, trough_date=date)
+    return worst

+    """Summarize each ticker of `vol.returns`, in its ticker order.
+
+    The annualized return is `periods_per_year` x the mean, the annualized volatility is
+    sqrt(`periods_per_year`) x the sample stdev, and the Sharpe ratio is their quotient (0 when
+    the volatility is 0; no risk-free rate). `last_vol` is the last rolling value that exists.
+    """
+    returns = vol.returns
+    if len(returns.dates) < 2:
+        raise ValueError(f"need at least two returns for a summary, got {len(returns.dates)}")
+    summaries = []
+    for ticker in returns.tickers:
+        series = returns.values[ticker]
+        ann_return = periods_per_year * statistics.fmean(series)
+        ann_vol = math.sqrt(periods_per_year) * statistics.stdev(series)
+        summaries.append(
+            TickerSummary(
+                ticker=ticker,
+                ann_return=ann_return,
+                ann_vol=ann_vol,
+                sharpe=0.0 if ann_vol == 0 else ann_return / ann_vol,
+                last_vol=_last_value(vol.values[ticker]),
+            )
+        )
+    return tuple(summaries)

+"""PL-02 step 3: volatility from earlier returns."""
+
+from __future__ import annotations
+
+import statistics
+
+from risk.types import Returns, VolSeries
+
+
+def rolling_vol(returns: Returns, window: int) -> VolSeries:
+    """Return each date's sample stdev of the `window` returns before it, None until they exist.
+
+    A date's own return never enters its value (RISK-02).
+    """
+    if window < 2:
+        raise ValueError(f"window must be at least 2, got {window}")
+    values = {
+        ticker: tuple(
+            None if i < window else statistics.stdev(series[i - window : i])
+            for i in range(len(series))
+        )
+        for ticker, series in returns.values.items()
+    }
+    return VolSeries(returns=returns, window=window, annualized=False, values=values)
