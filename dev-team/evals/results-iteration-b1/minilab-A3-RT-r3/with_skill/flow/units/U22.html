+"""Unit tests for risk.vol.rolling_vol (spine: expanding stdev of earlier returns)."""
+
+from __future__ import annotations
+
+import pytest
+
+from risk.types import Returns
+from risk.vol import rolling_vol
+
+
+@pytest.fixture
+def sample_returns(contract_sample) -> Returns:
+    raw = contract_sample("risk.Returns")
+    return Returns(
+        base_date=raw["base_date"],
+        dates=tuple(raw["dates"]),
+        tickers=tuple(raw["tickers"]),
+        values={t: tuple(v) for t, v in raw["values"].items()},
+    )
+
+
+def test_sample_is_none_until_window_earlier_returns(sample_returns: Returns) -> None:
+    vol = rolling_vol(sample_returns, 2)
+    for ticker in sample_returns.tickers:
+        assert len(vol.values[ticker]) == len(sample_returns.dates)
+        assert vol.values[ticker][:2] == (None, None)
+        assert all(v is not None and v > 0 for v in vol.values[ticker][2:])
+
+
+def test_sample_expanding_stdev_hand_values(sample_returns: Returns) -> None:
+    vol = rolling_vol(sample_returns, 2)
+    # AAPL earlier returns at index 2: 0.0137933, -0.0193644 -> |diff| / sqrt(2)
+    assert vol.values["AAPL"][2] == pytest.approx(0.0234462, abs=1e-6)
+    # MSFT index 4: stdev of the first four returns
+    assert vol.values["MSFT"][4] == pytest.approx(0.0023711, abs=1e-6)
+
+
+def test_output_carries_inputs_and_is_daily(sample_returns: Returns) -> None:
+    vol = rolling_vol(sample_returns, 3)
+    assert vol.returns is sample_returns
+    assert vol.window == 3
+    assert vol.annualized is False
+    assert vol.values["AAPL"][:3] == (None, None, None)
+
+
+def test_value_excludes_the_dates_own_return() -> None:
+    returns = Returns(
+        base_date="2024-01-01",
+        dates=("d1", "d2", "d3"),
+        tickers=("X",),
+        values={"X": (1.0, 3.0, 1000.0)},
+    )
+    vol = rolling_vol(returns, 2)
+    assert vol.values["X"] == (None, None, pytest.approx(1.4142136, abs=1e-6))
