10	def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
11	    """Closes on the dates every ticker has; no fill yet, so `max_gap` is not used."""
12	    by_ticker: dict[str, dict[str, float]] = {}
13	    for row in rows:
14	        if row.close is not None:

20	    return Prices(dates=dates, tickers=tickers, closes=closes)

@dataclass(frozen=True)

    """One CSV row; `close` is None for a blank cell."""

    date: str
    ticker: str
    close: float | None


@dataclass(frozen=True)
class Prices:
    """Daily closes on one calendar: dates ascending, tickers sorted, one positive close per date."""

    dates: tuple[str, ...]
    tickers: tuple[str, ...]
    closes: Mapping[str, tuple[float, ...]]

    def __post_init__(self) -> None:
        if list(self.dates) != sorted(set(self.dates)):
            raise ValueError("dates must be unique and ascending")
        if list(self.tickers) != sorted(set(self.tickers)):

            raise ValueError("closes must have exactly one series per ticker")

            if len(series) != len(self.dates):
                raise ValueError(f"{ticker}: expected one close per date")
            if any(close <= 0 for close in series):
                raise ValueError(f"{ticker}: closes must be positive")

"""PL-01 step 3: put every ticker's closes on one calendar."""

from __future__ import annotations

from collections.abc import Mapping, Sequence

from mdata.types import GapError, Prices, RawRow


def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
    """Closes on every date any ticker trades, from the first date every ticker has a close.

    A missing date takes that ticker's previous close; more than `max_gap` missing dates
    in a row raise `GapError` naming the ticker.
    """
    by_ticker: dict[str, dict[str, float]] = {}
    for row in rows:
        if row.close is not None:
            by_ticker.setdefault(row.ticker, {})[row.date] = row.close
    tickers = tuple(sorted(by_ticker))
    start = max((min(own) for own in by_ticker.values()), default="")
    dates = tuple(sorted({date for own in by_ticker.values() for date in own if date >= start}))
    closes = {ticker: _fill(ticker, by_ticker[ticker], dates, max_gap) for ticker in tickers}
    return Prices(dates=dates, tickers=tickers, closes=closes)


def _fill(
    ticker: str, own: Mapping[str, float], dates: Sequence[str], max_gap: int
) -> tuple[float, ...]:
    """One ticker's closes on `dates`, each missing date taking the previous close."""
    series: list[float] = []
    missing = 0
    for date in dates:
        if date in own:
            missing = 0
            series.append(own[date])
            continue
        missing += 1
        if missing > max_gap:
            raise GapError(f"{ticker}: more than {max_gap} dates in a row missing")
        series.append(series[-1])
    return tuple(series)
