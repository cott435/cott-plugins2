"""PL-01 step 2: keep the usable rows."""

from __future__ import annotations

from collections.abc import Sequence

from mdata.types import RawRow


def validate_rows(rows: Sequence[RawRow], tickers: Sequence[str] | None = None) -> list[RawRow]:
    """Positive closes only, one row per (date, ticker) (the later in file order), sorted.

    Given `tickers`, only those tickers' rows come back; every other ticker's rows are dropped.
    Raises KeyError for a requested ticker with no usable row.
    """
    wanted = None if tickers is None else set(tickers)
    latest = {
        (row.date, row.ticker): row
        for row in rows
        if row.close is not None and row.close > 0 and (wanted is None or row.ticker in wanted)
    }
    usable_tickers = {ticker for _, ticker in latest}
    for ticker in tickers or ():
        if ticker not in usable_tickers:
            raise KeyError(ticker)
    return [latest[key] for key in sorted(latest)]
