"""PL-02 step 3: rolling volatility from earlier returns."""

from __future__ import annotations

from statistics import stdev

from risk.types import Returns, VolSeries


def rolling_vol(returns: Returns, window: int) -> VolSeries:
    """Sample stdev (n - 1) of the `window` returns before each date, never its own.

    None until `window` earlier returns exist; ValueError if `window` < 2.
    """
    if window < 2:
        raise ValueError(f"window must be at least 2, got {window}")
    values = {
        ticker: tuple(_trailing_stdev(returns.values[ticker], window))
        for ticker in returns.tickers
    }
    return VolSeries(returns=returns, window=window, annualized=False, values=values)


def _trailing_stdev(series: tuple[float, ...], window: int) -> list[float | None]:
    return [stdev(series[i - window : i]) if i >= window else None for i in range(len(series))]
