67	**Status:** Accepted · **Date:** 2026-10-17 · **Scope:** `runs`
68	- **Context:** RUN-01 needs a run fully described by its settings. RUN-02 needs settings, seed, curve, checkpoint and games saved. RUN-05 needs reloading. The interrupted-run edge case needs partial records kept.

def _codec() -> types.ModuleType:
    name = "contracts_codec"
    if name in sys.modules:

    return module

class _Interrupted:
    """The replay agent, whose `learn` raises `KeyboardInterrupt` on call number `at`."""

    def __init__(self, at: int) -> None:
        codec = _codec()
        agents = codec.fake("core.Agent")
        self.inner = agents.ReplayAgent(
            codec.sample("core.Agent", AGENT_CASE, agents.AgentRecording)
        )
        self.at = at
        self.learns = 0

    def act(self, obs: np.ndarray, *, explore: bool) -> np.ndarray:
        action: np.ndarray = self.inner.act(obs, explore=explore)
        return action

    def learn(self, t: Transition) -> None:
        self.learns += 1
        if self.learns >= self.at:
            raise KeyboardInterrupt
        self.inner.learn(t)

    def save(self, path: Path) -> None:
        self.inner.save(path)

    def load(self, path: Path) -> None:
        self.inner.load(path)


def _run_folder(root: Path) -> Path:
    folders = [p for p in root.iterdir() if p.is_dir()]
    assert len(folders) == 1, folders

def _curve_rows(folder: Path) -> list[list[str]]:
    """The curve's data rows, dropping a header row naming `step` if there is one."""
    text = (folder / "curve.csv").read_text(encoding="utf-8")
