from rllab.core import Agent, Transition


class ReferencePlayer(Agent):
    """The library model's deterministic action, batch-first with one agent; never learns."""

    def __init__(self, checkpoint: Path) -> None:
        self._model = DQN.load(checkpoint, device="cpu")

    def act(self, obs: np.ndarray, *, explore: bool) -> np.ndarray:
        """The greedy action for each row of `obs`; `explore` is ignored, the reference never explores."""
        actions, _ = self._model.predict(obs, deterministic=True)

        """The reference is trained by the library, never here."""

    def save(self, path: Path) -> None:
        self._model.save(path)

    def load(self, path: Path) -> None:
        self._model = DQN.load(path, device="cpu")

from typing import Any

import gymnasium as gym
import numpy as np
import pytest
import torch
from stable_baselines3 import DQN

from rllab.core import Transition
from rllab.reference import ReferencePlayer

OBS_SIZE = 11
N_ACTIONS = 3


class _StillEnv(gym.Env[np.ndarray, np.int64]):
    """Snake-shaped spaces and nothing else: enough to build a DQN."""

    observation_space = gym.spaces.Box(0.0, 1.0, shape=(OBS_SIZE,), dtype=np.float32)
    action_space = gym.spaces.Discrete(N_ACTIONS)

    def reset(
        self, *, seed: int | None = None, options: dict[str, Any] | None = None
    ) -> tuple[np.ndarray, dict[str, Any]]:
        super().reset(seed=seed)
        return np.zeros(OBS_SIZE, dtype=np.float32), {}

    def step(self, action: np.int64) -> tuple[np.ndarray, float, bool, bool, dict[str, Any]]:
        return np.zeros(OBS_SIZE, dtype=np.float32), 0.0, True, False, {}


def _saved_dqn(tmp_path: Path, best: int) -> Path:
    """A DQN whose Q-values ignore the observation and rank `best` highest."""
    model = DQN("MlpPolicy", _StillEnv(), device="cpu", seed=0)
    head = model.q_net.q_net[-1]
    assert isinstance(head, torch.nn.Linear)
    with torch.no_grad():
        head.weight.zero_()
        head.bias.copy_(torch.tensor([float(a == best) for a in range(N_ACTIONS)]))
    path = tmp_path / "checkpoint.zip"
    model.save(path)
    return path


def _obs(rng: np.random.Generator) -> np.ndarray:
    return rng.integers(0, 2, size=(1, OBS_SIZE)).astype(np.float32)


@pytest.mark.parametrize("best", [0, 2])
def test_reference_player_acts_greedily_batch_first(tmp_path: Path, best: int) -> None:
    player = ReferencePlayer(_saved_dqn(tmp_path, best))
    rng = np.random.default_rng(0)
    for _ in range(5):
        action = player.act(_obs(rng), explore=False)
        assert action.shape == (1,)
        assert action.dtype == np.int64
        assert action[0] == best


def test_reference_player_never_explores(tmp_path: Path) -> None:
    player = ReferencePlayer(_saved_dqn(tmp_path, 1))
    rng = np.random.default_rng(1)
    actions = {int(player.act(_obs(rng), explore=True)[0]) for _ in range(50)}
    assert actions == {1}


def test_reference_player_learn_changes_nothing(tmp_path: Path) -> None:
    player = ReferencePlayer(_saved_dqn(tmp_path, 2))
    obs = np.ones((1, OBS_SIZE), dtype=np.float32)
    transition = Transition(
        obs=obs,
        actions=np.array([0], dtype=np.int64),
        reward=np.array([10.0], dtype=np.float32),
        next_obs=obs,
        terminated=np.array([True]),
        active=np.array([True]),
        truncated=False,
    )
    for _ in range(20):
        player.learn(transition)
    assert player.act(obs, explore=False)[0] == 2


def test_reference_player_save_then_load_keeps_its_actions(tmp_path: Path) -> None:
    first = ReferencePlayer(_saved_dqn(tmp_path / "a", 2))
    copy = tmp_path / "copy.zip"
    first.save(copy)

    other.load(copy)
    obs = np.zeros((1, OBS_SIZE), dtype=np.float32)
    assert other.act(obs, explore=False)[0] == 2
