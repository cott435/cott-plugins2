from rllab.charts.curve import learning_curve

__all__ = ["MixedRunsError", "learning_curve"]
"""Learning curves with a min-max band and baseline lines (LOG-03 to LOG-05)."""

from collections.abc import Sequence
from pathlib import Path

import numpy as np
from matplotlib.axes import Axes
from matplotlib.figure import Figure
from numpy.typing import NDArray

from rllab.core import BaselineMeans, Output, WorldSpec
from rllab.runlog import Run

STEPS_LABEL = "environment steps"
# The dataviz skill's reference palette (light mode): categorical slot 1 and the chart chrome.
SERIES = "#2a78d6"
SURFACE = "#fcfcfb"
INK = "#0b0b0b"
SECONDARY = "#52514e"
MUTED = "#898781"
GRID = "#e1e0d9"
AXIS = "#c3c2b7"


def learning_curve(
    runs: Sequence[Run], baselines: BaselineMeans, out: Path, *, measure: str | None = None
) -> Output:
    figure = draw_curves(runs, baselines, measure)
    out.parent.mkdir(parents=True, exist_ok=True)
    figure.savefig(out)
    figure.savefig(out.with_suffix(".svg"))
    return Output(out, "chart", tuple(run.record.config.name for run in runs))


def draw_curves(runs: Sequence[Run], baselines: BaselineMeans, measure: str | None) -> Figure:
    """One panel per team (or one for `measure`): mean line, min-max band, baseline lines."""
    if not runs:
        raise ValueError("no runs to chart")
    spec = runs[0].record.spec
    panels = _panels(spec, measure)
    figure = Figure(figsize=(6.4 * len(panels), 4.2), facecolor=SURFACE, layout="constrained")
    axes = figure.subplots(1, len(panels), squeeze=False)[0]
    for ax, (team, label) in zip(axes, panels, strict=True):
        steps, values = _series(runs, team, measure)
        _draw_band(ax, steps, values, f"{team}: {label}, {len(runs)} runs")
        if measure is None:
            _draw_baselines(ax, baselines, team)
        _style(ax, label)
    return figure


def _panels(spec: WorldSpec, measure: str | None) -> list[tuple[str, str]]:
    """(team, y-axis label) per panel: each team's score, or the declared measure's team."""
    if measure is None:
        return [(team.name, team.score_name) for team in spec.teams]
    declared = [m for m in spec.measures if m.name == measure]
    if not declared:
        raise ValueError(f"{spec.world} declares no measure {measure!r}")
    return [(declared[0].team, measure)]


def _series(
    runs: Sequence[Run], team: str, measure: str | None
) -> tuple[NDArray[np.int64], NDArray[np.float64]]:
    """The evaluation steps every run has, and each run's value there (runs x steps)."""
    by_run = [
        {p.step: p.scores[team] if measure is None else p.measures[measure] for p in run.evals()}
        for run in runs
    ]
    steps = sorted(set.intersection(*(set(values) for values in by_run)))
    rows = [[values[step] for step in steps] for values in by_run]
    return np.array(steps, dtype=np.int64), np.array(rows, dtype=np.float64)


def _draw_band(ax: Axes, steps: NDArray[np.int64], values: NDArray[np.float64], title: str) -> None:
    ax.fill_between(
        steps,
        values.min(axis=0),
        values.max(axis=0),
        color=SERIES,
        alpha=0.18,
        linewidth=0,
        label="min-max range",
    )
    ax.plot(steps, values.mean(axis=0), color=SERIES, linewidth=2, label="mean")
    ax.set_title(title, color=INK, loc="left")


def _draw_baselines(ax: Axes, baselines: BaselineMeans, team: str) -> None:
    """Both baselines' mean scores as labelled horizontal reference lines."""
    lines = (
        (baselines.random[team], "random play", "--", MUTED),
        (baselines.rule[team], "rule-based", ":", SECONDARY),
    )
    for value, label, style, color in lines:
        ax.axhline(value, color=color, linestyle=style, linewidth=1.5, label=label)
        ax.annotate(
            label,
            (1.0, value),
            xycoords=("axes fraction", "data"),
            xytext=(-4, 4),
            textcoords="offset points",
            ha="right",
            color=SECONDARY,
            fontsize=8,
        )


def _style(ax: Axes, label: str) -> None:
    ax.set_facecolor(SURFACE)
    ax.set_xlabel(STEPS_LABEL, color=SECONDARY)
    ax.set_ylabel(label, color=SECONDARY)
    ax.grid(color=GRID, linewidth=0.8)
    ax.tick_params(colors=MUTED)
    for side in ("top", "right"):
        ax.spines[side].set_visible(False)
    for side in ("left", "bottom"):
        ax.spines[side].set_color(AXIS)
    ax.legend(frameon=False, labelcolor=SECONDARY, fontsize=8, loc="upper left")

"""The settings check before charting: runs may differ only by seed (LOG-03)."""

from collections.abc import Sequence
from dataclasses import fields

from rllab.charts.types import MixedRunsError
from rllab.runlog import Run

# Settings that name or seed one run of a batch; every other setting must match.
PER_RUN = ("name", "seed")


def check_same_settings(runs: Sequence[Run]) -> None:
    """Raise `MixedRunsError` naming each setting the runs disagree on, in config order."""
    configs = [run.record.config for run in runs]
    if not configs:
        return
    differing = [
        field.name
        for field in fields(configs[0])
        if field.name not in PER_RUN
        and any(getattr(c, field.name) != getattr(configs[0], field.name) for c in configs)
    ]
    if differing:
        raise MixedRunsError(differing)
