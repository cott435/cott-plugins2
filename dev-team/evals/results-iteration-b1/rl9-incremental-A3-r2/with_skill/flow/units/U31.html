66	### AD-06: One folder per run; TOML settings in, plain files out
67	**Status:** Accepted · **Date:** 2026-10-17 · **Scope:** `runs`
68	- **Context:** RUN-01 needs a run fully described by its settings. RUN-02 needs settings, seed, curve, checkpoint and games saved. RUN-05 needs reloading. The interrupted-run edge case needs partial records kept.
69	- **Decision:** Settings are read from TOML into a frozen `core.RunSettings`. Each run writes `runs/<name>-s<seed>-<timestamp>/` holding `settings.json` (the resolved settings), `curve.csv`, `checkpoint.pt` or `checkpoint.npz`, `games.npz` and `status` (`running`, `complete` or `incomplete`). The curve is appended as the run goes. An interrupt marks the run `incomplete`.

def _codec() -> Any:
    spec = importlib.util.spec_from_file_location("contracts_codec", CONTRACTS / "codec.py")
    assert spec is not None and spec.loader is not None
    module = importlib.util.module_from_spec(spec)

    return module


def _settings() -> RunSettings:
    base = _codec().sample("core.RunSettings", "corridor_tabular", RunSettings)
    return replace(base, steps=STEPS, eval_every=EVAL_EVERY, eval_games=1)


def _make_env() -> Corridor:

class _RightAgent:
    """Always steps right; optionally raises KeyboardInterrupt on its n-th `learn` call."""

    def __init__(self, interrupt_on_learn: int | None) -> None:
        self.interrupt_on_learn = interrupt_on_learn
        self.learn_calls = 0

    def act(self, obs: np.ndarray, *, explore: bool) -> np.ndarray:
        return np.ones((obs.shape[0],), dtype=np.int64)

    def learn(self, t: Transition) -> None:
        self.learn_calls += 1
        if self.interrupt_on_learn is not None and self.learn_calls == self.interrupt_on_learn:
            raise KeyboardInterrupt

    def save(self, path: Path) -> None:
        path.parent.mkdir(parents=True, exist_ok=True)
        path.write_bytes(b"right")

    def load(self, path: Path) -> None:
        return None


class _InterruptingCorridor:
    """A corridor whose n-th training `step` raises KeyboardInterrupt (counted across envs)."""

    calls = 0
    interrupt_at = INTERRUPT_AT

    def __init__(self) -> None:
        self._env = _make_env()
        self.teams = self._env.teams
        self.board = self._env.board

    def reset(self, seed: int) -> tuple[dict[str, np.ndarray], dict[str, float]]:
        return self._env.reset(seed)

    def step(self, actions: Any) -> StepResult:
        type(self).calls += 1
        if type(self).calls == type(self).interrupt_at:
            raise KeyboardInterrupt
        return self._env.step(actions)

    def frame(self) -> np.ndarray:
        return self._env.frame()


def _run_folder(root: Path) -> Path:
    folders = [p for p in root.iterdir() if p.is_dir()]
    assert len(folders) == 1, f"expected one run folder under {root}, found {folders}"
    return folders[0]


def _curve_steps(folder: Path) -> list[int]:
    steps = []
    for line in (folder / "curve.csv").read_text().splitlines():
        first = line.split(",")[0].strip()
        if first.lstrip("-").isdigit():
            steps.append(int(first))
    return steps
