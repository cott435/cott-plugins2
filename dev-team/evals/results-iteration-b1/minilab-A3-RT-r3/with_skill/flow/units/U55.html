def write_report(
    summary: Sequence[TickerSummary], drawdowns: Sequence[Drawdown], out_path: str | Path
) -> Report:
    """Write `{"tickers": [...]}` to `out_path`, one row per ticker, sorted.

    Raises ValueError if summary and drawdowns cover different tickers.
    """
    by_ticker = {d.ticker: d for d in drawdowns}
    summary_tickers = {s.ticker for s in summary}
    if summary_tickers != set(by_ticker):
        msg = f"summary covers {sorted(summary_tickers)}, drawdowns cover {sorted(by_ticker)}"
        raise ValueError(msg)
    rows = tuple(_row(s, by_ticker[s.ticker]) for s in sorted(summary, key=lambda s: s.ticker))
    Path(out_path).write_text(json.dumps({"tickers": list(rows)}, indent=2) + "\n")
    return Report(path=str(out_path), rows=rows)

from __future__ import annotations

import math
from itertools import accumulate

from risk.types import Drawdown, Returns


def max_drawdown(returns: Returns) -> tuple[Drawdown, ...]:
    """The deepest fall below the running peak of wealth (1 on `base_date`), with its dates."""
    dates = (returns.base_date, *returns.dates)
    return tuple(
        _ticker_drawdown(ticker, dates, returns.values[ticker]) for ticker in returns.tickers
    )


def _ticker_drawdown(ticker: str, dates: tuple[str, ...], values: tuple[float, ...]) -> Drawdown:
    """Track log wealth against its running peak; keep the deepest gap and its dates."""
    log_wealth = accumulate(values, initial=0.0)
    peak_level, peak_i = 0.0, 0
    worst, worst_peak_i, worst_trough_i = 0.0, 0, 0
    for i, level in enumerate(log_wealth):
        if level > peak_level:
            peak_level, peak_i = level, i
        elif level - peak_level < worst:
            worst, worst_peak_i, worst_trough_i = level - peak_level, peak_i, i
    return Drawdown(ticker, math.expm1(worst), dates[worst_peak_i], dates[worst_trough_i])
