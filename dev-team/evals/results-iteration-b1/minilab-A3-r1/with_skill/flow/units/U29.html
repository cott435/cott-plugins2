72	def daily_returns(prices: pd.DataFrame) -> pd.DataFrame:

74	    return np.log(prices / prices.shift(1)).dropna(how="all")

77	def rolling_vol(returns: pd.DataFrame, window: int = 20) -> pd.DataFrame:
78	    """Rolling volatility over the last `window` days, today included."""
79	    return returns.rolling(window).apply(lambda x: np.std(x), raw=True)
80	
81	
82	def annualize(vol: pd.DataFrame) -> pd.DataFrame:
83	    """Annualize to a calendar year, like my brokerage statements: 365 days."""

88	    """Yearly return and volatility per ticker."""

"""PL-02 step 4: annualize the volatility."""

from __future__ import annotations

import math
from dataclasses import replace

from risk.types import VolSeries


def annualize(vol: VolSeries, periods_per_year: int = 252) -> VolSeries:
    """Scale each daily volatility by sqrt(periods_per_year), keeping gaps (RISK-03).

    Raises ValueError if `vol` is already annualized.
    """
    if vol.annualized:
        raise ValueError(f"vol is already annualized (periods_per_year={periods_per_year})")
    scale = math.sqrt(periods_per_year)
    values = {
        ticker: tuple(None if v is None else v * scale for v in series)
        for ticker, series in vol.values.items()
    }
    return replace(vol, annualized=True, values=values)
