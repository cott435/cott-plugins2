11	    """Prices on one calendar from the first date every ticker has a close."""
12	    by_ticker: dict[str, dict[str, float]] = {}
13	    for row in rows:
14	        series = by_ticker.setdefault(row.ticker, {})
15	        if row.close is not None:
16	            series[row.date] = row.close
17	    tickers = tuple(sorted(by_ticker))
18	    common = set.intersection(*(set(series) for series in by_ticker.values())) if tickers else set()

20	    closes = {ticker: tuple(by_ticker[ticker][date] for date in dates) for ticker in tickers}
