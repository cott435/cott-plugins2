def _key(obs: np.ndarray) -> bytes:
    """The table key of a `(1, *obs_shape)` observation."""
    if obs.shape[0] != 1:
        raise ValueError(f"TabularAgent takes a batch of 1, got {obs.shape[0]}")
    return np.ascontiguousarray(obs[0]).tobytes()


class TabularAgent(Agent):
    """One value per (observation bytes, action), starting at 0; batches of one agent."""

    def __init__(
        self,
        n_actions: int,
        params: AgentParams,
        exploration: Exploration,
        *,
        seed: int,
        explore_enabled: bool = True,
    ) -> None:
        self._n_actions = n_actions
        self._params = params
        self._exploration = exploration
        self._explore_enabled = explore_enabled
        self._rng = np.random.default_rng(seed)
        self._table: dict[bytes, np.ndarray] = {}
        self._learns = 0

    def values(self, obs: np.ndarray) -> np.ndarray:
        """The `(n_actions,)` values of a `(1, *obs_shape)` observation; zeros when unseen."""
        row = self._table.get(_key(obs))
        return np.zeros(self._n_actions) if row is None else row.copy()

    def epsilon(self, step: int) -> float:
        """The exploration rate after `step` `learn` calls."""
        if step < 0:
            raise ValueError(f"step must be non-negative, got {step}")
        return exploration_rate(self._exploration, step)

    def act(self, obs: np.ndarray, *, explore: bool) -> np.ndarray:
        """The best-valued action; with `explore` (unless disabled), random at the current rate."""
        explores = explore and self._explore_enabled
        if explores and self._rng.random() < self.epsilon(self._learns):
            return self._rng.integers(0, self._n_actions, size=1, dtype=np.int64)
        return np.array([np.argmax(self.values(obs))], dtype=np.int64)

    def learn(self, t: Transition) -> None:
        """Move Q(obs, action) toward `reward + gamma * max next value`, not bootstrapping at an end."""
        key = _key(t.obs)
        row = self._table.setdefault(key, np.zeros(self._n_actions))
        bootstrap = 0.0 if bool(t.terminated[0]) else float(np.max(self.values(t.next_obs)))
        target = float(t.reward[0]) + self._params.gamma * bootstrap
        action = int(t.actions[0])
        row[action] += self._params.lr * (target - row[action])
        self._learns += 1

    def save(self, path: Path) -> None:
        """The table as NPZ: one uint8 row of key bytes and one value row per observation."""
        keys = [np.frombuffer(key, dtype=np.uint8) for key in self._table]
        np.savez(
            path,
            keys=np.stack(keys) if keys else np.zeros((0, 0), dtype=np.uint8),
            values=np.stack(list(self._table.values()))
            if keys
            else np.zeros((0, self._n_actions)),
            learns=np.array(self._learns),
        )

    def load(self, path: Path) -> None:
        """Replace the table and the learn count with those saved at `path`."""
        with np.load(path) as archive:
            self._table = {
                key.tobytes(): row.copy()
                for key, row in zip(archive["keys"], archive["values"], strict=True)
            }
