def seed_range(text: str | None, *, default: int) -> list[int]:
    """`"0-4"` is seeds 0 to 4, `"3"` is seed 3, and no text is `default`."""
    if text is None:

    seeds = list(range(int(first), int(last or first) + 1))

    return seeds

class Replay:
    """Draws a recorded game frame by frame; space pauses, right arrow steps, +/- change speed.

    `index`, `fps` and `paused` are read-only to callers; a new replay starts at frame 0,
    not paused. Building one needs no display; only `show` opens a window.
    """

    index: int
    fps: float
    paused: bool

    def __init__(self, game: GameRecording, *, fps: float = 8.0) -> None:
        self._game = game
        self.index = 0
        self.fps = fps
        self.paused = False

    def advance(self) -> int:
        """Move to the next frame and return its index, stopping at the last frame."""
        self.index = min(self.index + 1, len(self._game.frames) - 1)
        return self.index

    def on_key(self, key: str) -> None:
        """One Matplotlib key name: `" "` pauses, `"right"` advances, `"+"`/`"-"` change fps."""
        if key == " ":
            self.paused = not self.paused
        elif key == "right":
            self.advance()
        elif key == "+":
            self.fps *= 2
        elif key == "-":
            self.fps /= 2

    def current(self) -> tuple[np.ndarray, dict[str, float], int]:
        """What frame `index` shows: the recorded frame, the scores and the step."""
        scores = {team: float(values[self.index]) for team, values in self._game.scores.items()}
        return self._game.frames[self.index], scores, self.index

    def show(self) -> None:
        """Play the game in a window, blocking until it closes."""
        figure, axes = plt.subplots()
        timer = figure.canvas.new_timer(interval=_interval_ms(self.fps))

        def redraw() -> None:
            timer.interval = _interval_ms(self.fps)
            self._draw(axes)
            figure.canvas.draw_idle()

        def tick() -> None:
            if not self.paused:
                self.advance()
            redraw()

        def key_pressed(event: KeyEvent) -> None:
            if event.key is not None:
                self.on_key(event.key)
            redraw()

        figure.canvas.mpl_connect("key_press_event", key_pressed)
        timer.add_callback(tick)
        self._draw(axes)
        timer.start()
        plt.show()

    def _draw(self, axes: Axes) -> None:
        """The current frame as grid cells or tank circles, titled with its step and scores."""
        frame, scores, step = self.current()
        axes.clear()
        if self._game.board == "grid":
            axes.imshow(frame, cmap="viridis", interpolation="nearest")
        else:
            _draw_circles(axes, frame)
        shown = "  ".join(f"{team}: {score:g}" for team, score in scores.items())
        axes.set_title(f"step {step}  {shown}")
        axes.set_xticks([])
        axes.set_yticks([])


def _draw_circles(axes: Axes, circles: np.ndarray) -> None:
    """Tank circles `(x, y, radius, kind)` (AD-12), coloured by kind."""
    for x, y, radius, kind in circles:
        axes.add_patch(Circle((float(x), float(y)), float(radius), color=f"C{int(kind)}"))
    axes.set_aspect("equal")
    axes.autoscale_view()


def _interval_ms(fps: float) -> int:
    """The timer interval for `fps` frames a second, at least one millisecond."""
    return max(1, round(1000 / fps))

def create_run(settings: RunSettings, *, root: Path = Path("runs")) -> RunWriter:
    """Start a uniquely named run folder holding `settings.json` and `status=running`."""
    stamp = datetime.now().strftime("%Y%m%dT%H%M%S")
    path = _new_folder(root, f"{settings.name}-s{settings.seed}-{stamp}")
    (path / "settings.json").write_text(json.dumps(asdict(settings), indent=2), encoding="utf-8")
    (path / "curve.csv").write_text(",".join(_CURVE_FIELDS) + "\n", encoding="utf-8")
    (path / "status").write_text("running", encoding="utf-8")
    return RunWriter(path, settings)


def open_run(path: Path) -> RunRecord:
    """Read a run folder back (PL-05 step 1)."""
    settings_file = path / "settings.json"
    if not settings_file.is_file():
        raise FileNotFoundError(f"run {path}: no settings.json")
    return RunRecord(
        path=path,
        settings=_settings_from_json(json.loads(settings_file.read_text(encoding="utf-8"))),
        curve=_read_curve(path / "curve.csv"),
        status=_read_status(path / "status"),
        checkpoints=_find_checkpoints(path),
        generations=_find_generations(path / "generations"),

def _new_folder(root: Path, stem: str) -> Path:
    """`root/stem`, or `root/stem-<n>` with the first free `n` when it exists."""
    root.mkdir(parents=True, exist_ok=True)
    n = 0
    while True:
        path = root / (stem if n == 0 else f"{stem}-{n}")
        try:
            path.mkdir()
        except FileExistsError:
            n += 1
            continue
        return path

def _read_games(path: Path) -> tuple[GameRecording, ...]:
