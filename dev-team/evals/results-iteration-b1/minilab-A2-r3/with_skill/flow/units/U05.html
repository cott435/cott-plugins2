"""PL-01 smoke: load_prices runs end to end on a synthetic CSV and returns well-shaped Prices."""

from __future__ import annotations

from pathlib import Path

from mdata import Prices, load_prices

CSV = """date,ticker,close
2024-03-04,AAA,101.25
2024-03-04,BBB,48.6
2024-03-05,AAA,102.1
2024-03-05,BBB,49.0
2024-03-06,AAA,100.8
2024-03-06,BBB,49.3
2024-03-07,AAA,103.4
2024-03-07,BBB,49.9
"""


def test_pl01_runs_end_to_end_and_returns_prices(tmp_path: Path) -> None:
    path = tmp_path / "prices.csv"
    path.write_text(CSV)
    prices = load_prices(path)
    assert isinstance(prices, Prices)
    assert prices.tickers == ("AAA", "BBB")
    assert len(prices.dates) == 4
    assert set(prices.closes) == {"AAA", "BBB"}
    assert all(len(closes) == len(prices.dates) for closes in prices.closes.values())


def test_pl01_narrows_to_the_requested_tickers(tmp_path: Path) -> None:
    path = tmp_path / "prices.csv"
    path.write_text(CSV)
    prices = load_prices(path, ["BBB"])
    assert prices.tickers == ("BBB",)
    assert set(prices.closes) == {"BBB"}

from __future__ import annotations

import json
from pathlib import Path
from typing import Any

import pytest

from mdata import RawRow
from mdata.validate import validate_rows

ROOT = Path(__file__).resolve().parents[3]
TC01 = ROOT / "tests" / "fixtures" / "mdata" / "TC-01"
SAMPLE = ROOT / "tests" / "contracts" / "mdata.RawRow" / "sample.json"


def _load(path: Path) -> dict[str, Any]:
    data: dict[str, Any] = json.loads(path.read_text())
    return data


def _rows(triples: list[list[Any]]) -> list[RawRow]:
    return [RawRow(date=d, ticker=t, close=c) for d, t, c in triples]


def _sample_rows() -> list[RawRow]:
    return [RawRow(**r) for r in _load(SAMPLE)["rows"]]


@pytest.mark.intent
def test_tc01_drops_unusable_keeps_later_duplicate_and_sorts() -> None:
    given = _load(TC01 / "input.json")
    expected = _rows(_load(TC01 / "expected.json")["rows"])
    result = validate_rows(_rows(given["rows"]), given["tickers"])
    # Tolerance: exact
    assert result == expected


@pytest.mark.intent
def test_tc01_later_duplicate_wins_in_file_order() -> None:
    given = _load(TC01 / "input.json")
    rows = _rows(given["rows"])
    result = validate_rows(rows, given["tickers"])
    keys = [(r.date, r.ticker) for r in result]
    assert len(keys) == len(set(keys))
    expected = _rows(_load(TC01 / "expected.json")["rows"])
    # Reversing input order must flip which duplicate survives.
    reversed_result = validate_rows(list(reversed(rows)), given["tickers"])
    assert reversed_result != expected
    assert sorted(reversed_result, key=lambda r: (r.date, r.ticker)) == reversed_result
