```python
def validate_snapshot(snapshot: SnapshotId, *, promote: bool = True) -> ValidationResult:
    run = open_run("validation", scope=snapshot)
    snap = resolve_snapshot(snapshot)
    cal = calendar(snap.id)
    returns = returns_panel(*snap.coverage, snapshot=snap.id)
    prices = check_prices(snap)
    gaps = check_gaps(snap, cal)
    moves = check_moves(returns)
    actions = check_actions(snap, returns)
    members = check_membership(snap, cal)
    dupes = check_duplicates(snap)
    result = decide((prices, gaps, moves, actions, members, dupes), run)
    promote_snapshot(result, promote=promote)
    inputs = gather_quality_inputs(snap, returns, result)
    write_quality_report(inputs)
    close_run(run, result)
    return result
```
