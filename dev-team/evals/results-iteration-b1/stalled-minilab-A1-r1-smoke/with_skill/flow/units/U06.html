from __future__ import annotations

import math

import pytest

from mdata import Prices
from risk import Returns, log_returns


def _prices(payload: dict) -> Prices:
    return Prices(
        dates=tuple(payload["dates"]),
        tickers=tuple(payload["tickers"]),
        closes={ticker: tuple(closes) for ticker, closes in payload["closes"].items()},
    )


@pytest.mark.intent
def test_tc04_geometric_closes_give_the_closed_form_returns(golden):
    given, expected = golden("risk", "TC-04")
    returns = log_returns(_prices(given["prices"]))
    assert isinstance(returns, Returns)
    assert returns.base_date == expected["base_date"]
    assert returns.dates == tuple(expected["dates"])
    assert returns.tickers == tuple(expected["tickers"])
    for ticker in expected["tickers"]:
        assert returns.values[ticker] == pytest.approx(
            tuple(expected["values"][ticker]), abs=expected["tol"], rel=0
        )


@pytest.mark.intent
def test_risk01_each_return_is_the_log_of_its_close_over_the_previous(contract_sample):
    payload = contract_sample("mdata.Prices")
    prices = _prices(payload)
    returns = log_returns(prices)
    assert returns.base_date == prices.dates[0]
    assert returns.dates == prices.dates[1:]
    assert returns.tickers == prices.tickers
    for ticker in prices.tickers:
        closes = prices.closes[ticker]
        assert len(returns.values[ticker]) == len(returns.dates)
        for t, value in enumerate(returns.values[ticker], start=1):
            assert value == pytest.approx(math.log(closes[t] / closes[t - 1]), abs=1e-12, rel=0)


@pytest.mark.intent
def test_fewer_than_two_dates_is_a_value_error(contract_sample):
    payload = contract_sample("mdata.Prices")
    one_date = Prices(

    )
    with pytest.raises(ValueError, match=r"(?i)two|2"):
        log_returns(one_date)
