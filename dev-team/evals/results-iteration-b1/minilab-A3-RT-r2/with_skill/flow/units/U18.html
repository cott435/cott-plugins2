+from __future__ import annotations

+
+from mdata import Prices, RawRow, align_calendar

+def _sample_rows() -> list[RawRow]:
+    data = json.loads(SAMPLE.read_text())
+    return [RawRow(**row) for row in data["rows"]]
+
+
+def test_sample_keeps_only_dates_every_ticker_has() -> None:
+    prices = align_calendar(_sample_rows())
+
+    assert prices == Prices(
+        dates=("2024-03-01", "2024-03-04", "2024-03-05"),
+        tickers=("AAA", "BBB"),
+        closes={"AAA": (100.0, 102.0, 100.98), "BBB": (50.0, 50.5, 51.51)},
+    )
+
+
+def test_dates_ascending_and_tickers_sorted_from_unsorted_rows() -> None:
+    rows = [
+        RawRow("2024-01-03", "ZZZ", 3.0),
+        RawRow("2024-01-02", "AAA", 1.0),
+        RawRow("2024-01-03", "AAA", 2.0),
+        RawRow("2024-01-02", "ZZZ", 4.0),
+    ]
+
+    prices = align_calendar(rows)
+
+    assert prices.dates == ("2024-01-02", "2024-01-03")
+    assert prices.tickers == ("AAA", "ZZZ")
+    assert prices.closes == {"AAA": (1.0, 2.0), "ZZZ": (4.0, 3.0)}
+
+
+def test_date_missing_for_one_ticker_is_dropped_not_filled() -> None:
+    rows = [

+        RawRow("2024-01-04", "BBB", 6.0),

+    prices = align_calendar(rows)
+
+    assert prices.dates == ("2024-01-02", "2024-01-04")

from __future__ import annotations

from collections.abc import Mapping
from dataclasses import dataclass


@dataclass(frozen=True)
class RawRow:

class Prices:
