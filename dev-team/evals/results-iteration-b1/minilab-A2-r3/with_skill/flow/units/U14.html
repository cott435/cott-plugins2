11	def log_returns(prices: Prices) -> Returns:
12	    """One return per ticker for each date after the first."""
13	    if len(prices.dates) < 2:
14	        raise ValueError("log_returns needs at least two dates")
15	    values = {
16	        ticker: tuple(now / before - 1.0 for before, now in pairwise(closes))
17	        for ticker, closes in prices.closes.items()
18	    }
19	    return Returns(
20	        base_date=prices.dates[0],
21	        dates=prices.dates[1:],
22	        tickers=prices.tickers,
23	        values=values,
24	    )

"""PL-02 step 2: daily log returns from prices (RISK-01)."""

from __future__ import annotations

import math
from itertools import pairwise

from mdata.types import Prices
from risk.types import Returns


def log_returns(prices: Prices) -> Returns:
    """r_t = ln(close_t / close_t-1) per ticker, dated t, for each date after the first."""
    if len(prices.dates) < 2:
        raise ValueError("log_returns needs at least two dates")
    values = {
        ticker: tuple(math.log(now / before) for before, now in pairwise(closes))
        for ticker, closes in prices.closes.items()
    }
    return Returns(
        base_date=prices.dates[0],
        dates=prices.dates[1:],
        tickers=prices.tickers,
        values=values,
    )
