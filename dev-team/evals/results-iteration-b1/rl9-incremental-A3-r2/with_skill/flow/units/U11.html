def _held_out(train_seed: int, n: int) -> list[int]:
    """The contract's closed form (DATA-01), written out independently of the code."""
    return [1_000_000 + 1000 * train_seed + i for i in range(n)]


class _Players:
    """The corridor replay double and one replay agent per team, keeping each reset seed."""

    def __init__(self, recording: Any, teams: tuple[str, ...] = (SOLO,)) -> None:
        codec = _codec()
        self.envs = codec.fake("core.Env")
        agents = codec.fake("core.Agent")
        self.recording = recording
        agent_recording = codec.sample("core.Agent", AGENT_CASE, agents.AgentRecording)
        self.agents = {team: agents.ReplayAgent(agent_recording) for team in teams}
        self.seeds: list[int] = []
        self.made = 0

    def make_env(self) -> Env:
        self.made += 1
        env = self.envs.ReplayEnv(self.recording)
        reset = env.reset

        def recording_reset(seed: int) -> tuple[dict[str, np.ndarray], dict[str, float]]:
            self.seeds.append(seed)
            return reset(seed)

        env.reset = recording_reset
        return env


def _corridor() -> Any:
    codec = _codec()
    return codec.sample("core.Env", CORRIDOR, codec.fake("core.Env").EnvRecording)


def _two_team_corridor() -> Any:
    """The corridor game with a second team mirroring the first, scored by its step count."""
    base = _corridor()
    spec: TeamSpec = base.teams[SOLO]

    def with_rival(info: dict[str, float]) -> dict[str, float]:
        return {**info, f"score:{RIVAL}": info["steps"]}

    def mirror(result: StepResult) -> StepResult:
        return replace(
            result,
            obs={**result.obs, RIVAL: result.obs[SOLO].copy()},
            reward={**result.reward, RIVAL: result.reward[SOLO].copy()},
            done={**result.done, RIVAL: result.done[SOLO].copy()},
            info=with_rival(result.info),
        )

    return replace(
        base,
        teams={SOLO: spec, RIVAL: spec},
        obs={SOLO: base.obs[SOLO].copy(), RIVAL: base.obs[SOLO].copy()},
        info=with_rival(base.info),
        actions=tuple({**a, RIVAL: a[SOLO].copy()} for a in base.actions),
        results=tuple(mirror(r) for r in base.results),
    )


def _run(players: _Players, spec: EvalSpec) -> ScoreSet:
    make_env: Callable[[], Env] = players.make_env
    return evaluate(make_env, players.agents, spec)
