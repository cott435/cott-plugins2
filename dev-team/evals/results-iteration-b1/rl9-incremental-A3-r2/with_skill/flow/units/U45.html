from rllab.core import SOLO, ActionSpace, AgentParams, Exploration, Rewards, TeamSpec
from rllab.envs import Snake

GRID_TEAM = TeamSpec(1, (3, 10, 10), ActionSpace("discrete", n=3))
CNN = AgentParams(lr=0.001, gamma=0.99, net="cnn")


def _agent(team: TeamSpec = GRID_TEAM, params: AgentParams = CNN) -> DQNAgent:
    return DQNAgent(team, params, Exploration(), seed=0)


def _weight_shapes(agent: DQNAgent) -> list[tuple[int, ...]]:
    state = agent.online_state()
    return [tuple(t.shape) for name, t in state.items() if name.endswith("weight")]


def _grid_obs() -> np.ndarray:
    env = Snake(view="grid", rewards=Rewards(), size=10)
    env.reset(0)
    obs, _ = env.reset_to([(5, 5), (5, 4), (5, 3)], "right", (2, 7))
    return np.asarray(obs, dtype=np.float32).reshape(1, 3, 10, 10)


@pytest.mark.intent
def test_cnn_acts_on_a_grid_observation_with_one_of_three_actions():
    agent = _agent()
    obs = np.zeros((1, 3, 10, 10), dtype=np.float32)
    obs[0, 0, 5, 5] = 1.0
    obs[0, 1, 5, 4] = 1.0
    obs[0, 2, 2, 7] = 1.0
    action = np.asarray(agent.act(obs, explore=False))
    assert action.shape == (1,)
    assert int(action[0]) in {0, 1, 2}


@pytest.mark.intent

    shapes = _weight_shapes(_agent())
    convs = [s for s in shapes if len(s) == 4]
    assert convs == [(16, 3, 3, 3), (32, 16, 3, 3)]

    assert linears[1] == (3, 128)

def _weight_shapes(agent: DQNAgent, tmp_path: Path) -> list[tuple[int, ...]]:
    """The online network's weight shapes, read from the checkpoint's `online` state dict."""
    path = tmp_path / "dqn.pt"
    agent.save(path)
    state = torch.load(path, weights_only=True)["online"]
    return [tuple(t.shape) for name, t in state.items() if name.endswith("weight")]


def _grid_obs() -> np.ndarray:
    env = Snake(view="grid", rewards=Rewards(), size=10)
    env.reset(0)
    obs, _ = env.reset_to([(5, 5), (5, 4), (5, 3)], "right", (2, 7))
    return np.asarray(obs, dtype=np.float32).reshape(1, 3, 10, 10)
