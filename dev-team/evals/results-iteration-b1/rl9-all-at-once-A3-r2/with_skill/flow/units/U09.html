def encode(value: Any) -> Any:
    """A contract value as JSON data, the inverse of `decode`."""
    if dataclasses.is_dataclass(value) and not isinstance(value, type):
        return {f.name: encode(getattr(value, f.name)) for f in dataclasses.fields(value)}
    if isinstance(value, Mapping):
        return {encode(k): encode(v) for k, v in value.items()}
    if isinstance(value, (list, tuple, np.ndarray)):
        return [encode(v) for v in value]
    if isinstance(value, np.generic):
        return value.item()
    return _encode_scalar(value)


def _encode_scalar(value: Any) -> Any:
    if isinstance(value, Path):
        return str(value)
    if isinstance(value, datetime):
        return value.isoformat()
    if isinstance(value, timedelta):
        return value.total_seconds()
    return value


# ------------------------------------------------------------------- JSON Schema (a subset)
_JSON_TYPES: dict[str, tuple[type, ...]] = {
    "object": (dict,),
    "array": (list,),
    "string": (str,),
    "integer": (int,),
    "number": (int, float),
    "boolean": (bool,),
    "null": (type(None),),
}


def errors(instance: Any, rules: Mapping[str, Any], where: str = "$") -> list[str]:
    """Every way `instance` breaks `rules`: type, enum, const, minimum, maximum, minItems,
    maxItems, items, properties, required, additionalProperties, anyOf, and `$ref` to another
    folder's schema by name (`"$ref": "core.TeamSpec"`, or one of its properties with
    `"$ref": "core.Policy#calls"`)."""
    if "$ref" in rules:
        folder, _, prop = rules["$ref"].partition("#")
        target = schema(folder)
        return errors(instance, target["properties"][prop] if prop else target, where)
    if "anyOf" in rules:
        if all(errors(instance, option, where) for option in rules["anyOf"]):
            return [f"{where}: matches no anyOf option"]
        return []
    if "type" in rules and not _type_ok(instance, rules["type"]):
        return [f"{where}: {instance!r} is not {rules['type']}"]
    found = _value_errors(instance, rules, where)
    if isinstance(instance, list):
        found += _array_errors(instance, rules, where)
    if isinstance(instance, dict):
        found += _object_errors(instance, rules, where)
    return found


def _type_ok(instance: Any, wanted: str | list[str]) -> bool:
    names = [wanted] if isinstance(wanted, str) else wanted
    if isinstance(instance, bool):
        return "boolean" in names
    return any(isinstance(instance, _JSON_TYPES[n]) for n in names)


def _value_errors(instance: Any, rules: Mapping[str, Any], where: str) -> list[str]:
    found = []
    if "enum" in rules and instance not in rules["enum"]:
        found.append(f"{where}: {instance!r} not in {rules['enum']}")
    if "const" in rules and instance != rules["const"]:
        found.append(f"{where}: {instance!r} != {rules['const']!r}")
    if "minimum" in rules and instance < rules["minimum"]:
        found.append(f"{where}: {instance!r} < {rules['minimum']}")
    if "maximum" in rules and instance > rules["maximum"]:
        found.append(f"{where}: {instance!r} > {rules['maximum']}")
    return found


def _array_errors(instance: list[Any], rules: Mapping[str, Any], where: str) -> list[str]:
    found = []
    if len(instance) < rules.get("minItems", 0):
        found.append(f"{where}: fewer than {rules['minItems']} items")
    if len(instance) > rules.get("maxItems", len(instance)):
        found.append(f"{where}: more than {rules['maxItems']} items")
    for i, item in enumerate(instance):
        found += errors(item, rules.get("items", {}), f"{where}[{i}]")
    return found


def _object_errors(instance: dict[str, Any], rules: Mapping[str, Any], where: str) -> list[str]:
    props = rules.get("properties", {})
    found = [f"{where}: missing {k!r}" for k in rules.get("required", []) if k not in instance]
    for key, value in instance.items():
        if key in props:
            found += errors(value, props[key], f"{where}.{key}")
        elif rules.get("additionalProperties", True) is False:
            found.append(f"{where}: unexpected {key!r}")
        elif isinstance(rules.get("additionalProperties"), dict):
            found += errors(value, rules["additionalProperties"], f"{where}.{key}")
    return found


# ------------------------------------------------------------------ decoding to contract types
def decode(tp: Any, data: Any) -> Any:
    """`data` (parsed JSON) as an instance of the contract type `tp`; TypeError if it isn't one."""
    if isinstance(tp, typing.TypeAliasType):
        tp = tp.__value__
    if tp is Any or tp is object:
        return data
    decoder = next((d for test, d in _DECODERS if test(tp)), _decode_scalar)
    return decoder(tp, data)


def _decode_dataclass(tp: type, data: Any) -> Any:
    fields = {f.name: f for f in dataclasses.fields(tp)}
    required = {
        n
        for n, f in fields.items()
        if f.default is dataclasses.MISSING and f.default_factory is dataclasses.MISSING
    }
    if not isinstance(data, dict) or not required <= data.keys() <= fields.keys():
        raise TypeError(f"{tp.__name__} needs an object of {sorted(fields)}, got {data!r}")
    hints = typing.get_type_hints(tp)
    return tp(**{k: decode(hints[k], v) for k, v in data.items()})


def _decode_union(tp: Any, data: Any) -> Any:
    options = typing.get_args(tp)
    if data is None and type(None) in options:
        return None
    problems = []
    for option in (o for o in options if o is not type(None)):
        try:
            return decode(option, data)
        except TypeError as exc:
            problems.append(str(exc))
    raise TypeError(f"{data!r} matches no option: {problems}")


def _decode_literal(tp: Any, data: Any) -> Any:
    if data not in typing.get_args(tp):
        raise TypeError(f"{data!r} is not one of {typing.get_args(tp)}")
    return data


def _is_array(tp: Any) -> bool:
    """`np.ndarray`, `np.ndarray[...]`, or a subscripted alias of it such as `NDArray[...]`."""
    origin = typing.get_origin(tp)
    if isinstance(origin, typing.TypeAliasType):
        origin = typing.get_origin(origin.__value__)
    return tp is np.ndarray or origin is np.ndarray


def _decode_array(_tp: Any, data: Any) -> np.ndarray[Any, Any]:
    array = np.asarray(data)
    if array.dtype.kind == "f":
        return array.astype(np.float32)
    if array.dtype.kind not in "biu":
        raise TypeError(f"{data!r} is not a numeric array")
    return array


def _decode_generic(tp: Any, data: Any) -> Any:
    origin, args = typing.get_origin(tp), typing.get_args(tp)
    is_mapping = isinstance(origin, type) and issubclass(origin, Mapping)
    if not isinstance(data, dict if is_mapping else list):
        raise TypeError(f"{tp!r} can't hold {data!r}")
    if is_mapping:
        return {decode(args[0], k): decode(args[1], v) for k, v in data.items()}
    if origin is list or (len(args) == 2 and args[1] is Ellipsis):
        items = [decode(args[0], d) for d in data]
        return items if origin is list else tuple(items)
    if len(args) != len(data):
        raise TypeError(f"{data!r} does not have {len(args)} items")
    return tuple(decode(a, d) for a, d in zip(args, data, strict=True))


def _decode_scalar(tp: Any, data: Any) -> Any:
    if tp is Path:
        return Path(_checked(str, data))
    if tp is datetime:
        return datetime.fromisoformat(_checked(str, data))
    if tp is timedelta:
        return timedelta(seconds=_checked(float, data))
    if tp is float:
        return float(_checked(float, data))
    if tp in {int, str, bool}:
        return _checked(tp, data)
    raise TypeError(f"no decoder for {tp!r}")


def _checked(tp: type, data: Any) -> Any:
    numeric = tp is float and isinstance(data, int)
    if (isinstance(data, bool) and tp is not bool) or not (isinstance(data, tp) or numeric):
        raise TypeError(f"{data!r} is not {tp.__name__}")
    return data


_DECODERS: list[tuple[typing.Callable[[Any], bool], typing.Callable[[Any, Any], Any]]] = [
    (lambda tp: isinstance(tp, type) and dataclasses.is_dataclass(tp), _decode_dataclass),
    (lambda tp: typing.get_origin(tp) in {types.UnionType, typing.Union}, _decode_union),
    (lambda tp: typing.get_origin(tp) is typing.Literal, _decode_literal),
    (_is_array, _decode_array),
    (lambda tp: typing.get_origin(tp) is not None, _decode_generic),
]

from rllab.core import Array


def row_key(row: Array) -> tuple[float, ...]:
    return tuple(np.round(np.asarray(row, np.float64), 5).tolist())


class ReplayPolicy:
    def __init__(self, calls: list[dict[str, Any]]) -> None:
        self.table: dict[tuple[float, ...], Array] = {}
        for call in calls:
            for row, action in zip(call["obs"], call["actions"], strict=True):
                self.table[row_key(np.asarray(row))] = np.asarray(action)

    def act(self, obs: Array, rng: np.random.Generator) -> Array:
        return np.stack([self.table[row_key(row)] for row in obs])

def summarize(
    runs: Sequence[Run],
    spec: WorldSpec,
    team: TeamSpec,
    scores: tuple[Array, Array, Array],
    comparisons: tuple[Comparison, ...],
) -> Summary:
    """`scores` is (learned runs x seeds, random per seed, rule per seed) for `team`."""

    }

        comparisons=comparisons,
        verdict=verdict(run_means, rule_mean, comparisons, team.direction),
        collapse=collapse_step(runs, team, random_mean),
    )


def collapse_step(runs: Sequence[Run], team: TeamSpec, random_mean: float) -> int | None:
    """The earliest step from which some run's last `COLLAPSE_POINTS`+ eval scores are no better
    than `random_mean` in the team's direction; `None` if no run collapsed."""
    starts = [step for run in runs if (step := _collapse_start(run.evals(), team, random_mean))]
    return min(starts, default=None)


def _collapse_start(evals: Sequence[EvalPoint], team: TeamSpec, random_mean: float) -> int | None:
    def no_better(point: EvalPoint) -> bool:
        score = point.scores[team.name]
        return score <= random_mean if team.direction == "higher" else score >= random_mean

    tail = list(takewhile(no_better, reversed(evals)))
    return tail[-1].step if len(tail) >= COLLAPSE_POINTS else None

    path = out / "summary.json"
    path.write_text(json.dumps([asdict(s) for s in summaries], indent=1))
    return path

MIN_RUNS = 5

def final_evaluation(
    runs: Sequence[Run], out: Path, *, workers: int | None = None
) -> list[Summary]:
    """One `Summary` per team of the runs' world and variant, also written to `out/summary.json`.

    Plays serially; `workers` is kept for spreading the seeds over processes (QR-05).
    """
    spec = runs[0].record.spec
    world = make(spec.world, spec.variant)
    seeds = np.random.SeedSequence(PERMUTATION_SEED).spawn(len(spec.teams))
    summaries = [
        evaluate_team(runs, world, team, np.random.default_rng(seed))
        for team, seed in zip(spec.teams, seeds, strict=True)
    ]
    write_summaries(summaries, out)
    return summaries


def evaluate_team(
    runs: Sequence[Run], world: World, team: TeamSpec, rng: np.random.Generator
) -> Summary:
    """Plays `team`'s learned policies and baselines, each against rule-based opponents (PREY-08),
    and tests learned against each baseline once there are `MIN_RUNS` runs."""
    learned, random, rule = play_team(runs, world, team.name)
    comparisons = (
        compare_baselines(
            team.name, learned, {"random": random, "rule": rule}, team.direction, rng=rng
        )
        if len(runs) >= MIN_RUNS
        else ()
    )
    return summarize(runs, world.spec, team, (learned, random, rule), comparisons=comparisons)


def play_team(runs: Sequence[Run], world: World, team: str) -> tuple[Array, Array, Array]:
    """`team`'s scores on `FINAL_SEEDS`: learned (runs x seeds), random and rule (per seed)."""
    spec = world.spec
    opponents: dict[str, Policy] = {
        t.name: rule_for(spec.world, spec.variant, t.name) for t in spec.teams if t.name != team
    }
    learned = np.stack([
        play_seeds(world, {**opponents, team: load_policy(final_checkpoint(run), team, spec)}, team)
        for run in runs
    ])
    random = play_seeds(world, {**opponents, team: RandomPolicy(spec.action)}, team)
    rule = play_seeds(world, {**opponents, team: rule_for(spec.world, spec.variant, team)}, team)
    return learned, random, rule


def play_seeds(world: World, policies: Mapping[str, Policy], team: str) -> Array:
    """`team`'s score in one episode per final seed."""
    return np.array(
        [play_episode(world, policies, seed).result.scores[team] for seed in FINAL_SEEDS],
        dtype=np.float64,
    )


def final_checkpoint(run: Run) -> Checkpoint:
    return max(run.checkpoints(), key=lambda ckpt: ckpt.step)
