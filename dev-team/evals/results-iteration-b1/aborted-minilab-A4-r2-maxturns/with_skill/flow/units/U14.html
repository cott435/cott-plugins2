10	def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
11	    """Prices from the first date every ticker has a close; a gap takes the previous close.
12	
13	    Raises GapError naming the ticker when more than `max_gap` dates in a row are missing.
14	    Spine version: keeps only the dates every ticker has a close for, with no fill.
15	    """
16	    closes_by_ticker: dict[str, dict[str, float]] = {}
17	    for row in rows:
18	        by_date = closes_by_ticker.setdefault(row.ticker, {})
19	        if row.close is not None:
20	            by_date[row.date] = row.close
21	    tickers = tuple(sorted(closes_by_ticker))
22	    common = set.intersection(*(set(closes_by_ticker[t]) for t in tickers)) if tickers else set()
23	    dates = tuple(sorted(common))
24	    closes = {t: tuple(closes_by_ticker[t][d] for d in dates) for t in tickers}
25	    return Prices(dates=dates, tickers=tickers, closes=closes)

10	def validate_rows(rows: Sequence[RawRow], tickers: Sequence[str] | None = None) -> list[RawRow]:
11	    """Rows with a positive close, one per (date, ticker), the later in file order wins,
12	    sorted by (date, ticker); only `tickers` when given.
13	
14	    Raises KeyError for a requested ticker with no usable row.
15	    """

17	        (row.date, row.ticker): row for row in rows if row.close is not None and row.close > 0
18	    }
19	    if tickers is not None:
20	        wanted = set(tickers)
21	        latest = {key: row for key, row in latest.items() if row.ticker in wanted}
22	        missing = wanted - {row.ticker for row in latest.values()}
23	        if missing:
24	            raise KeyError(f"no usable rows for tickers: {sorted(missing)}")
25	    return [latest[key] for key in sorted(latest)]
