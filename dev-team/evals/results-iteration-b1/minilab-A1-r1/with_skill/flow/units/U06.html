from __future__ import annotations

import math

import pytest

from mdata import Prices
from risk import Returns, log_returns


def _prices(payload: dict) -> Prices:
    return Prices(
        dates=tuple(payload["dates"]),
        tickers=tuple(payload["tickers"]),
        closes={t: tuple(c) for t, c in payload["closes"].items()},
    )


@pytest.mark.intent
def test_tc04_geometric_closes_match_the_closed_form(golden):
    given, expected = golden("risk", "TC-04")
    returns = log_returns(_prices(given["prices"]))

    assert isinstance(returns, Returns)
    assert returns.base_date == expected["base_date"]
    assert returns.dates == tuple(expected["dates"])
    assert returns.tickers == tuple(expected["tickers"])
    assert set(returns.values) == set(expected["tickers"])
    for ticker in expected["tickers"]:
        assert list(returns.values[ticker]) == pytest.approx(
            expected["values"][ticker], abs=expected["tol"], rel=0
        )


@pytest.mark.intent
def test_tc04_every_return_is_its_ticker_closed_form(golden):
    given, expected = golden("risk", "TC-04")
    returns = log_returns(_prices(given["prices"]))

    closed_form = {"X": math.log(1.01), "Y": math.log(0.98)}
    for ticker, rate in closed_form.items():
        assert all(
            v == pytest.approx(rate, abs=expected["tol"], rel=0) for v in returns.values[ticker]
        )


@pytest.mark.intent
def test_each_return_is_dated_at_its_later_close(contract_sample, golden):
    # RISK-01: r_t = ln(close_t / close_{t-1}), dated t. Non-constant closes catch a shift.
    _, expected = golden("risk", "TC-04")
    prices = _prices(contract_sample("mdata.Prices"))
    returns = log_returns(prices)

    assert returns.base_date == prices.dates[0]
    assert returns.dates == prices.dates[1:]
    assert returns.tickers == prices.tickers
    for ticker in prices.tickers:
        closes = prices.closes[ticker]
        values = returns.values[ticker]
        assert len(values) == len(returns.dates)
        for i, value in enumerate(values):
            assert closes[i] * math.exp(value) == pytest.approx(
                closes[i + 1], rel=expected["tol"] * 10
            )


@pytest.mark.intent
def test_returns_are_logarithmic_not_simple(contract_sample):
    prices = _prices(contract_sample("mdata.Prices"))
    returns = log_returns(prices)

    closes = prices.closes["AAA"]
    simple = closes[1] / closes[0] - 1.0
    assert returns.values["AAA"][0] != pytest.approx(simple, abs=1e-6)
    assert returns.values["AAA"][0] == pytest.approx(math.log(closes[1] / closes[0]), abs=1e-12)


@pytest.mark.intent
def test_one_date_raises_value_error():

        log_returns(prices)
