class ReplayDouble:
    """Frame indices only: `advance` moves on and stops at the last frame; `show` plays out."""

    def __init__(self, game: GameRecording, *, fps: float = 8.0) -> None:
        self.game = game
        self.fps = fps
        self.index = 0
        self.paused = False

    def advance(self) -> int:
        self.index = min(self.index + 1, len(self.game.frames) - 1)
        return self.index

    def on_key(self, key: str) -> None:
        if key == " ":
            self.paused = not self.paused
        elif key == "right":
            self.advance()
        elif key == "+":
            self.fps *= 2
        elif key == "-":
            self.fps /= 2

    def current(self) -> tuple[np.ndarray, dict[str, float], int]:
        scores = {team: float(values[self.index]) for team, values in self.game.scores.items()}
        return self.game.frames[self.index], scores, self.index

    def show(self) -> None:
        while self.index < len(self.game.frames) - 1:
            self.advance()
