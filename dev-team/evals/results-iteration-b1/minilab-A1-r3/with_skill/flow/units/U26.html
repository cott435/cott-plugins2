from collections.abc import Mapping, Sequence

from mdata.types import GapError, Prices, RawRow


def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
    """One calendar from the first date every ticker has a close; gaps take the previous close (DATA-03, DATA-04)."""
    by_ticker = _closes_by_ticker(rows)
    tickers = tuple(sorted(by_ticker))
    traded = sorted({day for closes in by_ticker.values() for day in closes})
    filled = {ticker: _forward_fill(ticker, by_ticker[ticker], traded, max_gap) for ticker in tickers}
    dates = _common_dates(traded, filled)
    return Prices(
        dates=dates,
        tickers=tickers,
        closes={ticker: tuple(filled[ticker][day] for day in dates) for ticker in tickers},
    )


def _closes_by_ticker(rows: Sequence[RawRow]) -> dict[str, dict[str, float]]:
    """Each ticker's closes by date, skipping blank cells."""
    by_ticker: dict[str, dict[str, float]] = {}
    for row in rows:
        closes = by_ticker.setdefault(row.ticker, {})
        if row.close is not None:
            closes[row.date] = row.close
    return by_ticker


def _forward_fill(
    ticker: str, closes: Mapping[str, float], traded: Sequence[str], max_gap: int
) -> dict[str, float]:
    """The ticker's close on each traded date from its first close, a missing date taking the previous one.

    Raises GapError naming the ticker when more than `max_gap` traded dates in a row are missing.
    """
    filled: dict[str, float] = {}
    previous: float | None = None
    missing = 0
    for day in traded:
        if day in closes:
            previous = closes[day]
            missing = 0
        elif previous is not None:
            missing += 1
            if missing > max_gap:
                raise GapError(f"{ticker} misses more than {max_gap} dates in a row, through {day}")
        if previous is not None:
            filled[day] = previous
    return filled


def _common_dates(traded: Sequence[str], filled: Mapping[str, Mapping[str, float]]) -> tuple[str, ...]:
    """The traded dates on which every ticker has a (possibly filled) close."""
    return tuple(day for day in traded if all(day in closes for closes in filled.values()))
