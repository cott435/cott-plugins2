```python
def run_forecast(*, known_on: date | None = None, score_holdout: bool = False, settings_path: Path | None = None) -> Report:
    today = date.today()
    phase: Phase = "holdout" if score_holdout else "development"
    settings = load_settings(settings_path or SETTINGS_PATH)
    inputs = load_inputs(known_on=known_on or today, phase=phase)
    samples, formations, shrinkage, holdings = run_months(inputs)
    schedule = refits(inputs.study.panel.rebalances)
    forecasts = run_forecasters(samples, schedule, settings)
    scores, forecast_tests = run_forecast_tests(forecasts, samples, settings, phase=phase)
    results, gamma, best = run_backtest(formations, shrinkage, holdings, forecasts, settings)
    portfolio_tests = run_portfolio_tests(results, settings, phase=phase)
    profiled = profile(samples, forecasts["network"], results)
    record = holdout_record(settings, phase=phase, today=today)
    sections = report_sections(scores, forecast_tests, profiled, results, portfolio_tests, record)
    params = report_params(inputs, record, settings, gamma=gamma, best=best)
    frames = tape_frames(sleeves(results), forecasts["network"])
    return save_report("forecast", sections, params=params, frames=frames)
```

def run_agents(*, score_holdout: bool = False, settings_path: Path | None = None) -> Report:
    today = date.today()
    phase: Phase = "holdout" if score_holdout else "development"
