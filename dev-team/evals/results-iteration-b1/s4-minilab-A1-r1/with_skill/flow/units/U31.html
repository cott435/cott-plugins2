8	def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
9	    """Prices from the first date every ticker has; `GapError` past `max_gap` missing dates."""
10	    by_ticker: dict[str, dict[str, float]] = {}

12	        if row.close is not None:
13	            by_ticker.setdefault(row.ticker, {})[row.date] = row.close

17	    closes = {ticker: tuple(by_ticker[ticker][date] for date in dates) for ticker in tickers}
18	    return Prices(dates=dates, tickers=tickers, closes=closes)

from __future__ import annotations

import pytest

import mdata
from mdata import GapError, Prices, RawRow, align_calendar


def _rows(payload: dict) -> list[RawRow]:
    return [RawRow(date=d, ticker=t, close=c) for d, t, c in payload["rows"]]


def _prices(payload: dict) -> Prices:
    return Prices(
        dates=tuple(payload["dates"]),
        tickers=tuple(payload["tickers"]),
        closes={t: tuple(v) for t, v in payload["closes"].items()},
    )


@pytest.mark.intent
def test_tc02_one_calendar_with_filled_gaps(golden):
    given, expected = golden("mdata", "TC-02")
    prices = align_calendar(_rows(given), max_gap=given["max_gap"])
    want = _prices(expected)
    assert prices.dates == want.dates
    assert prices.tickers == want.tickers
    assert {t: tuple(v) for t, v in prices.closes.items()} == want.closes

    given, _ = golden("mdata", "TC-02")

"""PL-01 step 3: put every ticker on one calendar."""

from collections.abc import Mapping, Sequence

from mdata.types import GapError, Prices, RawRow


def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
    """Prices from the first date every ticker has; `GapError` past `max_gap` missing dates."""
    by_ticker: dict[str, dict[str, float]] = {}
    for row in rows:
        if row.close is not None:
            by_ticker.setdefault(row.ticker, {})[row.date] = row.close
    tickers = tuple(sorted(by_ticker))
    start = max((min(closes) for closes in by_ticker.values()), default="")
    traded = {date for closes in by_ticker.values() for date in closes}
    dates = tuple(sorted(date for date in traded if date >= start))
    closes = {ticker: _fill(ticker, by_ticker[ticker], dates, max_gap) for ticker in tickers}
    return Prices(dates=dates, tickers=tickers, closes=closes)


def _fill(
    ticker: str, known: Mapping[str, float], dates: Sequence[str], max_gap: int
) -> tuple[float, ...]:
    """The ticker's closes on `dates`, each missing date taking the previous close."""
    filled: list[float] = []
    missing = 0
    for date in dates:
        if date in known:
            filled.append(known[date])
            missing = 0
            continue
        missing += 1
        if missing > max_gap:
            raise GapError(f"{ticker} is missing more than {max_gap} dates in a row at {date}")
        filled.append(filled[-1])
    return tuple(filled)
