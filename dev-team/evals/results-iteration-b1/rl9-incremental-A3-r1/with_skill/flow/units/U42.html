def tc28_game(inp: dict[str, Any]) -> tuple[GameRecording, list[np.ndarray]]:
    """Plays the TC-06 start board with the repeated action; returns the recording and Snake's frames."""
    env = Snake(view="grid", size=inp["size"])
    env.reset(0)
    env.reset_to([tuple(cell) for cell in inp["body"]], inp["heading"], tuple(inp["food"]))
    snake_frames = [env.frame().copy()]
    rewards: list[float] = []
    for _ in range(inp["n_steps"]):
        result = env.step({SOLO: np.array([inp["repeat_action"]], dtype=np.int64)})
        rewards.append(float(result.reward[SOLO][0]))
        snake_frames.append(env.frame().copy())
    score = np.concatenate([[0.0], np.cumsum(rewards)]).astype(np.float32)
    game = GameRecording(
        seed=0,
        board="grid",
        frames=np.stack(snake_frames).astype(np.int8),
        rewards=np.asarray(rewards, dtype=np.float32),
        scores={SOLO: score},
    )
    return game, snake_frames
