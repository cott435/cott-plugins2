"""PL-01 step 3: put every ticker on one calendar."""

from __future__ import annotations

from collections.abc import Sequence

from mdata.types import Prices, RawRow


def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
    """One calendar from the first date every ticker has a close; gaps take the previous close.

    Raises GapError naming the ticker when more than `max_gap` dates in a row are missing
    (DATA-03, DATA-04). Spine version: keeps only the dates every ticker has a close for.
    """
    by_ticker: dict[str, dict[str, float]] = {}
    for row in rows:
        closes = by_ticker.setdefault(row.ticker, {})
        if row.close is not None:
            closes[row.date] = row.close
    tickers = tuple(sorted(by_ticker))
    shared = set.intersection(*(set(closes) for closes in by_ticker.values())) if tickers else set()
    dates = tuple(sorted(shared))
    return Prices(
        dates=dates,
        tickers=tickers,
        closes={t: tuple(by_ticker[t][d] for d in dates) for t in tickers},
    )
