+def _ticker_drawdown(returns: Returns, ticker: str) -> Drawdown:
+    """Track log wealth and its running peak; the first peak and trough win ties."""
+    log_wealth = peak = 0.0
+    peak_date = returns.base_date
+    worst = Drawdown(ticker, 0.0, returns.base_date, returns.base_date)

+        log_wealth += value

+"""Unit tests for risk.max_drawdown."""
+
+from __future__ import annotations
+
+import itertools
+import json
+import math
+from pathlib import Path
+
+import pytest
+
+from risk import Drawdown, Returns, max_drawdown
+
+SAMPLE = Path(__file__).parents[2] / "contracts" / "risk.Returns" / "sample.json"
+
+
+def _sample_returns() -> Returns:
+    data = json.loads(SAMPLE.read_text())
+    return Returns(
+        base_date=data["base_date"],
+        dates=tuple(data["dates"]),
+        tickers=tuple(data["tickers"]),
+        values={t: tuple(v) for t, v in data["values"].items()},
+    )
+
+
+def _from_closes(closes: tuple[float, ...]) -> Returns:
+    dates = tuple(f"2024-01-{day:02d}" for day in range(1, 1 + len(closes)))
+    logs = tuple(math.log(b / a) for a, b in itertools.pairwise(closes))
+    return Returns(base_date=dates[0], dates=dates[1:], tickers=("X",), values={"X": logs})
+
+
+def test_sample_gives_one_fall_and_one_rising_ticker() -> None:
+    aaa, bbb = max_drawdown(_sample_returns())
+
+    assert aaa.ticker == "AAA"
+    assert aaa.depth == pytest.approx(-0.01, abs=1e-12)
+    assert (aaa.peak_date, aaa.trough_date) == ("2024-03-04", "2024-03-05")
+    assert bbb == Drawdown("BBB", 0.0, "2024-03-01", "2024-03-01")
+
+
+def test_deepest_fall_below_running_peak() -> None:
+    (dd,) = max_drawdown(_from_closes((100.0, 110.0, 99.0, 88.0, 95.0, 120.0)))
+
+    assert dd.depth == pytest.approx(-0.2, abs=1e-12)
+    assert (dd.peak_date, dd.trough_date) == ("2024-01-02", "2024-01-04")
+
+
+def test_first_day_fall_peaks_at_base_date() -> None:
+    (dd,) = max_drawdown(_from_closes((100.0, 90.0, 95.0)))
+
+    assert dd.depth == pytest.approx(-0.1, abs=1e-12)
+    assert (dd.peak_date, dd.trough_date) == ("2024-01-01", "2024-01-02")
+
