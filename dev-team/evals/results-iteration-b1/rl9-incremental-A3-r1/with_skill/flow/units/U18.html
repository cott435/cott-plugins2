86	### AD-02: Hand-written agents on NumPy arrays and PyTorch tensors
87	**Status:** Accepted · **Date:** 2026-10-17 · **Scope:** `agents`
88	- **Context:** AGENT-06 needs the agents written by hand from array and tensor operations, with automatic gradients allowed. The lesson is value learning, not backpropagation.
89	- **Decision:** The table-based agent is pure NumPy. The neural agents use PyTorch tensors, `torch.nn` layers, autograd and the Adam optimizer. Everything RL-specific is written by hand in `agents`: the memory of past steps, sampling, the target copy, the learning targets, the loss and the exploration schedule.
90	- **Alternatives:** NumPy only, with hand-written backpropagation — more to learn, but debugging gradients would crowd out the RL lesson, and the CNN in AD-05 becomes a project of its own. JAX — capable, but less familiar CPU tooling on a Mac.
91	- **Consequences:** PyTorch is a dependency of `agents` only. Swapping in hand-written backpropagation later stays local to the network module.

94	### AD-05: An MLP for the compact view, a small CNN for the full grid
95	**Status:** Accepted · **Date:** 2026-10-17 · **Scope:** `agents`
96	- **Context:** AGENT-03 takes an 11-value compact view. AGENT-05 takes the full grid. QR-02 and QR-03 give each 60 CPU minutes.
97	- **Decision:** Compact view: two hidden layers of 128 units, the same shape the reference uses. Full grid: two 3×3 convolution layers (16 and 32 channels) and one 128-unit hidden layer, over a 3×10×10 head/body/food grid.
98	- **Alternatives:** An MLP on the flattened grid — cheaper per step but slower to learn spatial patterns. It stays the fallback if the CNN misses QR-03.
99	- **Consequences:** The network is chosen in settings (`net = "mlp"` or `"cnn"`), so the fallback is a settings change, not a code change.

102	### AD-08: Single-threaded, deterministic CPU training
103	**Status:** Accepted · **Date:** 2026-10-17 · **Scope:** system
104	- **Context:** RUN-03 needs identical curves from identical settings and seed on one machine. QR-06 rules out NVIDIA GPUs. The networks are small.
105	- **Decision:** `torch.set_num_threads(1)`, `torch.use_deterministic_algorithms(True)`, CPU device everywhere. The Apple GPU backend is not used.
106	- **Alternatives:** The Apple GPU backend — no gain on networks this small, and its determinism is weaker. Several threads — little gain, and reduction order can vary.
107	- **Consequences:** The 5 seeds of one ladder rung can run as 5 parallel processes without fighting over threads.

import math

GRAD_CLIP = 10.0


def mlp(n_inputs: int, hidden: int, n_actions: int) -> nn.Sequential:
    """Two ReLU hidden layers of `hidden` units over the flattened observation (AD-05)."""
    return nn.Sequential(
        nn.Flatten(),
        nn.Linear(n_inputs, hidden),
        nn.ReLU(),
        nn.Linear(hidden, hidden),
        nn.ReLU(),
        nn.Linear(hidden, n_actions),
    )


def epsilon(exploration: Exploration, step: int) -> float:
    """The exploration rate after `step` transitions: linear to `end`, then flat (AGENT-02)."""
    if exploration.decay_steps <= 0:
        return exploration.end
    frac = min(step / exploration.decay_steps, 1.0)
    return exploration.start + frac * (exploration.end - exploration.start)


class Memory:
    """A uniform ring buffer of single-agent transitions."""

    def __init__(self, size: int, obs_shape: tuple[int, ...]) -> None:
        self.obs = np.zeros((size, *obs_shape), dtype=np.float32)
        self.next_obs = np.zeros((size, *obs_shape), dtype=np.float32)
        self.actions = np.zeros(size, dtype=np.int64)
        self.reward = np.zeros(size, dtype=np.float32)
        self.terminated = np.zeros(size, dtype=np.float32)
        self.count = 0

    def add(self, t: Transition) -> None:
        """Store agent 0's step in the next slot, overwriting the oldest when full."""
        i = self.count % len(self.actions)
        self.obs[i] = t.obs[0]
        self.next_obs[i] = t.next_obs[0]
        self.actions[i] = t.actions[0]
        self.reward[i] = t.reward[0]
        self.terminated[i] = float(t.terminated[0])
        self.count += 1

    def sample(self, rng: np.random.Generator, n: int) -> tuple[torch.Tensor, ...]:
        """`n` stored steps drawn uniformly with replacement, as tensors."""
        idx = rng.integers(0, min(self.count, len(self.actions)), size=n)
        arrays = (self.obs, self.actions, self.reward, self.next_obs, self.terminated)
        return tuple(torch.from_numpy(a[idx]) for a in arrays)


class DQNAgent(Agent):
    """A single-agent DQN sized from its team's observation shape and action count (AD-18)."""

    def __init__(
        self,
        team: TeamSpec,
        params: AgentParams,
        exploration: Exploration,
        *,
        seed: int,
        ablation: Literal["none", "no_memory", "no_target"] = "none",
    ) -> None:
        if team.n_agents != 1:
            raise ValueError(f"DQNAgent is single-agent, got n_agents={team.n_agents}")
        if team.actions.kind != "discrete":
            raise ValueError("DQNAgent needs discrete actions, got continuous")
        if params.net == "cnn" and len(team.obs_shape) != 3:
            raise ValueError(f"net 'cnn' needs a 3-D obs_shape, got {team.obs_shape}")
        torch.set_num_threads(1)
        torch.use_deterministic_algorithms(True)
        torch.manual_seed(seed)
        self._params = params
        self._exploration = exploration
        self._ablation = ablation
        self._n_actions = team.actions.n
        self._rng = np.random.default_rng(seed)
        self._online = mlp(math.prod(team.obs_shape), params.hidden, team.actions.n)
        self._target = mlp(math.prod(team.obs_shape), params.hidden, team.actions.n)
        self._target.load_state_dict(self._online.state_dict())
        self._optimizer = torch.optim.Adam(self._online.parameters(), lr=params.lr)
        self._memory = Memory(params.memory_size, team.obs_shape)
        self._updates = 0

    def act(self, obs: np.ndarray, *, explore: bool) -> np.ndarray:
        """Agent 0's action: random with the current exploration rate if exploring, else argmax."""
        if explore and self._rng.random() < epsilon(self._exploration, self._memory.count):
            return np.array([self._rng.integers(self._n_actions)], dtype=np.int64)
        with torch.no_grad():
            q = self._online(torch.as_tensor(obs, dtype=torch.float32))
        return q.argmax(dim=1).numpy().astype(np.int64)

    def learn(self, t: Transition) -> None:
        """Remember the step, then once past `learn_start` fit one random batch of memory."""
        if not t.active[0]:
            return
        self._memory.add(t)
        if self._memory.count < self._params.learn_start:
            return
        self._update(self._memory.sample(self._rng, self._params.batch_size))
        self._updates += 1
        if self._updates % self._params.target_every == 0:

        loss.backward()
        nn.utils.clip_grad_norm_(self._online.parameters(), GRAD_CLIP)
        self._optimizer.step()

        """Write the online network's PyTorch state dict."""
        torch.save(self._online.state_dict(), path)

        """Read a state dict into both the online and the target network."""
        state = torch.load(path, weights_only=True)

        self._target.load_state_dict(state)

def _array(d: dict[str, object]) -> np.ndarray:
    return np.asarray(d["data"], dtype=str(d["dtype"])).reshape(d["shape"])  # type: ignore[arg-type]


def _transition(d: dict[str, dict[str, object]]) -> Transition:
    return Transition(
        obs=_array(d["obs"]),
        actions=_array(d["actions"]),
        reward=_array(d["reward"]),
        next_obs=_array(d["next_obs"]),
        terminated=_array(d["terminated"]),
        active=_array(d["active"]),
        truncated=bool(d["truncated"]),
    )

    path = CONTRACTS / "agents.DQNAgent" / "samples" / "compact_after_learn_start.json"
    calls = json.loads(path.read_text())["calls"]
    return [_transition(c["transition"]) for c in calls if c["method"] == "learn"]

    return AgentParams(lr=1e-3, gamma=0.9, **kw)  # type: ignore[arg-type]


def _weights(agent: DQNAgent, attr: str) -> list[torch.Tensor]:
    return [p.detach().clone() for p in getattr(agent, attr).parameters()]


def _same(a: list[torch.Tensor], b: list[torch.Tensor]) -> bool:
    return all(torch.equal(x, y) for x, y in zip(a, b, strict=True))
