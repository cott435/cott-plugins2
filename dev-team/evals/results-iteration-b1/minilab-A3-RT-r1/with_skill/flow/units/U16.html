+"""Unit tests for mdata.io.read_csv (PL-01 step 1, DATA-01)."""

+from __future__ import annotations
+
+from pathlib import Path
+
+import pytest
+
+from mdata.io import read_csv
+from mdata.types import RawRow
+
+
+def _write(tmp_path: Path, text: str) -> Path:
+    path = tmp_path / "prices.csv"
+    path.write_text(text)
+    return path
+
+
+def test_rows_come_back_in_file_order(tmp_path: Path) -> None:
+    path = _write(
+        tmp_path,
+        "date,ticker,close\n2024-01-03,BBB,20.5\n2024-01-02,AAA,10\n2024-01-02,BBB,19.25\n",
+    )
+    assert read_csv(path) == [
+        RawRow(date="2024-01-03", ticker="BBB", close=20.5),
+        RawRow(date="2024-01-02", ticker="AAA", close=10.0),
+        RawRow(date="2024-01-02", ticker="BBB", close=19.25),
+    ]
+
+
+def test_blank_close_is_none_not_zero(tmp_path: Path) -> None:
+    path = _write(tmp_path, "date,ticker,close\n2024-01-02,AAA,\n2024-01-03,AAA,0\n")
+    rows = read_csv(str(path))
+    assert rows[0].close is None
+    assert rows[1].close == 0.0
+
+
+def test_column_order_does_not_matter(tmp_path: Path) -> None:
+    path = _write(tmp_path, "ticker,close,date\nAAA,1.5,2024-01-02\n")
+    assert read_csv(path) == [RawRow(date="2024-01-02", ticker="AAA", close=1.5)]
+
+
+def test_header_only_gives_no_rows(tmp_path: Path) -> None:
+    assert read_csv(_write(tmp_path, "date,ticker,close\n")) == []
+
+
+def test_missing_column_raises(tmp_path: Path) -> None:
+    path = _write(tmp_path, "date,ticker\n2024-01-02,AAA\n")

+        read_csv(path)
