61	import numpy as np
62	import pandas as pd
63	
64	
65	def daily_returns(prices: pd.DataFrame) -> pd.DataFrame:
66	    """Log returns of a dates x tickers close panel."""
67	    return np.log(prices / prices.shift(1)).dropna(how="all")
68	
69	
70	def rolling_vol(returns: pd.DataFrame, window: int = 20) -> pd.DataFrame:
71	    """Rolling volatility over the last `window` days, today included."""
72	    return returns.rolling(window).apply(lambda x: np.std(x), raw=True)
73	
74	
75	def annualize(vol: pd.DataFrame) -> pd.DataFrame:
76	    """Annualize to a calendar year, like my brokerage statements: 365 days."""
77	    return vol * np.sqrt(365)
78	
79	
80	def summary(returns: pd.DataFrame) -> pd.DataFrame:
81	    """Yearly return and volatility per ticker."""
82	    return pd.DataFrame({"ann_return": returns.mean() * 365, "ann_vol": returns.std(ddof=0) * np.sqrt(365)})
83	
84	
85	def max_drawdown(prices: pd.DataFrame) -> pd.Series:
86	    """Deepest fall below the running high."""
87	    return (prices / prices.cummax() - 1).min()
88	```

from __future__ import annotations

import pytest

from risk.annualize import annualize
from risk.types import Returns, VolSeries


def _returns(data: dict) -> Returns:
    return Returns(
        base_date=data["base_date"],
        dates=tuple(data["dates"]),
        tickers=tuple(data["tickers"]),
        values={ticker: tuple(values) for ticker, values in data["values"].items()},
    )


def _vol(data: dict) -> VolSeries:
    return VolSeries(
        returns=_returns(data["returns"]),
        window=data["window"],
        annualized=data["annualized"],
        values={ticker: tuple(values) for ticker, values in data["values"].items()},
    )


@pytest.mark.intent
def test_tc06_scales_each_value_and_keeps_gaps(golden):
    given, expected = golden("risk", "TC-06")
    vol = _vol(given["vol"])

    result = annualize(vol)

    assert result.annualized is expected["annualized"]
    assert set(result.values) == set(expected["values"])
    for ticker, want in expected["values"].items():
        got = result.values[ticker]
        assert len(got) == len(want)
        for g, w in zip(got, want, strict=True):
            if w is None:
                assert g is None
            else:
                assert g == pytest.approx(w, abs=expected["tol"], rel=0)


@pytest.mark.intent
def test_tc06_keeps_returns_and_window(golden):
    given, _ = golden("risk", "TC-06")
    vol = _vol(given["vol"])

    result = annualize(vol)

    assert result.returns == vol.returns
    assert result.window == given["vol"]["window"]
