def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
    """Every date any ticker trades, from the first date all have a close; gaps take the previous close.

    by_ticker: dict[str, dict[str, float]] = {}

from __future__ import annotations

from collections.abc import Sequence

from mdata.types import GapError, Prices, RawRow


def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
    """Every date any ticker trades, from the first date all have a close; gaps take the previous close.

    A ticker missing more than `max_gap` calendar dates in a row raises `GapError` naming it.
    """
    by_ticker: dict[str, dict[str, float]] = {}
    for row in rows:
        if row.close is not None:
            by_ticker.setdefault(row.ticker, {})[row.date] = row.close
    tickers = tuple(sorted(by_ticker))
    calendar = sorted({date for series in by_ticker.values() for date in series})
    start = max((min(series) for series in by_ticker.values()), default="")
    dates = tuple(date for date in calendar if date >= start)
    closes = {ticker: _fill(ticker, by_ticker[ticker], calendar, start, max_gap) for ticker in tickers}
    return Prices(dates=dates, tickers=tickers, closes=closes)


def _fill(
    ticker: str, series: dict[str, float], calendar: list[str], start: str, max_gap: int
) -> tuple[float, ...]:
    """The ticker's close on each calendar date from `start`, a missing date taking the previous one.

    The walk begins at the ticker's own first close, so a date missing at `start` still has one;
    a run of missing dates counts from there, but only a run reaching `start` or later is an error.
    """
    filled: list[float] = []
    previous = 0.0
    missing = 0
    for date in calendar[calendar.index(min(series)) :]:
        if date in series:
            previous = series[date]
            missing = 0
        else:
            missing += 1
        if date < start:
            continue
        if missing > max_gap:
            raise GapError(f"{ticker}: more than {max_gap} dates in a row missing, through {date}")
        filled.append(previous)
    return tuple(filled)
