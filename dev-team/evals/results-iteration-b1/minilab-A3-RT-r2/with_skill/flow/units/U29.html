"""PL-02 entry: annualized rolling volatility of a CSV."""

from __future__ import annotations

from collections.abc import Sequence
from pathlib import Path

import mdata
from risk.annualize import annualize
from risk.returns import log_returns
from risk.types import VolSeries
from risk.vol import rolling_vol


def vol_series(
    path: str | Path, *, window: int = 20, tickers: Sequence[str] | None = None
) -> VolSeries:
    """load_prices -> log_returns -> rolling_vol -> annualize.

    Errors as `mdata.load_prices`; ValueError if `window` < 2.
    """
    prices = mdata.load_prices(path, tickers)
    returns = log_returns(prices)
    vol = rolling_vol(returns, window)
    return annualize(vol)

def test_pipeline_runs_and_returns_aligned_annualized_series(fake_prices) -> None:
    prices = fake_prices.load_prices(CSV)

    vol = vol_series(CSV, window=2)

    assert isinstance(vol.returns, Returns)
    assert vol.window == 2
    assert vol.annualized is True
    assert vol.returns.tickers == ("AAA", "BBB")
    assert vol.returns.base_date == prices.dates[0]
    assert vol.returns.dates == prices.dates[1:]
    assert set(vol.values) == {"AAA", "BBB"}
    for ticker in ("AAA", "BBB"):
        assert len(vol.values[ticker]) == len(prices.dates) - 1


def test_tickers_are_passed_to_load_prices(fake_prices) -> None:
    vol = vol_series(CSV, window=2, tickers=["BBB"])

    assert vol.returns.tickers == ("BBB",)

def test_unknown_ticker_raises_key_error(fake_prices) -> None:
    with pytest.raises(KeyError):

    with pytest.raises(ValueError, match="window"):
