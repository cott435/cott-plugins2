65	def daily_returns(prices: pd.DataFrame) -> pd.DataFrame:
66	    """Log returns of a dates x tickers close panel."""
67	    return np.log(prices / prices.shift(1)).dropna(how="all")
68

70	def rolling_vol(returns: pd.DataFrame, window: int = 20) -> pd.DataFrame:
71	    """Rolling volatility over the last `window` days, today included."""
72	    return returns.rolling(window).apply(lambda x: np.std(x), raw=True)
73	
74	
75	def annualize(vol: pd.DataFrame) -> pd.DataFrame:
76	    """Annualize to a calendar year, like my brokerage statements: 365 days."""
77	    return vol * np.sqrt(365)
78	
79	
80	def summary(returns: pd.DataFrame) -> pd.DataFrame:
81	    """Yearly return and volatility per ticker."""
82	    return pd.DataFrame({"ann_return": returns.mean() * 365, "ann_vol": returns.std(ddof=0) * np.sqrt(365)})
83	
84	
85	def max_drawdown(prices: pd.DataFrame) -> pd.Series:
86	    """Deepest fall below the running high."""
