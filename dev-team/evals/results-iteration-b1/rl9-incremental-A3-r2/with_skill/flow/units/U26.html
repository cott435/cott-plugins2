def eval_seeds(train_seed: int, n: int) -> list[int]:
    """`1_000_000 + 1000 * train_seed + i` for `i < n`: never a training seed."""
    if not 0 <= n <= SEEDS_PER_TRAIN_SEED:

class TwoTeamTank:
    """A tank game of `length` steps: each step every agent earns 0.5, `SOLO` scores the step
    count and `PREY` minus it; it ends by truncation."""

    teams = {SOLO: SPEC, PREY: SPEC}
    board = "tank"

    def __init__(self, length: int) -> None:
        self.length = length
        self.t = 0
        self.seeds: list[int] = []

    def _obs(self) -> dict[str, np.ndarray]:
        return {team: np.full((2, 1), self.t, dtype=np.float32) for team in self.teams}

    def _info(self) -> dict[str, float]:
        return {"steps": self.t, "score": self.t, f"score:{SOLO}": self.t, f"score:{PREY}": -self.t}

    def reset(self, seed: int) -> tuple[dict[str, np.ndarray], dict[str, float]]:
        self.seeds.append(seed)
        self.t = 0
        return self._obs(), self._info()

    def step(self, actions: Mapping[str, np.ndarray]) -> StepResult:
        assert set(actions) == set(self.teams)
        self.t += 1
        over = self.t == self.length
        return StepResult(
            obs=self._obs(),
            reward={team: np.full(2, 0.5, dtype=np.float32) for team in self.teams},
            done={team: np.full(2, over) for team in self.teams},
            terminated=False,
            truncated=over,
            info=self._info(),
        )

    def frame(self) -> np.ndarray:
        return np.array([[0.0, 0.0, 1.0, 0.0], [self.t, 0.0, 0.1, 3.0]], dtype=np.float32)


class Still:
    """Always action 0 for both agents; counts its greedy and exploring calls."""
