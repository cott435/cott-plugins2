79	    """Log returns of a dates x tickers close panel."""
80	    return np.log(prices / prices.shift(1)).dropna(how="all")

83	def rolling_vol(returns: pd.DataFrame, window: int = 20) -> pd.DataFrame:
84	    """Rolling volatility over the last `window` days, today included."""
85	    return returns.rolling(window).apply(lambda x: np.std(x), raw=True)
86	
87	
88	def annualize(vol: pd.DataFrame) -> pd.DataFrame:
89	    """Annualize to a calendar year, like my brokerage statements: 365 days."""
90	    return vol * np.sqrt(365)
91

93	def summary(returns: pd.DataFrame) -> pd.DataFrame:
94	    """Yearly return and volatility per ticker."""
95	    return pd.DataFrame({"ann_return": returns.mean() * 365, "ann_vol": returns.std(ddof=0) * np.sqrt(365)})
96	
97	
98	def max_drawdown(prices: pd.DataFrame) -> pd.Series:
99	    """Deepest fall below the running high."""
100	    return (prices / prices.cummax() - 1).min()

from __future__ import annotations

from typing import Any

import pytest

from risk import Returns, TickerSummary, VolSeries, summary_stats


def _returns(d: dict[str, Any]) -> Returns:
    return Returns(
        base_date=d["base_date"],
        dates=tuple(d["dates"]),
        tickers=tuple(d["tickers"]),
        values={t: tuple(v) for t, v in d["values"].items()},
    )


def _vol(d: dict[str, Any]) -> VolSeries:
    return VolSeries(
        returns=_returns(d["returns"]),
        window=d["window"],
        annualized=d["annualized"],
        values={t: tuple(v) for t, v in d["values"].items()},
    )


def _tc07(golden) -> tuple[VolSeries, list[dict[str, Any]], float]:
    data, expected = golden("risk", "TC-07")
    return _vol(data["vol"]), expected["summary"], expected["tol"]


@pytest.mark.intent
def test_tc07_one_summary_per_ticker_in_ticker_order(golden):
    vol, expected, _ = _tc07(golden)
    result = summary_stats(vol)
    assert isinstance(result, tuple)
    assert all(isinstance(s, TickerSummary) for s in result)
    assert [s.ticker for s in result] == [e["ticker"] for e in expected]


@pytest.mark.intent
@pytest.mark.parametrize("field", ["ann_return", "ann_vol", "sharpe", "last_vol"])
def test_tc07_summary_values_within_tolerance(golden, field):
    vol, expected, tol = _tc07(golden)
    result = {s.ticker: s for s in summary_stats(vol)}
    for row in expected:
        got = getattr(result[row["ticker"]], field)
        assert got == pytest.approx(row[field], abs=tol), (row["ticker"], field)


@pytest.mark.intent
def test_tc07_zero_volatility_gives_zero_sharpe(golden):

    assert result.ann_vol == pytest.approx(0.0, abs=tol)
    assert result.sharpe == pytest.approx(flat["sharpe"], abs=tol)
