+"""PL-02 step 4: annualized volatility, scaled by the square root of trading days per year."""
+
+from __future__ import annotations
+
+import dataclasses
+import math
+
+from risk.types import VolSeries
+
+
+def annualize(vol: VolSeries, periods_per_year: int = 252) -> VolSeries:
+    """Each daily value times sqrt(periods_per_year), gaps kept; ValueError if already annualized."""
+    if vol.annualized:
+        raise ValueError("volatility is already annualized")
+    scale = math.sqrt(periods_per_year)
+    values = {
+        ticker: tuple(None if v is None else v * scale for v in series)
+        for ticker, series in vol.values.items()
+    }
+    return dataclasses.replace(vol, annualized=True, values=values)

+"""PL-03 step 3: maximum drawdown with the dates of its peak and trough."""
+
+from __future__ import annotations
+
+import math
+
+from risk.types import Drawdown, Returns
+
+
+def max_drawdown(returns: Returns) -> tuple[Drawdown, ...]:
+    """Each ticker's deepest fall of wealth below its running peak, wealth 1 at `base_date`.
+
+    Depth is trough / peak - 1 (at most 0); ties keep the first peak and trough. With no fall,
+    depth is 0 and both dates are `base_date`.
+    """
+    return tuple(
+        _drawdown(ticker, returns.base_date, returns.dates, returns.values[ticker])
+        for ticker in returns.tickers
+    )
+
+
+def _drawdown(
+    ticker: str, base_date: str, dates: tuple[str, ...], series: tuple[float, ...]
+) -> Drawdown:
+    log_wealth = log_peak = 0.0
+    running_peak_date = peak_date = trough_date = base_date
+    depth = 0.0
+    for date, value in zip(dates, series, strict=True):
+        log_wealth += value
+        if log_wealth > log_peak:
+            log_peak, running_peak_date = log_wealth, date
+            continue
+        fall = math.expm1(log_wealth - log_peak)
+        if fall < depth:
+            depth, peak_date, trough_date = fall, running_peak_date, date
+    return Drawdown(ticker=ticker, depth=depth, peak_date=peak_date, trough_date=trough_date)

+from __future__ import annotations

+import statistics
+
+from risk.types import TickerSummary, VolSeries

+def summary_stats(vol: VolSeries, periods_per_year: int = 252) -> tuple[TickerSummary, ...]:
+    """Per ticker: periods x mean, sqrt(periods) x sample stdev, their ratio, the last rolling vol.
+
+    The Sharpe ratio is 0 when the volatility is 0 (no risk-free rate). ValueError under 2 returns.
+    """
+    if len(vol.returns.dates) < 2:
+        raise ValueError(f"need at least two returns, got {len(vol.returns.dates)}")
+    return tuple(_summary(ticker, vol, periods_per_year) for ticker in vol.returns.tickers)
+
+
+def _summary(ticker: str, vol: VolSeries, periods_per_year: int) -> TickerSummary:
+    series = vol.returns.values[ticker]
+    ann_return = periods_per_year * statistics.fmean(series)
+    ann_vol = math.sqrt(periods_per_year) * statistics.stdev(series)
+    return TickerSummary(
+        ticker=ticker,
+        ann_return=ann_return,
+        ann_vol=ann_vol,
+        sharpe=ann_return / ann_vol if ann_vol > 0 else 0.0,
+        last_vol=_last_value(vol.values[ticker]),
+    )
+
+
+def _last_value(series: tuple[float | None, ...]) -> float | None:
+    """The latest rolling value that exists, or None if there is none."""
+    return next((v for v in reversed(series) if v is not None), None)

+"""PL-02 step 3: rolling volatility from the returns before each date."""
+
+from __future__ import annotations
+
+import statistics
+
+from risk.types import Returns, VolSeries
+
+
+def rolling_vol(returns: Returns, window: int) -> VolSeries:
+    """Each date's sample stdev of the `window` returns before it: None until they exist."""
+    if window < 2:
+        raise ValueError(f"window must be at least 2, got {window}")
+    values = {
+        ticker: tuple(
+            statistics.stdev(series[i - window : i]) if i >= window else None
+            for i in range(len(series))
+        )
+        for ticker, series in returns.values.items()
+    }
+    return VolSeries(returns=returns, window=window, annualized=False, values=values)
