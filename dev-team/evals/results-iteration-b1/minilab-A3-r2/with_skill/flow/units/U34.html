"""Unit tests for risk.report_pipeline.risk_report on real CSVs (no fakes)."""

from __future__ import annotations

import json
from pathlib import Path

import pytest

from mdata import GapError
from risk.report_pipeline import risk_report

DATES = [f"2024-02-{day:02d}" for day in range(1, 10)]


def _csv(tmp_path: Path, rows: list[tuple[str, str, str]]) -> Path:
    path = tmp_path / "prices.csv"
    lines = ["date,ticker,close", *(",".join(row) for row in rows)]
    path.write_text("\n".join(lines) + "\n", encoding="utf-8")
    return path


def _rising(ticker: str, start: float) -> list[tuple[str, str, str]]:
    return [(day, ticker, str(start + i)) for i, day in enumerate(DATES)]


def test_default_window_leaves_last_vol_empty_on_a_short_history(tmp_path: Path) -> None:
    path = _csv(tmp_path, _rising("AAA", 10.0) + _rising("BBB", 20.0))

    report = risk_report(path, tmp_path / "report.json")

    assert [row["ticker"] for row in report.rows] == ["AAA", "BBB"]
    assert all(row["last_vol"] is None for row in report.rows)
    assert all(row["max_drawdown"] == 0.0 for row in report.rows)
    assert all(row["peak_date"] == DATES[0] for row in report.rows)


def test_requested_tickers_narrow_the_report_and_the_file(tmp_path: Path) -> None:
    path = _csv(tmp_path, _rising("AAA", 10.0) + _rising("BBB", 20.0))
    out = tmp_path / "report.json"

    report = risk_report(path, out, window=3, tickers=["BBB"])

    assert [row["ticker"] for row in report.rows] == ["BBB"]
    assert report.rows[0]["last_vol"] is not None
    assert json.loads(out.read_text(encoding="utf-8")) == {"tickers": list(report.rows)}


def test_a_short_window_raises_before_writing(tmp_path: Path) -> None:
    path = _csv(tmp_path, _rising("AAA", 10.0))
    out = tmp_path / "report.json"

    with pytest.raises(ValueError, match="window"):
        risk_report(path, out, window=1)
    assert not out.exists()


def test_an_unknown_ticker_raises_key_error(tmp_path: Path) -> None:
    path = _csv(tmp_path, _rising("AAA", 10.0))

    with pytest.raises(KeyError, match="ZZZ"):
        risk_report(path, tmp_path / "report.json", tickers=["ZZZ"])


def test_a_long_gap_raises_gap_error_naming_the_ticker(tmp_path: Path) -> None:
    gappy = [row for row in _rising("BBB", 20.0) if row[0] not in DATES[1:7]]
    path = _csv(tmp_path, _rising("AAA", 10.0) + gappy)
    out = tmp_path / "report.json"

    with pytest.raises(GapError, match="BBB"):
        risk_report(path, out, window=2)
    assert not out.exists()
