_ONE_RECORDED_GAME = EvalOptions(n_games=1, record=1)


def pick_game(
    record: RunRecord, game: int, new_seed: int | None, pair: tuple[int, int] | None
) -> GameRecording:
    """Recorded game `game`, or one new game on `eval_seeds(new_seed, 1)`, or a generation pair."""
    if pair is not None:
        raise ValueError(f"run {record.path}: replaying a generation pair needs self-play")
    if new_seed is not None:
        return _play_new_game(record, new_seed)
    if not 0 <= game < len(record.games):
        raise IndexError(f"run {record.path}: no game {game}, it recorded {len(record.games)}")
    return record.games[game]


def _play_new_game(record: RunRecord, seed: int) -> GameRecording:
    """One greedy game of the run's final policy on the held-out seed for `seed` (REPLAY-03)."""
    agents, settings = load_player(
        str(record.path), seed, env=record.settings.env, team=SOLO, vs="scripted"
    )
    scores = evaluate(
        make_env(settings),
        agents,
        player=settings.name,
        train_seed=seed,
        options=_ONE_RECORDED_GAME,
    )
    return scores.games[0]
