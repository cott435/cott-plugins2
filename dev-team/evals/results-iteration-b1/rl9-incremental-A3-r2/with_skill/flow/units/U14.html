79	**Status:** Accepted · **Date:** 2026-10-17 · **Scope:** `runs`

310	class MemoryRunWriter:
311	    """Keeps every call in memory; agents save their checkpoints and generations under `path`."""
312	
313	    def __init__(self, path: Path, settings: RunSettings) -> None:
314	        self.path = path
315	        self.settings = settings
316	        self.points: list[CurvePoint] = []
317	        self.checkpoints: dict[str, Path] = {}
318	        self.generations: dict[str, list[Path]] = {}
319	        self.games: tuple[GameRecording, ...] = ()
320	        self.status = "running"
321	
322	    def add_point(self, p: CurvePoint) -> None:
323	        self.points.append(p)
324	
325	    def save_checkpoint(self, agent: Agent, *, team: str = SOLO) -> None:
326	        path = self.path / f"checkpoint-{team}"
327	        agent.save(path)
328	        self.checkpoints[team] = path
329	
330	    def save_generation(self, agent: Agent, *, team: str, index: int) -> Path:
331	        path = self.path / "generations" / team / str(index)
332	        path.parent.mkdir(parents=True, exist_ok=True)
333	        agent.save(path)
334	        self.generations.setdefault(team, []).append(path)
335	        return path
336	
337	    def save_games(self, games: Sequence[GameRecording]) -> None:
338	        self.games = tuple(games)
339	
340	    def finish(self, status: Literal["complete", "incomplete"]) -> RunRecord:
341	        self.status = status
342	        return RunRecord(
343	            path=self.path,
344	            settings=self.settings,
345	            curve=tuple(self.points),
346	            status=status,
347	            checkpoints=dict(self.checkpoints),
348	            generations={team: tuple(paths) for team, paths in self.generations.items()},
349	            games=self.games,
350	        )

_CURVE_FIELDS = ("step", "games", "mean_score", "elapsed_s")
_STATUSES: dict[str, Literal["running", "complete", "incomplete"]] = {
    "running": "running",
    "complete": "complete",
    "incomplete": "incomplete",
}

class RunWriter:
    """Writes one run folder; each curve point is flushed at once."""

    def __init__(self, path: Path, settings: RunSettings) -> None:
        self.path = path
        self.settings = settings

    def add_point(self, p: CurvePoint) -> None:
        with (self.path / "curve.csv").open("a", newline="", encoding="utf-8") as f:
            csv.writer(f).writerow([p.step, p.games, repr(p.mean_score), repr(p.elapsed_s)])

    def save_checkpoint(self, agent: Agent, *, team: str = SOLO) -> None:
        stem = "checkpoint" if team == SOLO else f"checkpoint-{team}"
        agent.save(self.path / (stem + _CHECKPOINT_SUFFIX.get(self.settings.agent, ".pt")))

    def save_generation(self, agent: Agent, *, team: str, index: int) -> Path:
        path = self.path / "generations" / team / str(index)
        path.parent.mkdir(parents=True, exist_ok=True)
        agent.save(path)
        return path

    def save_games(self, games: Sequence[GameRecording]) -> None:
        arrays: dict[str, np.ndarray] = {}
        for i, game in enumerate(games):
            arrays[f"{i}.seed"] = np.asarray(game.seed, dtype=np.int64)
            arrays[f"{i}.board"] = np.asarray(game.board)
            arrays[f"{i}.frames"] = game.frames
            arrays[f"{i}.rewards"] = game.rewards
            for team, scores in game.scores.items():
                arrays[f"{i}.scores.{team}"] = scores
        np.savez(self.path / "games.npz", **arrays)

    def finish(self, status: Literal["complete", "incomplete"]) -> RunRecord:
        (self.path / "status").write_text(status, encoding="utf-8")
        return open_run(self.path)


def create_run(settings: RunSettings, *, root: Path = Path("runs")) -> RunWriter:
    """Start a uniquely named run folder holding `settings.json` and `status=running`."""
    stamp = datetime.now().strftime("%Y%m%dT%H%M%S")
    path = _new_folder(root, f"{settings.name}-s{settings.seed}-{stamp}")
    (path / "settings.json").write_text(json.dumps(asdict(settings), indent=2), encoding="utf-8")
    (path / "curve.csv").write_text(",".join(_CURVE_FIELDS) + "\n", encoding="utf-8")
    (path / "status").write_text("running", encoding="utf-8")
    return RunWriter(path, settings)


def open_run(path: Path) -> RunRecord:
    """Read a run folder back (PL-05 step 1)."""
    settings_file = path / "settings.json"
    if not settings_file.is_file():
        raise FileNotFoundError(f"run {path}: no settings.json")
    return RunRecord(
        path=path,
        settings=_settings_from_json(json.loads(settings_file.read_text(encoding="utf-8"))),
        curve=_read_curve(path / "curve.csv"),
        status=_read_status(path / "status"),

        generations=_find_generations(path / "generations"),

    )


def _new_folder(root: Path, stem: str) -> Path:
    """`root/stem`, or `root/stem-<n>` with the first free `n` when it exists."""
    root.mkdir(parents=True, exist_ok=True)
    n = 0
    while True:
        path = root / (stem if n == 0 else f"{stem}-{n}")
        try:
            path.mkdir()
        except FileExistsError:
            n += 1
            continue
        return path


def _settings_from_json(raw: dict[str, Any]) -> RunSettings:
    return RunSettings(
        **{
            **raw,
            "rewards": Rewards(**raw["rewards"]),
            "exploration": Exploration(**raw["exploration"]),
            "params": AgentParams(**raw["params"]),
        }

def _read_curve(path: Path) -> tuple[CurvePoint, ...]:
    if not path.is_file():
        return ()
    with path.open(newline="", encoding="utf-8") as f:
        return tuple(

            for row in csv.DictReader(f)

def _read_status(path: Path) -> Literal["running", "complete", "incomplete"]:
    text = path.read_text(encoding="utf-8").strip()
    if text not in _STATUSES:
        raise ValueError(f"{path}: unknown status {text!r}")
    return _STATUSES[text]


def _find_checkpoints(run: Path) -> dict[str, Path]:
    """Each team's final checkpoint: `checkpoint.<ext>` is `SOLO`'s, `checkpoint-<team>.<ext>`."""
    found = {}
    for path in sorted(run.glob("checkpoint*")):
        team = path.stem.removeprefix("checkpoint").removeprefix("-")
        found[team or SOLO] = path
    return found


def _find_generations(folder: Path) -> dict[str, tuple[Path, ...]]:

        return {}
    return {
        team.name: tuple(sorted(team.iterdir(), key=lambda p: int(p.name)))
        for team in sorted(folder.iterdir())
    }
