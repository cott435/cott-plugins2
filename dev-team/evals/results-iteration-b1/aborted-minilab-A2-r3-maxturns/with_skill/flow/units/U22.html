    return Returns(

def _brute_force_depth(values: tuple[float, ...]) -> float:
    """Independent oracle: wealth 1 at the base date, worst trough/peak - 1 over all pairs."""
    wealth = [1.0]
    for r in values:
        wealth.append(wealth[-1] * math.exp(r))
    worst = 0.0
    for i, peak in enumerate(wealth):
        for trough in wealth[i:]:
            worst = min(worst, trough / peak - 1.0)
    return worst
