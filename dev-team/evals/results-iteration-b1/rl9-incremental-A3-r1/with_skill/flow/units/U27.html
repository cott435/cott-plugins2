import json
from pathlib import Path

import numpy as np

from rllab.core import Agent, Transition


class RandomPlayer(Agent):
    """Uniform over `n_actions` whatever `explore` says; never learns."""

    def __init__(self, n_actions: int, *, seed: int) -> None:
        self._n_actions = n_actions
        self._rng = np.random.default_rng(seed)

    def act(self, obs: np.ndarray, *, explore: bool) -> np.ndarray:
        """One uniform action per row of `obs`; `explore` is ignored, the player is always uniform."""
        return self._rng.integers(self._n_actions, size=len(obs), dtype=np.int64)

    def learn(self, t: Transition) -> None:
        """Discard `t`: the random player never learns."""
        del t

    def save(self, path: Path) -> None:
        """Write the generator's state, so a loaded player continues the same action sequence."""
        path.write_text(json.dumps(self._rng.bit_generator.state))

    def load(self, path: Path) -> None:
        self._rng.bit_generator.state = json.loads(path.read_text())

"""RandomPlayer: a uniform action per agent from a seeded generator (BASE-01)."""

from pathlib import Path

import numpy as np

from rllab.baselines import RandomPlayer
from rllab.core import Transition

OBS = np.zeros((1, 11), dtype=np.float32)


def test_random_player_replays_the_seeded_ten_sample() -> None:
    player = RandomPlayer(3, seed=0)
    actions = [int(player.act(OBS, explore=i % 2 == 0)[0]) for i in range(10)]
    assert actions == [2, 1, 1, 0, 0, 0, 0, 0, 0, 2]


def test_random_player_one_action_per_agent_in_range() -> None:
    player = RandomPlayer(4, seed=3)
    out = player.act(np.zeros((5, 11), dtype=np.float32), explore=False)
    assert out.shape == (5,)
    assert out.dtype == np.int64
    assert set(out.tolist()) <= {0, 1, 2, 3}


def test_random_player_different_seeds_differ() -> None:
    a = RandomPlayer(3, seed=1)
    b = RandomPlayer(3, seed=2)
    first = [int(a.act(OBS, explore=True)[0]) for _ in range(30)]
    second = [int(b.act(OBS, explore=True)[0]) for _ in range(30)]
    assert first != second


def test_random_player_learn_changes_nothing() -> None:
    learner = RandomPlayer(3, seed=5)
    untouched = RandomPlayer(3, seed=5)
    transition = Transition(
        obs=OBS,
        actions=np.array([0], dtype=np.int64),
        reward=np.array([1.0], dtype=np.float32),
        next_obs=OBS,
        terminated=np.array([False]),
        active=np.array([True]),
        truncated=False,
    )
    for _ in range(10):
        learner.learn(transition)
    got = [int(learner.act(OBS, explore=False)[0]) for _ in range(20)]
    want = [int(untouched.act(OBS, explore=False)[0]) for _ in range(20)]
    assert got == want
