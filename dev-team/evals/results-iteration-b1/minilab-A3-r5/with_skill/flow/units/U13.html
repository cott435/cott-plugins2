"""Put every ticker on one calendar (DATA-03, DATA-04)."""

from collections.abc import Sequence

from mdata.types import Prices, RawRow


def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
    """One calendar from the first date every ticker has a close; gaps take the previous close."""
    by_ticker: dict[str, dict[str, float]] = {}
    for row in rows:
        closes = by_ticker.setdefault(row.ticker, {})
        if row.close is not None:
            closes[row.date] = row.close
    tickers = tuple(sorted(by_ticker))
    shared = set.intersection(*(set(closes) for closes in by_ticker.values()))
    dates = tuple(sorted(shared))
    return Prices(
        dates=dates,
        tickers=tickers,
        closes={ticker: tuple(by_ticker[ticker][d] for d in dates) for ticker in tickers},
    )

"""Unit tests for mdata.calendar.align_calendar (spine: dates every ticker has, no fill)."""

from mdata.calendar import align_calendar
from mdata.types import RawRow


def test_sample_keeps_only_dates_every_ticker_has(contract_sample):
    rows = [RawRow(**row) for row in contract_sample("mdata.RawRow")["rows"]]

    prices = align_calendar(rows)

    assert prices.dates == ("2024-03-04", "2024-03-05", "2024-03-07")
    assert prices.tickers == ("ACME", "BOLT")
    assert prices.closes == {
        "ACME": (50.0, 51.2, 52.3),
        "BOLT": (120.0, 118.5, 121.4),
    }


def test_dates_ascending_and_tickers_sorted_from_unsorted_rows():
    rows = [
        RawRow("2024-01-03", "ZED", 3.0),
        RawRow("2024-01-02", "ABC", 2.0),
        RawRow("2024-01-03", "ABC", 4.0),
        RawRow("2024-01-02", "ZED", 1.0),
    ]

    prices = align_calendar(rows)

    assert prices.dates == ("2024-01-02", "2024-01-03")
    assert prices.tickers == ("ABC", "ZED")
    assert prices.closes == {"ABC": (2.0, 4.0), "ZED": (1.0, 3.0)}


def test_date_one_ticker_lacks_is_dropped():
    rows = [
        RawRow("2024-01-02", "ABC", 2.0),
        RawRow("2024-01-02", "ZED", 1.0),
        RawRow("2024-01-03", "ABC", 4.0),
    ]

    prices = align_calendar(rows)

    assert prices.dates == ("2024-01-02",)
    assert prices.closes == {"ABC": (2.0,), "ZED": (1.0,)}
