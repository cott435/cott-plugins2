class Replay:
    """Draws a recorded game frame by frame; space pauses, right arrow steps, +/- change fps."""

    def __init__(self, game: GameRecording, *, fps: float = 8.0) -> None:
        self._game = game
        self._fps = fps
        self._index = 0
        self._paused = False

    @property
    def fps(self) -> float:
        """Frames per second while playing."""
        return self._fps

    @property
    def index(self) -> int:
        """The frame drawn now."""
        return self._index

    @property
    def paused(self) -> bool:
        """Whether the timer stopped advancing."""
        return self._paused

    def current(self) -> tuple[np.ndarray, dict[str, float], int]:
        """The frame drawn now, each team's score at it, and its step."""
        i = self._index
        scores = {team: float(values[i]) for team, values in self._game.scores.items()}
        return self._game.frames[i], scores, i

    def advance(self) -> int:
        """Move one frame on and return its index, stopping at the last frame."""
        self._index = min(self._index + 1, self._game.frames.shape[0] - 1)
        return self._index

    def on_key(self, key: str) -> None:
        """Space toggles pause, right arrow steps, `+`/`-` double or halve fps; others do nothing."""
        if key == " ":
            self._paused = not self._paused
        elif key == "right":
            self.advance()
        elif key == "+":
            self._fps *= 2
        elif key == "-":
            self._fps /= 2

    def show(self) -> None:
        """Open the window and play; blocks until it closes."""
        fig, ax = plt.subplots()
        timer = fig.canvas.new_timer(interval=self._interval_ms())

        def redraw() -> None:
            frame, scores, step = self.current()
            draw_frame(ax, self._game.board, frame, title(scores, step))
            fig.canvas.draw_idle()

        def key_pressed(event: KeyEvent) -> None:
            if event.key is not None:
                self.on_key(event.key)
                redraw()

        def tick(timer: TimerBase) -> None:
            timer.interval = self._interval_ms()
            if not self._paused:
                self.advance()
                redraw()

        fig.canvas.mpl_connect("key_press_event", key_pressed)
        timer.add_callback(tick, timer)
        redraw()
        timer.start()
        plt.show()
        timer.stop()
        plt.close(fig)

    def _interval_ms(self) -> int:
        return max(1, round(1000 / self._fps))


def title(scores: dict[str, float], step: int) -> str:
