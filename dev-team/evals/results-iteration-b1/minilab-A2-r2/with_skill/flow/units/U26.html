+"""PL-01 step 3: align the rows on one calendar."""
+
+from __future__ import annotations
+
+from collections.abc import Mapping, Sequence
+
+from mdata.types import GapError, Prices, RawRow
+
+
+def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
+    """Prices on every date any ticker trades, from the first date every ticker has a close.
+
+    A missing date takes that ticker's previous close; more than `max_gap` missing dates
+    in a row raise `GapError` naming the ticker.
+    """
+    by_ticker: dict[str, dict[str, float]] = {}
+    for row in rows:
+        if row.close is not None:
+            by_ticker.setdefault(row.ticker, {})[row.date] = row.close
+    tickers = tuple(sorted(by_ticker))
+    start = max((min(closes) for closes in by_ticker.values()), default="")
+    dates = tuple(
+        sorted({date for closes in by_ticker.values() for date in closes if date >= start})
+    )
+    return Prices(
+        dates=dates,
+        tickers=tickers,
+        closes={ticker: _fill(ticker, by_ticker[ticker], dates, max_gap) for ticker in tickers},
+    )
+
+
+def _fill(
+    ticker: str, closes: Mapping[str, float], dates: Sequence[str], max_gap: int
+) -> tuple[float, ...]:
+    """One close per date, each missing date taking the previous close (even one before `dates`)."""
+    previous = closes[max(date for date in closes if date <= dates[0])]
+    filled: list[float] = []
+    missing = 0
+    for date in dates:
+        if date in closes:
+            missing = 0
+            previous = closes[date]
+        else:
+            missing += 1
+            if missing > max_gap:
+                raise GapError(f"{ticker} misses more than {max_gap} dates in a row up to {date}")
+        filled.append(previous)
+    return tuple(filled)

+    """Rows with a positive close (of the requested tickers, if any), the later of duplicates

+        if missing:

+    return [latest[key] for key in sorted(latest)]
