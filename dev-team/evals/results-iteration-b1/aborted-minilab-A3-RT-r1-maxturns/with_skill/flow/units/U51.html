"""End to end: risk_report runs PL-03 -> PL-02 -> PL-01 on a real CSV, no fakes."""

import json
from math import log

import pytest

from mdata import load_prices
from risk import risk_report, vol_series

CONTRACT_CSV = "tests/contracts/mdata.load_prices/prices.csv"

# Unsorted, with a blank, a zero and a negative close, a duplicate (the later 110 wins),
# B starting a day late and missing 2024-01-05 (filled from 2024-01-04).
MESSY_CSV = """date,ticker,close
2024-01-04,A,99
2024-01-02,A,100
2024-01-03,A,90
2024-01-03,A,110
2024-01-03,B,20
2024-01-04,B,
2024-01-04,B,22
2024-01-05,A,88
2024-01-05,B,0
2024-01-08,A,95
2024-01-08,B,-1
2024-01-08,B,24
2024-01-09,A,120
2024-01-09,B,25
"""


@pytest.fixture
def messy_csv(tmp_path):
    path = tmp_path / "prices.csv"
    path.write_text(MESSY_CSV)
    return path


def test_report_on_a_messy_csv(messy_csv, tmp_path):
    out = tmp_path / "report.json"
    report = risk_report(messy_csv, out, window=2)
    a, b = report.rows
    assert json.loads(out.read_text()) == {"tickers": [dict(row) for row in report.rows]}
    # The calendar starts on B's first date, 2024-01-03.
    assert a["ticker"] == "A"
    assert a["ann_return"] == pytest.approx(252 * log(120 / 110) / 4, abs=1e-12)
    assert a["max_drawdown"] == pytest.approx(88 / 110 - 1, abs=1e-12)
    assert (a["peak_date"], a["trough_date"]) == ("2024-01-03", "2024-01-05")
    assert b["ticker"] == "B"
    assert b["ann_return"] == pytest.approx(252 * log(25 / 20) / 4, abs=1e-12)
    assert (b["max_drawdown"], b["peak_date"], b["trough_date"]) == (0.0, "2024-01-03", "2024-01-03")


def test_same_csv_and_window_write_identical_files(messy_csv, tmp_path):
    risk_report(messy_csv, tmp_path / "a.json", window=2)
    risk_report(messy_csv, tmp_path / "b.json", window=2)
    assert (tmp_path / "a.json").read_bytes() == (tmp_path / "b.json").read_bytes()


def test_real_load_prices_agrees_with_its_fake(contract_fake):
    fake = contract_fake("mdata.load_prices")
    assert load_prices(CONTRACT_CSV) == fake.load_prices(CONTRACT_CSV)


def test_real_vol_series_agrees_with_its_fake(contract_fake):
    fake = contract_fake("risk.vol_series").vol_series(CONTRACT_CSV, window=2)
    real = vol_series(CONTRACT_CSV, window=2)
    assert real.returns.dates == fake.returns.dates
    assert real.returns.tickers == fake.returns.tickers
    assert (real.window, real.annualized) == (fake.window, fake.annualized)
    for ticker in real.returns.tickers:
        assert real.returns.values[ticker] == pytest.approx(fake.returns.values[ticker], abs=1e-6)
        for got, want in zip(real.values[ticker], fake.values[ticker], strict=True):
            assert (got is None) == (want is None)
            if got is not None:
                assert got == pytest.approx(want, abs=1e-6)
