+def vol_series(
+    path: str | Path, *, window: int = 20, tickers: Sequence[str] | None = None
+) -> VolSeries:
+    """The sample series, narrowed to `tickers`.
+
+    ValueError if `window` < 2; KeyError for a ticker the sample lacks. Any other window serves
+    the sample's own window of 2.

+        tickers=tuple(chosen),
+        values={ticker: tuple(raw["values"][ticker]) for ticker in chosen},
+    )
+    return VolSeries(
+        returns=returns,
+        window=data["window"],
+        annualized=data["annualized"],
+        values={ticker: tuple(data["values"][ticker]) for ticker in chosen},
+    )
