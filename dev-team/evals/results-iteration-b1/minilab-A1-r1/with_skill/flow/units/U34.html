def test_real_load_prices_agrees_with_its_fake(contract_fake) -> None:

def test_real_vol_series_agrees_with_its_fake(contract_fake) -> None:

def test_unusable_rows_and_gaps_are_handled_before_the_report(tmp_path: Path) -> None:
    csv = _write(

def test_a_long_gap_stops_the_report(tmp_path: Path) -> None:
    rows = ["date,ticker,close"]
    for day in range(1, 10):
        rows.append(f"2025-02-{day:02d},A,{100 + day}")
        if day in (1, 9):
            rows.append(f"2025-02-{day:02d},B,50")
    out = tmp_path / "out.json"
    with pytest.raises(GapError, match="B"):
        risk_report(_write(tmp_path, "\n".join(rows) + "\n"), out)
