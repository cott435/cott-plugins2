75	import pandas as pd

77	
78	def daily_returns(prices: pd.DataFrame) -> pd.DataFrame:
79	    """Log returns of a dates x tickers close panel."""
80	    return np.log(prices / prices.shift(1)).dropna(how="all")
81	
82	
83	def rolling_vol(returns: pd.DataFrame, window: int = 20) -> pd.DataFrame:
84	    """Rolling volatility over the last `window` days, today included."""
85	    return returns.rolling(window).apply(lambda x: np.std(x), raw=True)
86	
87	
88	def annualize(vol: pd.DataFrame) -> pd.DataFrame:
89	    """Annualize to a calendar year, like my brokerage statements: 365 days."""
90	    return vol * np.sqrt(365)
91	
92	
93	def summary(returns: pd.DataFrame) -> pd.DataFrame:
94	    """Yearly return and volatility per ticker."""
95	    return pd.DataFrame({"ann_return": returns.mean() * 365, "ann_vol": returns.std(ddof=0) * np.sqrt(365)})
96	
97	
98	def max_drawdown(prices: pd.DataFrame) -> pd.Series:
99	    """Deepest fall below the running high."""
100	    return (prices / prices.cummax() - 1).min()

def returns_of(payload: dict) -> Returns:
    return Returns(

        dates=tuple(payload["dates"]),
        tickers=tuple(payload["tickers"]),
        values={t: tuple(v) for t, v in payload["values"].items()},

def by_ticker(drawdowns: tuple[Drawdown, ...]) -> dict[str, Drawdown]:
    return {d.ticker: d for d in drawdowns}
