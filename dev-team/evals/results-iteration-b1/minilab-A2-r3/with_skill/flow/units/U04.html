    """Every row in file order; a blank close is None. ValueError for a missing column or bad date."""

from mdata.types import Prices, RawRow


def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
    """Spine: only the dates every ticker has a close on; nothing is filled, so no gap applies."""
    del max_gap
    closes: dict[str, dict[str, float]] = {}
    for row in rows:
        if row.close is not None:
            closes.setdefault(row.ticker, {})[row.date] = row.close
    tickers = tuple(sorted(closes))
    shared = set.intersection(*(set(by_date) for by_date in closes.values())) if closes else set()
    dates = tuple(sorted(shared))
    return Prices(
        dates=dates,
        tickers=tickers,
        closes={ticker: tuple(closes[ticker][day] for day in dates) for ticker in tickers},
    )
