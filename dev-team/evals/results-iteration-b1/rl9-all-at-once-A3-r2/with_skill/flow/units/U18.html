    record_file = path / "run.json"

def _status(status: str, pid: int | None) -> RunStatus:
    """A `running` run whose writing process (`pid` in `run.json`) is gone reads as incomplete."""
    if status == "running" and not (pid is not None and _alive(pid)):

    try:
