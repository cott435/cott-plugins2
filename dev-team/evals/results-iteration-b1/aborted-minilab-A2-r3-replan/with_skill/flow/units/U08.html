from __future__ import annotations

import pytest

from mdata.types import Prices
from risk.returns import log_returns
from risk.types import Returns


def _prices(payload: dict) -> Prices:
    return Prices(
        dates=tuple(payload["dates"]),
        tickers=tuple(payload["tickers"]),
        closes={ticker: tuple(series) for ticker, series in payload["closes"].items()},
    )


@pytest.mark.intent
def test_tc04_geometric_closes_give_closed_form_log_returns(golden):
    inp, expected = golden("risk", "TC-04")
    returns = log_returns(_prices(inp["prices"]))
    tol = expected["tol"]

    assert isinstance(returns, Returns)
    assert returns.base_date == expected["base_date"]
    assert returns.dates == tuple(expected["dates"])
    assert returns.tickers == tuple(expected["tickers"])
    assert set(returns.values) == set(expected["values"])
    for ticker, series in expected["values"].items():
        assert len(returns.values[ticker]) == len(series)
        assert tuple(returns.values[ticker]) == pytest.approx(tuple(series), abs=tol, rel=0)


@pytest.mark.intent
def test_returns_are_dated_after_the_base_date(contract_sample):
    prices = _prices(contract_sample("mdata.Prices"))
    returns = log_returns(prices)

    assert returns.base_date == prices.dates[0]
    assert returns.dates == prices.dates[1:]
    assert returns.tickers == prices.tickers
    for ticker in prices.tickers:
        assert len(returns.values[ticker]) == len(prices.dates) - 1
