def load_settings(path: Path) -> RunSettings:
    """Read a TOML settings file into fully resolved `RunSettings`."""
    table = _with_defaults(tomllib.loads(path.read_text(encoding="utf-8")))
    _check_names(RunSettings, table, "")
    for section, cls in _SECTIONS.items():
        _check_names(cls, table[section], f"{section}.")
    _check_opponents(table["opponents"])
    settings = RunSettings(
        **{
            **table,
            "opponents": dict(table["opponents"]),
            "rewards": Rewards(**table["rewards"]),
            "exploration": Exploration(**table["exploration"]),
            "params": AgentParams(**table["params"]),
        }
    )
    _check_values(settings)
    return settings


def with_seed(settings: RunSettings, seed: int) -> RunSettings:

def _check_names(cls: type, table: dict[str, Any], prefix: str) -> None:
    """Raise `ValueError` naming the first unknown, missing or unknown-valued key of `table`."""
    known = {f.name: f for f in fields(cls)}
    for key in table:
        if key not in known:
            raise ValueError(f"{prefix}{key}: unknown setting")
    for name, field in known.items():
        if name not in table:
            if field.default is MISSING:
                raise ValueError(f"{prefix}{name}: missing setting")
            continue
        if get_origin(field.type) is Literal and table[name] not in get_args(field.type):
            raise ValueError(f"{prefix}{name}: unknown value {table[name]!r}")


def _check_opponents(opponents: dict[str, Any]) -> None:
    for team, kind in opponents.items():
        if kind not in _OPPONENT_KINDS:
            raise ValueError(f"opponents.{team}: unknown value {kind!r}")

import tomllib
from pathlib import Path
from typing import Any

from rllab.compare import ClaimSpec
from rllab.lab.types import PlayerTarget, ReportTargets

_DIRECTIONS = ("greater", "not_worse")


def load_report_spec(path: Path) -> tuple[tuple[ClaimSpec, ...], ReportTargets]:
    """The claims and the targets a report TOML names."""
    raw = tomllib.loads(path.read_text(encoding="utf-8"))
    players = tuple(_player(name, table) for name, table in raw.get("players", {}).items())
    named = {p.name for p in players}
    claims = tuple(_claim(table, named) for table in raw.get("claims", []))
    ladder = tuple(raw.get("ladder", []))
    ablations = dict(raw.get("ablations", {}))
    _check_listed("ladder", ladder, named)
    _check_listed("ablations", (*ablations, *ablations.values()), named)
    baselines = tuple(name for name, table in raw.get("players", {}).items() if "baseline" in table)
    targets = ReportTargets(
        players=players, baselines=baselines, ladder=ladder, ablations=ablations
    )
    return claims, targets


def _player(name: str, table: dict[str, Any]) -> PlayerTarget:
    """One player from its table: `runs` (one per seed) or a `baseline` name, and `seeds`."""
    seeds = tuple(table["seeds"])
    if ("runs" in table) == ("baseline" in table):
        raise ValueError(f"players.{name}: give exactly one of `runs` or `baseline`")
    if "baseline" in table:
        return PlayerTarget(name=name, targets=(table["baseline"],) * len(seeds), seeds=seeds)
    runs = tuple(table["runs"])
    if len(runs) != len(seeds):
        raise ValueError(f"players.{name}: {len(runs)} runs for {len(seeds)} seeds")
    return PlayerTarget(name=name, targets=runs, seeds=seeds)


def _claim(table: dict[str, Any], named: set[str]) -> ClaimSpec:
    """One claim; `ValueError` if it names an unlisted player or an unknown direction."""
    _check_listed(f"claims.{table['name']}", (table["a"], table["b"]), named)
    direction = table["direction"]
    if direction not in _DIRECTIONS:
        raise ValueError(f"claims.{table['name']}: unknown direction {direction!r}")
    return ClaimSpec(
        name=table["name"],
        a=table["a"],
        b=table["b"],
        direction="greater" if direction == "greater" else "not_worse",
        stretch=table.get("stretch", False),
        min_ratio=table.get("min_ratio"),
    )

    """Raise `ValueError` naming each of `names` that is not a listed player."""
    missing = sorted(set(names) - named)
    if missing:
        raise ValueError(f"{where}: unlisted players {', '.join(missing)}")
