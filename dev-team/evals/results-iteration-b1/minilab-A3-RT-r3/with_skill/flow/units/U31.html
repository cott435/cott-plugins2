+import math

+import pytest

+from risk.drawdown import max_drawdown
+from risk.types import Drawdown, Returns

+def _returns(contract_sample) -> Returns:
+    raw = contract_sample("risk.Returns")
+    return Returns(
+        base_date=raw["base_date"],
+        dates=tuple(raw["dates"]),
+        tickers=tuple(raw["tickers"]),
+        values={t: tuple(v) for t, v in raw["values"].items()},
+    )
+
+
+def _series(values: tuple[float, ...]) -> Returns:
+    dates = tuple(f"2024-03-{d:02d}" for d in range(2, 2 + len(values)))
+    return Returns("2024-03-01", dates, ("X",), {"X": values})
+
+
+def test_sample_matches_the_drawdown_sample(contract_sample):
+    expected = contract_sample("risk.Drawdown")["drawdowns"]
+
+    out = max_drawdown(_returns(contract_sample))
+
+    assert [d.ticker for d in out] == [e["ticker"] for e in expected]
+    for got, want in zip(out, expected, strict=True):
+        assert got.depth == pytest.approx(want["depth"], abs=1e-6)
+        assert (got.peak_date, got.trough_date) == (want["peak_date"], want["trough_date"])
+
+
+def test_hand_computed_fall_is_a_simple_return_from_the_peak():
+    # wealth 1 -> 2 -> 1 -> 1.5: peak 2 on 03-02, trough 1 on 03-03, depth -50%
+    (d,) = max_drawdown(_series((math.log(2), math.log(0.5), math.log(1.5))))
+
+    assert d.depth == pytest.approx(-0.5)
