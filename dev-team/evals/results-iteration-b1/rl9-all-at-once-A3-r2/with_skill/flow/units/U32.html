from collections.abc import Sequence
from pathlib import Path

from rllab.cli.pipelines.pl04_watch import run_watch

def parse_seeds(text: str) -> list[int]:
    """`3`, `1-5` or `1,3,5` to a list of seeds."""
    seeds: list[int] = []
    for part in text.split(","):
        first, _, last = part.partition("-")
        seeds.extend(range(int(first), int(last or first) + 1))
    return seeds


def _check(args: argparse.Namespace) -> object:
    return run_check(args.config, args.seed)


def _train(args: argparse.Namespace) -> object:
    return run_batch(args.config, [args.seed], args.root, check_seed=args.check_seed)


def _batch(args: argparse.Namespace) -> object:
    return run_batch(
        args.config, args.seeds, args.root, workers=args.workers, check_seed=args.check_seed
    )


def _watch(args: argparse.Namespace) -> object:
    return run_watch(args.run_path)


def _chart(args: argparse.Namespace) -> object:
    return run_chart(
        args.root, args.world, args.variant, include=args.include, measure=args.measure
    )


def _evaluate(args: argparse.Namespace) -> object:
    return run_evaluate(
        args.root, args.world, args.variant, include=args.include, workers=args.workers
    )


def _parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(prog="rllab")
    sub = parser.add_subparsers(dest="command", required=True)
    root = argparse.ArgumentParser(add_help=False)
    root.add_argument("--root", type=Path, default=Path("runs"), help="the runs folder")
    runs_of = argparse.ArgumentParser(add_help=False, parents=[root])
    runs_of.add_argument("world")
    runs_of.add_argument("variant")
    runs_of.add_argument("--include", action="append", default=[], metavar="NAME")

    check = sub.add_parser("check", help="run the method check of a config")
    check.add_argument("config", type=Path)
    check.add_argument("--seed", type=int, default=0)
    check.set_defaults(entry=_check)

    train = sub.add_parser("train", parents=[root], help="train one seed of a config")
    train.add_argument("config", type=Path)
    train.add_argument("--seed", type=int, required=True)
    train.add_argument("--check-seed", type=int, default=0)
    train.set_defaults(entry=_train)

    batch = sub.add_parser("batch", parents=[root], help="train several seeds of a config")
    batch.add_argument("config", type=Path)
    batch.add_argument("--seeds", type=parse_seeds, required=True, help="1-5 or 1,3,5")
    batch.add_argument("--workers", type=int, default=None)
    batch.add_argument("--check-seed", type=int, default=0)
    batch.set_defaults(entry=_batch)

    watch = sub.add_parser("watch", help="show a run's progress")
    watch.add_argument("run_path", type=Path)
    watch.set_defaults(entry=_watch)

    chart = sub.add_parser("chart", parents=[runs_of], help="draw a world's learning curves")
    chart.add_argument("--measure", default=None)
    chart.set_defaults(entry=_chart)

    evaluate = sub.add_parser("evaluate", parents=[runs_of], help="evaluate a world's runs")
    evaluate.add_argument("--workers", type=int, default=None)
    evaluate.set_defaults(entry=_evaluate)
    return parser


def _show(result: object) -> None:
    for item in result if isinstance(result, list) else [result]:
        print(item)


def main(argv: Sequence[str] | None = None) -> int:
    args = _parser().parse_args(argv)
    try:
        _show(args.entry(args))
    except Exception as exc:  # every error is one line naming its cause (cli.main)
        print(f"rllab {args.command}: {type(exc).__name__}: {exc}", file=sys.stderr)
        return 1
    return 0
"""A live view of one run's progress (LOG-06)."""

import json
from datetime import datetime
from pathlib import Path

from rllab.runlog.types import EvalPoint, Progress


def progress(path: Path) -> Progress:
    """One snapshot of the run at `path`: its record plus the last evaluation point written."""
    record_file = path / "run.json"
    if not record_file.is_file():
        raise FileNotFoundError(f"no run at {path}: {record_file} is missing")
    record = json.loads(record_file.read_text())
    latest = _latest_eval(path / "evals.jsonl")
    started = datetime.fromisoformat(record["started"])
    ended = record.get("ended")
    until = datetime.fromisoformat(ended) if ended else datetime.now(started.tzinfo)
    return Progress(
        step=latest.step if latest else 0,
        budget=int(record["config"]["budget"]),
        latest=latest,
        elapsed=until - started,
        status=str(record["status"]),
    )


def _latest_eval(log: Path) -> EvalPoint | None:
    """The last complete line of `evals.jsonl`, or None before the first evaluation point."""
    if not log.is_file():
        return None
    lines = [line for line in log.read_text().splitlines() if line.strip()]
    for line in reversed(lines):
        try:
            data = json.loads(line)
        except json.JSONDecodeError:
            continue
        return EvalPoint(int(data["step"]), dict(data["scores"]), dict(data["measures"]))
    return None
