"""PL-01 step 2: keep the usable rows, sorted by (date, ticker)."""

from __future__ import annotations

from collections.abc import Sequence

from mdata.types import RawRow


def validate_rows(rows: Sequence[RawRow], tickers: Sequence[str] | None = None) -> list[RawRow]:
    """Rows with a close (of the requested tickers, if any), sorted by (date, ticker)."""
    wanted = set(tickers) if tickers is not None else None
    kept = [
        row for row in rows if row.close is not None and (wanted is None or row.ticker in wanted)
    ]
    return sorted(kept, key=lambda row: (row.date, row.ticker))

"""The mdata contract types: one CSV row, aligned daily closes, and the gap error."""

from __future__ import annotations

from collections.abc import Mapping
from dataclasses import dataclass


@dataclass(frozen=True)
class RawRow:
    """One CSV row: `date` is ISO yyyy-mm-dd; `close` is None for a blank cell."""

    date: str
    ticker: str
    close: float | None


@dataclass(frozen=True)
class Prices:

from __future__ import annotations

from collections.abc import Sequence

from mdata.types import RawRow


def validate_rows(rows: Sequence[RawRow], tickers: Sequence[str] | None = None) -> list[RawRow]:
    """Rows with a positive close (of the requested tickers, if any), the later of duplicates
    in file order winning, sorted by (date, ticker).

    Raises KeyError naming each requested ticker left with no usable row.
    """
    wanted = set(tickers) if tickers is not None else None
    latest: dict[tuple[str, str], RawRow] = {}
    for row in rows:
        if row.close is not None and row.close > 0 and (wanted is None or row.ticker in wanted):
            latest[(row.date, row.ticker)] = row
    if tickers is not None:
        found = {ticker for _, ticker in latest}
        missing = [ticker for ticker in tickers if ticker not in found]
        if missing:
            raise KeyError(f"no usable rows for tickers: {', '.join(missing)}")
    return [latest[key] for key in sorted(latest)]
