def assert_summary_matches(got: TickerSummary, want: dict[str, Any], tol: float) -> None:
    assert got.ticker == want["ticker"]
    assert got.ann_return == pytest.approx(want["ann_return"], abs=tol)
    assert got.ann_vol == pytest.approx(want["ann_vol"], abs=tol)
    assert got.sharpe == pytest.approx(want["sharpe"], abs=tol)
    if want["last_vol"] is None:
        assert got.last_vol is None
    else:
        assert got.last_vol == pytest.approx(want["last_vol"], abs=tol)

def test_tc07_whole_period_summary(golden) -> None:
    raw, expected = golden("risk", "TC-07")
    result = summary_stats(vol_of(raw["vol"]))
    want = expected["summary"]

    assert [s.ticker for s in result] == [w["ticker"] for w in want]

        assert_summary_matches(got, item, expected["tol"])

@pytest.mark.intent
def test_tc07_periods_per_year_scales_return_and_vol(golden) -> None:
    raw, expected = golden("risk", "TC-07")
    tol = expected["tol"]
    result = summary_stats(vol_of(raw["vol"]), periods_per_year=365)
    for got, item in zip(result, expected["summary"], strict=True):
        ann_return = item["ann_return"] / 252 * 365
        ann_vol = item["ann_vol"] / math.sqrt(252) * math.sqrt(365)
        assert got.ticker == item["ticker"]
        assert got.ann_return == pytest.approx(ann_return, abs=tol)
        assert got.ann_vol == pytest.approx(ann_vol, abs=tol)
        sharpe = ann_return / ann_vol if ann_vol != 0 else 0.0
        assert got.sharpe == pytest.approx(sharpe, abs=tol)
