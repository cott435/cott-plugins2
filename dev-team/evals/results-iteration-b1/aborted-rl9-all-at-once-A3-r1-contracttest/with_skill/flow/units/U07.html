Status = Literal["running", "finished", "incomplete", "failed"]
ACTION_KINDS: dict[type, str] = {Discrete: "discrete", Box: "box"}


def _is_status(status: str) -> TypeGuard[Status]:
    return status in get_args(Status)


def to_json(value: object) -> object:
    """The JSON form of a run-folder value: dataclasses as objects of their fields (`Discrete`
    and `Box` add `"kind"`), tuples as lists, a `Path` or `datetime` as a string."""
    if dataclasses.is_dataclass(value) and not isinstance(value, type):
        out = {f.name: to_json(getattr(value, f.name)) for f in dataclasses.fields(value)}
        kind = ACTION_KINDS.get(type(value))
        return {"kind": kind, **out} if kind else out
    if isinstance(value, Mapping):
        return {str(k): to_json(v) for k, v in value.items()}
    if isinstance(value, (tuple, list)):
        return [to_json(v) for v in value]
    if isinstance(value, datetime):
        return value.isoformat()
    if isinstance(value, Path):
        return str(value)
    if isinstance(value, np.generic):
        return value.item()
    return value

def lines(path: Path) -> list[dict]:
