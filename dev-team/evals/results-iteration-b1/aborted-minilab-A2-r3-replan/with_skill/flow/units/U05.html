"""The mdata contract samples build their types, and the load_prices fake serves its sample."""

from __future__ import annotations

import pytest

from mdata import GapError, Prices, RawRow


def build_prices(payload: dict) -> Prices:
    closes = {ticker: tuple(series) for ticker, series in payload["closes"].items()}
    return Prices(dates=tuple(payload["dates"]), tickers=tuple(payload["tickers"]), closes=closes)


def test_prices_sample_builds_two_tickers_over_six_dates(contract_sample):
    prices = build_prices(contract_sample("mdata.Prices"))
    assert prices.tickers == ("AAA", "BBB")
    assert len(prices.dates) == 6
    assert all(len(prices.closes[t]) == 6 for t in prices.tickers)


def test_rawrow_sample_builds_eight_rows_with_one_blank_close(contract_sample):
    rows = [RawRow(**row) for row in contract_sample("mdata.RawRow")["rows"]]
    assert len(rows) == 8
    assert {row.ticker for row in rows} == {"AAA", "BBB"}
    assert [row for row in rows if row.close is None] == [RawRow("2024-01-03", "BBB", None)]


@pytest.mark.parametrize(
    ("change", "message"),
    [
        ({"dates": ["2024-01-03", "2024-01-02"]}, "ascending"),
        ({"dates": ["2024-01-02", "2024-01-02"]}, "ascending"),
        ({"tickers": ["BBB", "AAA"]}, "sorted"),
        ({"tickers": ["AAA"]}, "one series per ticker"),
        ({"closes": {"AAA": [1.0], "BBB": [1.0, 2.0]}}, "one close per date"),
        ({"closes": {"AAA": [1.0, 0.0], "BBB": [1.0, 2.0]}}, "positive"),

@pytest.mark.intent
def test_later_duplicate_wins_in_file_order(golden):
    given, _ = golden("mdata", "TC-01")
    rows = rows_from_triples(given["rows"])
    forward = validate_rows(rows, given["tickers"])
    backward = validate_rows(list(reversed(rows)), given["tickers"])
    dupes = {(r.date, r.ticker) for r in rows if sum((o.date, o.ticker) == (r.date, r.ticker) and o.close is not None and o.close > 0 for o in rows) > 1}
    assert dupes
    for key in dupes:
        usable = [r for r in rows if (r.date, r.ticker) == key and r.close is not None and r.close > 0]
        assert [r for r in forward if (r.date, r.ticker) == key] == [usable[-1]]
        assert [r for r in backward if (r.date, r.ticker) == key] == [usable[0]]
