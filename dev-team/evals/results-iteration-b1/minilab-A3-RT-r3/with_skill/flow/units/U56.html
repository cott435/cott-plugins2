+"""PL-01 step 3: put every ticker on one calendar."""
+
+from __future__ import annotations
+
+from collections.abc import Sequence
+
+from mdata.types import GapError, Prices, RawRow
+
+
+def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
+    """Prices from the first date every ticker has a close; a gap takes the previous close.
+
+    The calendar is every date any ticker has a close on. Raises GapError naming the ticker
+    when more than `max_gap` dates in a row are missing. A ticker with no close at all
+    leaves no date every ticker has, so the calendar is empty.
+    """
+    closes_by_ticker: dict[str, dict[str, float]] = {}
+    for row in rows:
+        ticker_closes = closes_by_ticker.setdefault(row.ticker, {})
+        if row.close is not None:
+            ticker_closes[row.date] = row.close
+    tickers = tuple(sorted(closes_by_ticker))
+    if not all(closes_by_ticker.values()):
+        return Prices(dates=(), tickers=tickers, closes=dict.fromkeys(tickers, ()))
+    start = max((min(c) for c in closes_by_ticker.values()), default="")
+    dates = tuple(sorted({d for c in closes_by_ticker.values() for d in c if d >= start}))
+    closes = {t: _forward_fill(t, closes_by_ticker[t], dates, max_gap) for t in tickers}
+    return Prices(dates=dates, tickers=tickers, closes=closes)
+
+
+def _forward_fill(
+    ticker: str, closes: dict[str, float], dates: tuple[str, ...], max_gap: int
+) -> tuple[float, ...]:
+    """The ticker's close on each date, a missing date taking the previous close.
+
+    Raises GapError when more than `max_gap` dates in a row are missing.
+    """
+    previous = closes[max(d for d in closes if d <= dates[0])]
+    filled: list[float] = []
+    missing = 0
+    for date in dates:
+        if date in closes:
+            previous = closes[date]
+            missing = 0
+        else:
+            missing += 1
+            if missing > max_gap:
+                raise GapError(f"{ticker} misses more than {max_gap} dates in a row")
+        filled.append(previous)
+    return tuple(filled)
