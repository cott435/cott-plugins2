def validate_rows(rows: Sequence[RawRow], tickers: Sequence[str] | None = None) -> list[RawRow]:
    """Positive closes only, one row per (date, ticker), sorted by (date, ticker) (DATA-02)."""
    usable = [row for row in rows if row.close is not None]
    return sorted(usable, key=lambda row: (row.date, row.ticker))

"""PL-01 step 2: keep the usable rows, sorted."""

from __future__ import annotations

from collections.abc import Sequence

from mdata.types import RawRow


def validate_rows(rows: Sequence[RawRow], tickers: Sequence[str] | None = None) -> list[RawRow]:
    """Positive closes only, one row per (date, ticker), sorted by (date, ticker) (DATA-02).

    Of two usable rows for one (date, ticker), the later one in file order wins. With `tickers`,
    only those tickers are kept, and one with no usable row is a `KeyError` naming it.
    """
    latest = {
        (row.date, row.ticker): row for row in rows if row.close is not None and row.close > 0
    }
    if tickers is not None:
        missing = [t for t in tickers if t not in {row.ticker for row in latest.values()}]
        if missing:
            raise KeyError(f"no usable rows for ticker(s): {', '.join(missing)}")
        wanted = set(tickers)
        latest = {key: row for key, row in latest.items() if row.ticker in wanted}
    return [latest[key] for key in sorted(latest)]
