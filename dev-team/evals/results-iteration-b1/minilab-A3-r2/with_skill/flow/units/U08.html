72	def daily_returns(prices: pd.DataFrame) -> pd.DataFrame:
73	    """Log returns of a dates x tickers close panel."""
74	    return np.log(prices / prices.shift(1)).dropna(how="all")
75	
76	
77	def rolling_vol(returns: pd.DataFrame, window: int = 20) -> pd.DataFrame:
78	    """Rolling volatility over the last `window` days, today included."""
79	    return returns.rolling(window).apply(lambda x: np.std(x), raw=True)
80	
81	
82	def annualize(vol: pd.DataFrame) -> pd.DataFrame:
83	    """Annualize to a calendar year, like my brokerage statements: 365 days."""
84	    return vol * np.sqrt(365)
85	
86	
87	def summary(returns: pd.DataFrame) -> pd.DataFrame:
88	    """Yearly return and volatility per ticker."""
89	    return pd.DataFrame({"ann_return": returns.mean() * 365, "ann_vol": returns.std(ddof=0) * np.sqrt(365)})
90	
91	
92	def max_drawdown(prices: pd.DataFrame) -> pd.Series:
93	    """Deepest fall below the running high."""
94	    return (prices / prices.cummax() - 1).min()

def _assert_values_match(actual, expected: dict, tol: float) -> None:
    assert set(actual) == set(expected)
    for ticker, exp_values in expected.items():
        got = tuple(actual[ticker])
        assert len(got) == len(exp_values)
        for got_v, exp_v in zip(got, exp_values, strict=True):
            if exp_v is None:
                assert got_v is None
            else:
                assert got_v == pytest.approx(exp_v, abs=tol)
