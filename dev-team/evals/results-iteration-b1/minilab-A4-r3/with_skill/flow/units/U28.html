+"""PL-01 step 3: put every ticker on one calendar."""
+
+from __future__ import annotations
+
+from collections.abc import Sequence
+
+from mdata.types import GapError, Prices, RawRow
+
+
+def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
+    """Prices from the first date every ticker has a close; gaps take the previous close.
+
+    The calendar is every date any ticker has a close on, from the first date on which
+    every ticker has one. Raises GapError naming the ticker when more than `max_gap`
+    calendar dates in a row are missing.
+    """
+    by_ticker: dict[str, dict[str, float]] = {}
+    for row in rows:
+        closes = by_ticker.setdefault(row.ticker, {})
+        if row.close is not None:
+            closes[row.date] = row.close
+    tickers = tuple(sorted(by_ticker))
+    dates = _calendar(by_ticker)
+    return Prices(
+        dates=dates,
+        tickers=tickers,
+        closes={t: _filled(t, by_ticker[t], dates, max_gap) for t in tickers},
+    )
+
+
+def _calendar(by_ticker: dict[str, dict[str, float]]) -> tuple[str, ...]:
+    """Every date any ticker has a close on, from the first date every ticker has one."""
+    if not by_ticker:
+        return ()
+    shared = set.intersection(*(set(closes) for closes in by_ticker.values()))
+    if not shared:
+        return ()
+    start = min(shared)
+    every = set().union(*(set(closes) for closes in by_ticker.values()))
+    return tuple(sorted(d for d in every if d >= start))
+
+
+def _filled(
+    ticker: str, closes: dict[str, float], dates: tuple[str, ...], max_gap: int
+) -> tuple[float, ...]:
+    """The ticker's closes on `dates`, each missing date taking the previous close."""
+    filled: list[float] = []
+    missing = 0
+    for date in dates:
+        if date in closes:
+            filled.append(closes[date])
+            missing = 0
+            continue
+        missing += 1
+        if missing > max_gap:
+            raise GapError(f"{ticker} misses more than {max_gap} dates in a row by {date}")
+        filled.append(filled[-1])
+    return tuple(filled)

+import csv

+def read_csv(path: str | Path) -> list[RawRow]:
+    """Every `date,ticker,close` row in file order; a blank close is None.
+
+    Raises ValueError for a missing column or a malformed date.
+    """
+    with Path(path).open(newline="") as handle:
+        reader = csv.DictReader(handle)
+        missing = [c for c in COLUMNS if c not in (reader.fieldnames or [])]
+        if missing:
+            raise ValueError(f"{path}: missing column(s) {', '.join(missing)}")
+        return [_to_row(record) for record in reader]

+    """One record as a RawRow; a cell a short line leaves out reads as blank."""
+    day, ticker, close = ((record[c] or "").strip() for c in COLUMNS)
+    if not _is_iso_date(day):
+        raise ValueError(f"malformed date {day!r}: expected yyyy-mm-dd")
+    return RawRow(date=day, ticker=ticker, close=float(close) if close else None)

+    try:
+        return date.fromisoformat(text).isoformat() == text

+        return False

+"""PL-01 step 2: keep the usable rows, sorted."""
+
+from __future__ import annotations
+
+from collections.abc import Sequence
+
+from mdata.types import RawRow
+
+
+def validate_rows(rows: Sequence[RawRow], tickers: Sequence[str] | None = None) -> list[RawRow]:
+    """Positive closes only, one row per (date, ticker) (the later in file order), sorted.
+
+    Raises KeyError for a requested ticker with no usable row.
+    """
+    wanted = None if tickers is None else set(tickers)
+    latest: dict[tuple[str, str], RawRow] = {}
+    for row in rows:
+        if row.close is not None and row.close > 0 and (wanted is None or row.ticker in wanted):
+            latest[(row.date, row.ticker)] = row
+    missing = sorted(set(tickers or ()) - {ticker for _, ticker in latest})
+    if missing:
+        raise KeyError(f"no usable rows for ticker(s): {', '.join(missing)}")
+    return [latest[key] for key in sorted(latest)]
