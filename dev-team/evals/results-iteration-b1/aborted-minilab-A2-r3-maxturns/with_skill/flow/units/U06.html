def _check_series(tickers: tuple[str, ...], values: Mapping[str, Sequence[object]], length: int) -> None:
    """Exactly one series per ticker, each `length` long."""
    if set(values) != set(tickers):
        raise ValueError("values must have exactly one entry per ticker")
    for ticker in tickers:
        if len(values[ticker]) != length:
            raise ValueError(f"{ticker}: one value per date required")

class Returns:
    """Daily returns: `dates` are the prices' dates after the first, which is `base_date`."""

    base_date: str
    dates: tuple[str, ...]
    tickers: tuple[str, ...]
