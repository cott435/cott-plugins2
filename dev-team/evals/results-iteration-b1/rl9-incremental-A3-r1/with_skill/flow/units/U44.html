98	    """An Agent replay double that expects, for each of `n_games` games, the recorded game's
99	    observations with `explore=False`, returns the recorded actions, and has no `learn` call."""

105	        for obs, act in zip(observations, actions, strict=True)

class _LoopEnv:
    """A corridor-shaped Env: the recording's start observation, then the recording's first
    step result for every step, ending the game after GAME_LENGTH steps."""

    def __init__(self) -> None:
        self.teams: dict[str, TeamSpec] = kit.decode(dict[str, TeamSpec], RECORDING["teams"])
        self.board = RECORDING["board"]
        self._start = kit.decode(dict[str, np.ndarray], RECORDING["reset"]["obs"])
        self._info = kit.decode(dict[str, float], RECORDING["reset"]["info"])
        self._frame = kit.decode(np.ndarray, RECORDING["frames"][0])
        self._t = -1

    def reset(self, seed: int) -> tuple[dict[str, np.ndarray], dict[str, float]]:
        self._t = 0
        return dict(self._start), dict(self._info)

    def step(self, actions: Any) -> StepResult:
        if self._t < 0 or self._t >= GAME_LENGTH:
            raise RuntimeError("step called before reset or after the game is over")
        self._t += 1
        raw = copy.deepcopy(RECORDING["steps"][0]["result"])
        raw["info"]["steps"] = float(self._t)
        if self._t == GAME_LENGTH:
            raw["truncated"] = True
            raw["done"][SOLO]["data"] = [True]
        result: StepResult = kit.decode(StepResult, raw)
        return result

    def frame(self) -> np.ndarray:
        return self._frame


class _InterruptedAgent:
    """A corridor Agent that always moves right and raises KeyboardInterrupt (a Ctrl-C) on its
    INTERRUPT_AT_LEARN-th `learn` call."""

    def __init__(self) -> None:
        self.learns = 0
        self.saved: list[Path] = []

    def act(self, obs: np.ndarray, *, explore: bool) -> np.ndarray:
        return np.ones((obs.shape[0],), dtype=np.int64)

    def learn(self, t: Transition) -> None:
        self.learns += 1

            raise KeyboardInterrupt

    def save(self, path: Path) -> None:
        path.write_bytes(b"agent")

    def load(self, path: Path) -> None:
        path.read_bytes()
