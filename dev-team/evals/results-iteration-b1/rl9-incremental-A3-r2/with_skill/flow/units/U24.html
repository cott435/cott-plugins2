import json
from pathlib import Path

import numpy as np

from rllab.core import Agent, Transition


class RandomPlayer(Agent):
    """Uniform over `n_actions` from a seeded generator, whatever `explore` says."""

    def __init__(self, n_actions: int, *, seed: int) -> None:
        self.n_actions = n_actions
        self._rng = np.random.default_rng(seed)

    def act(self, obs: np.ndarray, *, explore: bool) -> np.ndarray:
        """One uniform action per agent in `obs`; `explore` changes nothing."""
        return self._rng.integers(self.n_actions, size=obs.shape[0], dtype=np.int64)

    def learn(self, t: Transition) -> None:
        """Nothing to learn: the random player never changes."""

    def save(self, path: Path) -> None:
        """Write the generator's state, so `load` replays the same actions."""
        path.write_text(json.dumps(self._rng.bit_generator.state))

    def load(self, path: Path) -> None:
        """Restore the generator's state that `save` wrote."""
        self._rng.bit_generator.state = json.loads(path.read_text())
