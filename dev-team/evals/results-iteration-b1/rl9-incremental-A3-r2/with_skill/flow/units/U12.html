def _poisson_scores(
    players: dict[str, float], seeds: int, games: int, seed: int
) -> dict[str, tuple[ScoreSet, ...]]:
    """Generator `poisson_scores`: per player, `seeds` ScoreSets of `games` Poisson draws."""
    rng = np.random.default_rng(seed)
    out: dict[str, tuple[ScoreSet, ...]] = {}
    for name, lam in players.items():
        out[name] = tuple(
            ScoreSet(
                player=name,
                train_seed=train_seed,
                scores=rng.poisson(lam, games).astype(np.float64),
                games=(),
            )
            for train_seed in range(seeds)
        )
    return out


def _fixed_scores(name: str, values: list[float], seeds: int = 5) -> tuple[ScoreSet, ...]:
    """Split a fixed score list evenly over training seeds (pooling restores the list)."""
    chunks = np.array_split(np.asarray(values, dtype=np.float64), seeds)
    return tuple(
        ScoreSet(player=name, train_seed=i, scores=chunk, games=())
        for i, chunk in enumerate(chunks)
    )


def _seed_range(text: str) -> range:
    lo, hi = (int(part) for part in text.split("-"))
    return range(lo, hi + 1)


def _only(claims: list[Claim]) -> Claim:
    assert len(claims) == 1
    return claims[0]
