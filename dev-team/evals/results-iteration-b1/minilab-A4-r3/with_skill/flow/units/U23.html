"""PL-03 step 2: whole-period summary per ticker."""

from __future__ import annotations

import math
from statistics import fmean, stdev

from risk.types import TickerSummary, VolSeries


def summary_stats(vol: VolSeries, periods_per_year: int = 252) -> tuple[TickerSummary, ...]:
    """Annualized return, volatility, Sharpe and the last rolling volatility (RISK-04).

    The annualized return is `periods_per_year` times the daily mean, the annualized volatility
    is its square root times the sample standard deviation, and the Sharpe ratio is their
    quotient with no risk-free rate (0 when the volatility is 0). The last rolling volatility is
    the latest value that is not None.

    Raises ValueError with fewer than two returns.
    """
    returns = vol.returns
    if len(returns.dates) < 2:
        msg = f"summary_stats needs at least two returns, got {len(returns.dates)}"
        raise ValueError(msg)
    summaries = []
    for ticker in returns.tickers:
        daily = returns.values[ticker]
        ann_return = periods_per_year * fmean(daily)
        ann_vol = math.sqrt(periods_per_year) * stdev(daily)
        sharpe = ann_return / ann_vol if ann_vol else 0.0
        last = next((v for v in reversed(vol.values[ticker]) if v is not None), None)
        summaries.append(TickerSummary(ticker, ann_return, ann_vol, sharpe, last))
    return tuple(summaries)

"""PL-03 step 2: summary_stats annualizes each ticker's mean and sample stdev, with a Sharpe."""

from __future__ import annotations

import math

import pytest

from risk.summary import summary_stats
from risk.types import Returns, VolSeries


@pytest.fixture
def sample_vol(contract_sample) -> VolSeries:
    raw = contract_sample("risk.VolSeries")
    r = raw["returns"]
    returns = Returns(
        base_date=r["base_date"],
        dates=tuple(r["dates"]),
        tickers=tuple(r["tickers"]),
        values={t: tuple(v) for t, v in r["values"].items()},
    )
    return VolSeries(
        returns=returns,
        window=raw["window"],
        annualized=raw["annualized"],
        values={t: tuple(v) for t, v in raw["values"].items()},
    )


def test_one_summary_per_ticker_in_order(sample_vol):
    assert [s.ticker for s in summary_stats(sample_vol)] == ["A", "B"]


def test_annualized_mean_and_sample_stdev(sample_vol):
    a, b = summary_stats(sample_vol)
    # A: daily mean 0.010 / 5; squared deviations sum to 252e-6, over n - 1 = 4.
    assert a.ann_return == pytest.approx(252 * 0.002)
    assert a.ann_vol == pytest.approx(math.sqrt(252 * 63e-6))
    # B: daily mean 0.015 / 5; squared deviations sum to 330e-6, over 4.
    assert b.ann_return == pytest.approx(252 * 0.003)
    assert b.ann_vol == pytest.approx(math.sqrt(252 * 82.5e-6))


def test_periods_per_year_scales_return_and_vol(sample_vol):
    a, _ = summary_stats(sample_vol, periods_per_year=12)
    assert a.ann_return == pytest.approx(12 * 0.002)
    assert a.ann_vol == pytest.approx(math.sqrt(12 * 63e-6))


def test_sharpe_is_annual_return_over_annual_vol(sample_vol):
    a, _ = summary_stats(sample_vol)
    assert a.sharpe == pytest.approx(math.sqrt(252) * 0.002 / math.sqrt(63e-6))


def test_last_vol_is_the_final_rolling_value(sample_vol):
    a, b = summary_stats(sample_vol)
    assert a.last_vol == pytest.approx(0.06734983296193095)
    assert b.last_vol == pytest.approx(0.11224972160321824)


def test_last_vol_skips_trailing_none():
    returns = Returns("2024-03-04", ("d1", "d2", "d3"), ("A",), {"A": (0.01, 0.02, 0.03)})
    vol = VolSeries(returns, 2, annualized=True, values={"A": (None, 0.3, None)})
    (a,) = summary_stats(vol)
    assert a.last_vol == 0.3


def test_last_vol_is_none_without_any_rolling_value():
    returns = Returns("2024-03-04", ("d1", "d2"), ("A",), {"A": (0.01, 0.02)})
    vol = VolSeries(returns, 2, annualized=True, values={"A": (None, None)})
    (a,) = summary_stats(vol)
    assert a.last_vol is None


def test_constant_returns_give_zero_vol_and_zero_sharpe():
    returns = Returns("2024-03-04", ("d1", "d2", "d3"), ("A",), {"A": (0.004, 0.004, 0.004)})
    vol = VolSeries(returns, 2, annualized=True, values={"A": (None, None, 0.0)})
    (a,) = summary_stats(vol)
    assert a.ann_return == pytest.approx(252 * 0.004)
    assert a.ann_vol == 0.0
    assert a.sharpe == 0.0


def test_fewer_than_two_returns_raises():
    returns = Returns("2024-03-04", ("2024-03-05",), ("A",), {"A": (0.01,)})
    vol = VolSeries(returns, 2, annualized=True, values={"A": (None,)})
    with pytest.raises(ValueError, match="at least two returns"):
        summary_stats(vol)
