+"""Unit tests for the summary_stats spine: daily mean and sample stdev per ticker."""
+
+from __future__ import annotations
+
+import json
+import math
+from pathlib import Path
+
+import pytest
+
+from risk.summary import summary_stats
+from risk.types import Returns, TickerSummary, VolSeries
+
+SAMPLE = Path(__file__).parents[2] / "contracts" / "risk.VolSeries" / "sample.json"
+
+
+def _sample_vol() -> VolSeries:
+    data = json.loads(SAMPLE.read_text())
+    r = data["returns"]
+    returns = Returns(
+        base_date=r["base_date"],
+        dates=tuple(r["dates"]),
+        tickers=tuple(r["tickers"]),
+        values={t: tuple(v) for t, v in r["values"].items()},
+    )
+    return VolSeries(
+        returns=returns,
+        window=data["window"],
+        annualized=data["annualized"],
+        values={t: tuple(v) for t, v in data["values"].items()},
+    )
+
+
+def test_sample_gives_one_summary_per_ticker_in_order() -> None:
+    out = summary_stats(_sample_vol())
+    assert [s.ticker for s in out] == ["AAA", "BBB"]
+    assert all(isinstance(s, TickerSummary) for s in out)
+
+
+def test_sample_daily_mean_and_stdev() -> None:
+    aaa, bbb = summary_stats(_sample_vol())
+    # AAA: mean 0.005 / 5; squared deviations sum to 770e-6, over n - 1 = 4
+    assert aaa.ann_return == pytest.approx(0.001)
+    assert aaa.ann_vol == pytest.approx(math.sqrt(192.5e-6))
+    assert aaa.sharpe == pytest.approx(0.001 / math.sqrt(192.5e-6))
+    # BBB: mean 0.006; squared deviations sum to 20e-6, over 4
+    assert bbb.ann_return == pytest.approx(0.006)
+    assert bbb.ann_vol == pytest.approx(math.sqrt(5e-6))
+
+
+def test_last_vol_is_the_last_rolling_value() -> None:
+    aaa, bbb = summary_stats(_sample_vol())
+    assert aaa.last_vol == pytest.approx(0.11224972160321824)
+    assert bbb.last_vol == pytest.approx(0.05612486080160912)
+
+
+def test_flat_returns_give_zero_sharpe() -> None:
+    returns = Returns("2024-01-02", ("2024-01-03", "2024-01-04"), ("AAA",), {"AAA": (0.01, 0.01)})
+    vol = VolSeries(returns, 2, False, {"AAA": (None, None)})
+    (s,) = summary_stats(vol)
+    assert s.ann_vol == 0.0
+    assert s.sharpe == 0.0
+    assert s.last_vol is None
+
+
+def test_fewer_than_two_returns_raises() -> None:
+    returns = Returns("2024-01-02", ("2024-01-03",), ("AAA",), {"AAA": (0.01,)})
+    vol = VolSeries(returns, 2, False, {"AAA": (None,)})
+    with pytest.raises(ValueError, match="at least two returns"):
+        summary_stats(vol)
