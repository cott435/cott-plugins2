from __future__ import annotations

from statistics import fmean, stdev

from risk.types import TickerSummary, VolSeries


def summary_stats(vol: VolSeries, periods_per_year: int = 252) -> tuple[TickerSummary, ...]:
    """One summary per ticker; `ValueError` with fewer than two returns.

    Spine version: `ann_return` and `ann_vol` hold the daily mean and sample stdev of the
    returns, not yet annualized; `sharpe` is their ratio (0 when the stdev is 0).
    """
    if len(vol.returns.dates) < 2:
        raise ValueError(f"summary_stats needs at least two returns, got {len(vol.returns.dates)}")
    summaries = []
    for ticker in vol.returns.tickers:
        returns = vol.returns.values[ticker]
        mean = fmean(returns)
        spread = stdev(returns)
        summaries.append(
            TickerSummary(
                ticker=ticker,
                ann_return=mean,
                ann_vol=spread,
                sharpe=mean / spread if spread else 0.0,
                last_vol=vol.values[ticker][-1],
            )
        )
    return tuple(summaries)
"""PL-03 step 3: maximum drawdown."""

from __future__ import annotations

from math import exp

from risk.types import Drawdown, Returns


def max_drawdown(returns: Returns) -> tuple[Drawdown, ...]:
    """Each ticker's worst fall from a peak, with its peak and trough dates."""
    return tuple(
        _worst_fall(ticker, returns.base_date, returns.dates, returns.values[ticker])
        for ticker in returns.tickers
    )


def _worst_fall(
    ticker: str, base_date: str, dates: tuple[str, ...], values: tuple[float, ...]
) -> Drawdown:
    """Largest drop of the cumulative log level below its running peak, as trough / peak - 1."""
    level = 0.0
    peak, peak_date = 0.0, base_date
    worst, worst_peak_date, worst_trough_date = 0.0, base_date, base_date
    for date, value in zip(dates, values, strict=True):
        level += value
        if level > peak:
            peak, peak_date = level, date
        elif level - peak < worst:
            worst, worst_peak_date, worst_trough_date = level - peak, peak_date, date
    return Drawdown(
        ticker=ticker,
        depth=exp(worst) - 1.0,
        peak_date=worst_peak_date,
        trough_date=worst_trough_date,
    )

from __future__ import annotations

import json
from collections.abc import Mapping, Sequence
from dataclasses import asdict
from pathlib import Path

from risk.types import Drawdown, Report, TickerSummary


def write_report(
    summary: Sequence[TickerSummary], drawdowns: Sequence[Drawdown], out_path: str | Path
) -> Report:
    """The JSON report at `out_path`; `ValueError` if the two cover different tickers."""
    by_ticker = {d.ticker: d for d in drawdowns}
    summary_tickers = {s.ticker for s in summary}
    if summary_tickers != set(by_ticker):
        raise ValueError(
            f"summary covers {sorted(summary_tickers)} but drawdowns cover {sorted(by_ticker)}"
        )
    rows: tuple[Mapping[str, object], ...] = tuple(
        _row(s, by_ticker[s.ticker]) for s in sorted(summary, key=lambda s: s.ticker)
    )
    path = Path(out_path)
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps({"tickers": list(rows)}, indent=2) + "\n", encoding="utf-8")
    return Report(path=str(out_path), rows=rows)


def _row(summary: TickerSummary, drawdown: Drawdown) -> dict[str, object]:
    """One ticker's report row: its summary fields, then its drawdown and both dates."""
    return {
        **asdict(summary),
        "max_drawdown": drawdown.depth,
        "peak_date": drawdown.peak_date,
        "trough_date": drawdown.trough_date,
    }
