def run_open_holdout(spec: StudySpec, *, confirm: bool, home: Path | None = None) -> StudyResult:
    version = resolve_version(spec.version, home=home)
    ledger = open_ledger(version, home=home)
    confirm_opening(ledger, spec, confirm=confirm)
    run_spec = window_spec(spec, version)
    returns = load_returns(run_spec, home)
    windows = build_windows(returns, run_spec)
    structures = analyse_structure(windows)
    targets = lock_holdout(build_targets(returns, windows, structures, spec), ledger)
    sections = build_features(returns, load_volume(version, home), windows, structures)
    split = holdout_split(sections, ledger, spec)
    verdicts = test_feature_holdout(sections, targets, split, ledger)
    models = test_model_holdout(sections, targets, split, ledger)
    return write_holdout_report(verdicts, models, ledger, spec)
