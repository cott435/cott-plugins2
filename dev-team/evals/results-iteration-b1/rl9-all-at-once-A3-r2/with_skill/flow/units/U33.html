def load_kit() -> Any:
    if "contracts_kit" not in sys.modules:
        spec = importlib.util.spec_from_file_location("contracts_kit", CONTRACTS / "kit.py")
        assert spec is not None
        assert spec.loader is not None
        module = importlib.util.module_from_spec(spec)
        sys.modules["contracts_kit"] = module
        spec.loader.exec_module(module)
    return sys.modules["contracts_kit"]


class Calls:
    """What the faked steps were called with."""

    def __init__(self) -> None:
        self.find: list[tuple[Path, str, str, tuple[str, ...]]] = []
        self.final: list[tuple[list[Run], Path, int | None]] = []
        self.found: list[Run] = []


@pytest.fixture
def calls(monkeypatch: pytest.MonkeyPatch) -> Calls:
    kit = load_kit()
    find_fake = kit.fake("runlog.find_runs")
    final_fake = kit.fake("evaluate.final_evaluation")
    seen = Calls()

    def find_runs(
        root: Path, world: str, variant: str, *, include: Sequence[str] = ()
    ) -> list[Run]:
        seen.find.append((root, world, variant, tuple(include)))
        seen.found = find_fake(root, world, variant, include=include)
        return seen.found

    def final_evaluation(
        runs: Sequence[Run], out: Path, *, workers: int | None = None
    ) -> list[Summary]:
        seen.final.append((list(runs), out, workers))
        result: list[Summary] = final_fake(runs, out, workers=workers)
        return result

    monkeypatch.setattr(pl06_evaluate, "find_runs", find_runs)
    monkeypatch.setattr(pl06_evaluate, "final_evaluation", final_evaluation)
    return seen


def test_pl06_smoke_evaluate_command_prints_the_summary(
    calls: Calls, tmp_path: Path, capsys: pytest.CaptureFixture[str]
) -> None:
    code = main(["evaluate", "snake", "default", "--root", str(tmp_path), "--workers", "2"])

    assert code == 0
    printed = capsys.readouterr().out
    assert "world='snake'" in printed
    assert "beats_baselines=True" in printed
    assert calls.find == [(tmp_path, "snake", "default", ())]
    assert len(calls.final) == 1
    assert calls.final[0][2] == 2


def test_pl06_smoke_passes_the_found_runs_to_the_final_evaluation(
    calls: Calls, tmp_path: Path
) -> None:
    summaries = pl06_evaluate.run_evaluate(tmp_path, "snake", "default")

    runs, _out, workers = calls.final[0]
    assert [r.record.config.name for r in runs] == [f"snake-s{i}" for i in range(1, 6)]
    assert runs == calls.found
    assert workers is None
    assert [(s.world, s.variant, s.team) for s in summaries] == [("snake", "default", "snake")]


def test_pl06_smoke_include_adds_the_named_incomplete_run(calls: Calls, tmp_path: Path) -> None:
    load_kit()
    names = [
        r["record"]["config"]["name"]
        for r in load_kit().read("runlog.find_runs", "snake_5_plus_incomplete")
    ]
    incomplete = names[-1]

    pl06_evaluate.run_evaluate(tmp_path, "snake", "default", include=[incomplete])

    assert calls.find[0][3] == (incomplete,)
    assert len(calls.final[0][0]) == 6
    assert calls.final[0][0][-1].record.config.name == incomplete


def test_pl06_smoke_writes_under_the_dated_eval_folder(calls: Calls, tmp_path: Path) -> None:
    before = datetime.now(UTC).date().isoformat()
    pl06_evaluate.run_evaluate(tmp_path, "snake", "default")
    after = datetime.now(UTC).date().isoformat()

    out = calls.final[0][1]
    assert out.parent == tmp_path / "snake" / "default" / "_eval"
    assert out.name in {before, after}
