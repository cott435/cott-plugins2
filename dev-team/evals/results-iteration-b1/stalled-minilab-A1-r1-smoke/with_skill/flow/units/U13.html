import json
from pathlib import Path

from mdata.validate import validate_rows

def _sample_rows() -> list[RawRow]:
    data = json.loads(SAMPLE.read_text(encoding="utf-8"))
    return [RawRow(**row) for row in data["rows"]]


def test_sample_drops_blank_close_and_sorts() -> None:
    assert validate_rows(_sample_rows()) == [
        RawRow(date="2024-03-01", ticker="AAA", close=40.0),

        RawRow(date="2024-03-06", ticker="AAA", close=41.00802),
        RawRow(date="2024-03-06", ticker="BBB", close=24.7401),
    ]


def test_sort_is_by_date_then_ticker() -> None:
    rows = [
        RawRow(date="2024-03-02", ticker="AAA", close=1.0),
        RawRow(date="2024-03-01", ticker="CCC", close=2.0),
        RawRow(date="2024-03-01", ticker="BBB", close=3.0),
    ]
    assert [(r.date, r.ticker) for r in validate_rows(rows)] == [
        ("2024-03-01", "BBB"),
        ("2024-03-01", "CCC"),
        ("2024-03-02", "AAA"),
    ]


def test_tickers_keeps_only_requested() -> None:
    kept = validate_rows(_sample_rows(), ["BBB"])
    assert {row.ticker for row in kept} == {"BBB"}
    assert len(kept) == 3


def test_empty_input_gives_empty_output() -> None:
    assert validate_rows([]) == []
