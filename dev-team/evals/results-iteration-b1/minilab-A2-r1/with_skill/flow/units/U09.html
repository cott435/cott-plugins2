60	"""My old notebook code for volatility and drawdown (pandas). Kept for the formulas, not to import."""
61	import numpy as np
62	import pandas as pd
63	
64	
65	def daily_returns(prices: pd.DataFrame) -> pd.DataFrame:
66	    """Log returns of a dates x tickers close panel."""
67	    return np.log(prices / prices.shift(1)).dropna(how="all")
68	
69	
70	def rolling_vol(returns: pd.DataFrame, window: int = 20) -> pd.DataFrame:
71	    """Rolling volatility over the last `window` days, today included."""
72	    return returns.rolling(window).apply(lambda x: np.std(x), raw=True)
73	
74	
75	def annualize(vol: pd.DataFrame) -> pd.DataFrame:
76	    """Annualize to a calendar year, like my brokerage statements: 365 days."""
77	    return vol * np.sqrt(365)
78	
79	
80	def summary(returns: pd.DataFrame) -> pd.DataFrame:
81	    """Yearly return and volatility per ticker."""
82	    return pd.DataFrame({"ann_return": returns.mean() * 365, "ann_vol": returns.std(ddof=0) * np.sqrt(365)})
83	
84	
85	def max_drawdown(prices: pd.DataFrame) -> pd.Series:
86	    """Deepest fall below the running high."""
87	    return (prices / prices.cummax() - 1).min()
88	```

from __future__ import annotations

import pytest

from risk import Returns, VolSeries, rolling_vol


def _returns(payload: dict) -> Returns:
    return Returns(
        base_date=payload["base_date"],
        dates=tuple(payload["dates"]),
        tickers=tuple(payload["tickers"]),
        values={t: tuple(v) for t, v in payload["values"].items()},
    )


@pytest.mark.intent
def test_tc05_rolling_vol_of_earlier_returns(golden):
    given, expected = golden("risk", "TC-05")
    returns = _returns(given["returns"])

    vol = rolling_vol(returns, given["window"])

    assert isinstance(vol, VolSeries)
    assert vol.window == expected["window"]
    assert vol.annualized is expected["annualized"]
    assert vol.returns == returns
    assert set(vol.values) == set(expected["values"])
    tol = expected["tol"]
    for ticker, want in expected["values"].items():
        got = vol.values[ticker]
        assert len(got) == len(returns.dates)
        for got_v, want_v in zip(got, want, strict=True):
            if want_v is None:
                assert got_v is None
            else:
                assert got_v == pytest.approx(want_v, abs=tol)


@pytest.mark.intent
def test_tc05_no_value_until_window_earlier_returns(golden):
    given, _ = golden("risk", "TC-05")
    returns = _returns(given["returns"])
    window = given["window"]

    vol = rolling_vol(returns, window)

    for ticker in returns.tickers:
        assert vol.values[ticker][:window] == (None,) * window
        assert all(v is not None for v in vol.values[ticker][window:])
