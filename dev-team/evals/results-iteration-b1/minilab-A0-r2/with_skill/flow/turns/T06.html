import numpy as np
import pandas as pd


def daily_returns(prices: pd.DataFrame) -> pd.DataFrame:
    """Log returns of a dates x tickers close panel."""
    return np.log(prices / prices.shift(1)).dropna(how="all")


def rolling_vol(returns: pd.DataFrame, window: int = 20) -> pd.DataFrame:
    """Rolling volatility over the last `window` days, today included."""
    return returns.rolling(window).apply(lambda x: np.std(x), raw=True)


def annualize(vol: pd.DataFrame) -> pd.DataFrame:
    """Annualize to a calendar year, like my brokerage statements: 365 days."""
    return vol * np.sqrt(365)


def summary(returns: pd.DataFrame) -> pd.DataFrame:
    """Yearly return and volatility per ticker."""
    return pd.DataFrame({"ann_return": returns.mean() * 365, "ann_vol": returns.std(ddof=0) * np.sqrt(365)})


def max_drawdown(prices: pd.DataFrame) -> pd.Series:
    """Deepest fall below the running high."""
    return (prices / prices.cummax() - 1).min()

from __future__ import annotations

from collections.abc import Sequence
from itertools import accumulate
from math import exp

from risk.types import Drawdown, Returns


def max_drawdown(returns: Returns) -> tuple[Drawdown, ...]:
    """The deepest fall below the running peak of wealth, which starts at 1 on `base_date`.

    Each drawdown carries the date of its peak and of its trough, the first if tied; with no
    fall, both are `base_date`.
    """
    dates = (returns.base_date, *returns.dates)
    return tuple(_drawdown(t, dates, returns.values[t]) for t in returns.tickers)


def _drawdown(ticker: str, dates: Sequence[str], values: Sequence[float]) -> Drawdown:
    wealth = [1.0, *(exp(x) for x in accumulate(values))]
    peak, peak_date = wealth[0], dates[0]
    worst = Drawdown(ticker=ticker, depth=0.0, peak_date=dates[0], trough_date=dates[0])
    for day, w in zip(dates, wealth, strict=True):
        if w > peak:
            peak, peak_date = w, day
        if w / peak - 1 < worst.depth:
            worst = Drawdown(ticker=ticker, depth=w / peak - 1, peak_date=peak_date, trough_date=day)
    return worst

    lines = ["date,ticker,close"]
