"""The `mdata.load_prices` fake (M1-T22): given `tickers`, only those tickers, on their calendar."""

from __future__ import annotations

from mdata import Prices


def test_one_requested_ticker_is_served_alone_on_its_dates(contract_fake, contract_sample):
    fake = contract_fake("mdata.load_prices")
    sample = contract_sample("mdata.load_prices")
    narrowed = fake.load_prices(fake.CSV, ["AAPL"])
    assert narrowed == Prices(
        dates=tuple(sample["dates"]),
        tickers=("AAPL",),
        closes={"AAPL": tuple(sample["closes"]["AAPL"])},
    )
    assert narrowed.closes["AAPL"] == (180.0, 182.5, 179.0, 176.5, 178.0, 181.25)


def test_requested_tickers_come_back_sorted_once(contract_fake):
    fake = contract_fake("mdata.load_prices")
    both = fake.load_prices(fake.CSV, ["MSFT", "AAPL", "MSFT"])
    assert both.tickers == ("AAPL", "MSFT")
    assert both == fake.load_prices(fake.CSV)
