78	**Status:** Accepted · **Date:** 2026-10-17 · **Scope:** `reference`
79	- **Context:** For Snake, REF-01–REF-03 need an established implementation trained with the same view, rewards, step budget and seeds as the compact-view neural agent; AD-10 covers the tank. The Stable-Baselines3 DQN docs (https://stable-baselines3.readthedocs.io/en/master/modules/dqn.html, read 2026-10-17) say it supports Discrete actions and Box observations, has `buffer_size`, `target_update_interval`, `exploration_fraction`, `seed` and `device`, and offers `MlpPolicy` for vector observations. The custom-environment guide (https://stable-baselines3.readthedocs.io/en/master/guide/custom_env.html, read 2026-10-17) requires a `gymnasium.Env` with `reset(seed, options) → (obs, info)` and a five-field `step`, checked by `check_env`.
80	- **Decision:** `reference` trains Stable-Baselines3 DQN with `MlpPolicy` on the compact view through the AD-03 adapter, with `device="cpu"`. Its hidden layers, learning rate, discount, batch size, memory size, target interval and exploration schedule are mapped one-to-one from the same `RunSettings` the hand-written agent uses. Its curve points come from `evaluate`, exactly as the lab's own agents' do.
81	- **Alternatives:** CleanRL's single-file DQN — very readable, but a script to copy rather than a library to call, so "matching" it blurs into copying it. RLlib — far heavier than one laptop needs.
82	- **Consequences:** Stable-Baselines3 measures exploration as a fraction of training rather than in decay steps, and updates every `train_freq` steps. The mapping fixes `train_freq=1` and `gradient_steps=1`, and sets `exploration_fraction = decay_steps / steps`. Stable-Baselines3 and Gymnasium are an optional install extra, so the rest of the lab never carries them.
83	- **Probe:** none — the two pages above settle the spaces, parameters and environment interface. No account or plan is involved.

import importlib.util

from rllab.core import AgentParams, Exploration, RunSettings

TESTS = Path(__file__).resolve().parents[2]

AD04_KEYS = {
    "learning_rate",

    "batch_size",
    "buffer_size",

def _load(path: Path) -> dict:
    return json.loads(path.read_text(encoding="utf-8"))


def _reference_settings() -> RunSettings:
    raw = codec.samples("reference.dqn_kwargs")["reference_compact"]["settings"]
    return codec.build(RunSettings, raw)


def _distinct_settings() -> RunSettings:
    """Reference settings whose every mapped field has a value no other field shares."""
    base = _reference_settings()
    params = replace(
        base.params,
        lr=0.00037,
        gamma=0.93,
        batch_size=48,
        memory_size=12_345,
        target_every=777,
        learn_start=321,
        hidden=96,
    )
    exploration = Exploration(start=0.9, end=0.11, decay_steps=30_000)
    return replace(base, params=params, exploration=exploration, steps=120_000, seed=4242)


def _expected(settings: RunSettings) -> dict[str, object]:
    """AD-04's mapping, read straight off the spec table."""
    p: AgentParams = settings.params
    e: Exploration = settings.exploration
    return {
        "learning_rate": p.lr,
        "gamma": p.gamma,
        "batch_size": p.batch_size,
        "buffer_size": p.memory_size,
        "target_update_interval": p.target_every,
        "learning_starts": p.learn_start,
        "policy_kwargs": {"net_arch": [p.hidden, p.hidden]},
        "exploration_initial_eps": e.start,
        "exploration_final_eps": e.end,
        "exploration_fraction": e.decay_steps / settings.steps,
        "train_freq": 1,
        "gradient_steps": 1,
        "max_grad_norm": 10,
        "seed": settings.seed,
        "device": "cpu",
    }


# --- TC-17: the adapter passes the library's environment checker ---------------------------


@pytest.mark.intent
def test_tc17_adapter_passes_check_env_without_warnings():
    inp = _load(TC17 / "input.json")
    exp = _load(TC17 / "expected.json")
    assert inp["env"] == "snake"
    assert exp["check_env"] == "passes, no warnings"
    adapter = GymAdapter(Snake(view=inp["view"]))
    with warnings.catch_warnings(record=True) as caught:
        warnings.simplefilter("always")
        check_env(adapter, warn=True)
    assert [str(w.message) for w in caught] == []


@pytest.mark.intent
def test_tc17_adapter_spaces_match_the_compact_view():
    inp = _load(TC17 / "input.json")
    exp = _load(TC17 / "expected.json")
    adapter = GymAdapter(Snake(view=inp["view"]))
    assert exp["observation_space"]["type"] == "Box"
    assert isinstance(adapter.observation_space, gym.spaces.Box)
    assert adapter.observation_space.shape == tuple(exp["observation_space"]["shape"])
    assert exp["action_space"]["type"] == "Discrete"
    assert isinstance(adapter.action_space, gym.spaces.Discrete)
    assert int(adapter.action_space.n) == exp["action_space"]["n"]


# --- dqn_kwargs: AD-04's one-to-one mapping ------------------------------------------------


@pytest.mark.intent
def test_dqn_kwargs_has_exactly_the_ad04_keys():
    kwargs = dqn_kwargs(_reference_settings())
    assert set(kwargs) == AD04_KEYS
    assert "loss" not in kwargs
    assert "loss_fn" not in kwargs


@pytest.mark.intent
def test_dqn_kwargs_maps_the_reference_sample():
    settings = _reference_settings()
    kwargs = dqn_kwargs(settings)
    expected = _expected(settings)
    assert kwargs["exploration_fraction"] == pytest.approx(
        expected.pop("exploration_fraction"), rel=1e-12
    )
    kwargs.pop("exploration_fraction")
    assert kwargs == expected


@pytest.mark.intent
@pytest.mark.parametrize("key", sorted(AD04_KEYS - {"exploration_fraction"}))
def test_dqn_kwargs_maps_each_field_one_to_one(key):
    settings = _distinct_settings()
    assert dqn_kwargs(settings)[key] == _expected(settings)[key]


@pytest.mark.intent
def test_dqn_kwargs_exploration_fraction_is_decay_steps_over_steps():
    settings = _distinct_settings()
    kwargs = dqn_kwargs(settings)
    assert kwargs["exploration_fraction"] == pytest.approx(0.25, rel=1e-12)
    longer = replace(settings, steps=settings.steps * 2)
    assert dqn_kwargs(longer)["exploration_fraction"] == pytest.approx(0.125, rel=1e-12)


@pytest.mark.intent
def test_dqn_kwargs_fixes_the_update_cadence_and_clip():
    kwargs = dqn_kwargs(_distinct_settings())
    assert (kwargs["train_freq"], kwargs["gradient_steps"], kwargs["max_grad_norm"]) == (1, 1, 10)
    assert kwargs["device"] == "cpu"


@pytest.mark.intent
def test_dqn_kwargs_is_pure():
    settings = _distinct_settings()
    before = replace(settings)
    first = dqn_kwargs(settings)
    second = dqn_kwargs(settings)
    assert first == second
    assert settings == before
