def q_network(obs_len: int, hidden: int, n_actions: int, gen: torch.Generator) -> nn.Sequential:
    """An MLP from observations to one value per action, initialised from `gen` (CR-13)."""
    net = nn.Sequential(nn.Linear(obs_len, hidden), nn.ReLU(), nn.Linear(hidden, n_actions))
    with torch.no_grad():
        for layer in (net[0], net[2]):
            bound = 1.0 / math.sqrt(layer.in_features)
            nn.init.uniform_(layer.weight, -bound, bound, generator=gen)
            nn.init.zeros_(layer.bias)
    return net


class GreedyPolicy:
    """A frozen copy of a Q network acting greedily: the highest-valued action per agent."""

    def __init__(self, q: nn.Sequential) -> None:
        self._q = copy.deepcopy(q).requires_grad_(False)

    def act(self, obs: Array, rng: np.random.Generator) -> Array:
        return greedy_actions(self._q, obs)


def greedy_actions(q: nn.Sequential, obs: Array) -> Array:
    """The argmax action of `q` for each row of `obs`, as int64."""
    with torch.no_grad():
        values = q(torch.as_tensor(obs, dtype=torch.float32))
    return np.asarray(values.argmax(dim=1).numpy(), dtype=np.int64)


def policy_from_state(state: Mapping[str, object]) -> Policy:
    """The greedy policy of a saved `DQN.state_dict()`; layer sizes come from the weights."""
    weights = {
        key.removeprefix("q."): torch.as_tensor(value, dtype=torch.float32)
        for key, value in state.items()
        if key.startswith("q.")
    }
    hidden, obs_len = weights["0.weight"].shape
    q = q_network(obs_len, hidden, weights["2.weight"].shape[0], torch.Generator())
    q.load_state_dict(weights)
    return GreedyPolicy(q)


class DQN:
    """Online one-step Q-learning on an MLP with epsilon-greedy exploration (spine)."""

    def __init__(
        self, config: MethodConfig, spec: WorldSpec, team: str, rng: np.random.Generator
    ) -> None:
        if not isinstance(spec.action, Discrete):
            raise ValueError(f"dqn needs a Discrete action space, not {type(spec.action).__name__}")
        settings = {**DEFAULTS, **config.settings}
        self.n_actions = spec.action.n
        self.gamma = float(settings["gamma"])
        self.epsilon_start = float(settings["epsilon_start"])
        self.epsilon_end = float(settings["epsilon_end"])
        self.epsilon_decay_steps = int(settings["epsilon_decay_steps"])
        gen = torch.Generator().manual_seed(int(rng.integers(2**62)))
        self.q = q_network(spec.obs_len, int(settings["hidden"]), self.n_actions, gen)
        self.optimizer = torch.optim.Adam(self.q.parameters(), lr=float(settings["lr"]))
        self.steps = 0

    def epsilon(self) -> float:
        """The exploration rate: linear from `epsilon_start` to `epsilon_end` over the decay."""
        frac = min(1.0, self.steps / max(1, self.epsilon_decay_steps))
        return self.epsilon_start + frac * (self.epsilon_end - self.epsilon_start)

    def act(self, obs: Array, rng: np.random.Generator) -> Array:
        """Per agent: a uniform random action with probability epsilon, else the greedy one."""
        greedy = greedy_actions(self.q, obs)
        explore = rng.random(len(greedy)) < self.epsilon()
        random = rng.integers(self.n_actions, size=len(greedy))
        return np.where(explore, random, greedy).astype(np.int64)

    def observe(self, t: Transition) -> None:
        """One gradient step on the squared one-step TD error of the active agents."""
        self.steps += 1
        active = torch.as_tensor(t.active, dtype=torch.bool)
        if not bool(active.any()):
            return
        obs = torch.as_tensor(t.obs, dtype=torch.float32)[active]
        next_obs = torch.as_tensor(t.next_obs, dtype=torch.float32)[active]
        actions = torch.as_tensor(t.actions, dtype=torch.int64)[active]
        rewards = torch.as_tensor(t.rewards, dtype=torch.float32)[active]
        with torch.no_grad():
            bootstrap = 0.0 if t.done else self.gamma
            target = rewards + bootstrap * self.q(next_obs).max(dim=1).values
        chosen = self.q(obs).gather(1, actions.unsqueeze(1)).squeeze(1)
        loss = nn.functional.mse_loss(chosen, target)
        self.optimizer.zero_grad()
        loss.backward()
        require_finite(loss, self.q)
        self.optimizer.step()

    def policy(self) -> Policy:
        """A frozen greedy copy of the current Q network."""
        return GreedyPolicy(self.q)

    def state_dict(self) -> dict[str, object]:
        """The Q network's weights under `q.` and the number of transitions observed."""
        state: dict[str, object] = {
            f"q.{key}": value.detach().clone() for key, value in self.q.state_dict().items()
        }
        state["steps"] = self.steps
        return state

    """`NonFiniteError` if the loss, a parameter or a gradient is NaN or infinite."""

        tensors.append(param.detach().reshape(-1))
        if param.grad is not None:
            tensors.append(param.grad.reshape(-1))
    values = torch.cat(tensors)
    bad = values[~torch.isfinite(values)]
    if bad.numel():
        raise NonFiniteError(float(bad[0]))
