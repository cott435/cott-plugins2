def collapse_step(runs: Sequence[Run], team: TeamSpec, random_mean: float) -> int | None:
    """The earliest step from which some run's last `COLLAPSE_POINTS`+ eval scores are no better
    than `random_mean` in the team's direction; `None` if no run collapsed."""
    starts = [_collapse_start(run.evals(), team, random_mean) for run in runs]
    return min((step for step in starts if step is not None), default=None)


def _collapse_start(evals: Sequence[EvalPoint], team: TeamSpec, random_mean: float) -> int | None:
    def no_better(point: EvalPoint) -> bool:
        score = point.scores[team.name]
        return score <= random_mean if team.direction == "higher" else score >= random_mean

    tail = list(takewhile(no_better, reversed(evals)))
    return tail[-1].step if len(tail) >= COLLAPSE_POINTS else None

"""Paired permutation tests against each baseline, with Holm's correction (EVAL-04).

Designed with the null-hypothesis-testing skill. H0: the learned agent's mean score is no
better than the baseline's; H1: it is better in the team's score direction (one-sided). The
statistic is the mean over seeds of the learned score (averaged over runs) minus the baseline
score, signed so positive means better. The null swaps the learned and baseline labels within
each seed, which flips the sign of that seed's difference.
"""

from collections.abc import Mapping, Sequence
from typing import Literal

import numpy as np

from rllab.core import Array, Direction
from rllab.evaluate.types import Comparison

ALPHA = 0.05


def compare(
    learned: Array,
    baseline: Array,
    direction: Direction,
    *,
    permutations: int = 10_000,
    rng: np.random.Generator,
) -> tuple[float, float]:
    """(difference, one-sided p) of learned (runs x seeds) against baseline (one per seed).

    The p-value counts the observed labelling as one of the permutations, so it is never 0.
    """
    per_seed = np.asarray(learned, dtype=np.float64).mean(axis=0)
    base = np.asarray(baseline, dtype=np.float64)
    if per_seed.shape != base.shape:
        raise ValueError(
            f"learned has {per_seed.shape[0]} seeds but baseline has {base.shape[0]} seeds"
        )
    sign = 1.0 if direction == "higher" else -1.0
    differences = sign * (per_seed - base)
    observed = float(differences.mean())
    swaps = rng.choice(np.array([-1.0, 1.0]), size=(permutations, differences.size))
    null = (swaps * differences).mean(axis=1)
    at_least = int(np.count_nonzero(null >= observed))
    return observed, (at_least + 1) / (permutations + 1)


def holm(p_values: Sequence[float]) -> tuple[float, ...]:
    """Holm's step-down adjusted p-values, in the input order."""
    m = len(p_values)
    adjusted = [0.0] * m
    running = 0.0
    for rank, i in enumerate(sorted(range(m), key=lambda j: p_values[j])):
        running = max(running, min(1.0, (m - rank) * p_values[i]))
        adjusted[i] = running
    return tuple(adjusted)


def compare_baselines(
    team: str,
    learned: Array,
    baselines: Mapping[Literal["random", "rule"], Array],
    direction: Direction,
    *,
    rng: np.random.Generator,
) -> tuple[Comparison, ...]:
    """One `Comparison` per baseline, with Holm's correction across them at `ALPHA`."""
    names = list(baselines)
    tests = [compare(learned, baselines[name], direction, rng=rng) for name in names]
    corrected = holm([p for _, p in tests])
    return tuple(
        Comparison(
            baseline=name,
            team=team,
            learned_mean=float(np.asarray(learned, dtype=np.float64).mean()),
            baseline_mean=float(np.asarray(baselines[name], dtype=np.float64).mean()),
            difference=difference,
            p_raw=p_raw,
            p_holm=p_holm,
            rejected=p_holm < ALPHA,
        )
        for name, (difference, p_raw), p_holm in zip(names, tests, corrected, strict=True)
    )

import json
from pathlib import Path
from typing import Literal

from rllab.evaluate import Comparison, Verdict
from rllab.evaluate.final.verdict import verdict

TC20 = Path(__file__).parents[2] / "fixtures" / "rllab" / "TC-20"
FIVE_RUNS = {"s1": 14.0, "s2": 13.0, "s3": 15.0, "s4": 12.0, "s5": 11.0}


def comparison(baseline: Literal["random", "rule"], *, rejected: bool) -> Comparison:
    return Comparison(
        baseline=baseline,
        team="snake",
        learned_mean=13.0,
        baseline_mean=10.0,
        difference=3.0,
        p_raw=0.01 if rejected else 0.4,
        p_holm=0.02 if rejected else 0.4,
        rejected=rejected,
    )


BOTH_REJECTED = (comparison("random", rejected=True), comparison("rule", rejected=True))


def test_tc20_one_run_below_the_rule_fails_the_world() -> None:
    given = json.loads((TC20 / "input.json").read_text())
    expected = json.loads((TC20 / "expected.json").read_text())
    comparisons = tuple(
        comparison(name, rejected=rejected)
        for name, rejected in given["comparisons_rejected"].items()
    )
    result = verdict(
        {name: float(mean) for name, mean in given["run_means"].items()},
        float(given["rule_mean"]),
        comparisons,
        given["direction"],
    )
    assert result.beats_baselines is expected["beats_baselines"]
    assert result.failing_runs == tuple(expected["failing_runs"])
    assert result.runs_needed == 0


def test_five_runs_all_beating_and_both_rejected_pass() -> None:
    assert verdict(FIVE_RUNS, 10.0, BOTH_REJECTED, "higher") == Verdict(True, (), 0)


def test_a_run_equal_to_the_rule_mean_fails() -> None:
    result = verdict({**FIVE_RUNS, "s5": 10.0}, 10.0, BOTH_REJECTED, "higher")
    assert result == Verdict(False, ("s5",), 0)


def test_three_runs_need_two_more_and_never_pass() -> None:
    result = verdict({"s1": 14.0, "s2": 13.0, "s3": 15.0}, 10.0, (), "higher")
    assert result == Verdict(False, (), 2)


def test_a_kept_null_against_a_baseline_fails() -> None:
    kept = (comparison("random", rejected=True), comparison("rule", rejected=False))
    assert verdict(FIVE_RUNS, 10.0, kept, "higher") == Verdict(False, (), 0)


def test_five_runs_without_comparisons_fail() -> None:
    assert verdict(FIVE_RUNS, 10.0, (), "higher") == Verdict(False, (), 0)

    result = verdict(means, 7.0, BOTH_REJECTED, "lower")
