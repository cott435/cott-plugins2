SAMPLE_TOL = 1e-5  # the samples' derived values are rounded to about six digits


def _returns(d: dict[str, Any]) -> Returns:
    return Returns(
        base_date=d["base_date"],
        dates=tuple(d["dates"]),
        tickers=tuple(d["tickers"]),
        values={t: tuple(v) for t, v in d["values"].items()},
    )


def _vol(d: dict[str, Any]) -> VolSeries:
    return VolSeries(
        returns=_returns(d["returns"]),
        window=d["window"],
        annualized=d["annualized"],
        values={t: tuple(v) for t, v in d["values"].items()},
    )


def _prices(d: dict[str, Any]) -> Prices:
    return Prices(
        dates=tuple(d["dates"]),
        tickers=tuple(d["tickers"]),
        closes={t: tuple(v) for t, v in d["closes"].items()},
    )


# ---- types: each sample builds its type and holds its invariants -----------------------------


def test_raw_row_sample(contract_sample):
    rows = [RawRow(**r) for r in contract_sample("mdata.RawRow")["rows"]]
    assert len(rows) == 8
    assert {r.ticker for r in rows} == {"AAPL", "MSFT"}
    assert all(ISO_DATE.match(r.date) for r in rows)
    assert [r for r in rows if r.close is None] == [RawRow("2024-03-05", "AAPL", None)]


def test_prices_sample(contract_sample):
    prices = _prices(contract_sample("mdata.Prices"))
    assert len(prices.dates) == 6
    assert list(prices.dates) == sorted(set(prices.dates))
    assert list(prices.tickers) == sorted(prices.tickers) == ["AAPL", "MSFT"]
    assert set(prices.closes) == set(prices.tickers)
    for closes in prices.closes.values():
        assert len(closes) == len(prices.dates)
        assert all(c > 0 for c in closes)


def test_returns_sample(contract_sample):
    returns = _returns(contract_sample("risk.Returns"))
    assert len(returns.dates) == 5
    assert returns.base_date < returns.dates[0]
    assert set(returns.values) == set(returns.tickers) == {"AAPL", "MSFT"}
    assert all(len(v) == len(returns.dates) for v in returns.values.values())


def test_vol_series_sample(contract_sample):
    vol = _vol(contract_sample("risk.VolSeries"))
    assert vol.window == 2
    assert vol.annualized is True
    for ticker in vol.returns.tickers:
        values = vol.values[ticker]
        assert len(values) == len(vol.returns.dates)
        assert values[: vol.window] == (None, None)
        assert all(v is not None and v >= 0 for v in values[vol.window :])


def test_ticker_summary_sample(contract_sample):
    summary = TickerSummary(**contract_sample("risk.TickerSummary"))
    assert summary.ticker == "AAPL"
    assert summary.last_vol is not None
    assert summary.ann_vol > 0
    assert summary.sharpe == pytest.approx(summary.ann_return / summary.ann_vol, abs=SAMPLE_TOL)


def test_drawdown_sample(contract_sample):
    fall, flat = (Drawdown(**d) for d in contract_sample("risk.Drawdown")["drawdowns"])
    assert fall.depth < 0
    assert fall.peak_date < fall.trough_date
    assert flat.depth == 0.0
    assert flat.peak_date == flat.trough_date == "2024-03-01"


def test_report_sample(contract_sample):
    data = contract_sample("risk.Report")
    report = Report(path=data["path"], rows=tuple(data["rows"]))
    tickers = [row["ticker"] for row in report.rows]
    assert tickers == sorted(tickers) == ["AAPL", "MSFT"]
    keys = {"ticker", "ann_return", "ann_vol", "sharpe", "last_vol"}
    keys |= {"max_drawdown", "peak_date", "trough_date"}
    assert all(set(row) == keys for row in report.rows)


# ---- the samples agree with one another (PL-01 -> PL-02 -> PL-03) -----------------------------


def test_returns_are_the_log_returns_of_the_prices(contract_sample):
    prices = _prices(contract_sample("mdata.Prices"))
    returns = _returns(contract_sample("risk.Returns"))
    assert returns.base_date == prices.dates[0]
    assert returns.dates == prices.dates[1:]
    assert returns.tickers == prices.tickers
    for ticker, closes in prices.closes.items():
        expected = [math.log(b / a) for a, b in zip(closes, closes[1:], strict=False)]
        assert returns.values[ticker] == pytest.approx(expected, abs=SAMPLE_TOL)


def test_vol_series_is_the_annualized_window_2_vol_of_the_returns(contract_sample):
    vol = _vol(contract_sample("risk.VolSeries"))
    assert vol.returns == _returns(contract_sample("risk.Returns"))
    for ticker, rets in vol.returns.values.items():
        for i in range(vol.window, len(rets)):
            expected = statistics.stdev(rets[i - vol.window : i]) * math.sqrt(252)
            assert vol.values[ticker][i] == pytest.approx(expected, abs=SAMPLE_TOL)


def test_summary_and_drawdowns_are_the_report_rows(contract_sample):
    summary = contract_sample("risk.TickerSummary")
    drawdowns = contract_sample("risk.Drawdown")["drawdowns"]
    rows = {row["ticker"]: row for row in contract_sample("risk.Report")["rows"]}
    assert {k: rows["AAPL"][k] for k in summary} == summary
    for d in drawdowns:
        row = rows[d["ticker"]]
        assert (row["max_drawdown"], row["peak_date"], row["trough_date"]) == (
            d["depth"],
            d["peak_date"],
            d["trough_date"],
        )

def test_vol_series_fake_errors(contract_fake, tmp_path):

    with pytest.raises(ValueError, match="at least 2"):

        fake.vol_series(fake.CSV, window=20)

        fake.vol_series(fake.CSV, window=2, tickers=["NVDA"])
    with pytest.raises(FileNotFoundError):
        fake.vol_series(tmp_path / "other.csv", window=2)
