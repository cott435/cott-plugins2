"""PL-03 step 3: maximum drawdown per ticker."""

from __future__ import annotations

import math
from itertools import accumulate

from risk.types import Drawdown, Returns


def max_drawdown(returns: Returns) -> tuple[Drawdown, ...]:
    """The deepest fall below the running peak, with its peak and trough dates (RISK-05).

    Spine version: depth only; both dates are `base_date`.
    """
    return tuple(
        Drawdown(
            ticker=ticker,
            depth=_depth(returns.values[ticker]),
            peak_date=returns.base_date,
            trough_date=returns.base_date,
        )
        for ticker in returns.tickers
    )


def _depth(log_returns: tuple[float, ...]) -> float:
    """Worst simple-return fall from the running peak of the cumulative log level (base 0)."""
    worst = 0.0
    peak = 0.0
    for level in accumulate(log_returns):
        peak = max(peak, level)
        worst = min(worst, level - peak)
    return math.expm1(worst)
