Each `tests/contracts/<component>.<Symbol>/` folder holds `schema.json` and `samples/<case>.json`;
a function's folder also holds `fake.py`. Folder names hold dots, so load this kit by path:

    from pathlib import Path

# ------------------------------------------------------------------------ decoding and encoding
def decode(tp: Any, data: Any) -> Any:
    """`data` (parsed JSON) as an instance of the type `tp`; dataclass defaults fill gaps."""
    if tp is np.ndarray:
        return np.asarray(data["data"], dtype=data["dtype"]).reshape(data["shape"])
    if dataclasses.is_dataclass(tp) and isinstance(tp, type):
        hints = get_type_hints(tp)
        return tp(**{key: decode(hints[key], value) for key, value in data.items()})
    origin = get_origin(tp)
    if origin is not None:
        return _decode_generic(origin, get_args(tp), data)
    return _decode_scalar(tp, data)


def _decode_generic(origin: Any, args: tuple[Any, ...], data: Any) -> Any:
    if origin is Literal:
        if data not in args:
            raise ValueError(f"{data!r} is not one of {args}")
        return data
    if origin in {Union, types.UnionType}:
        inner = [a for a in args if a is not type(None)]
        return None if data is None and len(inner) < len(args) else decode(inner[0], data)
    if origin is tuple:
        if len(args) == 2 and args[1] is Ellipsis:
            args = (args[0],) * len(data)
        return tuple(decode(a, x) for a, x in zip(args, data, strict=True))
    if origin in {list, Sequence}:
        return [decode(args[0], x) for x in data]
    if origin in {dict, Mapping}:
        return {key: decode(args[1], value) for key, value in data.items()}
    raise TypeError(f"cannot decode {data!r} as {origin}")


def _decode_scalar(tp: Any, data: Any) -> Any:
    if tp is Any or (tp in {int, str, bool} and type(data) is tp):
        return data
    if tp is Path:
        return Path(data)
    if tp is float and isinstance(data, int | float) and not isinstance(data, bool):
        return float(data)
    raise TypeError(f"cannot decode {data!r} as {tp}")


def encode(value: Any) -> Any:
    """The JSON form `decode` reads back."""
    if dataclasses.is_dataclass(value) and not isinstance(value, type):
        return {f.name: encode(getattr(value, f.name)) for f in dataclasses.fields(value)}
    if isinstance(value, np.ndarray):
        return {"dtype": str(value.dtype), "shape": list(value.shape), "data": value.tolist()}
    if isinstance(value, Mapping):
        return {str(k): encode(v) for k, v in value.items()}
    if isinstance(value, list | tuple):
        return [encode(v) for v in value]
    return _encode_scalar(value)


def _encode_scalar(value: Any) -> Any:
    if isinstance(value, np.generic):
        return value.item()
    if isinstance(value, Path):
        return value.as_posix()
    return value


# ---------------------------------------------------------------------------- schema validation
_JSON_TYPES: dict[str, tuple[type, ...]] = {
    "object": (dict,),
    "array": (list,),
    "string": (str,),
    "integer": (int,),
    "number": (int, float),
    "boolean": (bool,),
    "null": (type(None),),
}


def validate(schema_: dict[str, Any], data: Any) -> list[str]:
    """Every way `data` breaks `schema_` (a JSON Schema subset: $ref, type, enum, const,
    properties, required, additionalProperties, items, prefixItems, min/maxItems,
    minimum/maximum, anyOf). Empty when valid."""
    return _check(schema_, data, schema_, "$")


def _is_type(data: Any, name: str) -> bool:
    if isinstance(data, bool) and name in {"integer", "number"}:
        return False
    return isinstance(data, _JSON_TYPES[name])


def _check(s: dict[str, Any], data: Any, root: dict[str, Any], at: str) -> list[str]:
    if "$ref" in s:
        target: Any = root
        for part in s["$ref"].removeprefix("#/").split("/"):
            target = target[part]
        return _check(target, data, root, at)
    if "anyOf" in s:
        if any(not _check(option, data, root, at) for option in s["anyOf"]):
            return []
        return [f"{at}: matches none of anyOf"]
    errors: list[str] = []
    if "type" in s:
        names = s["type"] if isinstance(s["type"], list) else [s["type"]]
        if not any(_is_type(data, n) for n in names):
            return [f"{at}: expected {s['type']}, got {type(data).__name__}"]
    if "enum" in s and data not in s["enum"]:
        errors.append(f"{at}: {data!r} not in {s['enum']}")
    if "const" in s and data != s["const"]:
        errors.append(f"{at}: {data!r} != {s['const']!r}")
    if "minimum" in s and isinstance(data, int | float) and data < s["minimum"]:
        errors.append(f"{at}: {data} < {s['minimum']}")
    if "maximum" in s and isinstance(data, int | float) and data > s["maximum"]:
        errors.append(f"{at}: {data} > {s['maximum']}")
    if isinstance(data, dict):
        errors += _check_object(s, data, root, at)
    if isinstance(data, list):
        errors += _check_array(s, data, root, at)
    return errors


def _check_object(
    s: dict[str, Any], data: dict[str, Any], root: dict[str, Any], at: str
) -> list[str]:
    errors = [f"{at}: missing {key!r}" for key in s.get("required", []) if key not in data]
    props = s.get("properties", {})
    extra = s.get("additionalProperties", True)
    for key, value in data.items():
        if key in props:
            errors += _check(props[key], value, root, f"{at}.{key}")
        elif extra is False:
            errors.append(f"{at}: unexpected {key!r}")
        elif isinstance(extra, dict):
            errors += _check(extra, value, root, f"{at}.{key}")
    return errors


def _check_array(s: dict[str, Any], data: list[Any], root: dict[str, Any], at: str) -> list[str]:
    errors = []
    if len(data) < s.get("minItems", 0):
        errors.append(f"{at}: fewer than {s['minItems']} items")
    if "maxItems" in s and len(data) > s["maxItems"]:
        errors.append(f"{at}: more than {s['maxItems']} items")
    prefix = s.get("prefixItems", [])
    for i, item in enumerate(data):
        item_schema = prefix[i] if i < len(prefix) else s.get("items")
        if item_schema is not None:
            errors += _check(item_schema, item, root, f"{at}[{i}]")
    return errors

class EnvReplay:
    """A core.Env replaying one recorded game: `reset` returns the recorded start, and each
    `step` must get the recorded actions (`recorded_actions`), returning the recorded result."""

    def __init__(self, recording: dict[str, Any]) -> None:
        self.teams: dict[str, TeamSpec] = decode(dict[str, TeamSpec], recording["teams"])
        self.board: Literal["grid", "tank"] = recording["board"]
        self.seed: int = recording["seed"]
        self._start = decode(dict[str, np.ndarray], recording["reset"]["obs"])
        self._info = decode(dict[str, float], recording["reset"]["info"])
        self._frames = [decode(np.ndarray, f) for f in recording["frames"]]
        self.recorded_actions: list[dict[str, np.ndarray]] = [
            decode(dict[str, np.ndarray], s["actions"]) for s in recording["steps"]
        ]
        self._results: list[StepResult] = [
            decode(StepResult, s["result"]) for s in recording["steps"]
        ]
        self.reset_seeds: list[int] = []
        self._t = -1

    def reset(self, seed: int) -> tuple[dict[str, np.ndarray], dict[str, float]]:
        self.reset_seeds.append(seed)
        self._t = 0
        return dict(self._start), dict(self._info)

    def step(self, actions: Mapping[str, np.ndarray]) -> StepResult:
        if self._t < 0 or self._t >= len(self._results):
            raise RuntimeError("step called before reset or after the game is over")
        expected = self.recorded_actions[self._t]
        if set(actions) != set(expected):
            raise ValueError(
                f"teams {sorted(actions)} differ from the recording's {sorted(expected)}"
            )
        for team, action in expected.items():
            if not np.array_equal(np.asarray(actions[team]), action):
                raise ValueError(f"step {self._t}: {team} actions differ from the recording")
        result = self._results[self._t]
        self._t += 1
        return result

    def frame(self) -> np.ndarray:
        return self._frames[max(self._t, 0)]


class AgentReplay:
    """A core.Agent replaying recorded calls in order: each `act` must get the recorded
    observation and `explore`, and returns the recorded actions; each `learn` must get the
    recorded Transition. `calls` lists what it was asked."""

    def __init__(self, recording: dict[str, Any]) -> None:
        self.team: TeamSpec = decode(TeamSpec, recording["team"])
        self._calls: list[dict[str, Any]] = recording["calls"]
        self._next = 0
        self.calls: list[str] = []

    def _expect(self, method: str) -> dict[str, Any]:
        if self._next >= len(self._calls):
            raise RuntimeError(f"{method} called after the {len(self._calls)} recorded calls")
        call = self._calls[self._next]
        if call["method"] != method:
            raise ValueError(f"call {self._next}: {method}, but the recording has {call['method']}")
        self._next += 1
        self.calls.append(method)
        return call

    def act(self, obs: np.ndarray, *, explore: bool) -> np.ndarray:
        call = self._expect("act")
        if not np.array_equal(obs, decode(np.ndarray, call["obs"])) or explore != call["explore"]:
            raise ValueError(f"act {self._next - 1}: observation or explore differ")
        return decode(np.ndarray, call["returns"])

    def learn(self, t: Transition) -> None:
        call = self._expect("learn")
        recorded = decode(Transition, call["transition"])
        same = all(
            np.array_equal(getattr(t, f.name), getattr(recorded, f.name))
            for f in dataclasses.fields(Transition)
        )
        if not same:
            raise ValueError(f"learn {self._next - 1}: the transition differs from the recording")

    def save(self, path: Path) -> None:
        path.write_text(json.dumps({"next": self._next}), encoding="utf-8")
        self.calls.append("save")

    def load(self, path: Path) -> None:
        self._next = json.loads(path.read_text(encoding="utf-8"))["next"]
        self.calls.append("load")


class RunWriterReplay:
    """A runs.RunWriter that keeps everything in memory and builds the RunRecord on `finish`.
    `calls` lists the calls it got, as the runs.RunWriter sample records them."""

    def __init__(self, path: Path, settings: Any) -> None:
        self.path = path
        self.settings = settings
        self.points: list[CurvePoint] = []
        self.checkpoints: dict[str, Path] = {}
        self.generations: dict[str, list[Path]] = {}
        self.games: tuple[GameRecording, ...] = ()
        self.calls: list[dict[str, Any]] = []

    def add_point(self, p: CurvePoint) -> None:
        self.points.append(p)
        self.calls.append({"method": "add_point", "step": p.step})

    def save_checkpoint(self, agent: Agent, *, team: str = "agents") -> None:
        self.checkpoints[team] = self.path / f"checkpoint-{team}"
        self.calls.append({"method": "save_checkpoint", "team": team})

    def save_generation(self, agent: Agent, *, team: str, index: int) -> Path:

        self.calls.append({"method": "finish", "status": status})

            path=self.path,
            settings=self.settings,
            curve=tuple(self.points),
            status=status,
            checkpoints=dict(self.checkpoints),
            generations={team: tuple(paths) for team, paths in self.generations.items()},
            games=self.games,
        )


class ReplayViewerDouble:
    """A viz.Replay that draws nothing: `advance` moves one frame on, stopping at the last."""

    def __init__(self, game: GameRecording, *, fps: float = 8.0) -> None:
        self.game = game
        self.fps = fps
        self.index = 0
        self.shown = False

    def advance(self) -> int:
        self.index = min(self.index + 1, self.game.frames.shape[0] - 1)
        return self.index

    def show(self) -> None:
        while self.index < self.game.frames.shape[0] - 1:
            self.advance()
        self.shown = True
