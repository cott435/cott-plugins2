def summary_stats(vol: VolSeries, periods_per_year: int = 252) -> tuple[TickerSummary, ...]:
    """Annualized return, volatility and Sharpe of `vol.returns`, with the last rolling vol.

    Raises ValueError with fewer than two returns.
    """
    if len(vol.returns.dates) < 2:
        msg = f"summary_stats needs at least two returns, got {len(vol.returns.dates)}"
        raise ValueError(msg)
    return tuple(_summarize(vol, ticker) for ticker in vol.returns.tickers)


def _summarize(vol: VolSeries, ticker: str) -> TickerSummary:
    """Spine: daily mean and sample stdev, Sharpe as their ratio (0.0 with no spread)."""
    returns = vol.returns.values[ticker]
    mean = fmean(returns)
    spread = stdev(returns)
    return TickerSummary(
        ticker=ticker,
        ann_return=mean,
        ann_vol=spread,
        sharpe=mean / spread if spread else 0.0,

"""PL-03 step 2: whole-period summary per ticker."""

from __future__ import annotations

from math import sqrt
from statistics import fmean, stdev

from risk.types import TickerSummary, VolSeries


def summary_stats(vol: VolSeries, periods_per_year: int = 252) -> tuple[TickerSummary, ...]:
    """Annualized return, volatility and Sharpe of `vol.returns`, with the last rolling vol.

    Raises ValueError with fewer than two returns.
    """
    if len(vol.returns.dates) < 2:
        msg = f"summary_stats needs at least two returns, got {len(vol.returns.dates)}"
        raise ValueError(msg)
    return tuple(_summarize(vol, ticker, periods_per_year) for ticker in vol.returns.tickers)


def _summarize(vol: VolSeries, ticker: str, periods_per_year: int) -> TickerSummary:
    """`periods_per_year` x mean, sqrt(`periods_per_year`) x sample stdev; Sharpe 0.0 with no spread."""
    returns = vol.returns.values[ticker]
    ann_return = periods_per_year * fmean(returns)
    ann_vol = sqrt(periods_per_year) * stdev(returns)
    return TickerSummary(
        ticker=ticker,
        ann_return=ann_return,
        ann_vol=ann_vol,
        sharpe=ann_return / ann_vol if ann_vol else 0.0,
        last_vol=_last_value(vol.values[ticker]),
    )


def _last_value(values: tuple[float | None, ...]) -> float | None:
    """The latest rolling value that exists, or None when there is none."""
    return next((v for v in reversed(values) if v is not None), None)
