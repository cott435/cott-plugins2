"""PL-03 step 2: whole-period summary."""

from math import sqrt
from statistics import fmean, stdev

from risk.types import TickerSummary, VolSeries


def summary_stats(vol: VolSeries, periods_per_year: int = 252) -> tuple[TickerSummary, ...]:
    """One summary per ticker; `ValueError` with fewer than two returns.

    The annualized return is `periods_per_year` times the mean, the annualized volatility
    the square root of `periods_per_year` times the sample stdev, and the Sharpe ratio their
    quotient (0 when the volatility is 0; no risk-free rate).
    """
    if len(vol.returns.dates) < 2:
        raise ValueError(f"need at least two returns, got {len(vol.returns.dates)}")
    summaries = []
    for ticker in vol.returns.tickers:
        series = vol.returns.values[ticker]
        ann_return = periods_per_year * fmean(series)
        ann_vol = sqrt(periods_per_year) * stdev(series)
        summaries.append(
            TickerSummary(
                ticker=ticker,
                ann_return=ann_return,
                ann_vol=ann_vol,
                sharpe=ann_return / ann_vol if ann_vol else 0.0,
                last_vol=vol.values[ticker][-1],
            )
        )
    return tuple(summaries)
