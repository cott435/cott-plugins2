def run_validate(settings: Settings, full: bool, seed: int | None) -> RunSummary:
    run = start_run("validate_full" if full else "validate")
    try:
        progress = StoreProgress(run)
        source = make_check_source(settings, full)
        members = members_between(FIRST_DAY, settings.last_loaded_day)
        sample = draw_sample(members, full, seed)
        histories = fetch_histories(sample, source, progress)
        returns = total_returns(sample.ids, sample.start, sample.end, as_of=sample.end)
        compared = compare(returns, histories)
        save_source_diffs(run.id, compared.daily)
        found = compared + internal_checks(returns, sample.universe) + reference_check(sample.universe)
