from __future__ import annotations

def log_returns(prices: Prices) -> Returns:
    """Each ticker's return from one date to the next; ValueError with fewer than two dates."""
    if len(prices.dates) < 2:
        raise ValueError(f"need at least two dates for returns, got {len(prices.dates)}")
    values = {
        ticker: tuple(now / before - 1.0 for before, now in pairwise(prices.closes[ticker]))
        for ticker in prices.tickers
    }
    return Returns(
        base_date=prices.dates[0],
        dates=prices.dates[1:],
        tickers=prices.tickers,
        values=values,
    )

from __future__ import annotations

import statistics

from risk.types import Returns, VolSeries


def rolling_vol(returns: Returns, window: int) -> VolSeries:
    """Each date's stdev of the earlier returns: None until `window` of them exist."""
    if window < 2:
        raise ValueError(f"window must be at least 2, got {window}")
    values = {
        ticker: tuple(
            statistics.stdev(series[:i]) if i >= window else None for i in range(len(series))
        )
        for ticker, series in returns.values.items()
    }
    return VolSeries(returns=returns, window=window, annualized=False, values=values)

def _vol(values: dict[str, tuple[float, ...]], last: float | None) -> VolSeries:
    n = len(next(iter(values.values())))
    dates = tuple(f"2024-01-{day:02d}" for day in range(3, 3 + n))
    returns = Returns(base_date="2024-01-02", dates=dates, tickers=tuple(values), values=values)
    vols = {ticker: (None,) * (n - 1) + (last,) for ticker in values}
    return VolSeries(returns=returns, window=2, annualized=True, values=vols)
