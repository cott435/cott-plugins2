from __future__ import annotations

from statistics import fmean, stdev

from risk.types import TickerSummary, VolSeries


def summary_stats(vol: VolSeries, periods_per_year: int = 252) -> tuple[TickerSummary, ...]:
    """Annualized return, volatility and Sharpe, and the last rolling volatility.

    Spine version: the daily mean and sample stdev, not yet scaled by `periods_per_year`.
    Raises ValueError with fewer than two returns.
    """
    returns = vol.returns
    if len(returns.dates) < 2:
        msg = f"summary_stats needs at least two returns, got {len(returns.dates)}"
        raise ValueError(msg)
    return tuple(_ticker_summary(vol, ticker) for ticker in returns.tickers)


def _ticker_summary(vol: VolSeries, ticker: str) -> TickerSummary:
    values = vol.returns.values[ticker]
    mean = fmean(values)
    spread = stdev(values)
    rolling = vol.values[ticker]
    return TickerSummary(
        ticker=ticker,
        ann_return=mean,
        ann_vol=spread,
        sharpe=mean / spread if spread > 0 else 0.0,
        last_vol=rolling[-1] if rolling else None,
    )
