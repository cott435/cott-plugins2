+"""PL-01 step 3: put every ticker on one calendar."""
+
+from __future__ import annotations
+
+from collections.abc import Sequence
+
+from mdata.types import Prices, RawRow
+
+
+def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
+    """Spine: the dates every ticker has a close on, no fill (`max_gap` is used by the thicken)."""
+    by_ticker: dict[str, dict[str, float]] = {}
+    for row in rows:
+        if row.close is not None:
+            by_ticker.setdefault(row.ticker, {})[row.date] = row.close
+    tickers = tuple(sorted(by_ticker))
+    shared = set.intersection(*(set(series) for series in by_ticker.values())) if tickers else set()
+    dates = tuple(sorted(shared))
+    closes = {ticker: tuple(by_ticker[ticker][date] for date in dates) for ticker in tickers}
+    return Prices(dates=dates, tickers=tickers, closes=closes)

+"""PL-01 step 1: read the `date,ticker,close` rows of a CSV."""
+
+from __future__ import annotations
+
+import csv
+import datetime as dt
+from pathlib import Path
+
+from mdata.types import RawRow
+
+COLUMNS = ("date", "ticker", "close")
+
+
+def read_csv(path: str | Path) -> list[RawRow]:
+    """Every row in file order; a blank close is None, not zero (DATA-01)."""
+    with Path(path).open(newline="", encoding="utf-8") as handle:
+        reader = csv.DictReader(handle, restval="")
+        missing = [column for column in COLUMNS if column not in (reader.fieldnames or ())]
+        if missing:
+            raise ValueError(f"{path}: missing column(s) {', '.join(missing)}")
+        return [_parse_row(record, reader.line_num) for record in reader]
+
+
+def _parse_row(record: dict[str, str], line: int) -> RawRow:
+    date = record["date"].strip()
+    if not _is_iso_date(date):
+        raise ValueError(f"line {line}: malformed date {date!r}")
+    close = record["close"].strip()
+    try:
+        value = float(close) if close else None
+    except ValueError as error:
+        raise ValueError(f"line {line}: malformed close {close!r}") from error
+    return RawRow(date=date, ticker=record["ticker"].strip(), close=value)
+
+
+def _is_iso_date(text: str) -> bool:
+    try:
+        return dt.date.fromisoformat(text).isoformat() == text
+    except ValueError:
+        return False

+
+from mdata.types import RawRow
+

"""mdata contract types: RawRow, Prices, GapError."""

from __future__ import annotations

from collections.abc import Mapping
from dataclasses import dataclass


@dataclass(frozen=True)
class RawRow:
    """One CSV row; `date` is ISO yyyy-mm-dd, `close` is None for a blank cell."""

    date: str
    ticker: str
    close: float | None


@dataclass(frozen=True)
class Prices:
    """Aligned daily closes: dates ascending, tickers sorted, one positive close per date."""

    dates: tuple[str, ...]
    tickers: tuple[str, ...]
    closes: Mapping[str, tuple[float, ...]]

    def __post_init__(self) -> None:
        if list(self.dates) != sorted(set(self.dates)):
            raise ValueError("dates must be strictly ascending")
        if list(self.tickers) != sorted(set(self.tickers)):
            raise ValueError("tickers must be sorted and unique")
        if set(self.closes) != set(self.tickers):
            raise ValueError("closes must have exactly one entry per ticker")
        for ticker in self.tickers:
            series = self.closes[ticker]
            if len(series) != len(self.dates):
                raise ValueError(f"{ticker}: one close per date required")
            if any(close <= 0 for close in series):
                raise ValueError(f"{ticker}: closes must be positive")


class GapError(ValueError):
    """A gap too long to fill."""
