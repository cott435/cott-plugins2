from __future__ import annotations

from risk import Returns, VolSeries, annualize


def _vol_sample(data: dict) -> VolSeries:
    returns = data["returns"]
    return VolSeries(
        returns=Returns(
            base_date=returns["base_date"],
            dates=tuple(returns["dates"]),
            tickers=tuple(returns["tickers"]),
            values={t: tuple(v) for t, v in returns["values"].items()},
        ),
        window=data["window"],
        annualized=data["annualized"],
        values={t: tuple(v) for t, v in data["values"].items()},
    )


def test_sample_values_pass_through(contract_sample):
    data = contract_sample("risk.VolSeries")
    out = annualize(_vol_sample(data))
    assert isinstance(out, VolSeries)
    assert {t: list(v) for t, v in out.values.items()} == data["values"]
    assert out.window == 2
    assert out.returns.dates == tuple(data["returns"]["dates"])


def test_periods_per_year_leaves_spine_values_alone(contract_sample):
    vol = _vol_sample(contract_sample("risk.VolSeries"))
    assert annualize(vol, periods_per_year=12) == vol
