# ------------------------------------------------------------------------ schema
def schema_errors(rule: dict[str, Any], value: Any, where: str = "$") -> list[str]:
    """Every way `value` breaks `rule`, as `path: message` lines (empty when it conforms)."""
    if "anyOf" in rule:
        if any(not schema_errors(option, value, where) for option in rule["anyOf"]):
            return []
        return [f"{where}: matches no anyOf option"]
    errors = _scalar_errors(rule, value, where)
    if errors:
        return errors
    if isinstance(value, dict):
        errors += _object_errors(rule, value, where)
    if isinstance(value, list):
        errors += _array_errors(rule, value, where)
    return errors


def _type_ok(name: str, value: Any) -> bool:
    if isinstance(value, bool) and name in {"integer", "number"}:
        return False
    return isinstance(value, _JSON_TYPES[name])


def _scalar_errors(rule: dict[str, Any], value: Any, where: str) -> list[str]:
    kinds = rule.get("type")
    names = [kinds] if isinstance(kinds, str) else kinds
    errors = []
    if names is not None and not any(_type_ok(n, value) for n in names):
        errors.append(f"{where}: expected {kinds}, got {type(value).__name__}")
    elif "const" in rule and value != rule["const"]:
        errors.append(f"{where}: expected {rule['const']!r}, got {value!r}")
    elif "enum" in rule and value not in rule["enum"]:
        errors.append(f"{where}: {value!r} not one of {rule['enum']}")
    elif isinstance(value, (int, float)) and not isinstance(value, bool):
        errors += _range_errors(rule, value, where)
    return errors


def _range_errors(rule: dict[str, Any], value: float, where: str) -> list[str]:
    errors = []
    if "minimum" in rule and value < rule["minimum"]:
        errors.append(f"{where}: {value} below {rule['minimum']}")
    if "maximum" in rule and value > rule["maximum"]:
        errors.append(f"{where}: {value} above {rule['maximum']}")
    if "exclusiveMinimum" in rule and value <= rule["exclusiveMinimum"]:
        errors.append(f"{where}: {value} not above {rule['exclusiveMinimum']}")
    return errors


def _object_errors(rule: dict[str, Any], value: dict[str, Any], where: str) -> list[str]:
    errors = [f"{where}: missing {key!r}" for key in rule.get("required", []) if key not in value]
    known = rule.get("properties", {})
    extra = rule.get("additionalProperties", True)
    for key, item in value.items():
        if key in known:
            errors += schema_errors(known[key], item, f"{where}.{key}")
        elif extra is False:
            errors.append(f"{where}: unexpected {key!r}")
        elif isinstance(extra, dict):
            errors += schema_errors(extra, item, f"{where}.{key}")
    return errors


def _array_errors(rule: dict[str, Any], value: list[Any], where: str) -> list[str]:
    errors = []
    if "minItems" in rule and len(value) < rule["minItems"]:
        errors.append(f"{where}: {len(value)} items, fewer than {rule['minItems']}")
    if "maxItems" in rule and len(value) > rule["maxItems"]:
        errors.append(f"{where}: {len(value)} items, more than {rule['maxItems']}")
    if "items" in rule:
        for i, item in enumerate(value):
            errors += schema_errors(rule["items"], item, f"{where}[{i}]")
    return errors

# ------------------------------------------------------------------- typed values
def build(hint: Any, data: Any) -> Any:
    """`data` (decoded JSON) as a value of type `hint`."""
    if is_dataclass(hint) and isinstance(hint, type):
        return _build_dataclass(hint, data)
    if hint is np.ndarray:
        return _build_array(data)
    if get_origin(hint) is not None:
        return _build_generic(hint, data)
    return _build_scalar(hint, data)


def _build_generic(hint: Any, data: Any) -> Any:
    origin = get_origin(hint)
    if origin is Literal:
        if data not in get_args(hint):
            raise ValueError(f"{data!r} is not one of {get_args(hint)}")
        return data
    if origin in {Union, types.UnionType}:
        return _build_union(hint, data)
    if origin is tuple:
        return _build_tuple(hint, data)
    if origin in {dict, collections.abc.Mapping}:
        value_hint = get_args(hint)[1]
        return {key: build(value_hint, item) for key, item in data.items()}
    raise TypeError(f"cannot build {hint}")


def _build_dataclass(cls: type, data: dict[str, Any]) -> Any:
    hints = get_type_hints(cls)
    names = {f.name for f in fields(cls)}
    unknown = set(data) - names
    if unknown:
        raise ValueError(f"{cls.__name__} has no fields {sorted(unknown)}")
    return cls(**{name: build(hints[name], data[name]) for name in names if name in data})


def _build_array(data: dict[str, Any]) -> np.ndarray:
    array = np.asarray(data["data"], dtype=data["dtype"])
    if array.size == 0:
        array = array.reshape(data["shape"])
    if list(array.shape) != list(data["shape"]):
        raise ValueError(f"array shape {array.shape} is not {data['shape']}")
    return array


def _build_union(hint: Any, data: Any) -> Any:
    options = [a for a in get_args(hint) if a is not type(None)]
    if data is None and len(options) < len(get_args(hint)):
        return None
    if len(options) != 1:
        raise TypeError(f"cannot build the union {hint}")
    return build(options[0], data)


def _build_tuple(hint: Any, data: list[Any]) -> tuple[Any, ...]:
    args = get_args(hint)
    if len(args) == 2 and args[1] is Ellipsis:
        return tuple(build(args[0], item) for item in data)
    if len(args) != len(data):
        raise ValueError(f"expected {len(args)} items, got {len(data)}")
    return tuple(build(a, item) for a, item in zip(args, data, strict=True))


def _build_scalar(hint: Any, data: Any) -> Any:
    if hint is Any:
        return data
    if hint is Path:
        return Path(data)
    if hint is float and isinstance(data, (int, float)) and not isinstance(data, bool):
        return float(data)
    if hint in {int, str, bool} and type(data) is hint:
        return data
    raise TypeError(f"cannot build {hint} from {data!r}")


def encode(value: Any) -> Any:
    """A typed value as JSON: the inverse of `build`."""
    if is_dataclass(value) and not isinstance(value, type):
        return {f.name: encode(getattr(value, f.name)) for f in fields(value)}
    if isinstance(value, np.ndarray):
        return {"dtype": str(value.dtype), "shape": list(value.shape), "data": value.tolist()}
    if isinstance(value, collections.abc.Mapping):
        return {str(key): encode(item) for key, item in value.items()}
    if isinstance(value, (tuple, list)):
        return [encode(item) for item in value]
    return _encode_scalar(value)


def _encode_scalar(value: Any) -> Any:
    if isinstance(value, Path):
        return str(value)
    if isinstance(value, np.generic):
        return value.item()
    return value

def plot_curves(
    groups: Mapping[str, Sequence[RunRecord]],
    *,
    scores: Mapping[str, Sequence[ScoreSet]],
    baselines: Sequence[str],
    out: Path,
) -> Chart:
    """One mean line with a min-max band per group, one horizontal line per baseline."""

    if missing:

    figure = Figure(figsize=(8, 5))

    axes.legend()
    out.parent.mkdir(parents=True, exist_ok=True)
    figure.savefig(out, format="png")
    return Chart(path=out, groups=tuple(groups), baselines=tuple(baselines))


def _band(
    runs: Sequence[RunRecord],
) -> tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
    """Steps every run evaluated at, and the mean, min and max score over the runs there."""
    common = set.intersection(*({point.step for point in run.curve} for run in runs))
    steps = np.array(sorted(common), dtype=np.int64)
    table = np.array(
        [[point.mean_score for point in run.curve if point.step in common] for run in runs],
        dtype=np.float64,
    ).reshape(len(runs), len(steps))
    return steps, table.mean(axis=0), table.min(axis=0), table.max(axis=0)


def _pooled_mean(sets: Sequence[ScoreSet]) -> float:
    """The mean over every game of every seed."""
    return float(np.concatenate([s.scores for s in sets]).mean())
