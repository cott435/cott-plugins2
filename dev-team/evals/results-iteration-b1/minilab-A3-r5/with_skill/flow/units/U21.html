import pytest

from risk.types import VolSeries

def fake_prices(monkeypatch, contract_fake, contract_sample):

def test_vol_series_runs_on_the_fake_and_keeps_the_shape(fake_prices):
    result = vol_pipeline.vol_series(FAKE_CSV, window=2)

    assert isinstance(result, VolSeries)
    assert result.window == 2
    assert result.returns.tickers == tuple(sorted(fake_prices["tickers"]))
    assert result.returns.dates == tuple(fake_prices["dates"][1:])
    assert set(result.values) == set(fake_prices["tickers"])
    for ticker in fake_prices["tickers"]:
        assert len(result.values[ticker]) == len(fake_prices["dates"]) - 1


def test_vol_series_passes_the_ticker_selection_through(fake_prices):
    chosen = sorted(fake_prices["tickers"])[0]

    result = vol_pipeline.vol_series(FAKE_CSV, window=2, tickers=[chosen])

    assert result.returns.tickers == (chosen,)
    assert set(result.values) == {chosen}
