+"""Fake `risk.vol_series`: serves the risk.VolSeries sample (two tickers, window 2, annualized)."""
+
+from __future__ import annotations
+
+import json
+from collections.abc import Sequence
+from pathlib import Path
+from typing import Any
+
+from risk import Returns, VolSeries
+
+SAMPLE = Path(__file__).resolve().parent.parent / "risk.VolSeries" / "sample.json"
+
+
+def vol_series(
+    path: str | Path, *, window: int = 20, tickers: Sequence[str] | None = None
+) -> VolSeries:
+    """The sample's series, whatever the path and window; `ValueError` if `window` < 2,
+    `KeyError` for a requested ticker it lacks."""
+    if window < 2:
+        raise ValueError(f"window must be at least 2, got {window}")
+    data = json.loads(SAMPLE.read_text(encoding="utf-8"))
+    ret: dict[str, Any] = data["returns"]
+    chosen = sorted(set(tickers)) if tickers is not None else ret["tickers"]
+    for ticker in chosen:
+        if ticker not in ret["values"]:
+            raise KeyError(ticker)
+    returns = Returns(
+        base_date=ret["base_date"],
+        dates=tuple(ret["dates"]),
+        tickers=tuple(chosen),
+        values={t: tuple(ret["values"][t]) for t in chosen},
+    )
+    return VolSeries(
+        returns=returns,
+        window=data["window"],
+        annualized=data["annualized"],
+        values={t: tuple(data["values"][t]) for t in chosen},
+    )
