from __future__ import annotations

import math
from dataclasses import dataclass
from itertools import accumulate

from risk.types import Returns


@dataclass(frozen=True)
class Drawdown:
    """The deepest fall below the running peak: `depth` ≤ 0, `peak_date` ≤ `trough_date`."""

    ticker: str
    depth: float
    peak_date: str
    trough_date: str

    def __post_init__(self) -> None:
        if not self.depth <= 0:
            raise ValueError(f"{self.ticker}: depth must be at most 0")
        if self.peak_date > self.trough_date:
            raise ValueError(f"{self.ticker}: peak_date must not come after trough_date")


def max_drawdown(returns: Returns) -> tuple[Drawdown, ...]:
    """One drawdown per ticker, wealth 1 on `base_date`; the first peak and trough if tied."""
    dates = (returns.base_date, *returns.dates)
    drawdowns = []
    for ticker in returns.tickers:
        wealth = [1.0, *(math.exp(total) for total in accumulate(returns.values[ticker]))]
        peak_i, trough_i = _deepest_fall(wealth)
        drawdowns.append(
            Drawdown(
                ticker=ticker,
                depth=wealth[trough_i] / wealth[peak_i] - 1,
                peak_date=dates[peak_i],
                trough_date=dates[trough_i],
            )
        )
    return tuple(drawdowns)


def _deepest_fall(wealth: list[float]) -> tuple[int, int]:
    """Indices of the peak and trough of the deepest fall; (0, 0) with no fall."""
    best = (0, 0)
    best_depth = 0.0
    running = 0
    for i, value in enumerate(wealth):
        if value > wealth[running]:
            running = i
        depth = value / wealth[running] - 1
        if depth < best_depth:
            best, best_depth = (running, i), depth
    return best
