+    """The Q-value MLP: one hidden ReLU layer, one output per action."""

+def greedy_actions(net: nn.Module, obs: Array) -> Array:
+    """The highest-valued action of each observation row."""
+    with torch.no_grad():

+    actions: Array = values.argmax(dim=1).numpy()

+class DQN:
+    """A `methods.Learner` updating its Q-network once per observed transition."""

+    def __init__(

+        self._updates = 0

+        """Epsilon-greedy per row; draws the same random numbers whatever it chooses."""
+        explore = rng.random(len(obs)) < self._epsilon
+        random_actions = rng.integers(self._n_actions, size=len(obs))
+        actions: Array = np.where(explore, random_actions, greedy_actions(self._net, obs))
+        return actions
+
+    def observe(self, t: Transition) -> None:
+        """One gradient step towards `r + gamma * max_a Q(s', a)` over the active rows."""
+        active = torch.as_tensor(t.active, dtype=torch.bool)
+        if not bool(active.any()):
+            return
+        obs = torch.as_tensor(t.obs, dtype=torch.float32)[active]
+        actions = torch.as_tensor(t.actions, dtype=torch.int64)[active]
+        rewards = torch.as_tensor(t.rewards, dtype=torch.float32)[active]
+        next_obs = torch.as_tensor(t.next_obs, dtype=torch.float32)[active]
+        with torch.no_grad():
+            bootstrap = 0.0 if t.done else self._gamma
+            targets = rewards + bootstrap * self._net(next_obs).max(dim=1).values
+        taken = self._net(obs).gather(1, actions.unsqueeze(1)).squeeze(1)
+        loss = nn.functional.smooth_l1_loss(taken, targets)
+        self._optimiser.zero_grad()
+        torch.autograd.backward(loss)
+        _raise_if_non_finite(loss, self._net)
+        self._optimiser.step()
+        self._updates += 1
+
+    def policy(self) -> GreedyPolicy:
+        return GreedyPolicy(copy.deepcopy(self._net))
+
+    def state_dict(self) -> dict[str, object]:
+        state: dict[str, object] = {
+            STATE_PREFIX + name: value.detach().clone()
+            for name, value in self._net.state_dict().items()
+        }
+        state["updates"] = self._updates
+        return state
+
+
+def load_greedy(

+) -> GreedyPolicy:
+    """The greedy policy of a `DQN.state_dict()`."""
+    net = q_network(obs_len, n_actions, int(settings["hidden"]))
+    weights = {

+        for name, value in state.items()

+    }
+    net.load_state_dict(weights)
+    return GreedyPolicy(net)
+
+

+    if len(bad) > 0:
+        raise NonFiniteError(float(bad[0]))
