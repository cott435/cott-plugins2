+from __future__ import annotations

+from collections.abc import Mapping, Sequence
+
+from mdata.types import GapError, Prices, RawRow

+def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
+    """Put every ticker on the dates any ticker trades, from the first date all have a close.
+
+    A missing date takes that ticker's previous close; more than `max_gap` missing dates
+    in a row raise `GapError` naming the ticker.
+    """
+    by_ticker: dict[str, dict[str, float]] = {}
+    for row in rows:
+        closes = by_ticker.setdefault(row.ticker, {})
+        if row.close is not None:
+            closes[row.date] = row.close
+    tickers = tuple(sorted(by_ticker))
+    dates = _calendar(by_ticker)
+    return Prices(
+        dates=dates,
+        tickers=tickers,
+        closes={ticker: _fill(ticker, by_ticker[ticker], dates, max_gap) for ticker in tickers},
+    )
+
+
+def _calendar(by_ticker: Mapping[str, Mapping[str, float]]) -> tuple[str, ...]:
+    """The dates any ticker has a close, from the first date on which every ticker has one."""
+    shared = (
+        set.intersection(*(set(closes) for closes in by_ticker.values())) if by_ticker else set()
+    )
+    if not shared:
+        return ()
+    start = min(shared)
+    traded = {date for closes in by_ticker.values() for date in closes}
+    return tuple(sorted(date for date in traded if date >= start))
+
+
+def _fill(
+    ticker: str, closes: Mapping[str, float], dates: Sequence[str], max_gap: int
+) -> tuple[float, ...]:
+    """One close per date, a missing date taking the previous close, up to `max_gap` in a row."""
+    filled: list[float] = []
+    missing = 0
+    for date in dates:
+        if date in closes:
+            filled.append(closes[date])
+            missing = 0
+            continue
+        missing += 1
+        if missing > max_gap:
+            msg = f"{ticker} misses more than {max_gap} dates in a row (through {date})"
+            raise GapError(msg)
+        filled.append(filled[-1])
+    return tuple(filled)
