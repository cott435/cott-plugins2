67	"""My old notebook code for volatility and drawdown (pandas). Kept for the formulas, not to import."""
68	import numpy as np
69	import pandas as pd
70	
71	
72	def daily_returns(prices: pd.DataFrame) -> pd.DataFrame:
73	    """Log returns of a dates x tickers close panel."""
74	    return np.log(prices / prices.shift(1)).dropna(how="all")
75	
76	
77	def rolling_vol(returns: pd.DataFrame, window: int = 20) -> pd.DataFrame:
78	    """Rolling volatility over the last `window` days, today included."""
79	    return returns.rolling(window).apply(lambda x: np.std(x), raw=True)
80	
81	
82	def annualize(vol: pd.DataFrame) -> pd.DataFrame:
83	    """Annualize to a calendar year, like my brokerage statements: 365 days."""
84	    return vol * np.sqrt(365)
85	
86	
87	def summary(returns: pd.DataFrame) -> pd.DataFrame:
88	    """Yearly return and volatility per ticker."""
89	    return pd.DataFrame({"ann_return": returns.mean() * 365, "ann_vol": returns.std(ddof=0) * np.sqrt(365)})
90	
91	
92	def max_drawdown(prices: pd.DataFrame) -> pd.Series:
93	    """Deepest fall below the running high."""
94	    return (prices / prices.cummax() - 1).min()
95	```

from __future__ import annotations

import pytest

from risk import Returns, VolSeries, annualize


def _returns(payload: dict) -> Returns:
    return Returns(
        base_date=payload["base_date"],
        dates=tuple(payload["dates"]),
        tickers=tuple(payload["tickers"]),
        values={k: tuple(v) for k, v in payload["values"].items()},
    )


def _vol(payload: dict) -> VolSeries:
    return VolSeries(
        returns=_returns(payload["returns"]),
        window=payload["window"],
        annualized=payload["annualized"],
        values={k: tuple(v) for k, v in payload["values"].items()},
    )


@pytest.mark.intent
def test_tc06_each_value_scaled_and_gaps_kept(golden):
    inp, expected = golden("risk", "TC-06")
    vol = _vol(inp["vol"])
    tol = expected["tol"]

    result = annualize(vol)

    assert result.annualized is expected["annualized"]
    assert set(result.values) == set(expected["values"])
    for ticker, want in expected["values"].items():
        got = result.values[ticker]
        assert len(got) == len(want)
        for g, w in zip(got, want, strict=True):
            if w is None:
                assert g is None
            else:
                assert g == pytest.approx(w, abs=tol)
