"""PL-03 step 3: maximum drawdown."""

from __future__ import annotations

import math

from risk.types import Drawdown, Returns


def max_drawdown(returns: Returns) -> tuple[Drawdown, ...]:
    """One Drawdown per ticker: the deepest fall's peak and trough dates, the first if tied.

    Spine version: `depth` is the deepest fall of the value curve `exp(cumulative log
    return)` below its running peak (starting at 1 on `base_date`), as a fraction; both
    dates are the first date. `ValueError` with no returns.
    """
    if not returns.dates:
        raise ValueError("need at least one return")
    first = returns.dates[0]
    drawdowns = []
    for ticker in returns.tickers:
        level = 0.0
        peak = 0.0
        depth = 0.0
        for value in returns.values[ticker]:
            level += value
            peak = max(peak, level)
            depth = min(depth, math.expm1(level - peak))
        drawdowns.append(Drawdown(ticker=ticker, depth=depth, peak_date=first, trough_date=first))
    return tuple(drawdowns)
