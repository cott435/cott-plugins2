from rllab.core import SOLO, ActionSpace, Env, Rewards, StepResult, TeamSpec

_DEFAULT_REWARDS = Rewards()
_AGENT_CELL = 1
_GOAL_CELL = 3


class Corridor(Env):
    """One team, `SOLO`, of one agent; actions 0 left, 1 right; one-hot position view."""

    board: Literal["grid", "tank"] = "grid"

    def __init__(
        self, *, length: int = 10, max_steps: int = 50, rewards: Rewards = _DEFAULT_REWARDS
    ) -> None:
        if length < 2:
            raise ValueError(f"corridor length must be at least 2, got {length}")
        if max_steps < 1:
            raise ValueError(f"max_steps must be at least 1, got {max_steps}")
        self.teams: Mapping[str, TeamSpec] = {
            SOLO: TeamSpec(1, (length,), ActionSpace("discrete", n=2))
        }
        self._length = length
        self._max_steps = max_steps
        self._rewards = rewards
        self._position = 0
        self._steps = 0
        self._score = 0.0
        self._over = True

    def reset(self, seed: int) -> tuple[dict[str, np.ndarray], dict[str, float]]:
        """Start at cell 0; the corridor has no randomness, so `seed` changes nothing."""
        self._position = 0
        self._steps = 0
        self._score = 0.0
        self._over = False
        return self._obs(), self._info()

    def step(self, actions: Mapping[str, np.ndarray]) -> StepResult:
        if self._over:
            raise RuntimeError("step called after the game is over; call reset first")
        move = 1 if _read_action(actions) == 1 else -1
        self._position = min(max(self._position + move, 0), self._length - 1)
        self._steps += 1
        terminated = self._position == self._length - 1
        truncated = not terminated and self._steps >= self._max_steps
        reward = self._rewards.food if terminated else 0.0
        self._score += reward
        self._over = terminated or truncated
        return StepResult(
            obs=self._obs(),
            reward={SOLO: np.array([reward], dtype=np.float32)},
            done={SOLO: np.array([terminated], dtype=np.bool_)},
            terminated=terminated,
            truncated=truncated,
            info=self._info(),
        )

    def frame(self) -> np.ndarray:
        """A `1 × length` grid: the goal in the last cell, the agent drawn over it."""
        grid = np.zeros((1, self._length), dtype=np.int8)
        grid[0, -1] = _GOAL_CELL
        grid[0, self._position] = _AGENT_CELL
        return grid

    def _obs(self) -> dict[str, np.ndarray]:
        one_hot = np.zeros((1, self._length), dtype=np.float32)
        one_hot[0, self._position] = 1.0
        return {SOLO: one_hot}

    def _info(self) -> dict[str, float]:
        return {"steps": float(self._steps), f"score:{SOLO}": self._score, "score": self._score}


def _read_action(actions: Mapping[str, np.ndarray]) -> int:
    """The one agent's action, checked: team present, shape `(1,)`, finite, 0 or 1."""
    if SOLO not in actions:

        raise ValueError(f"action for team {SOLO!r} must be 0 (left) or 1 (right), got {value}")
    return int(value)

def _act(action: float, dtype: type = np.int64) -> dict[str, np.ndarray]:

def test_frame_shows_agent_and_goal_then_agent_on_goal():

    env.step(_act(1))
    assert env.frame().tolist() == [[0, 1, 3]]
    env.step(_act(1))
    assert env.frame().tolist() == [[0, 0, 1]]


def test_right_then_left_returns_to_start():
    env = Corridor(length=4)
    obs0, _ = env.reset(0)
    env.step(_act(1))
    result = env.step(_act(0))
    assert result.obs[SOLO].tolist() == obs0[SOLO].tolist()
    assert result.info["steps"] == 2.0


def test_reaching_goal_on_the_cap_step_terminates_not_truncates():
    env = Corridor(length=3, max_steps=2)
    env.reset(0)
    env.step(_act(1))
    result = env.step(_act(1))
    assert (result.terminated, result.truncated) == (True, False)

    env = Corridor(length=2, rewards=Corridor.__init__.__kwdefaults__["rewards"])
    _, info = env.reset(0)
    assert info == {"steps": 0.0, "score:agents": 0.0, "score": 0.0}
    result = env.step(_act(1))
    assert result.info == {"steps": 1.0, "score:agents": 1.0, "score": 1.0}
    assert result.reward[SOLO].dtype == np.float32
    assert result.done[SOLO].dtype == np.bool_


def test_reset_starts_a_new_game_after_the_end():
    env = Corridor(length=2)
    env.reset(0)
    env.step(_act(1))
    obs, info = env.reset(1)
    assert int(np.argmax(obs[SOLO][0])) == 0
    assert info["score"] == 0.0
    assert env.step(_act(0)).terminated is False
