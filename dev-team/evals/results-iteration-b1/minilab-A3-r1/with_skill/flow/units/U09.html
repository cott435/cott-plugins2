78	def daily_returns(prices: pd.DataFrame) -> pd.DataFrame:
79	    """Log returns of a dates x tickers close panel."""
80	    return np.log(prices / prices.shift(1)).dropna(how="all")
81	
82	
83	def rolling_vol(returns: pd.DataFrame, window: int = 20) -> pd.DataFrame:
84	    """Rolling volatility over the last `window` days, today included."""
85	    return returns.rolling(window).apply(lambda x: np.std(x), raw=True)
86	
87	
88	def annualize(vol: pd.DataFrame) -> pd.DataFrame:
89	    """Annualize to a calendar year, like my brokerage statements: 365 days."""
90	    return vol * np.sqrt(365)
91	
92	
93	def summary(returns: pd.DataFrame) -> pd.DataFrame:
94	    """Yearly return and volatility per ticker."""
95	    return pd.DataFrame({"ann_return": returns.mean() * 365, "ann_vol": returns.std(ddof=0) * np.sqrt(365)})
96	
97	
98	def max_drawdown(prices: pd.DataFrame) -> pd.Series:
99	    """Deepest fall below the running high."""
100	    return (prices / prices.cummax() - 1).min()

from __future__ import annotations

import math

import pytest

from risk import Returns, TickerSummary, VolSeries, summary_stats


def _returns(data: dict) -> Returns:
    return Returns(
        base_date=data["base_date"],
        dates=tuple(data["dates"]),
        tickers=tuple(data["tickers"]),
        values={k: tuple(v) for k, v in data["values"].items()},
    )


def _vol(data: dict) -> VolSeries:
    return VolSeries(
        returns=_returns(data["returns"]),
        window=data["window"],
        annualized=data["annualized"],
        values={k: tuple(v) for k, v in data["values"].items()},
    )


def _truncate(data: dict, n: int) -> dict:
    returns = data["returns"]
    return {
        "returns": {
            "base_date": returns["base_date"],
            "dates": returns["dates"][:n],
            "tickers": returns["tickers"],
            "values": {k: v[:n] for k, v in returns["values"].items()},
        },
        "window": data["window"],
        "annualized": data["annualized"],
        "values": {k: v[:n] for k, v in data["values"].items()},
    }

@pytest.mark.intent
def test_tc07_zero_volatility_gives_zero_sharpe(golden):
    data, expected = golden("risk", "TC-07")
    tol = expected["tol"]
    flat = next(row for row in expected["summary"] if row["ann_vol"] == 0)
    result = {s.ticker: s for s in summary_stats(_vol(data["vol"]))}
    summary = result[flat["ticker"]]
    assert summary.ann_vol == pytest.approx(0.0, abs=tol)
    assert summary.sharpe == pytest.approx(flat["sharpe"], abs=tol)
    assert math.isfinite(summary.sharpe)


@pytest.mark.intent
def test_tc07_last_rolling_vol_reported(golden):
    data, expected = golden("risk", "TC-07")
    tol = expected["tol"]
    result = {s.ticker: s for s in summary_stats(_vol(data["vol"]))}
    for row in expected["summary"]:
        assert result[row["ticker"]].last_vol == pytest.approx(row["last_vol"], abs=tol)


@pytest.mark.intent
def test_tc07_sharpe_is_return_over_vol(golden):
