10	def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
11	    """Prices from the first date every ticker has a close; a gap takes the previous close.
12	
13	    The calendar is every date any ticker has a close on. Raises GapError naming the ticker
14	    when more than `max_gap` dates in a row are missing. A ticker with no close at all
15	    leaves no date every ticker has, so the calendar is empty.
16	    """
17	    closes_by_ticker: dict[str, dict[str, float]] = {}
18	    for row in rows:
19	        ticker_closes = closes_by_ticker.setdefault(row.ticker, {})
20	        if row.close is not None:
21	            ticker_closes[row.date] = row.close
22	    tickers = tuple(sorted(closes_by_ticker))
23	    if not all(closes_by_ticker.values()):
24	        return Prices(dates=(), tickers=tickers, closes=dict.fromkeys(tickers, ()))
25	    start = max((min(c) for c in closes_by_ticker.values()), default="")
26	    dates = tuple(sorted({d for c in closes_by_ticker.values() for d in c if d >= start}))
27	    closes = {t: _forward_fill(t, closes_by_ticker[t], dates, max_gap) for t in tickers}
28	    return Prices(dates=dates, tickers=tickers, closes=closes)

32	    ticker: str, closes: dict[str, float], dates: tuple[str, ...], max_gap: int
33	) -> tuple[float, ...]:
34	    """The ticker's close on each date, a missing date taking the previous close.
35	
36	    Raises GapError when more than `max_gap` dates in a row are missing.
37	    """
