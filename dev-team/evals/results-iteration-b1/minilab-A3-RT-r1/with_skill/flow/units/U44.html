"""PL-03 step 4: write the JSON report."""

import json
from collections.abc import Mapping, Sequence
from pathlib import Path

from risk.types import Drawdown, Report, TickerSummary


def write_report(
    summary: Sequence[TickerSummary], drawdowns: Sequence[Drawdown], out_path: str | Path
) -> Report:
    """Write `{"tickers": [...]}`, one row per ticker, sorted.

    ValueError if summary and drawdowns cover different tickers.
    """
    by_ticker = {d.ticker: d for d in drawdowns}
    summary_tickers = sorted(s.ticker for s in summary)
    if summary_tickers != sorted(d.ticker for d in drawdowns):
        raise ValueError(
            f"summary covers {summary_tickers} but drawdowns cover {sorted(by_ticker)}"
        )
    rows = tuple(
        _row(s, by_ticker[s.ticker]) for s in sorted(summary, key=lambda s: s.ticker)
    )
    path = Path(out_path)
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps({"tickers": list(rows)}, indent=1) + "\n")
    return Report(path=str(path), rows=rows)


def _row(s: TickerSummary, d: Drawdown) -> Mapping[str, object]:
    return {
        "ticker": s.ticker,
        "ann_return": s.ann_return,
        "ann_vol": s.ann_vol,
        "sharpe": s.sharpe,
        "last_vol": s.last_vol,
        "max_drawdown": d.depth,
        "peak_date": d.peak_date,
        "trough_date": d.trough_date,
    }
