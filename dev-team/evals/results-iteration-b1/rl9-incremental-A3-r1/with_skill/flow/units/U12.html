"""Intent tests for M1-T16: the random baseline player (BASE-01, TC-20)."""

import json
from pathlib import Path
from typing import Any

import numpy as np
import pytest

from rllab.baselines import RandomPlayer

ROOT = Path(__file__).resolve().parents[2]
TC20 = ROOT / "fixtures" / "rllab" / "TC-20"
SAMPLE = ROOT / "contracts" / "baselines.RandomPlayer" / "samples" / "seeded_ten.json"


def _load(path: Path) -> Any:
    return json.loads(path.read_text())


def _sample_obs() -> np.ndarray:
    first = _load(SAMPLE)["calls"][0]["obs"]
    return np.asarray(first["data"], dtype=first["dtype"]).reshape(first["shape"])


def _shares(*, explore: bool) -> np.ndarray:
    spec = _load(TC20 / "input.json")
    player = RandomPlayer(spec["n_actions"], seed=spec["seed"])
    obs = _sample_obs()
    actions = [int(player.act(obs, explore=explore)[0]) for _ in range(spec["calls"])]
    counts = np.bincount(np.asarray(actions), minlength=spec["n_actions"])
    assert counts.shape == (spec["n_actions"],)
    return counts / spec["calls"]


@pytest.mark.intent
def test_random_player_tc20_each_action_share_in_band() -> None:
    expected = _load(TC20 / "expected.json")
    shares = _shares(explore=True)
    for share in shares:
        assert expected["low"] <= share <= expected["high"]


@pytest.mark.intent
def test_random_player_uniform_even_without_explore() -> None:
    expected = _load(TC20 / "expected.json")
    shares = _shares(explore=False)
    for share in shares:
        assert expected["low"] <= share <= expected["high"]


@pytest.mark.intent
def test_random_player_returns_one_valid_action_per_agent() -> None:
    sample = _load(SAMPLE)
    n = sample["team"]["actions"]["n"]
    player = RandomPlayer(n, seed=0)
    for call in sample["calls"]:
        obs = np.asarray(call["obs"]["data"], dtype=call["obs"]["dtype"])
        out = np.asarray(player.act(obs, explore=call["explore"]))
        assert out.shape == tuple(call["returns"]["shape"])
        assert np.issubdtype(out.dtype, np.integer)
        assert 0 <= int(out[0]) < n


@pytest.mark.intent
def test_random_player_same_seed_same_actions() -> None:
    obs = _sample_obs()
    first = RandomPlayer(3, seed=7)
    second = RandomPlayer(3, seed=7)
    a = [int(first.act(obs, explore=True)[0]) for _ in range(50)]
    b = [int(second.act(obs, explore=True)[0]) for _ in range(50)]
    assert a == b
    assert len(set(a)) > 1
