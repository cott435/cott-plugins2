from __future__ import annotations

import dataclasses

import pytest

from mdata import RawRow, validate_rows


def _rows_from_triples(triples: list[list]) -> list[RawRow]:
    return [RawRow(date=d, ticker=t, close=c) for d, t, c in triples]


def _sample_rows(contract_sample) -> list[RawRow]:
    return [RawRow(**row) for row in contract_sample("mdata.RawRow")["rows"]]


@pytest.mark.intent
def test_tc01_drops_unusable_resolves_duplicates_and_sorts(golden):
    data, expected = golden("mdata", "TC-01")
    rows = _rows_from_triples(data["rows"])

    result = validate_rows(rows, data["tickers"])

    assert result == _rows_from_triples(expected["rows"])


@pytest.mark.intent
def test_tc01_result_is_a_list_of_raw_rows(golden):
    data, _ = golden("mdata", "TC-01")
    rows = _rows_from_triples(data["rows"])

    result = validate_rows(rows)

    assert isinstance(result, list)
    assert all(isinstance(row, RawRow) for row in result)


@pytest.mark.intent
def test_only_positive_closes_survive(contract_sample):
    rows = _sample_rows(contract_sample)

    result = validate_rows(rows)

    assert all(row.close is not None and row.close > 0 for row in result)
    assert len(result) == len(rows) - 1


@pytest.mark.intent
def test_later_duplicate_in_file_order_wins(contract_sample):
    rows = _sample_rows(contract_sample)
    first = rows[0]
    later = dataclasses.replace(first, close=first.close + 1.0)

    result = validate_rows([*rows, later])

    matching = [r for r in result if (r.date, r.ticker) == (first.date, first.ticker)]
    assert matching == [later]


@pytest.mark.intent
def test_rows_sorted_by_date_then_ticker(contract_sample):
    rows = list(reversed(_sample_rows(contract_sample)))

    result = validate_rows(rows)

    keys = [(r.date, r.ticker) for r in result]
    assert keys == sorted(keys)
    assert len(keys) == len(set(keys))
