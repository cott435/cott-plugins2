"""PL-03 step 3: the maximum drawdown per ticker."""

from __future__ import annotations

import math

from risk.types import Drawdown, Returns


def max_drawdown(returns: Returns) -> tuple[Drawdown, ...]:
    """The deepest fall below the running peak of wealth, with its dates (RISK-05).

    Wealth starts at 1 on `base_date` and compounds the log returns. The first peak and
    trough win a tie; with no fall, both dates are `base_date`.
    """
    dates = (returns.base_date, *returns.dates)
    return tuple(_drawdown(ticker, dates, returns.values[ticker]) for ticker in returns.tickers)


def _drawdown(ticker: str, dates: tuple[str, ...], log_returns: tuple[float, ...]) -> Drawdown:
    """Walk log wealth, tracking the running peak and the deepest fall below it."""
    log_wealth = 0.0
    peak, peak_date = 0.0, dates[0]
    worst, worst_peak_date, trough_date = 0.0, dates[0], dates[0]
    for date, r in zip(dates[1:], log_returns, strict=True):
        log_wealth += r
        if log_wealth > peak:
            peak, peak_date = log_wealth, date
        elif log_wealth - peak < worst:
            worst, worst_peak_date, trough_date = log_wealth - peak, peak_date, date
    return Drawdown(
        ticker=ticker,
        depth=math.expm1(worst),
        peak_date=worst_peak_date,
        trough_date=trough_date,
    )
