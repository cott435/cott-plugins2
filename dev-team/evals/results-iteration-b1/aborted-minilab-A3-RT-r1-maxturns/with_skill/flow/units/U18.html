+"""Unit tests for the align_calendar spine: dates every ticker has, no fill."""
+
+import json
+from pathlib import Path
+
+from mdata.calendar import align_calendar
+from mdata.types import RawRow
+
+SAMPLE = Path(__file__).parents[2] / "contracts" / "mdata.RawRow" / "sample.json"
+
+
+def _sample_rows() -> list[RawRow]:
+    data = json.loads(SAMPLE.read_text())
+    return [RawRow(**row) for row in data["rows"]]
+
+
+def test_sample_keeps_only_dates_every_ticker_has() -> None:
+    prices = align_calendar(_sample_rows())
+
+    assert prices.dates == ("2024-01-02", "2024-01-03", "2024-01-05")
+    assert prices.tickers == ("A", "B")
+    assert prices.closes == {"A": (100.0, 101.0, 100.5), "B": (50.0, 50.5, 49.75)}
+
+
+def test_tickers_sorted_and_dates_ascending_from_unordered_rows() -> None:
+    rows = [
+        RawRow("2024-02-02", "Z", 3.0),
+        RawRow("2024-02-01", "M", 2.0),
+        RawRow("2024-02-02", "M", 4.0),
+        RawRow("2024-02-01", "Z", 1.0),
+    ]
+
+    prices = align_calendar(rows)
+
+    assert prices.dates == ("2024-02-01", "2024-02-02")
+    assert prices.tickers == ("M", "Z")
+    assert prices.closes == {"M": (2.0, 4.0), "Z": (1.0, 3.0)}
+
+
+def test_date_missing_for_one_ticker_is_dropped() -> None:
+    rows = [
+        RawRow("2024-03-01", "A", 10.0),
+        RawRow("2024-03-01", "B", 20.0),
+        RawRow("2024-03-04", "A", 11.0),
+    ]
+
+    prices = align_calendar(rows)
+
+    assert prices.dates == ("2024-03-01",)
+    assert prices.closes == {"A": (10.0,), "B": (20.0,)}
