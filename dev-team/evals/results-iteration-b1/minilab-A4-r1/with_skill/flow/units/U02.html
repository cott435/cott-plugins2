  - Given tickers that start on different dates and miss some dates, when aligned, then the calendar is the dates any ticker trades from the first date on which every ticker has a close, and a missing date takes that ticker's previous close.

### Risk (`RISK`)
**Serves:** Goal 2

**Requirements**
- **RISK-01** Daily returns are log returns.
  - Given closes, when returns are computed, then r_t = ln(close_t / close_t−1), dated t.
- **RISK-02** Rolling volatility uses only earlier returns.
  - Given a window of w, when the volatility for date t is computed, then it is the sample standard deviation (n − 1) of the w returns before t, never t's own return, and there is no value until w earlier returns exist.
- **RISK-03** Volatility is annualized over trading days.
  - Given a daily volatility σ, when annualized, then the value is σ·√252.
- **RISK-04** Each ticker gets a whole-period summary.
  - Given returns, when summarized, then the annualized return is 252 × the mean, the annualized volatility is √252 × the sample standard deviation, the Sharpe ratio is their quotient (0 when the volatility is 0; no risk-free rate), and the last rolling volatility is reported.
- **RISK-05** Maximum drawdown comes with its dates.
  - Given closes, when the drawdown is computed, then wealth starts at 1 on the first date, and the result is the deepest fall below the running peak (trough / peak − 1, at most 0) with the date of that peak and the date of the trough; the first if tied. **Requirements**
- **REP-01** The risk report is a JSON file.

    """Log returns of a dates x tickers close panel."""

def rolling_vol(returns: pd.DataFrame, window: int = 20) -> pd.DataFrame:
    """Rolling volatility over the last `window` days, today included."""
    return returns.rolling(window).apply(lambda x: np.std(x), raw=True)

    """Annualize to a calendar year, like my brokerage statements: 365 days."""

"""The frozen M1 contract: each sample builds its type and holds its invariants, the samples agree

from __future__ import annotations

import math
import statistics
from datetime import date

import pytest
from mdata import Prices, RawRow
from risk import Drawdown, Report, Returns, TickerSummary, VolSeries

TOL = 1e-12
REPORT_KEYS = {
    "ticker",
    "ann_return",
    "ann_vol",
    "sharpe",
    "last_vol",
    "max_drawdown",
    "peak_date",
    "trough_date",
}


def _returns(payload: dict) -> Returns:
    return Returns(
        base_date=payload["base_date"],
        dates=tuple(payload["dates"]),
        tickers=tuple(payload["tickers"]),
        values={t: tuple(v) for t, v in payload["values"].items()},
    )


@pytest.fixture
def prices(contract_fake) -> Prices:
    return contract_fake("mdata.load_prices").load_prices("unused.csv")


@pytest.fixture
def returns(contract_sample) -> Returns:
    return _returns(contract_sample("risk.Returns"))


@pytest.fixture
def vol(contract_fake) -> VolSeries:
    return contract_fake("risk.vol_series").vol_series("unused.csv", window=2)


def test_raw_row_sample_is_eight_rows_of_two_tickers_with_one_blank(contract_sample):
    rows = [RawRow(**row) for row in contract_sample("mdata.RawRow")["rows"]]
    assert len(rows) == 8
    assert {row.ticker for row in rows} == {"AAA", "BBB"}
    assert [row.close for row in rows].count(None) == 1
    for row in rows:
        assert date.fromisoformat(row.date).isoformat() == row.date


def test_prices_sample_is_two_tickers_on_six_aligned_dates(prices):
    assert isinstance(prices, Prices)
    assert len(prices.dates) == 6
    assert list(prices.dates) == sorted(prices.dates)
    assert prices.tickers == ("AAA", "BBB")
    for ticker in prices.tickers:
        closes = prices.closes[ticker]
        assert len(closes) == len(prices.dates)
        assert all(close > 0 for close in closes)


def test_returns_sample_is_the_log_returns_of_the_prices_sample(prices, returns):
    assert returns.base_date == prices.dates[0]
    assert returns.dates == prices.dates[1:]
    assert returns.tickers == prices.tickers
    for ticker in returns.tickers:
        closes = prices.closes[ticker]
        expected = [math.log(b / a) for a, b in zip(closes, closes[1:], strict=False)]
        assert returns.values[ticker] == pytest.approx(expected, abs=TOL)


def test_vol_series_sample_is_those_returns_window_two_annualized(returns, vol):
    assert vol.returns == returns
    assert vol.window == 2
    assert vol.annualized is True
    for ticker in returns.tickers:
        r = returns.values[ticker]
        values = vol.values[ticker]
        assert len(values) == len(returns.dates)
        assert values[:2] == (None, None)
        expected = [statistics.stdev(r[i - 2 : i]) * math.sqrt(252) for i in range(2, len(r))]
        assert values[2:] == pytest.approx(expected, abs=TOL)


def _expected_summary(ticker: str, vol: VolSeries) -> dict[str, object]:
    r = vol.returns.values[ticker]
    ann_return = 252 * statistics.mean(r)
    ann_vol = math.sqrt(252) * statistics.stdev(r)
    return {
        "ticker": ticker,
        "ann_return": pytest.approx(ann_return, abs=TOL),
        "ann_vol": pytest.approx(ann_vol, abs=TOL),
        "sharpe": pytest.approx(ann_return / ann_vol, abs=TOL),
        "last_vol": pytest.approx(vol.values[ticker][-1], abs=TOL),
    }


def test_ticker_summary_sample_is_one_ticker_with_a_last_vol(contract_sample, vol):
    summary = TickerSummary(**contract_sample("risk.TickerSummary"))
    assert summary.last_vol is not None
    assert vars(summary) == _expected_summary(summary.ticker, vol)


def test_drawdown_sample_is_one_fall_and_one_without(contract_sample, prices):
    fall, flat = (Drawdown(**d) for d in contract_sample("risk.Drawdown")["drawdowns"])
    closes = dict(zip(prices.dates, prices.closes[fall.ticker], strict=True))
    assert fall.depth < 0
    assert fall.trough_date > fall.peak_date
    assert fall.depth == pytest.approx(closes[fall.trough_date] / closes[fall.peak_date] - 1, abs=TOL)
    assert flat.depth == 0
    assert flat.peak_date == flat.trough_date == prices.dates[0]
    assert list(prices.closes[flat.ticker]) == sorted(prices.closes[flat.ticker])


def test_report_sample_has_one_sorted_row_per_ticker(contract_sample, vol):
    payload = contract_sample("risk.Report")
    report = Report(path=payload["path"], rows=tuple(payload["rows"]))
    drawdowns = {d["ticker"]: d for d in contract_sample("risk.Drawdown")["drawdowns"]}
    assert [row["ticker"] for row in report.rows] == list(vol.returns.tickers)
    for row in report.rows:
        assert set(row) == REPORT_KEYS
        ticker = str(row["ticker"])
        drawdown = drawdowns[ticker]
        assert {key: row[key] for key in _expected_summary(ticker, vol)} == _expected_summary(
            ticker, vol
        )
        assert row["max_drawdown"] == pytest.approx(drawdown["depth"], abs=TOL)
        assert (row["peak_date"], row["trough_date"]) == (
            drawdown["peak_date"],
            drawdown["trough_date"],
        )
