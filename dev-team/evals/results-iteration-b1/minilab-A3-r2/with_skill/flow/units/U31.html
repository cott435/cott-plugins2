"""PL-03 step 3: maximum drawdown."""

from __future__ import annotations

from math import exp

from risk.types import Drawdown, Returns


def max_drawdown(returns: Returns) -> tuple[Drawdown, ...]:
    """Each ticker's worst fall from a peak, with its peak and trough dates."""
    return tuple(
        _worst_fall(ticker, returns.base_date, returns.dates, returns.values[ticker])
        for ticker in returns.tickers
    )


def _worst_fall(
    ticker: str, base_date: str, dates: tuple[str, ...], values: tuple[float, ...]
) -> Drawdown:
    """Largest drop of the cumulative log level below its running peak, as trough / peak - 1."""
    level = 0.0
    peak, peak_date = 0.0, base_date
    worst, worst_peak_date, worst_trough_date = 0.0, base_date, base_date
    for date, value in zip(dates, values, strict=True):
        level += value
        if level > peak:
            peak, peak_date = level, date
        elif level - peak < worst:
            worst, worst_peak_date, worst_trough_date = level - peak, peak_date, date
    return Drawdown(
        ticker=ticker,
        depth=exp(worst) - 1.0,
        peak_date=worst_peak_date,
        trough_date=worst_trough_date,
    )

from __future__ import annotations

from collections.abc import Mapping
from dataclasses import dataclass


@dataclass(frozen=True)
class Returns:
    """Daily log returns on the prices' dates after the first, which is `base_date`."""

    base_date: str
    dates: tuple[str, ...]
    tickers: tuple[str, ...]
    values: Mapping[str, tuple[float, ...]]


@dataclass(frozen=True)
class VolSeries:
    """Rolling volatility aligned to `returns.dates`; None until `window` earlier returns exist."""

    returns: Returns
    window: int
    annualized: bool
    values: Mapping[str, tuple[float | None, ...]]


@dataclass(frozen=True)
class TickerSummary:
    """One ticker's whole-period statistics."""

    ticker: str
    ann_return: float
    ann_vol: float
    sharpe: float
    last_vol: float | None


@dataclass(frozen=True)
class Drawdown:
    """A ticker's worst fall from a peak: `depth` <= 0 (trough / peak - 1).

    `peak_date` <= `trough_date`; with no fall `depth` is 0 and both dates are `base_date`.
    """

    ticker: str
    depth: float
    peak_date: str
    trough_date: str


@dataclass(frozen=True)
class Report:
    """The written report: one row per ticker, sorted; the same rows as the file at `path`."""

    path: str
    rows: tuple[Mapping[str, object], ...]
