def check_plan(configs: Sequence[RunConfig]) -> dict[str, MethodConfig]:
    """Each distinct team method, keyed by method name, with `check_settings` applied."""
    plan: dict[str, MethodConfig] = {}

def train_checks(plan: Mapping[str, MethodConfig], seed: int) -> dict[str, Learner]:
    """A learner per method, trained on its check world for the fixed check budget."""
    learners: dict[str, Learner] = {}
    for method, config in plan.items():

def _train(world: CheckWorld, learner: Learner, rng: np.random.Generator) -> None:
    """Step `world` with `learner` for `CHECK_BUDGET` world steps, resetting after each end."""
    rows = list(world.spec.teams[0].agents)
    step = world.reset(int(rng.integers(2**31)))
    for _ in range(CHECK_BUDGET):
        actions = learner.act(step.obs[rows], rng)
        after = world.step(actions)

                actions=actions,
                rewards=after.rewards[rows],
                next_obs=after.obs[rows],
                done=after.done,
                truncated=after.truncated,
                active=after.active[rows],

def score_checks(
    learners: Mapping[str, Learner], plan: Mapping[str, MethodConfig]
) -> list[CheckResult]:
    """Per method, the share of check-world states where `policy().act` is the best action."""
    results = []
    for method in plan:

        rng = np.random.default_rng(np.random.SeedSequence(SCORE_SEED))
        actions = learners[method].policy().act(world.states(), rng)
        agreement = _agreement(actions, world.best_actions(), world.spec)
        results.append(
            CheckResult(method, world.spec.world, agreement, agreement >= PASS_AGREEMENT)
        )
    return results


def _agreement(actions: Array, best: Array, spec: WorldSpec) -> float:
    """Share of rows of `actions` equal to `best` (within `BOX_TOLERANCE` per dimension)."""
    if isinstance(spec.action, Box):
        hits = np.all(np.abs(actions - best) <= BOX_TOLERANCE, axis=-1)
    else:
        hits = np.asarray(actions) == best
    return float(np.mean(hits))

C = runpy.run_path(str(Path(__file__).parents[2] / "contracts" / "contracts.py"))
DQN = MethodConfig("dqn", {"lr": 0.001})


def valid(symbol, value):
    C["validate"](C["encode"](value), C["schema"](symbol))
    return value


class FixedPolicy:
    def __init__(self, actions: Array) -> None:
        self.actions = actions

    def act(self, obs: Array, rng: np.random.Generator) -> Array:
        assert obs.shape[0] == self.actions.shape[0]
        return self.actions


class PolicyOnly:
    def __init__(self, actions) -> None:
        self._policy = FixedPolicy(np.asarray(actions))

    def policy(self) -> FixedPolicy:
        return self._policy


def test_check_plan_applies_check_settings_over_team_settings():
    config = C["sample"]("train.RunConfig", "snake_default")
    plan = check.check_plan([config, config])
    assert list(plan) == ["dqn"]
    settings = valid("methods.MethodConfig", plan["dqn"]).settings
    assert settings["hidden"] == 16
    assert settings["epsilon_decay_steps"] == 500
    assert settings["lr"] == 0.001
    assert config.methods["snake"].settings["hidden"] == 64


def test_check_plan_keeps_one_entry_per_distinct_method():

def test_train_checks_steps_the_learner_for_the_budget(monkeypatch):
    world = C["double"]("core.World")
    learner = C["double"]("methods.Learner")
    made = []

    def fake_make_learner(config, spec, team, rng):
        made.append((config, spec.world, team))
        return learner

    monkeypatch.setattr(check, "check_world", lambda method: world)
    monkeypatch.setattr(check, "make_learner", fake_make_learner)
    monkeypatch.setattr(check, "CHECK_BUDGET", 10)

    learners = check.train_checks({"dqn": DQN}, seed=3)

    assert learners == {"dqn": learner}
    assert made == [(DQN, "snake", "snake")]
    assert len(learner.observed) == 10
    assert len(world.seeds) == 3  # two full 5-step episodes, then the reset after the second
    for got, want in zip(learner.observed, learner.recorded, strict=True):
        valid("methods.Transition", got)
        np.testing.assert_array_equal(got.obs, want.obs)
        np.testing.assert_array_equal(got.next_obs, want.next_obs)
        assert got.done == want.done


def test_train_checks_is_seeded(monkeypatch):
    seeds = []
    for _ in range(2):
        world = C["double"]("core.World")
        monkeypatch.setattr(check, "check_world", lambda method, w=world: w)
        monkeypatch.setattr(check, "make_learner", lambda *a: C["double"]("methods.Learner"))
        monkeypatch.setattr(check, "CHECK_BUDGET", 5)
        check.train_checks({"dqn": DQN}, seed=7)
        seeds.append(world.seeds)

def test_score_checks_counts_states_with_the_best_action(monkeypatch):
    monkeypatch.setattr(
        check, "check_world", lambda method: C["double"]("worlds.CheckWorld", "two_exit_corridor")
    )
    learners = {"dqn": PolicyOnly([0, 0, 1, 1, 1, 1])}
    [passing] = check.score_checks(learners, {"dqn": DQN})
    assert valid("train.CheckResult", passing) == check.CheckResult(
        "dqn", "check-discrete", 1.0, True
    )

    biased = {"dqn": PolicyOnly([1, 1, 1, 1, 1, 1])}
    [failing] = check.score_checks(biased, {"dqn": DQN})
    assert failing.agreement == pytest.approx(4 / 6)
    assert not failing.passed


def test_score_checks_allows_box_actions_within_a_tenth(monkeypatch):
    monkeypatch.setattr(
        check, "check_world", lambda method: C["double"]("worlds.CheckWorld", "target_point")
    )
    ppo = MethodConfig("ppo", {"lr": 0.0003})
    near = [[0.95], [1.0], [0.91], [-0.92], [-1.0], [-0.5]]
    [result] = check.score_checks({"ppo": PolicyOnly(near)}, {"ppo": ppo})
    assert valid("train.CheckResult", result).world == "check-continuous"
    assert result.agreement == pytest.approx(5 / 6)
    assert not result.passed


def test_check_world_refuses_a_method_without_one():
    with pytest.raises(ValueError, match="ppo"):
