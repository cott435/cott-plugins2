    """Two hidden ReLU layers of `hidden` units over the flattened observation."""
    return nn.Sequential(
        nn.Flatten(),
        nn.Linear(math.prod(obs_shape), hidden),
        nn.ReLU(),
        nn.Linear(hidden, hidden),
        nn.ReLU(),
        nn.Linear(hidden, n_actions),
    )


def _cnn(obs_shape: tuple[int, ...], n_actions: int, hidden: int) -> nn.Module:
    """Two padded 3x3 ReLU convolutions (16, 32 channels), then one hidden layer of `hidden` units.

    `obs_shape` is `(channels, height, width)`; padding keeps the height and width.
    """
    channels, height, width = obs_shape
    return nn.Sequential(
        nn.Conv2d(channels, 16, kernel_size=3, padding=1),
        nn.ReLU(),
        nn.Conv2d(16, 32, kernel_size=3, padding=1),
        nn.ReLU(),
        nn.Flatten(),
        nn.Linear(32 * height * width, hidden),
        nn.ReLU(),
        nn.Linear(hidden, n_actions),
    )

import torch
from torch import nn

from rllab.agents.dqn.networks import build_net
from rllab.core import AgentParams

def _layers(net: nn.Module, kind: type[nn.Module]) -> list[nn.Module]:
    return [m for m in net.modules() if isinstance(m, kind)]


def test_cnn_gives_one_value_per_action_for_each_grid_in_the_batch():
    net = build_net((3, 10, 10), 3, CNN)
    values = net(torch.zeros(4, 3, 10, 10))
    assert tuple(values.shape) == (4, 3)


def test_cnn_keeps_the_grid_size_so_its_hidden_layer_reads_32_channels_per_cell():
    net = build_net((3, 6, 8), 5, CNN)
    first, second = _layers(net, nn.Linear)
    assert (first.in_features, first.out_features) == (32 * 6 * 8, 128)
    assert (second.in_features, second.out_features) == (128, 5)
    assert tuple(net(torch.zeros(2, 3, 6, 8)).shape) == (2, 5)


def test_cnn_takes_its_input_channels_from_the_observation():
    net = build_net((4, 5, 5), 3, CNN)
    convs = _layers(net, nn.Conv2d)
    assert [tuple(c.weight.shape) for c in convs] == [(16, 4, 3, 3), (32, 16, 3, 3)]


def test_cnn_hidden_width_follows_params_hidden():
    params = AgentParams(lr=0.001, gamma=0.99, net="cnn", hidden=64)
    first, _ = _layers(build_net((3, 10, 10), 3, params), nn.Linear)
    assert first.out_features == 64


def test_mlp_has_no_convolutions_and_two_hidden_layers():

    shapes = [(m.in_features, m.out_features) for m in _layers(net, nn.Linear)]
    assert shapes == [(11, 128), (128, 128), (128, 3)]
