def draw_curves(runs: Sequence[Run], baselines: BaselineMeans, measure: str | None) -> Figure:
    """One panel per team (or one for `measure`): mean line, min-max band, baseline lines."""
    if not runs:
        raise ValueError("no runs to chart")
    spec = runs[0].record.spec
    panels = _panels(spec, measure)
    figure = Figure(figsize=(6.4 * len(panels), 4.2), facecolor=SURFACE, layout="constrained")
    axes = figure.subplots(1, len(panels), squeeze=False)[0]
    for ax, (team, label) in zip(axes, panels, strict=True):
        steps, values = _series(runs, team, measure)
        _draw_band(ax, steps, values, f"{team}: {label}, {len(runs)} runs")
        if measure is None:
            _draw_baselines(ax, baselines, team)
        _style(ax, label)
    return figure


def _panels(spec: WorldSpec, measure: str | None) -> list[tuple[str, str]]:
    """(team, y-axis label) per panel: each team's score, or the declared measure's team."""
    if measure is None:
        return [(team.name, team.score_name) for team in spec.teams]
    declared = [m for m in spec.measures if m.name == measure]
    if not declared:
        raise ValueError(f"{spec.world} declares no measure {measure!r}")
    return [(declared[0].team, measure)]


def _series(
    runs: Sequence[Run], team: str, measure: str | None
) -> tuple[NDArray[np.int64], NDArray[np.float64]]:
    """The evaluation steps every run has, and each run's value there (runs x steps)."""
    by_run = [
        {p.step: p.scores[team] if measure is None else p.measures[measure] for p in run.evals()}
        for run in runs
    ]
    steps = sorted(set.intersection(*(set(values) for values in by_run)))
    rows = [[values[step] for step in steps] for values in by_run]
    return np.array(steps, dtype=np.int64), np.array(rows, dtype=np.float64)

        values.max(axis=0),
        color=SERIES,

        linewidth=0,
        label="min-max range",
    )
