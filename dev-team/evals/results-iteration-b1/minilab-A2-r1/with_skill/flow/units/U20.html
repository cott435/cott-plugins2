"""PL-03 step 4: write the report as JSON rows, one per ticker, sorted."""

from __future__ import annotations

import json
from collections.abc import Mapping, Sequence
from pathlib import Path

from risk.types import Drawdown, Report, TickerSummary


def write_report(
    summary: Sequence[TickerSummary], drawdowns: Sequence[Drawdown], out_path: str | Path
) -> Report:
    """Join summary and drawdown by ticker, write `{"tickers": rows}`; ValueError if they differ."""
    summary_tickers = sorted(stats.ticker for stats in summary)
    drawdown_tickers = sorted(drawdown.ticker for drawdown in drawdowns)
    if summary_tickers != drawdown_tickers:
        raise ValueError(f"summary covers {summary_tickers}, drawdowns cover {drawdown_tickers}")
    by_ticker = {drawdown.ticker: drawdown for drawdown in drawdowns}
    rows = tuple(
        _row(stats, by_ticker[stats.ticker]) for stats in sorted(summary, key=lambda s: s.ticker)
    )
    Path(out_path).write_text(json.dumps({"tickers": rows}, indent=2) + "\n")
    return Report(path=str(out_path), rows=rows)


def _row(stats: TickerSummary, drawdown: Drawdown) -> Mapping[str, object]:
    return {
        "ticker": stats.ticker,
        "ann_return": stats.ann_return,
        "ann_vol": stats.ann_vol,
        "sharpe": stats.sharpe,
        "last_vol": stats.last_vol,
        "max_drawdown": drawdown.depth,
        "peak_date": drawdown.peak_date,
    }

import csv
import datetime as dt
from pathlib import Path

from mdata.types import RawRow

COLUMNS = ("date", "ticker", "close")


def read_csv(path: str | Path) -> list[RawRow]:
    """Every row in file order; a blank close is None, not zero."""
    with Path(path).open(newline="", encoding="utf-8") as handle:
        reader = csv.DictReader(handle)
        missing = [column for column in COLUMNS if column not in (reader.fieldnames or ())]

    series = vol.returns.values[ticker]
    mean = statistics.fmean(series)
    stdev = statistics.stdev(series)
    return TickerSummary(

        ann_return=mean,
        ann_vol=stdev,
        sharpe=mean / stdev if stdev > 0 else 0.0,
        last_vol=vol.values[ticker][-1],
    )

"""PL-03 step 3: maximum drawdown (spine: depth only; M1-T12 adds its dates)."""

from __future__ import annotations

from risk.types import Drawdown, Returns


def max_drawdown(returns: Returns) -> tuple[Drawdown, ...]:
    """Each ticker's worst fall of compounded returns from a running peak, as depth ≤ 0."""
    return tuple(
        Drawdown(ticker=ticker, depth=_depth(returns.values[ticker]), peak_date=returns.base_date)
        for ticker in returns.tickers
    )


def _depth(series: tuple[float, ...]) -> float:
    wealth = peak = 1.0
    depth = 0.0
    for value in series:
        wealth *= 1.0 + value
        peak = max(peak, wealth)
        depth = min(depth, wealth / peak - 1.0)
    return depth

def test_report_rows_are_the_csv_tickers(csv, tmp_path):
    out = tmp_path / "report.json"
    report = risk.risk_report(csv, out, window=2)
    assert [row["ticker"] for row in report.rows] == ["GOOG", "IBM"]
    assert json.loads(out.read_text())["tickers"] == [dict(row) for row in report.rows]


def test_report_peak_date_is_the_csv_first_date(csv, tmp_path):
    report = risk.risk_report(csv, tmp_path / "report.json", window=2)
    goog = report.rows[0]
    assert goog["peak_date"] == "2024-01-02"
    assert goog["max_drawdown"] < 0


def test_report_narrows_to_requested_ticker(csv, tmp_path):
    report = risk.risk_report(csv, tmp_path / "report.json", window=2, tickers=["IBM"])
    assert [row["ticker"] for row in report.rows] == ["IBM"]


def test_report_rejects_a_ticker_missing_from_the_csv(csv, tmp_path):
