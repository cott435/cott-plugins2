def _load(path: Path) -> dict:
    return json.loads(path.read_text(encoding="utf-8"))


def _player(
    tmp_path: Path, name: str, run_set: dict, records: dict[Path, RunRecord]
) -> PlayerTarget:
    """One run folder per seed of `run_set`, each served by open_run as a record with its settings."""
    base = codec.sample("runs.RunRecord", "snake_incomplete", RunRecord)
    folders = []
    for seed in run_set["seeds"]:
        folder = tmp_path / f"{name}-s{seed}"
        folder.mkdir()
        (folder / "settings.json").write_text("{}", encoding="utf-8")
        settings = replace(
            base.settings,
            agent=run_set["agent"],
            ablation=run_set["ablation"],
            steps=run_set["steps"],
            seed=seed,
        )
        records[folder] = replace(base, path=folder, settings=settings)
        folders.append(folder)
    return PlayerTarget(
        name=name, targets=tuple(str(f) for f in folders), seeds=tuple(run_set["seeds"])
    )


def _targets(monkeypatch: pytest.MonkeyPatch, tmp_path: Path, case: str) -> ReportTargets:
    data = _load(TC25 / "input.json")
    records: dict[Path, RunRecord] = {}
    full = _player(tmp_path, "dqn", data["full"], records)
    ablation = _player(tmp_path, "no_target", data["ablations"][case], records)

    def open_run(path: Path) -> RunRecord:
        return records[Path(path)]

    monkeypatch.setattr(players, "open_run", open_run)
    return ReportTargets(
        players=(full, ablation),
        baselines=(),
        ladder=("dqn",),
        ablations={"no_target": "dqn"},
    )
