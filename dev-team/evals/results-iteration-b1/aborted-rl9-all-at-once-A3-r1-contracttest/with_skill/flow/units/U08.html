from pathlib import Path

from rllab.runlog import Progress


def build(fx, case: str):
    rec = fx.recording(case)
    snapshot = fx.decode(Progress, rec["progress"])

    def progress(path) -> Progress:
        if Path(path).name != rec["run"]:
            raise FileNotFoundError(f"no run at {path}")
        return snapshot

    return progress
