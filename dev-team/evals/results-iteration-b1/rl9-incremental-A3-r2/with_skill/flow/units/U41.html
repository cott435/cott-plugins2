    seeds = list(range(int(first), int(last or first) + 1))

    return seeds

def load_report_spec(path: Path) -> tuple[tuple[ClaimSpec, ...], ReportTargets]:
    """The claims and the targets a report TOML names."""

def _player(name: str, table: dict[str, Any]) -> PlayerTarget:
    """One player from its table: `runs` (one per seed) or a `baseline` name, and `seeds`."""
    seeds = tuple(table["seeds"])
    if ("runs" in table) == ("baseline" in table):
        raise ValueError(f"players.{name}: give exactly one of `runs` or `baseline`")
    if "baseline" in table:

    runs = tuple(table["runs"])
    if len(runs) != len(seeds):
        raise ValueError(f"players.{name}: {len(runs)} runs for {len(seeds)} seeds")
    return PlayerTarget(name=name, targets=runs, seeds=seeds)

def plot_curves(
    groups: Mapping[str, Sequence[RunRecord]],
    *,
    scores: Mapping[str, Sequence[ScoreSet]],
    baselines: Sequence[str],
    out: Path,
) -> Chart:
    """One mean line with a min-max band per group, one horizontal line per baseline."""
    missing = [name for name in baselines if name not in scores]
    if missing:
        raise ValueError(f"no scores for baselines {missing}")
    figure = Figure(figsize=(8, 5))
    axes = figure.add_subplot()
    for name, runs in groups.items():
        steps, mean, low, high = _band(runs)
        (line,) = axes.plot(steps, mean, label=name)
        axes.fill_between(steps, low, high, color=line.get_color(), alpha=0.2)
    for i, name in enumerate(baselines):
        axes.axhline(_pooled_mean(scores[name]), color=f"C{len(groups) + i}", ls="--", label=name)
    axes.set_xlabel("training step")
    axes.set_ylabel("mean score")
    axes.legend()
    out.parent.mkdir(parents=True, exist_ok=True)
    figure.savefig(out, format="png")
    return Chart(path=out, groups=tuple(groups), baselines=tuple(baselines))


def _band(
    runs: Sequence[RunRecord],
) -> tuple[np.ndarray, np.ndarray, np.ndarray, np.ndarray]:
    """Steps every run evaluated at, and the mean, min and max score over the runs there."""
    common = set.intersection(*({point.step for point in run.curve} for run in runs))
    steps = np.array(sorted(common), dtype=np.int64)
    table = np.array(
        [[point.mean_score for point in run.curve if point.step in common] for run in runs],
        dtype=np.float64,
    ).reshape(len(runs), len(steps))
    return steps, table.mean(axis=0), table.min(axis=0), table.max(axis=0)


def _pooled_mean(sets: Sequence[ScoreSet]) -> float:
    """The mean over every game of every seed."""
    return float(np.concatenate([s.scores for s in sets]).mean())

"""PL-04 end to end: `python -m rllab.lab report reports/smoke.toml --smoke`."""

import shutil
from pathlib import Path

import pytest

from rllab.lab.cli import main
from rllab.lab.pipelines import pl04_report
from rllab.lab.types import CurveRuns
from rllab.runs import RunRecord

pytestmark = pytest.mark.unit

SMOKE = Path(__file__).parents[3] / "reports" / "smoke.toml"
PNG_MAGIC = b"\x89PNG\r\n\x1a\n"


def test_smoke_report_writes_report_and_both_charts(tmp_path: Path) -> None:
    spec = tmp_path / "smoke.toml"
    shutil.copy(SMOKE, spec)

    assert main(["report", str(spec), "--smoke"]) == 0

    report = (tmp_path / "report.md").read_text(encoding="utf-8")
    assert "- scripted: mean" in report
    assert "- random: mean" in report
    assert "scripted greater than random: PASSED" in report
    for chart in ("ladder.png", "ablations.png"):
        assert (tmp_path / chart).read_bytes().startswith(PNG_MAGIC)


def test_smoke_report_with_unknown_player_exits_2(tmp_path: Path) -> None:
    spec = tmp_path / "smoke.toml"
    text = SMOKE.read_text(encoding="utf-8")
    spec.write_text(text.replace('b = "random"', 'b = "nobody"'), encoding="utf-8")

    assert main(["report", str(spec), "--smoke"]) == 2
    assert not (tmp_path / "report.md").exists()


def test_run_report_draws_ladder_and_ablation_groups_to_their_own_charts(
    tmp_path: Path, monkeypatch: pytest.MonkeyPatch
) -> None:
    ladder: dict[str, tuple[RunRecord, ...]] = {"dqn-compact": ()}
    ablations: dict[str, tuple[RunRecord, ...]] = {"dqn-no-memory": (), "dqn-compact": ()}
    drawn: dict[str, list[str]] = {}

    def plot_curves(groups: dict[str, object], **kwargs: object) -> None:
        out = kwargs["out"]
        assert isinstance(out, Path)
        drawn[out.name] = list(groups)
        assert kwargs["baselines"] == ("scripted", "random")

    monkeypatch.setattr(pl04_report, "score_players", lambda targets: {})
    monkeypatch.setattr(
        pl04_report, "open_curve_runs", lambda targets: CurveRuns(ladder, ablations)
    )
    monkeypatch.setattr(pl04_report, "run_claims", lambda scores, specs, out: ["claim"])
    monkeypatch.setattr(pl04_report, "plot_curves", plot_curves)

    assert pl04_report.run_report(SMOKE, out=tmp_path) == ["claim"]
    assert drawn == {
        "ladder.png": ["dqn-compact"],
        "ablations.png": ["dqn-no-memory", "dqn-compact"],
    }
