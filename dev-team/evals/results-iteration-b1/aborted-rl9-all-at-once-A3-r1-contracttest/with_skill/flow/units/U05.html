# ------------------------------------------------------------------------------- helpers
def clockwise(v):
    return (-v[1], v[0])


def turn(heading, name):
    """The head's displacement for turn `name`, rotating TC-06's up-table to `heading`."""
    v, up = tuple(TC06["turns_from_up"][name]), UP
    for _ in range(4):
        if up == tuple(heading):
            return v
        up, v = clockwise(up), clockwise(v)
    raise AssertionError(f"{heading} is not a unit heading")


def act(a):
    return np.array([a], dtype=np.int64)


def board(world):
    """(head, body cells, food cells) from the world's draw state."""
    objs = world.draw().objects
    heads = [(int(o.x), int(o.y)) for o in objs if o.kind == "head"]
    body = [(int(o.x), int(o.y)) for o in objs if o.kind == "body"]
    food = [(int(o.x), int(o.y)) for o in objs if o.kind == "food"]
    assert len(heads) == 1
    assert len(food) == 1
    return heads[0], body, food[0]


def in_grid(c):
    return 0 <= c[0] < GRID and 0 <= c[1] < GRID


def adjacent(a, b):
    return abs(a[0] - b[0]) + abs(a[1] - b[1]) == 1


def ordered_snake(head, body):
    """Head-to-tail order of the snake's cells, walking adjacent cells from the head."""
    order, rest = [head], list(body)
    while rest:
        nxt = [c for c in rest if adjacent(c, order[-1])]
        assert len(nxt) >= 1, f"snake cells are not connected: {head} {body}"
        order.append(nxt[0])
        rest.remove(nxt[0])
    return order


def start_heading(seed):
    """The start heading: the head's displacement going straight from a fresh `reset(seed)`."""
    probe = Snake()
    probe.reset(seed)
    before = board(probe)[0]
    probe.step(act(TURNS.index("straight")))
    after = board(probe)[0]
    return (after[0] - before[0], after[1] - before[1])


@dataclass
class Tracker:
    """The snake as the spec says it must move, checked against the world step by step."""

    cells: deque
    heading: tuple
    food: tuple
    eaten: int = 0
    hunger: int = 0
    t: int = 0
    reason: str | None = None
    lengths: list = field(default_factory=list)

    def expect(self, target, max_steps):
        """The outcome the spec gives for moving the head onto `target`."""
        self.t += 1
        if not in_grid(target):
            return "wall"
        if target == self.food:
            return "eat"
        if target in list(self.cells)[:-1]:
            return "body"
        if target == self.cells[-1]:
            return "tail"
        if self.hunger + 1 >= HUNGER:
            return "hunger"
        return "max_steps" if self.t >= max_steps else "move"


def check_live_step(world, tr, target, ate):
    head, body, food = board(world)
    assert head == target, "the head moves exactly one cell by the relative-turn table"
    tr.cells.appendleft(target)
    if not ate:
        tr.cells.pop()
    assert {head, *body} == set(tr.cells)
    assert len(body) + 1 == len(tr.cells)
    assert in_grid(food)
    assert food not in tr.cells, "food is placed on a free cell"
    tr.food = food


def play_step(world, tr, action, spec):
    target_turn = turn(tr.heading, TURNS[action])
    head = tr.cells[0]
    target = (head[0] + target_turn[0], head[1] + target_turn[1])
    outcome = tr.expect(target, spec.max_steps)
    step = world.step(act(action))
    assert step.obs.shape == (1, spec.obs_len)
    assert step.obs.dtype == np.float32
    if outcome in ("wall", "body", "hunger"):
        assert step.done, f"the episode ends on a {outcome}"
        assert not step.truncated, "only max_steps truncates"
        tr.reason = outcome
        return step
    if outcome == "tail" and step.done:
        tr.reason = "body"
        return step
    if outcome == "max_steps":
        assert step.done
        assert step.truncated
        tr.reason = outcome
        return step
    assert not step.done, f"no end condition holds after a {outcome}"
    ate = outcome == "eat"
    tr.eaten += ate
    tr.hunger = 0 if ate else tr.hunger + 1
    check_live_step(world, tr, target, ate)
    tr.heading = target_turn
    return step


def rollout(seed, policy):
    """One episode under `policy`, every step checked against the spec; the final tracker."""
    world = Snake()
    world.reset(seed)
    head, body, food = board(world)
    tr = Tracker(deque(ordered_snake(head, body)), start_heading(seed), food)
    rng = np.random.default_rng(seed)
    for _ in range(world.spec.max_steps + 1):
        step = play_step(world, tr, policy(tr, rng), world.spec)
        if step.done:
            break
    assert tr.reason is not None, "the episode ended"
    result = world.result()
    assert result.scores == {"snake": pytest.approx(tr.eaten, abs=TC06["tol"])}
    final_length = len(tr.cells)
    assert result.scores["snake"] == final_length - TC06["start_length"]
    assert result.measures["length"] == final_length
    with pytest.raises(RuntimeError):
        world.step(act(TURNS.index("straight")))
    return tr


def target_of(tr, action):
    d = turn(tr.heading, TURNS[action])
    return (tr.cells[0][0] + d[0], tr.cells[0][1] + d[1])


def safe(tr, action):
    c = target_of(tr, action)
    return in_grid(c) and c not in tr.cells


def random_policy(tr, rng):
    return int(rng.integers(3))


def greedy_policy(tr, rng):
    options = [a for a in (2, 0, 1) if safe(tr, a)] or [2]

    def dist(a):
        c = target_of(tr, a)
        return abs(c[0] - tr.food[0]) + abs(c[1] - tr.food[1])

    return min(options, key=dist)


def avoid_policy(tr, rng):
    options = [a for a in (2, 0, 1) if safe(tr, a) and target_of(tr, a) != tr.food]
    return (options or [a for a in (2, 0, 1) if safe(tr, a)] or [2])[0]


def coil_policy(tr, rng):
    return greedy_policy(tr, rng) if len(tr.cells) < 5 else TURNS.index("left")
