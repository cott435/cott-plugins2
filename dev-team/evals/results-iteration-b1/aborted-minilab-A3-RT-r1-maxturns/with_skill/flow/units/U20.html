+"""rolling_vol spine: expanding sample stdev of the returns before each date."""
+
+import pytest
+
+from risk import Returns, rolling_vol
+
+
+def _sample(contract_sample) -> Returns:
+    data = contract_sample("risk.Returns")
+    return Returns(
+        base_date=data["base_date"],
+        dates=tuple(data["dates"]),
+        tickers=tuple(data["tickers"]),
+        values={t: tuple(v) for t, v in data["values"].items()},
+    )
+
+
+def test_none_until_window_earlier_returns(contract_sample):
+    vol = rolling_vol(_sample(contract_sample), 2)
+    for series in vol.values.values():
+        assert len(series) == 5
+        assert series[:2] == (None, None)
+        assert all(v is not None and v >= 0 for v in series[2:])
+
+
+def test_first_value_is_stdev_of_two_earlier_returns(contract_sample):
+    vol = rolling_vol(_sample(contract_sample), 2)
+    # A: stdev of 0.00995033 and -0.01496287 is |difference| / sqrt(2).
+    assert vol.values["A"][2] == pytest.approx(0.0249132 / 2**0.5, rel=1e-5)
+
+
+def test_expanding_uses_every_earlier_return():
+    returns = Returns(
+        base_date="2024-01-01",
+        dates=("2024-01-02", "2024-01-03", "2024-01-04", "2024-01-05"),
+        tickers=("X",),
+        values={"X": (1.0, 3.0, 5.0, 100.0)},
+    )
+    vol = rolling_vol(returns, 2)
+    assert vol.values["X"][2] == pytest.approx(2**0.5)
+    assert vol.values["X"][3] == pytest.approx(2.0)
+
+
+def test_keeps_returns_and_window_unannualized(contract_sample):
+    returns = _sample(contract_sample)
+    vol = rolling_vol(returns, 3)
+    assert vol.returns is returns
+    assert vol.window == 3
+    assert vol.annualized is False
+    assert set(vol.values) == {"A", "B"}
+    assert vol.values["B"][:3] == (None, None, None)
+
+
+def test_window_below_two_is_rejected(contract_sample):
+    with pytest.raises(ValueError, match="window"):
+        rolling_vol(_sample(contract_sample), 1)
