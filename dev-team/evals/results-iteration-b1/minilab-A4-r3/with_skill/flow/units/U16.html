10	def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
11	    """Prices from the first date every ticker has a close; gaps take the previous close.
12	
13	    Raises GapError naming the ticker when more than `max_gap` dates in a row are missing.
14	    Spine version: keeps only the dates every ticker has a close on, with no fill.
15	    """
16	    by_ticker: dict[str, dict[str, float]] = {}
17	    for row in rows:
18	        closes = by_ticker.setdefault(row.ticker, {})
19	        if row.close is not None:
20	            closes[row.date] = row.close
21	    tickers = tuple(sorted(by_ticker))
22	    shared = set.intersection(*(set(by_ticker[t]) for t in tickers)) if tickers else set()
23	    dates = tuple(sorted(shared))
24	    return Prices(
25	        dates=dates,
26	        tickers=tickers,
27	        closes={t: tuple(by_ticker[t][d] for d in dates) for t in tickers},
28	    )

"""PL-01 step 3: put every ticker on one calendar."""

from __future__ import annotations

from collections.abc import Sequence

from mdata.types import GapError, Prices, RawRow


def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
    """Prices from the first date every ticker has a close; gaps take the previous close.

    The calendar is every date any ticker has a close on, from the first date on which
    every ticker has one. Raises GapError naming the ticker when more than `max_gap`
    calendar dates in a row are missing.
    """
    by_ticker: dict[str, dict[str, float]] = {}
    for row in rows:
        closes = by_ticker.setdefault(row.ticker, {})
        if row.close is not None:
            closes[row.date] = row.close
    tickers = tuple(sorted(by_ticker))
    dates = _calendar(by_ticker)
    return Prices(
        dates=dates,
        tickers=tickers,
        closes={t: _filled(t, by_ticker[t], dates, max_gap) for t in tickers},
    )


def _calendar(by_ticker: dict[str, dict[str, float]]) -> tuple[str, ...]:
    """Every date any ticker has a close on, from the first date every ticker has one."""
    if not by_ticker:
        return ()
    shared = set.intersection(*(set(closes) for closes in by_ticker.values()))
    if not shared:
        return ()
    start = min(shared)
    every = set().union(*(set(closes) for closes in by_ticker.values()))
    return tuple(sorted(d for d in every if d >= start))


def _filled(
    ticker: str, closes: dict[str, float], dates: tuple[str, ...], max_gap: int
) -> tuple[float, ...]:
    """The ticker's closes on `dates`, each missing date taking the previous close."""
    filled: list[float] = []
    missing = 0
    for date in dates:
        if date in closes:
            filled.append(closes[date])
            missing = 0
            continue
        missing += 1
        if missing > max_gap:
            raise GapError(f"{ticker} misses more than {max_gap} dates in a row from {date}")
        filled.append(filled[-1])
    return tuple(filled)
