"""Intent tests for risk.summary_stats: TC-07 and RISK-04 (whole-period summary with Sharpe)."""

from __future__ import annotations

import pytest

from risk.summary import summary_stats
from risk.types import Returns, TickerSummary, VolSeries


def _returns(payload: dict) -> Returns:
    return Returns(
        base_date=payload["base_date"],
        dates=tuple(payload["dates"]),
        tickers=tuple(payload["tickers"]),
        values={t: tuple(v) for t, v in payload["values"].items()},
    )


def _vol(payload: dict) -> VolSeries:
    return VolSeries(
        returns=_returns(payload["returns"]),
        window=payload["window"],
        annualized=payload["annualized"],
        values={t: tuple(v) for t, v in payload["values"].items()},
    )


@pytest.mark.intent
def test_tc07_whole_period_summary(golden):
    given, expected = golden("risk", "TC-07")
    vol = _vol(given["vol"])

    summary = summary_stats(vol)

    tol = expected["tol"]
    assert isinstance(summary, tuple)
    assert [s.ticker for s in summary] == [w["ticker"] for w in expected["summary"]]
    for got, want in zip(summary, expected["summary"], strict=True):
        assert isinstance(got, TickerSummary)
        assert got.ann_return == pytest.approx(want["ann_return"], abs=tol)
        assert got.ann_vol == pytest.approx(want["ann_vol"], abs=tol)
        assert got.sharpe == pytest.approx(want["sharpe"], abs=tol)
        if want["last_vol"] is None:
            assert got.last_vol is None
        else:
            assert got.last_vol == pytest.approx(want["last_vol"], abs=tol)

from __future__ import annotations

import pytest

from risk.drawdown import max_drawdown
from risk.types import Drawdown, Returns


def _returns(payload: dict) -> Returns:
    return Returns(
        base_date=payload["base_date"],
        dates=tuple(payload["dates"]),
        tickers=tuple(payload["tickers"]),
        values={t: tuple(v) for t, v in payload["values"].items()},
    )


@pytest.mark.intent
def test_tc08_drawdown_with_dates(golden):
    given, expected = golden("risk", "TC-08")
    returns = _returns(given["returns"])

    drawdowns = max_drawdown(returns)

    tol = expected["tol"]
    assert isinstance(drawdowns, tuple)
    assert [d.ticker for d in drawdowns] == [w["ticker"] for w in expected["drawdowns"]]
    for got, want in zip(drawdowns, expected["drawdowns"], strict=True):
        assert isinstance(got, Drawdown)
        assert got.depth == pytest.approx(want["depth"], abs=tol)
        assert got.peak_date == want["peak_date"]
        assert got.trough_date == want["trough_date"]


@pytest.mark.intent
def test_invariants_hold_on_contract_sample(contract_sample):
    returns = _returns(contract_sample("risk.Returns"))
    all_dates = (returns.base_date, *returns.dates)

    drawdowns = max_drawdown(returns)

    assert [d.ticker for d in drawdowns] == list(returns.tickers)
    for d in drawdowns:
        assert d.depth <= 0.0
        assert d.peak_date in all_dates
        assert d.trough_date in all_dates
        assert d.peak_date <= d.trough_date


@pytest.mark.intent
def test_no_fall_gives_zero_depth_at_the_first_date(contract_sample):
    sample = contract_sample("risk.Returns")
    rising = {
        "base_date": sample["base_date"],
        "dates": sample["dates"],
        "tickers": sample["tickers"],
