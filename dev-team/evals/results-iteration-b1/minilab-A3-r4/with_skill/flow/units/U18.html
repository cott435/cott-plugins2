"""Rolling volatility from earlier returns (RISK-02)."""

from statistics import stdev

from risk.types import Returns, VolSeries


def rolling_vol(returns: Returns, window: int) -> VolSeries:
    """Sample stdev of the `window` returns before each date. ValueError if `window` < 2.

    Spine version: the stdev of all earlier returns, once `window` of them exist.
    """
    if window < 2:
        raise ValueError(f"window must be at least 2, got {window}")
    values = {
        ticker: tuple(stdev(series[:i]) if i >= window else None for i in range(len(series)))
        for ticker, series in returns.values.items()
    }
    return VolSeries(returns=returns, window=window, annualized=False, values=values)

    """The lowest wealth / running peak - 1, with wealth 1 before the first return."""
    wealth = [1.0, *(math.exp(total) for total in accumulate(log_returns))]
