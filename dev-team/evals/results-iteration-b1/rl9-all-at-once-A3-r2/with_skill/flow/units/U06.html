42	### AD-03: DQN for discrete actions, PPO for continuous ones, one learner per team
43	**Status:** Accepted · **Date:** 2026-10-17 · **Scope:** system
44	- **Context:** Snake has 3 discrete actions (SNAKE-02); fish and predator-prey push with two numbers in [−1, 1] (FISH-02, PREY-02). Many-fish has 8 agents (FISH-06), and predator-prey has two teams learning at once (TRAIN-07). The brief wants each world to teach one idea, and wants checking against a reference implementation (CHECK-02).
45	- **Decision:** `dqn` (replay buffer, target network, ε-greedy, Huber loss) for snake and the discrete check world. `ppo` (Gaussian policy, GAE, clipped objective) for fish, predator-prey and the continuous check world. All agents of a team share one learner (parameter sharing over a batch of agents). Teams have separate learners in the same run. Evaluation and replay act deterministically: greedy for DQN, the mean action for PPO.
46	- **Alternatives:** PPO everywhere: one method fewer, but snake would not teach value learning. SAC for fish: strong on continuous control, but a third method and harder to explain for many agents. Tabular Q-learning for snake: SNAKE-06 includes length, an unbounded number, so the table would need discretising.
47	- **Consequences:** Two methods to write, each with a check world (TRAIN-06) and a Stable-Baselines3 equivalent (AD-08). A prey that leaves the episode is masked out of its team's batch.

"""Replay double of `methods.Learner`: a DQN learner's recorded acts, policy and state.

from rllab.methods import Transition


class ReplayLearner:
    def __init__(self, kit: Any, recording: dict[str, Any]) -> None:
        replay_policy = kit.module("core.Policy", "double").ReplayPolicy
        self._explore = replay_policy(recording["acts"])
        self._greedy = recording["policy"]
        self._replay_policy = replay_policy
        self._state = dict(recording["state_dict"])
        self.observed: list[Transition] = []

    def act(self, obs: Array, rng: np.random.Generator) -> Array:
        return self._explore.act(obs, rng)

    def observe(self, t: Transition) -> None:
        self.observed.append(t)

    def policy(self) -> Policy:
        frozen: Policy = self._replay_policy(self._greedy)
        return frozen

    def state_dict(self) -> dict[str, object]:
        return dict(self._state)


def build(kit: Any, recording: dict[str, Any]) -> ReplayLearner:
    return ReplayLearner(kit, recording)

def encode(value: Any) -> Any:
    """A contract value as JSON data, the inverse of `decode`."""
    if dataclasses.is_dataclass(value) and not isinstance(value, type):

def errors(instance: Any, rules: Mapping[str, Any], where: str = "$") -> list[str]:
    """Every way `instance` breaks `rules`: type, enum, const, minimum, maximum, minItems,
    maxItems, items, properties, required, additionalProperties, anyOf, and `$ref` to another
    folder's schema by name (`"$ref": "core.TeamSpec"`, or one of its properties with
    `"$ref": "core.Policy#calls"`)."""
    if "$ref" in rules:
        folder, _, prop = rules["$ref"].partition("#")
        target = schema(folder)
        return errors(instance, target["properties"][prop] if prop else target, where)
    if "anyOf" in rules:
        if all(errors(instance, option, where) for option in rules["anyOf"]):
            return [f"{where}: matches no anyOf option"]
        return []
    if "type" in rules and not _type_ok(instance, rules["type"]):
        return [f"{where}: {instance!r} is not {rules['type']}"]
    found = _value_errors(instance, rules, where)
    if isinstance(instance, list):
        found += _array_errors(instance, rules, where)
    if isinstance(instance, dict):
        found += _object_errors(instance, rules, where)
    return found


def _type_ok(instance: Any, wanted: str | list[str]) -> bool:
    names = [wanted] if isinstance(wanted, str) else wanted
    if isinstance(instance, bool):
        return "boolean" in names
    return any(isinstance(instance, _JSON_TYPES[n]) for n in names)


def _value_errors(instance: Any, rules: Mapping[str, Any], where: str) -> list[str]:
    found = []
    if "enum" in rules and instance not in rules["enum"]:
        found.append(f"{where}: {instance!r} not in {rules['enum']}")
    if "const" in rules and instance != rules["const"]:
        found.append(f"{where}: {instance!r} != {rules['const']!r}")
    if "minimum" in rules and instance < rules["minimum"]:
        found.append(f"{where}: {instance!r} < {rules['minimum']}")
    if "maximum" in rules and instance > rules["maximum"]:
        found.append(f"{where}: {instance!r} > {rules['maximum']}")
    return found


def _array_errors(instance: list[Any], rules: Mapping[str, Any], where: str) -> list[str]:
    found = []
    if len(instance) < rules.get("minItems", 0):
        found.append(f"{where}: fewer than {rules['minItems']} items")
    if len(instance) > rules.get("maxItems", len(instance)):
        found.append(f"{where}: more than {rules['maxItems']} items")
    for i, item in enumerate(instance):
        found += errors(item, rules.get("items", {}), f"{where}[{i}]")
    return found


def _object_errors(instance: dict[str, Any], rules: Mapping[str, Any], where: str) -> list[str]:
    props = rules.get("properties", {})
    found = [f"{where}: missing {k!r}" for k in rules.get("required", []) if k not in instance]
    for key, value in instance.items():
        if key in props:
            found += errors(value, props[key], f"{where}.{key}")
        elif rules.get("additionalProperties", True) is False:
            found.append(f"{where}: unexpected {key!r}")
        elif isinstance(rules.get("additionalProperties"), dict):
            found += errors(value, rules["additionalProperties"], f"{where}.{key}")
    return found

def decode(tp: Any, data: Any) -> Any:
    """`data` (parsed JSON) as an instance of the contract type `tp`; TypeError if it isn't one."""
    if isinstance(tp, typing.TypeAliasType):
        tp = tp.__value__
    if tp is Any or tp is object:
        return data
    decoder = next((d for test, d in _DECODERS if test(tp)), _decode_scalar)
    return decoder(tp, data)


def _decode_dataclass(tp: type, data: Any) -> Any:
    fields = {f.name: f for f in dataclasses.fields(tp)}
    required = {
        n
        for n, f in fields.items()
        if f.default is dataclasses.MISSING and f.default_factory is dataclasses.MISSING
    }
    if not isinstance(data, dict) or not required <= data.keys() <= fields.keys():
        raise TypeError(f"{tp.__name__} needs an object of {sorted(fields)}, got {data!r}")
    hints = typing.get_type_hints(tp)
    return tp(**{k: decode(hints[k], v) for k, v in data.items()})


def _decode_union(tp: Any, data: Any) -> Any:
    options = typing.get_args(tp)
    if data is None and type(None) in options:
        return None
    problems = []
    for option in (o for o in options if o is not type(None)):
        try:
            return decode(option, data)
        except TypeError as exc:
            problems.append(str(exc))
    raise TypeError(f"{data!r} matches no option: {problems}")


def _decode_literal(tp: Any, data: Any) -> Any:
    if data not in typing.get_args(tp):
        raise TypeError(f"{data!r} is not one of {typing.get_args(tp)}")
    return data


def _is_array(tp: Any) -> bool:

    origin = typing.get_origin(tp)
    if isinstance(origin, typing.TypeAliasType):
        origin = typing.get_origin(origin.__value__)
    return tp is np.ndarray or origin is np.ndarray


def _decode_array(_tp: Any, data: Any) -> np.ndarray[Any, Any]:
    array = np.asarray(data)
    if array.dtype.kind == "f":
        return array.astype(np.float32)
    if array.dtype.kind not in "biu":

    return array


def _decode_generic(tp: Any, data: Any) -> Any:
    origin, args = typing.get_origin(tp), typing.get_args(tp)
    is_mapping = isinstance(origin, type) and issubclass(origin, Mapping)
    if not isinstance(data, dict if is_mapping else list):
        raise TypeError(f"{tp!r} can't hold {data!r}")
    if is_mapping:
        return {decode(args[0], k): decode(args[1], v) for k, v in data.items()}
    if origin is list or (len(args) == 2 and args[1] is Ellipsis):
        items = [decode(args[0], d) for d in data]
        return items if origin is list else tuple(items)
    if len(args) != len(data):
        raise TypeError(f"{data!r} does not have {len(args)} items")
    return tuple(decode(a, d) for a, d in zip(args, data, strict=True))

import numpy as np
import torch
from torch import nn

from rllab.core import Array, Settings
from rllab.methods.types import NonFiniteError, Transition

STATE_PREFIX = "net."


def q_network(obs_len: int, n_actions: int, hidden: int) -> nn.Sequential:
    """The Q-value MLP: one hidden ReLU layer, one output per action."""
    return nn.Sequential(nn.Linear(obs_len, hidden), nn.ReLU(), nn.Linear(hidden, n_actions))


def greedy_actions(net: nn.Module, obs: Array) -> Array:
    """The highest-valued action of each observation row."""
    with torch.no_grad():
        values = net(torch.as_tensor(obs, dtype=torch.float32))
    actions: Array = values.argmax(dim=1).numpy()
    return actions


class GreedyPolicy:
    """A deterministic policy over its own Q-network; nothing else holds that network."""

    def __init__(self, net: nn.Module) -> None:
        self._net = net

    def act(self, obs: Array, rng: np.random.Generator) -> Array:
        return greedy_actions(self._net, obs)


class DQN:
    """A `methods.Learner` updating its Q-network once per observed transition."""

    def __init__(
        self, settings: Settings, obs_len: int, n_actions: int, rng: np.random.Generator
    ) -> None:
        self._gamma = float(settings["gamma"])
        self._epsilon = float(settings["epsilon"])
        self._n_actions = n_actions
        self._net = q_network(obs_len, n_actions, int(settings["hidden"]))
        _seeded_init(self._net, rng)
        self._optimiser = torch.optim.Adam(self._net.parameters(), lr=float(settings["lr"]))
        self._updates = 0

    def act(self, obs: Array, rng: np.random.Generator) -> Array:
        """Epsilon-greedy per row; draws the same random numbers whatever it chooses."""
        explore = rng.random(len(obs)) < self._epsilon
        random_actions = rng.integers(self._n_actions, size=len(obs))
        actions: Array = np.where(explore, random_actions, greedy_actions(self._net, obs))
        return actions

    def observe(self, t: Transition) -> None:
        """One gradient step towards `r + gamma * max_a Q(s', a)` over the active rows."""
        active = torch.as_tensor(t.active, dtype=torch.bool)
        if not bool(active.any()):
            return
        obs = torch.as_tensor(t.obs, dtype=torch.float32)[active]
        actions = torch.as_tensor(t.actions, dtype=torch.int64)[active]
        rewards = torch.as_tensor(t.rewards, dtype=torch.float32)[active]
        next_obs = torch.as_tensor(t.next_obs, dtype=torch.float32)[active]
        with torch.no_grad():
            bootstrap = 0.0 if t.done else self._gamma
            targets = rewards + bootstrap * self._net(next_obs).max(dim=1).values
        taken = self._net(obs).gather(1, actions.unsqueeze(1)).squeeze(1)
        loss = nn.functional.smooth_l1_loss(taken, targets)
        self._optimiser.zero_grad()
        loss.backward()
        _raise_if_non_finite(loss, self._net)
        self._optimiser.step()
        self._updates += 1

    def policy(self) -> GreedyPolicy:
        return GreedyPolicy(copy.deepcopy(self._net))

    def state_dict(self) -> dict[str, object]:
        state: dict[str, object] = {
            STATE_PREFIX + name: value.detach().clone()
            for name, value in self._net.state_dict().items()
        }
        state["updates"] = self._updates
        return state

) -> GreedyPolicy:

    net = q_network(obs_len, n_actions, int(settings["hidden"]))

        for name, value in state.items()
        if name.startswith(STATE_PREFIX)
    }
    net.load_state_dict(weights)
    return GreedyPolicy(net)

    """PyTorch's default `Linear` ranges, drawn from a generator spawned from `rng` (CR-13)."""
    generator = torch.Generator().manual_seed(int(rng.integers(2**63 - 1)))

                nn.init.uniform_(layer.weight, -bound, bound, generator=generator)
                nn.init.uniform_(layer.bias, -bound, bound, generator=generator)


def _raise_if_non_finite(loss: torch.Tensor, net: nn.Module) -> None:
    """`NonFiniteError` on a non-finite loss, parameter or gradient, before the step."""

def dqn(config: MethodConfig, seed: int = 1, **settings: float) -> DQN:
    return DQN({**config.settings, **settings}, 11, 3, np.random.default_rng(seed))


def weights(learner: DQN) -> list[torch.Tensor]:
    return [v for k, v in learner.state_dict().items() if isinstance(v, torch.Tensor)]


def q_value(learner: DQN, obs: np.ndarray[Any, Any], action: int) -> float:
    policy_net = learner.policy()._net  # the frozen copy, read only
    with torch.no_grad():
        return float(policy_net(torch.as_tensor(obs))[0, action])
