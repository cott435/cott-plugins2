64	def daily_returns(prices: pd.DataFrame) -> pd.DataFrame:

66	    return np.log(prices / prices.shift(1)).dropna(how="all")

69	def rolling_vol(returns: pd.DataFrame, window: int = 20) -> pd.DataFrame:
70	    """Rolling volatility over the last `window` days, today included."""
71	    return returns.rolling(window).apply(lambda x: np.std(x), raw=True)
72	
73	
74	def annualize(vol: pd.DataFrame) -> pd.DataFrame:
75	    """Annualize to a calendar year, like my brokerage statements: 365 days."""
76	    return vol * np.sqrt(365)
77	
78	
79	def summary(returns: pd.DataFrame) -> pd.DataFrame:
80	    """Yearly return and volatility per ticker."""
81	    return pd.DataFrame({"ann_return": returns.mean() * 365, "ann_vol": returns.std(ddof=0) * np.sqrt(365)})

84	def max_drawdown(prices: pd.DataFrame) -> pd.Series:
85	    """Deepest fall below the running high."""
