+"""PL-01 step 3: one calendar for every ticker."""
+
+from __future__ import annotations
+
+from collections.abc import Mapping, Sequence
+
+from mdata.types import GapError, Prices, RawRow
+
+
+def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
+    """Prices from the first date every ticker has a close; gaps take the previous close.
+
+    The calendar is every date any ticker has a close, from the first date on which every
+    ticker has one. `GapError` naming the ticker when more than `max_gap` dates in a row
+    are missing.
+    """
+    by_ticker: dict[str, dict[str, float]] = {}
+    for row in rows:
+        if row.close is not None:
+            by_ticker.setdefault(row.ticker, {})[row.date] = row.close
+    tickers = tuple(sorted(by_ticker))
+    if not tickers:
+        return Prices(dates=(), tickers=(), closes={})
+    start = max(min(closes) for closes in by_ticker.values())
+    all_dates = {date for closes in by_ticker.values() for date in closes}
+    dates = tuple(sorted(date for date in all_dates if date >= start))
+    closes = {ticker: _fill(ticker, by_ticker[ticker], dates, max_gap) for ticker in tickers}
+    return Prices(dates=dates, tickers=tickers, closes=closes)
+
+
+def _fill(
+    ticker: str, known: Mapping[str, float], dates: Sequence[str], max_gap: int
+) -> tuple[float, ...]:
+    """One close per date, each missing date taking the previous close; GapError past max_gap."""
+    filled: list[float] = []
+    missing = 0
+    for date in dates:
+        if date in known:
+            missing = 0
+            filled.append(known[date])
+            continue
+        missing += 1
+        if missing > max_gap:
+            msg = f"{ticker}: more than {max_gap} dates in a row missing, through {date}"
+            raise GapError(msg)
+        filled.append(filled[-1])
+    return tuple(filled)
