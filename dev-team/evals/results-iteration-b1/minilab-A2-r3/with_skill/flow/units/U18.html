from __future__ import annotations

import statistics

from risk.types import Returns, VolSeries


def rolling_vol(returns: Returns, window: int) -> VolSeries:
    """Each date's sample stdev of the `window` returns before it, never its own.

    None until `window` earlier returns exist.
    """
    if window < 2:
        raise ValueError(f"window must be at least 2, got {window}")
    values = {
        ticker: tuple(
            None if i < window else statistics.stdev(series[i - window : i])
            for i in range(len(series))
        )
        for ticker, series in returns.values.items()
    }
    return VolSeries(returns=returns, window=window, annualized=False, values=values)

import math
from itertools import pairwise

from mdata.types import Prices
from risk.types import Returns


def log_returns(prices: Prices) -> Returns:
    """r_t = ln(close_t / close_t-1) per ticker, dated t, for each date after the first."""
    if len(prices.dates) < 2:
        raise ValueError("log_returns needs at least two dates")
    values = {

        for ticker, closes in prices.closes.items()
    }

        base_date=prices.dates[0],
        dates=prices.dates[1:],
        tickers=prices.tickers,
        values=values,
    )

from __future__ import annotations

import math
from dataclasses import replace

from risk.types import VolSeries


def annualize(vol: VolSeries, periods_per_year: int = 252) -> VolSeries:
    """Each daily volatility scaled by √periods_per_year; gaps (None) are kept."""
    if vol.annualized:
        raise ValueError("vol is already annualized")
    scale = math.sqrt(periods_per_year)
    values = {
        ticker: tuple(None if v is None else v * scale for v in series)
        for ticker, series in vol.values.items()
    }
    return replace(vol, annualized=True, values=values)

  - Given tickers that start on different dates and miss some dates, when aligned, then the calendar is the dates any ticker trades from the first date on which every ticker has a close, and a missing date takes that ticker's previous close.
- **DATA-04** A long gap is an error, not a fill.
  - Given a ticker missing more than 5 calendar dates in a row, when aligned, then a `GapError` names the ticker.

### Risk (`RISK`)
**Serves:** Goal 2

**Requirements**
- **RISK-01** Daily returns are log returns.
  - Given closes, when returns are computed, then r_t = ln(close_t / close_t−1), dated t.
- **RISK-02** Rolling volatility uses only earlier returns.
  - Given a window of w, when the volatility for date t is computed, then it is the sample standard deviation (n − 1) of the w returns before t, never t's own return, and there is no value until w earlier returns exist.
- **RISK-03** Volatility is annualized over trading days.
  - Given a daily volatility σ, when annualized, then the value is σ·√252.
- **RISK-04** Each ticker gets a whole-period summary.
  - Given returns, when summarized, then the annualized return is 252 × the mean, the annualized volatility is √252 × the sample standard deviation, the Sharpe ratio is their quotient (0 when the volatility is 0; no risk-free rate), and the last rolling volatility is reported.
- **RISK-05** Maximum drawdown comes with its dates.
  - Given closes, when the drawdown is computed, then wealth starts at 1 on the first date, and the result is the deepest fall below the running peak (trough / peak − 1, at most 0) with the date of that peak and the date of the trough; the first if tied. def write_report(
    summary: Sequence[TickerSummary], drawdowns: Sequence[Drawdown], out_path: str | Path

    """Write `{"tickers": [...]}` to `out_path`: each ticker's summary and drawdown, by ticker."""

    )
    path = Path(out_path)
    path.write_text(json.dumps({"tickers": list(rows)}, indent=2) + "\n", encoding="utf-8")

Any path gives the sample; `tickers` picks a subset of it, and a ticker the sample lacks is a

raises; any other window than the sample's is a `ValueError` too, since the fake has no other.
"""

import json
from collections.abc import Sequence
from pathlib import Path

from risk.types import Returns, VolSeries

SAMPLE = Path(__file__).resolve().parent.parent / "risk.VolSeries" / "sample.json"


def vol_series(
    path: str | Path, *, window: int = 20, tickers: Sequence[str] | None = None
) -> VolSeries:
    """The sample as `VolSeries`, narrowed to `tickers` when given; `path` is ignored."""
    del path
    payload = json.loads(SAMPLE.read_text())
    if window < 2:
        raise ValueError(f"window must be at least 2, got {window}")
    if window != payload["window"]:
        raise ValueError(f"the fake serves window {payload['window']} only, got {window}")
    raw = payload["returns"]
    chosen = sorted(raw["values"]) if tickers is None else sorted(set(tickers))
    for ticker in chosen:
        if ticker not in raw["values"]:
            raise KeyError(ticker)
    returns = Returns(
        base_date=raw["base_date"],
        dates=tuple(raw["dates"]),
        tickers=tuple(chosen),
        values={ticker: tuple(raw["values"][ticker]) for ticker in chosen},
    )
    return VolSeries(
        returns=returns,
        window=payload["window"],
        annualized=payload["annualized"],
        values={ticker: tuple(payload["values"][ticker]) for ticker in chosen},
    )
