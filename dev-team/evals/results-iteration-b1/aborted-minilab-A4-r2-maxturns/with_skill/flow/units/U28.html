+    """Prices from the first date every ticker has a close; a gap takes the previous close.

+    if not closes_by_ticker or not all(closes_by_ticker.values()):
+        return ()
+    start = max(min(by_date) for by_date in closes_by_ticker.values())
+    every = set().union(*closes_by_ticker.values())
+    return tuple(sorted(d for d in every if d >= start))
