+from __future__ import annotations
+
+from collections.abc import Sequence
+
+from mdata.types import GapError, Prices, RawRow

+def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
+    """Prices from the first date every ticker has a close; gaps take the previous close.
+
+    The calendar is every date any ticker has a close on, from the first date on which
+    every ticker has one. Raises GapError naming the ticker when more than `max_gap`
+    calendar dates in a row are missing.
+    """
+    by_ticker: dict[str, dict[str, float]] = {}
+    for row in rows:
+        closes = by_ticker.setdefault(row.ticker, {})
+        if row.close is not None:
+            closes[row.date] = row.close
+    tickers = tuple(sorted(by_ticker))
+    dates = _calendar(list(by_ticker.values()))
+    return Prices(
+        dates=dates,
+        tickers=tickers,
+        closes={t: _fill(t, by_ticker[t], dates, max_gap) for t in tickers},
+    )
+
+
+def _calendar(closes: list[dict[str, float]]) -> tuple[str, ...]:
+    """Every date any ticker has a close on, from the first date every ticker has one."""
+    if not closes:
+        return ()
+    shared = set.intersection(*(set(c) for c in closes))
+    if not shared:
+        return ()
+    start = min(shared)
+    return tuple(sorted({d for c in closes for d in c if d >= start}))
+
+
+def _fill(
+    ticker: str, closes: dict[str, float], dates: tuple[str, ...], max_gap: int
+) -> tuple[float, ...]:
+    """The ticker's close on each date, a missing date taking the previous close."""
+    filled: list[float] = []
+    missing = 0
+    for date in dates:
+        if date in closes:
+            filled.append(closes[date])
+            missing = 0
+            continue
+        missing += 1
+        if missing > max_gap:
+            raise GapError(f"{ticker} misses more than {max_gap} dates in a row before {date}")
+        filled.append(filled[-1])
+    return tuple(filled)
