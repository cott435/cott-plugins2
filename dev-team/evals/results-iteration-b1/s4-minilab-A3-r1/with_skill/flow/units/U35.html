+from collections.abc import Mapping, Sequence
+
+from mdata.types import GapError, Prices, RawRow

+def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
+    """One calendar from the first date every ticker has a close (DATA-03, DATA-04).
+
+    The calendar is every date any ticker has a close on, from the first date on which
+    every ticker has one. A missing date takes that ticker's previous close; more than
+    `max_gap` missing dates in a row raise `GapError` naming the ticker.
+    """
+    by_ticker: dict[str, dict[str, float]] = {}
+    for row in rows:
+        ticker_closes = by_ticker.setdefault(row.ticker, {})
+        if row.close is not None:
+            ticker_closes[row.date] = row.close
+    tickers = tuple(sorted(by_ticker))
+    dates = _calendar(by_ticker)
+    closes = {t: _fill(t, by_ticker[t], dates, max_gap) for t in tickers}
+    return Prices(dates=dates, tickers=tickers, closes=closes)
+
+
+def _calendar(by_ticker: Mapping[str, Mapping[str, float]]) -> tuple[str, ...]:
+    """Every date any ticker has a close on, from the first date every ticker has one."""
+    if not by_ticker or not all(by_ticker.values()):
+        return ()
+    start = max(min(closes) for closes in by_ticker.values())
+    traded = {date for closes in by_ticker.values() for date in closes}
+    return tuple(sorted(date for date in traded if date >= start))
+
+
+def _fill(
+    ticker: str, closes: Mapping[str, float], dates: Sequence[str], max_gap: int
+) -> tuple[float, ...]:
+    """The ticker's close on each date, a missing one taking the previous close.
+
+    The first date may take a close from before the calendar starts; only missing
+    dates on the calendar count towards `max_gap`.
+    """
+    if not dates:
+        return ()
+    previous = closes[max(date for date in closes if date <= dates[0])]
+    filled: list[float] = []
+    missing = 0
+    for date in dates:
+        if date in closes:
+            previous = closes[date]
+            missing = 0
+        else:
+            missing += 1
+            if missing > max_gap:
+                raise GapError(f"{ticker} misses more than {max_gap} dates in a row up to {date}")
+        filled.append(previous)
+    return tuple(filled)
