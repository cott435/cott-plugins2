+"""Fake `risk.vol_series`: serves the risk.VolSeries sample (window 2), whatever the path.
+
+The sample is the annualized window-2 volatility of `../mdata.load_prices/prices.csv`.
+"""
+
+from __future__ import annotations
+
+import json
+from collections.abc import Sequence
+from pathlib import Path
+
+from risk import Returns, VolSeries
+
+SAMPLE = Path(__file__).resolve().parent.parent / "risk.VolSeries" / "sample.json"
+
+
+def vol_series(
+    path: str | Path, *, window: int = 20, tickers: Sequence[str] | None = None
+) -> VolSeries:
+    """The VolSeries sample, narrowed to `tickers`.
+
+    Raises ValueError if `window` < 2 and KeyError for an unknown ticker, as the real one does.
+    """
+    if window < 2:
+        raise ValueError(f"window must be at least 2, got {window}")
+    data = json.loads(SAMPLE.read_text(encoding="utf-8"))
+    raw = data["returns"]
+    names = tuple(raw["tickers"]) if tickers is None else tuple(sorted(tickers))
+    for name in names:
+        if name not in raw["values"]:
+            raise KeyError(name)
+    returns = Returns(
+        base_date=raw["base_date"],
+        dates=tuple(raw["dates"]),
+        tickers=names,
+        values={t: tuple(raw["values"][t]) for t in names},
+    )
+    return VolSeries(
+        returns=returns,
+        window=data["window"],
+        annualized=data["annualized"],
+        values={t: tuple(data["values"][t]) for t in names},
+    )
