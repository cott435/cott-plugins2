"""Log returns (RISK-01)."""

import math
from itertools import pairwise

from mdata import Prices
from risk.types import Returns


def log_returns(prices: Prices) -> Returns:
    """r_t = ln(close_t / close_t-1), dated t, for every ticker."""
    if len(prices.dates) < 2:
        raise ValueError(f"need at least two dates for returns, got {len(prices.dates)}")
    values = {
        ticker: tuple(
            math.log(today / yesterday) for yesterday, today in pairwise(prices.closes[ticker])
        )
        for ticker in prices.tickers
    }
    return Returns(
        base_date=prices.dates[0],
        dates=prices.dates[1:],
        tickers=prices.tickers,
        values=values,
    )

LN2 = 0.6931471805599453


def test_log_returns_dated_after_the_base_date() -> None:
    prices = Prices(
        dates=("2024-01-02", "2024-01-03", "2024-01-04"),
        tickers=("AAA", "BBB"),
        closes={"AAA": (100.0, 200.0, 100.0), "BBB": (50.0, 50.0, 100.0)},
    )

    returns = log_returns(prices)

    assert returns.base_date == "2024-01-02"
    assert returns.dates == ("2024-01-03", "2024-01-04")
    assert returns.tickers == ("AAA", "BBB")
    assert returns.values["AAA"] == pytest.approx((LN2, -LN2), abs=1e-15)
    assert returns.values["BBB"] == pytest.approx((0.0, LN2), abs=1e-15)


def test_log_returns_sum_to_log_of_total_growth() -> None:
    prices = Prices(
        dates=("d0", "d1", "d2", "d3"),
        tickers=("AAA",),
        closes={"AAA": (10.0, 13.0, 7.0, 40.0)},
    )

    returns = log_returns(prices)

    # ln(40 / 10) = 2 ln 2: log returns telescope, simple returns would not.
    assert sum(returns.values["AAA"]) == pytest.approx(2 * LN2, abs=1e-12)


def test_sample_prices_give_one_value_per_date_and_ticker() -> None:
