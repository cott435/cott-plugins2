"""Intent tests for M1-T15: mdata.align_calendar forward fill with a gap limit."""

from __future__ import annotations

import json
from pathlib import Path
from typing import Any

import pytest

from mdata import GapError, Prices, RawRow, align_calendar

FIXTURES = Path(__file__).resolve().parents[2] / "fixtures" / "mdata"


def _load(case: str, name: str) -> dict[str, Any]:
    return json.loads((FIXTURES / case / name).read_text())


def _rows(case: str) -> tuple[list[RawRow], int]:
    data = _load(case, "input.json")
    rows = [RawRow(date=d, ticker=t, close=c) for d, t, c in data["rows"]]
    return rows, data["max_gap"]


@pytest.mark.intent
def test_tc02_one_calendar_with_filled_gaps():
    rows, max_gap = _rows("TC-02")
    expected = _load("TC-02", "expected.json")

    prices = align_calendar(rows, max_gap=max_gap)

    assert isinstance(prices, Prices)
    assert prices.dates == tuple(expected["dates"])
    assert prices.tickers == tuple(expected["tickers"])
    assert set(prices.closes) == set(expected["closes"])
    for ticker, closes in expected["closes"].items():
        assert tuple(prices.closes[ticker]) == tuple(closes), ticker


@pytest.mark.intent
def test_tc02_default_max_gap_matches_fixture():
    rows, _ = _rows("TC-02")
    expected = _load("TC-02", "expected.json")

    prices = align_calendar(rows)

    assert prices.dates == tuple(expected["dates"])
    assert tuple(prices.closes["B"]) == tuple(expected["closes"]["B"])


@pytest.mark.intent
def test_tc03_six_date_gap_raises_gap_error_naming_ticker():
