def run_validate(root: Path, snapshot_id: str, crosscheck: CrossCheckSource, expected: Path) -> RunRecord:
    run = claim_run(RunKind.VALIDATE, root, snapshot_id)
    loader = open_loader(snapshot_id, root)
    tables = read_tables(loader.info)
    flags = price_flags(loader, loader.calendar)
    flags += coverage_flags(loader, loader.calendar)
    flags += adjustment_flags(loader, tables)

    return close_run(run, info)

### Entry sketch (pre-build — removed by spine task)
```python
def run_derive(root: Path, snapshot_id: str) -> Loader:
    info = check_snapshot(snapshot_id, root)
    tables = read_tables(info)
    calendar = trading_days(info)
    securities = build_identities(tables)
    securities = build_membership(securities, tables, calendar)
    securities = build_exits(securities, tables)
    actions = build_actions(tables, securities, calendar)
    prices = build_raw(tables, securities)
    prices = adjust_prices(prices, actions)
    prices = compute_returns(prices)
    derived = write_derived(securities, actions, prices, root)
    return make_loader(derived, calendar)
```
