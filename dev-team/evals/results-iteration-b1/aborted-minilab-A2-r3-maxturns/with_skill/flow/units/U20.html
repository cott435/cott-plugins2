+
+from __future__ import annotations
+

+
+from risk.types import Returns, VolSeries
+
+
+def rolling_vol(returns: Returns, window: int) -> VolSeries:
+    """Per date, the stdev of the returns before it; None until `window` of them exist."""
+    if window < 2:
+        raise ValueError("window must be at least 2")
+    values = {
+        ticker: tuple(
+            statistics.stdev(series[:index]) if index >= window else None
+            for index in range(len(series))
+        )
+        for ticker, series in returns.values.items()
+    }
+    return VolSeries(returns=returns, window=window, annualized=False, values=values)

def write_report(

    report = Report(path=str(out_path), rows=rows)

def _returns(values: dict[str, tuple[float, ...]]) -> Returns:
    return Returns(
        base_date="2024-03-04",
        dates=("2024-03-05", "2024-03-06", "2024-03-07"),
        tickers=tuple(sorted(values)),
        values=values,
    )


def test_depth_is_the_deepest_fall_below_the_running_peak():
    # wealth 1 -> 1.25 -> 1.0 -> 1.1: the deepest fall is 1.0 / 1.25 - 1 = -0.2
    returns = _returns({"AAPL": (math.log(1.25), math.log(0.8), math.log(1.1))})
