+"""The mdata contract types: one CSV row, aligned daily closes, and the gap error."""
+
+from __future__ import annotations
+
+from collections.abc import Mapping
+from dataclasses import dataclass
+
+
+@dataclass(frozen=True)
+class RawRow:
+    """One CSV row; `close` is None for a blank cell."""
+
+    date: str
+    ticker: str
+    close: float | None
+
+
+@dataclass(frozen=True)
+class Prices:
+    """Daily closes on one calendar: dates ascending, tickers sorted, one positive close per date."""
+
+    dates: tuple[str, ...]
+    tickers: tuple[str, ...]
+    closes: Mapping[str, tuple[float, ...]]
+
+    def __post_init__(self) -> None:
+        if list(self.dates) != sorted(set(self.dates)):
+            raise ValueError("dates must be unique and ascending")
+        if list(self.tickers) != sorted(set(self.tickers)):
+            raise ValueError("tickers must be unique and sorted")
+        if set(self.closes) != set(self.tickers):
+            raise ValueError("closes must have exactly one series per ticker")
+        for ticker, series in self.closes.items():
+            if len(series) != len(self.dates):
+                raise ValueError(f"{ticker}: expected one close per date")
+            if any(close <= 0 for close in series):
+                raise ValueError(f"{ticker}: closes must be positive")
+
+
+class GapError(ValueError):
+    """A ticker misses more dates in a row than the fill allows."""

+"""The mdata contract samples build their types, and the load_prices fake serves its sample."""
+
+from __future__ import annotations
+
+import pytest
+
+from mdata import GapError, Prices, RawRow
+
+
+def build_prices(payload: dict) -> Prices:
+    closes = {ticker: tuple(series) for ticker, series in payload["closes"].items()}
+    return Prices(dates=tuple(payload["dates"]), tickers=tuple(payload["tickers"]), closes=closes)
+
+
+def test_prices_sample_builds_two_tickers_over_six_dates(contract_sample):
+    prices = build_prices(contract_sample("mdata.Prices"))
+    assert prices.tickers == ("AAA", "BBB")
+    assert len(prices.dates) == 6
+    assert all(len(prices.closes[t]) == 6 for t in prices.tickers)
+
+
+def test_rawrow_sample_builds_eight_rows_with_one_blank_close(contract_sample):
+    rows = [RawRow(**row) for row in contract_sample("mdata.RawRow")["rows"]]
+    assert len(rows) == 8
+    assert {row.ticker for row in rows} == {"AAA", "BBB"}
+    assert [row for row in rows if row.close is None] == [RawRow("2024-01-03", "BBB", None)]
+
+
+@pytest.mark.parametrize(
+    ("change", "message"),
+    [
+        ({"dates": ["2024-01-03", "2024-01-02"]}, "ascending"),
+        ({"dates": ["2024-01-02", "2024-01-02"]}, "ascending"),
+        ({"tickers": ["BBB", "AAA"]}, "sorted"),
+        ({"tickers": ["AAA"]}, "one series per ticker"),
+        ({"closes": {"AAA": [1.0], "BBB": [1.0, 2.0]}}, "one close per date"),
+        ({"closes": {"AAA": [1.0, 0.0], "BBB": [1.0, 2.0]}}, "positive"),
+    ],
+)
+def test_prices_rejects_broken_invariants(change, message):
+    payload = {
+        "dates": ["2024-01-02", "2024-01-03"],
+        "tickers": ["AAA", "BBB"],
+        "closes": {"AAA": [1.0, 2.0], "BBB": [3.0, 4.0]},
+    }
+    with pytest.raises(ValueError, match=message):
+        build_prices({**payload, **change})
+
+
