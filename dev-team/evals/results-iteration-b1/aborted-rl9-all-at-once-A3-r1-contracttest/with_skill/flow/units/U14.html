def _encode_dataclass(value: Any) -> dict[str, Any]:
    out = {f.name: encode(getattr(value, f.name)) for f in dataclasses.fields(value)}
    kind = next((k for k, cls in KINDS.items() if isinstance(value, cls)), None)
    return {"kind": kind, **out} if kind else out


def encode(value: Any) -> Any:
    """The JSON form of a contract value."""
    if dataclasses.is_dataclass(value) and not isinstance(value, type):
        return _encode_dataclass(value)
    if isinstance(value, np.ndarray):
        return {"dtype": str(value.dtype), "shape": list(value.shape), "data": value.tolist()}
    if isinstance(value, Mapping):
        return {str(k): encode(v) for k, v in value.items()}
    if isinstance(value, (tuple, list)):
        return [encode(v) for v in value]
    return next((form(value) for cls, form in LEAF_FORMS if isinstance(value, cls)), value)


def _fail(where: str, why: str) -> typing.NoReturn:
    raise ValueError(f"{where or '<root>'}: {why}")


def _decode_dataclass(tp: type, data: Any, where: str) -> Any:
    if not isinstance(data, dict):
        _fail(where, f"expected an object for {tp.__name__}")
    data = dict(data)
    if tp in KINDS.values() and KINDS.get(data.pop("kind", None)) is not tp:
        _fail(where, f"expected kind {tp.__name__.lower()!r}")
    hints = typing.get_type_hints(tp)
    fields = {f.name: f for f in dataclasses.fields(tp)}
    if extra := set(data) - set(fields):
        _fail(where, f"unknown fields {sorted(extra)} for {tp.__name__}")
    missing = [
        n
        for n, f in fields.items()
        if n not in data
        and f.default is dataclasses.MISSING
        and f.default_factory is dataclasses.MISSING
    ]
    if missing:
        _fail(where, f"missing fields {missing} for {tp.__name__}")
    return tp(**{n: decode(hints[n], v, f"{where}.{n}") for n, v in data.items()})


def _decode_union(args: tuple[Any, ...], data: Any, where: str) -> Any:
    if data is None and type(None) in args:
        return None
    if isinstance(data, dict) and KINDS.get(data.get("kind", "")) in args:
        return _decode_dataclass(KINDS[data["kind"]], data, where)
    errors = []
    for arg in args:
        if arg is type(None):
            continue
        try:
            return decode(arg, data, where)
        except ValueError as exc:
            errors.append(str(exc))
    _fail(where, f"matches no member of the union: {errors}")


def _is_number(data: Any) -> bool:
    return isinstance(data, (int, float)) and not isinstance(data, bool)


SCALAR_FORMS: dict[Any, tuple[Callable[[Any], bool], Callable[[Any], Any]]] = {
    bool: (lambda d: isinstance(d, bool), bool),
    int: (lambda d: isinstance(d, int) and not isinstance(d, bool), int),
    float: (_is_number, float),
    str: (lambda d: isinstance(d, str), str),
    Path: (lambda d: isinstance(d, str), Path),
    dt.datetime: (lambda d: isinstance(d, str), dt.datetime.fromisoformat),
    dt.timedelta: (_is_number, lambda d: dt.timedelta(seconds=d)),
}


def _decode_scalar(tp: Any, data: Any, where: str) -> Any:
    if tp not in SCALAR_FORMS:
        _fail(where, f"no JSON form for {tp!r}")
    accepts, convert = SCALAR_FORMS[tp]
    if not accepts(data):
        _fail(where, f"{data!r} is not a {getattr(tp, '__name__', tp)}")
    return convert(data)


def _decode_array(data: Any, where: str) -> Any:
    if not isinstance(data, dict) or set(data) != {"dtype", "shape", "data"}:
        _fail(where, "expected an array object {dtype, shape, data}")
    return np.asarray(data["data"], dtype=np.dtype(data["dtype"])).reshape(data["shape"])


def _decode_container(origin: Any, args: tuple[Any, ...], data: Any, where: str) -> Any:
    if origin is np.ndarray:
        return _decode_array(data, where)
    if origin in (Mapping, dict):
        if not isinstance(data, dict):
            _fail(where, "expected an object")
        key, value = args or (str, Any)
        return {decode(key, k, where): decode(value, v, f"{where}.{k}") for k, v in data.items()}
    if not isinstance(data, list):
        _fail(where, "expected a list")
    if origin is list or (len(args) == 2 and args[1] is Ellipsis):
        items = [decode(args[0], v, f"{where}[{i}]") for i, v in enumerate(data)]
        return items if origin is list else tuple(items)
    if len(args) != len(data):
        _fail(where, f"expected {len(args)} items, got {len(data)}")
    pairs = enumerate(zip(args, data, strict=True))
    return tuple(decode(a, v, f"{where}[{i}]") for i, (a, v) in pairs)


def decode(tp: Any, data: Any, where: str = "") -> Any:
    """`data` (a JSON form) as a value of type `tp`, checked strictly; `ValueError` if not."""
    if isinstance(typing.get_origin(tp), typing.TypeAliasType):  # numpy's NDArray[...]
        tp = typing.get_origin(tp).__value__
    if isinstance(tp, typing.TypeAliasType):
        tp = tp.__value__
    origin, args = typing.get_origin(tp) or tp, typing.get_args(tp)
    if tp is Any or tp is object or origin is typing.Literal:
        if origin is typing.Literal and data not in args:
            _fail(where, f"{data!r} is not one of {args}")
        return data
    if origin in (typing.Union, types.UnionType):
        return _decode_union(args, data, where)
    if origin in (Mapping, dict, list, tuple, np.ndarray):
        return _decode_container(origin, args, data, where)
    if isinstance(tp, type) and dataclasses.is_dataclass(tp):
        return _decode_dataclass(tp, data, where)
    return _decode_scalar(tp, data, where)

def _check_object(data: dict[str, Any], schema: dict[str, Any], root: dict[str, Any], where: str):
    for name in schema.get("required", []):
        if name not in data:
            _fail(where, f"missing required property {name!r}")
    props = schema.get("properties", {})
    extra = schema.get("additionalProperties", True)
    for key, value in data.items():
        if key in props:
            validate(value, props[key], root, f"{where}.{key}")
        elif extra is False:
            _fail(where, f"unexpected property {key!r}")
        elif isinstance(extra, dict):
            validate(value, extra, root, f"{where}.{key}")


def _check_array(data: list[Any], schema: dict[str, Any], root: dict[str, Any], where: str):
    if len(data) < schema.get("minItems", 0):
        _fail(where, f"fewer than {schema['minItems']} items")
    if "maxItems" in schema and len(data) > schema["maxItems"]:
        _fail(where, f"more than {schema['maxItems']} items")
    prefix = schema.get("prefixItems", [])
    for i, item in enumerate(data):
        sub = prefix[i] if i < len(prefix) else schema.get("items")
        if sub is False:
            _fail(where, f"unexpected item {i}")
        if isinstance(sub, dict):
            validate(item, sub, root, f"{where}[{i}]")


def _check_number(data: float, schema: dict[str, Any], where: str):
    if "minimum" in schema and data < schema["minimum"]:
        _fail(where, f"{data} is below the minimum {schema['minimum']}")
    if "maximum" in schema and data > schema["maximum"]:
        _fail(where, f"{data} is above the maximum {schema['maximum']}")


def _matches(data: Any, schema: dict[str, Any], root: dict[str, Any], where: str) -> bool:
    try:
        validate(data, schema, root, where)
    except ValueError:
        return False
    return True


def _check_choice(data: Any, schema: dict[str, Any], root: dict[str, Any], where: str):
    for key in ("oneOf", "anyOf"):
        if key not in schema:
            continue
        passed = sum(_matches(data, sub, root, where) for sub in schema[key])
        if passed == 0 or (key == "oneOf" and passed > 1):
            _fail(where, f"{key}: {passed} of {len(schema[key])} alternatives match")


def _check_keywords(data: Any, schema: dict[str, Any], where: str):
    if "type" in schema:
        names = schema["type"] if isinstance(schema["type"], list) else [schema["type"]]
        if not any(_is_type(data, n) for n in names):
            _fail(where, f"{data!r} is not of type {names}")
    if "enum" in schema and data not in schema["enum"]:
        _fail(where, f"{data!r} is not one of {schema['enum']}")
    if "const" in schema and data != schema["const"]:
        _fail(where, f"{data!r} is not {schema['const']!r}")
    if _is_number(data):
        _check_number(data, schema, where)


def validate(data: Any, schema: dict[str, Any], root: dict[str, Any] | None = None, where=""):
    """Check `data` against the JSON Schema subset these folders use; `ValueError` if not.

    Keywords: $ref (local), type, enum, const, minimum, maximum, properties, required,
    additionalProperties, items, prefixItems, minItems, maxItems, oneOf, anyOf."""
    root = schema if root is None else root
    if "$ref" in schema:
        target: Any = root
        for part in schema["$ref"].removeprefix("#/").split("/"):
            target = target[part]
        validate(data, target, root, where)
    _check_keywords(data, schema, where)
    if isinstance(data, dict):
        _check_object(data, schema, root, where)
    if isinstance(data, list):
        _check_array(data, schema, root, where)
    _check_choice(data, schema, root, where)

def summarize(

def compare_baselines(
    team: TeamSpec,
    learned: Mapping[str, Array],
    baselines: Mapping[Literal["random", "rule"], Array],
) -> tuple[Comparison, ...]:
    """The paired permutation test against each baseline, Holm-corrected across the two."""
    matrix = np.stack(list(learned.values()))
    rng = np.random.default_rng(np.random.SeedSequence(STATS_SEED))
    tests = {name: compare(matrix, base, team.direction, rng=rng) for name, base in baselines.items()}
    p_holm = holm([p for _, p in tests.values()])
    return tuple(
        Comparison(
            baseline=name,
            team=team.name,
            learned_mean=float(matrix.mean()),
            baseline_mean=float(np.mean(baselines[name])),
            difference=difference,
            p_raw=p_raw,
            p_holm=adjusted,
            rejected=adjusted < ALPHA,
        )
        for (name, (difference, p_raw)), adjusted in zip(tests.items(), p_holm, strict=True)
    )


def holm(p_raw: Sequence[float]) -> list[float]:
    """Holm-Bonferroni adjusted p-values, in the order given."""
    m = len(p_raw)
    adjusted = [0.0] * m
    running = 0.0
    for rank, i in enumerate(sorted(range(m), key=lambda k: p_raw[k])):
        running = max(running, min(1.0, (m - rank) * p_raw[i]))
        adjusted[i] = running
    return adjusted


def collapse_step(runs: Sequence[Run], team: TeamSpec, random_mean: float) -> int | None:
    """The step of the first of the last `COLLAPSE_POINTS`+ eval points where the runs' mean
    eval score is no better than `random_mean`; None if the tail is better."""
    by_step: dict[int, list[float]] = {}
    for run in runs:
        for point in run.evals():
            by_step.setdefault(point.step, []).append(point.scores[team.name])
    first: int | None = None
    count = 0
    for step in sorted(by_step, reverse=True):
        if not _no_better(float(np.mean(by_step[step])), random_mean, team.direction):
            break
        first, count = step, count + 1
    return first if count >= COLLAPSE_POINTS else None


def _no_better(score: float, baseline: float, direction: Direction) -> bool:
    return score <= baseline if direction == "higher" else score >= baseline

    return path
