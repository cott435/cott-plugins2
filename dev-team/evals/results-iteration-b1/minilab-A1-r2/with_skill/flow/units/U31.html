10	def max_drawdown(returns: Returns) -> tuple[Drawdown, ...]:
11	    """One Drawdown per ticker: the deepest fall's peak and trough dates, the first if tied.
12	
13	    Spine version: `depth` is the deepest fall of the value curve `exp(cumulative log
14	    return)` below its running peak (starting at 1 on `base_date`), as a fraction; both
15	    dates are the first date. `ValueError` with no returns.
16	    """
17	    if not returns.dates:

19	    first = returns.dates[0]
20	    drawdowns = []
21	    for ticker in returns.tickers:
22	        level = 0.0
23	        peak = 0.0
24	        depth = 0.0
25	        for value in returns.values[ticker]:
26	            level += value
27	            peak = max(peak, level)
28	            depth = min(depth, math.expm1(level - peak))
29	        drawdowns.append(Drawdown(ticker=ticker, depth=depth, peak_date=first, trough_date=first))
30	    return tuple(drawdowns)

"""PL-03 step 3: maximum drawdown."""

from __future__ import annotations

import math

from risk.types import Drawdown, Returns


def max_drawdown(returns: Returns) -> tuple[Drawdown, ...]:
    """One Drawdown per ticker: the deepest fall's peak and trough dates, the first if tied.

    Wealth `exp(cumulative log return)` starts at 1 on `base_date`; `depth` is the deepest
    fall below its running peak (trough / peak - 1). With no fall, `depth` is 0 and both
    dates are `base_date`. `ValueError` with no returns.
    """
    if not returns.dates:
        raise ValueError("need at least one return")
    return tuple(_ticker_drawdown(returns, ticker) for ticker in returns.tickers)


def _ticker_drawdown(returns: Returns, ticker: str) -> Drawdown:
    """The deepest fall of one ticker's wealth, with the first peak and trough reaching it."""
    level = peak = depth = 0.0
    peak_date = best_peak = best_trough = returns.base_date
    for date, value in zip(returns.dates, returns.values[ticker], strict=True):
        level += value
        if level > peak:
            peak, peak_date = level, date
        fall = math.expm1(level - peak)
        if fall < depth:
            depth, best_peak, best_trough = fall, peak_date, date
    return Drawdown(ticker=ticker, depth=depth, peak_date=best_peak, trough_date=best_trough)

    dates = tuple(f"d{i}" for i in range(1, len(values) + 1))
