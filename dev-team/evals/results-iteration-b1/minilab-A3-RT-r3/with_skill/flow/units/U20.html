"""PL-03 step 2: whole-period summary per ticker."""

from __future__ import annotations

from statistics import fmean, stdev

from risk.types import TickerSummary, VolSeries


def summary_stats(vol: VolSeries, periods_per_year: int = 252) -> tuple[TickerSummary, ...]:
    """Annualized return, volatility and Sharpe of `vol.returns`, with the last rolling vol.

    Raises ValueError with fewer than two returns.
    """
    if len(vol.returns.dates) < 2:
        msg = f"summary_stats needs at least two returns, got {len(vol.returns.dates)}"
        raise ValueError(msg)
    return tuple(_summarize(vol, ticker) for ticker in vol.returns.tickers)


def _summarize(vol: VolSeries, ticker: str) -> TickerSummary:
    """Spine: daily mean and sample stdev, Sharpe as their ratio (0.0 with no spread)."""
    returns = vol.returns.values[ticker]
    mean = fmean(returns)
    spread = stdev(returns)
    return TickerSummary(
        ticker=ticker,
        ann_return=mean,
        ann_vol=spread,
        sharpe=mean / spread if spread else 0.0,
        last_vol=vol.values[ticker][-1],
    )

import pytest

from risk.summary import summary_stats
from risk.types import Returns, TickerSummary, VolSeries


def _vol(contract_sample) -> VolSeries:
    raw = contract_sample("risk.VolSeries")
    r = raw["returns"]
    returns = Returns(
        base_date=r["base_date"],
        dates=tuple(r["dates"]),
        tickers=tuple(r["tickers"]),
        values={t: tuple(v) for t, v in r["values"].items()},
    )
    return VolSeries(
        returns=returns,
        window=raw["window"],
        annualized=raw["annualized"],
        values={t: tuple(v) for t, v in raw["values"].items()},
    )


def _series(values: dict[str, tuple[float, ...]]) -> VolSeries:
    dates = tuple(f"2024-03-{d:02d}" for d in range(2, 2 + len(next(iter(values.values())))))
    returns = Returns("2024-03-01", dates, tuple(values), values)
    return VolSeries(returns, 2, False, {t: (None,) * len(dates) for t in values})


def test_sample_gives_one_summary_per_ticker_in_order(contract_sample):
    out = summary_stats(_vol(contract_sample))

    assert [s.ticker for s in out] == ["AAPL", "MSFT"]
    assert all(isinstance(s, TickerSummary) for s in out)


def test_sample_daily_mean_stdev_and_last_vol(contract_sample):
    aapl, msft = summary_stats(_vol(contract_sample))

    # AAPL: sum 0.0069205 / 5; sample stdev 0.269455 / sqrt(252) from the TickerSummary sample.
    assert aapl.ann_return == pytest.approx(0.0013841, abs=1e-7)
    assert aapl.ann_vol == pytest.approx(0.016974, abs=1e-6)
    assert aapl.sharpe == pytest.approx(0.0013841 / 0.016974, rel=1e-3)
    assert aapl.last_vol == pytest.approx(0.252872)
    assert msft.ann_return == pytest.approx(0.00591176, abs=1e-8)
    assert msft.last_vol == pytest.approx(0.047833)


def test_hand_computed_case_uses_the_sample_stdev():
    # mean 2, sample variance ((-1)^2 + 0 + 1^2) / 2 = 1
    (s,) = summary_stats(_series({"X": (1.0, 2.0, 3.0)}))

    assert s == TickerSummary("X", 2.0, 1.0, 2.0, None)


def test_flat_returns_give_zero_vol_and_zero_sharpe():
    (s,) = summary_stats(_series({"X": (0.01, 0.01)}))

    assert s.ann_vol == 0.0
    assert s.sharpe == 0.0


def test_fewer_than_two_returns_raise():
    with pytest.raises(ValueError, match="at least two returns"):
        summary_stats(_series({"X": (0.01,)}))
