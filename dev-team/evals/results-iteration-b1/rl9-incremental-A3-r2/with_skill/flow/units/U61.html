+The fake raises what the contract names: `ValueError` naming the run for a run that is not a
+complete reference run, `FileNotFoundError` when the team's checkpoint file is missing; else it

+from rllab.core import SOLO, Agent

+def make_from_run(player: Agent) -> Callable[..., Agent]:
+    """A `from_run` returning `player` for a complete reference run with the team's checkpoint."""
+
+    def from_run(record: RunRecord, *, team: str = SOLO) -> Agent:
+        if record.settings.agent != "reference":
+            raise ValueError(f"run {record.path.name} is not a reference run")
+        if record.status != "complete":
+            raise ValueError(f"run {record.path.name} is {record.status}, not complete")
+        checkpoint = record.checkpoints.get(team)
+        if checkpoint is None or not checkpoint.is_file():
+            raise FileNotFoundError(f"run {record.path.name}: no checkpoint for team {team}")
+        return player
+
+    return from_run
+
+
+def check_from_run(from_run: Callable[..., Agent], record: RunRecord) -> Agent:
+    """`record`, a complete reference run, loads as an Agent; the refused runs raise."""
+    player = from_run(record)
+    assert Agent in type(player).__mro__
+    name = re.escape(record.path.name)
+    with pytest.raises(ValueError, match=name):
+        from_run(replace(record, settings=replace(record.settings, agent="dqn")))
+    with pytest.raises(ValueError, match=name):
+        from_run(replace(record, status="incomplete"))
+    missing = {SOLO: record.path / "no-such-checkpoint.zip"}
+    with pytest.raises(FileNotFoundError):
+        from_run(replace(record, checkpoints=missing))
+    return player
