from __future__ import annotations

import json
from collections.abc import Callable
from pathlib import Path
from types import ModuleType

import pytest

import risk.report_pipeline
from risk import risk_report


@pytest.fixture
def fake(monkeypatch: pytest.MonkeyPatch, contract_fake: Callable[[str], ModuleType]) -> ModuleType:
    module = contract_fake("risk.vol_series")
    monkeypatch.setattr(risk.report_pipeline, "vol_series", module.vol_series)
    return module


def test_risk_report_runs_summary_drawdown_write(fake: ModuleType, tmp_path: Path) -> None:
    out = tmp_path / "report.json"

    report = risk_report(fake.FIXTURE, out, window=2)

    assert report.path == str(out)
    assert [row["ticker"] for row in report.rows] == ["AAA", "BBB"]
    aaa, bbb = report.rows
    # AAA's log returns: ln 1.01, ln 1.005, ln 1.01, ln 1.02, ln 1.005; it only rises.
    assert aaa["ann_return"] == pytest.approx(0.0099356744, rel=1e-8)
    assert aaa["ann_vol"] == pytest.approx(0.00604825, rel=1e-4)
    assert aaa["sharpe"] == pytest.approx(1.642735, rel=1e-4)
    assert aaa["last_vol"] == pytest.approx(0.1105917533)
    assert aaa["max_drawdown"] == 0.0
    # BBB's prices run 100, 102, 99.96, 98.9604, ...: the worst fall is 0.98 * 0.99 - 1.
    assert bbb["ann_return"] == pytest.approx(0.0008974913, rel=1e-7)
    assert bbb["last_vol"] == pytest.approx(0.224506927)
    assert bbb["max_drawdown"] == pytest.approx(-0.0298)
    assert json.loads(out.read_text(encoding="utf-8")) == {"tickers": [aaa, bbb]}


def test_risk_report_narrows_to_tickers(fake: ModuleType, tmp_path: Path) -> None:
    report = risk_report(str(fake.FIXTURE), tmp_path / "bbb.json", window=2, tickers=["BBB"])

    assert [row["ticker"] for row in report.rows] == ["BBB"]


def test_risk_report_rejects_window_below_two(fake: ModuleType, tmp_path: Path) -> None:
    out = tmp_path / "report.json"

    with pytest.raises(ValueError, match="window"):
        risk_report(fake.FIXTURE, out, window=1)
    assert not out.exists()
