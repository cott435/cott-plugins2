135	def parse_diff(text: str) -> Diff:
136	    """Added lines (with new line numbers), removed lines and deleted files, per file."""
137	    diff = Diff()
138	    old = new = ""
139	    line_no = 0
140	    in_header = False
141	    for line in text.splitlines():
142	        if line.startswith("diff --git "):
143	            old, new, in_header = "", "", True
144	        elif in_header and line.startswith("--- "):
145	            old = _diff_path(line[4:])
146	        elif in_header and line.startswith("+++ "):
147	            new = _diff_path(line[4:])
148	            if new == "/dev/null":
149	                diff.deleted.add(old)
150	        elif match := HUNK.match(line):
151	            in_header, line_no = False, int(match[1])
152	        elif in_header:
153	            continue
154	        elif line.startswith("+"):
155	            diff.added.setdefault(new, []).append((line_no, line[1:]))
156	            line_no += 1
157	        elif line.startswith("-"):
158	            diff.removed.setdefault(old, []).append(line[1:])
159	    return diff

245	    """Only pass, `...` or `raise NotImplementedError`, after an optional docstring."""

250	def _task_stub(body: list[ast.stmt]) -> str | None:
251	    """The task an interface stub names: a body that is only `raise NotImplementedError("M1-T05")`."""
252	    stmts = body[1:] if body and _is_docstring(body[0]) else body
253	    if len(stmts) != 1 or not isinstance(stmts[0], ast.Raise):
254	        return None
255	    exc = stmts[0].exc
256	    if not (isinstance(exc, ast.Call) and _name(exc.func) == "NotImplementedError"):
257	        return None
258	    arg = exc.args[0] if len(exc.args) == 1 and not exc.keywords else None
259	    if (
260	        isinstance(arg, ast.Constant)
261	        and isinstance(arg.value, str)
262	        and TASK_ID.fullmatch(arg.value)
263	    ):
264	        return arg.value
265	    return None

277	    if isinstance(stmt, ast.Pass):

298	    """Abstract methods, overloads and Protocol members may have placeholder bodies."""
299	    exempt: set[ast.AST] = set()

307	            )

345	    """sys.path.insert/append/extend, site.addsitedir, or an assignment to sys.path."""

381	def ast_findings(path: str, lines: set[int]) -> list[Finding]:
382	    try:
383	        tree = ast.parse(Path(path).read_text(encoding="utf-8"))
384	    except (SyntaxError, UnicodeDecodeError, OSError):
385	        return []  # ruff reports syntax errors; a deleted file has nothing to check
386	    exempt, test = _exempt_functions(tree), is_test_file(path)
387	    found: list[Finding] = []
388	    for node in ast.walk(tree):
389	        if not _touches(node, lines):
390	            continue
391	        if outside := _outside_finding(path, node):
392	            found.append(outside)
393	        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
394	            if (
395	                node not in exempt
396	                and not path.endswith(".pyi")
397	                and _is_trivial(node.body)
398	                and not _task_stub(node.body)
399	            ):
400	                found.append(
401	                    Finding("stub", path, node.lineno, f"`{node.name}` has a placeholder body")
402	                )
403	            if test and node.name.startswith("test") and not _asserts(node):

405	                    Finding(
406	                        "test-without-assertion",
407	                        path,
408	                        node.lineno,
409	                        f"`{node.name}` asserts nothing",

414	                found.append(
415	                    Finding("swallowed-exception", path, node.lineno, "exception swallowed")
416	                )

503	def _limit_findings(old: dict[str, Any], new: dict[str, Any]) -> list[str]:
504	    pylint = _get(old, "tool.ruff.lint.pylint") or {}
505	    maxes = [*MAX_KEYS, *(f"tool.ruff.lint.pylint.{k}" for k in pylint)]
506	    out = [
507	        f"{k}: {_get(old, k)} -> {_get(new, k)}"
508	        for k in maxes
509	        if _loosened(_get(old, k), _get(new, k), +1)
510	    ]
511	    out += [
512	        f"{k}: {_get(old, k)} -> {_get(new, k)}"
513	        for k in MIN_KEYS
514	        if _loosened(_get(old, k), _get(new, k), -1)
515	    ]
516	    out += [
517	        f"{k} turned off" for k in TRUE_KEYS if _get(old, k) is True and _get(new, k) is not True
518	    ]
519	    out += [f"{k} turned on" for k in OFF_KEYS if _get(new, k) is True and _get(old, k) is not True]
520	    return out

523	def _loosened(was: Any, now: Any, direction: int) -> bool:
524	    if not isinstance(was, (int, float)):
525	        return False
526	    if not isinstance(now, (int, float)):
527	        return True  # a limit that disappeared
528	    return (now - was) * direction > 0

531	def _list_findings(old: dict[str, Any], new: dict[str, Any]) -> list[str]:
532	    out = [f"{k} lost {sorted(_as_set(_get(old, k)) - _as_set(_get(new, k)))}" for k in NO_SHRINK]
533	    for k in NO_GROW:
534	        gained = _as_set(_get(new, k)) - _as_set(_get(old, k))
535	        out.append(f"{k} gained {sorted(g for g in gained if not OUTSIDE_EXCLUDE.fullmatch(g))}")
536	    for table in NO_GROW_TABLES:
537	        was, now = _get(old, table) or {}, _get(new, table) or {}
538	        out += [
539	            f"{table}.{key} gained {sorted(_as_set(codes) - _as_set(was.get(key)))}"
540	            for key, codes in now.items()
541	            if _as_set(codes) - _as_set(was.get(key))
542	        ]
543	    flags = _as_set((_get(new, "tool.pytest.ini_options.addopts") or "").split())
544	    out += [
545	        f"pytest addopts dropped {f}"
546	        for f in STRICT_FLAGS
547	        if f in str(_get(old, "tool.pytest.ini_options.addopts")) and f not in flags
548	    ]
549	    out += [
550	        f"{k} changed (boundaries and type config change only through build-setup)"
551	        for k in MUST_MATCH
552	        if _get(old, k) != _get(new, k)
553	    ]
554	    return [o for o in out if not o.endswith("[]")]

def returns_from_json(payload: dict[str, Any]) -> Returns:
    """A Returns from its JSON form (lists become tuples)."""
    return Returns(
        base_date=payload["base_date"],
        dates=tuple(payload["dates"]),
        tickers=tuple(payload["tickers"]),
        values={ticker: tuple(values) for ticker, values in payload["values"].items()},
    )


def vol_from_json(payload: dict[str, Any]) -> VolSeries:
    """A VolSeries from its JSON form (lists become tuples)."""
    return VolSeries(
        returns=returns_from_json(payload["returns"]),
        window=payload["window"],
        annualized=payload["annualized"],
        values={ticker: tuple(values) for ticker, values in payload["values"].items()},
    )


def vol_series(
    path: str | Path, *, window: int = 20, tickers: Sequence[str] | None = None
) -> VolSeries:
    """The sample volatility, narrowed to `tickers`. It serves window 2 only; ValueError if < 2."""
    if not Path(path).is_file():
        raise FileNotFoundError(path)
    vol = vol_from_json(json.loads(SAMPLE.read_text()))
    if window < 2:
        raise ValueError(f"window must be at least 2, got {window}")
    if window != vol.window:
        raise ValueError(f"the fake serves window {vol.window} only, got {window}")
    if tickers is None:
        return vol
    for ticker in tickers:
        if ticker not in vol.values:
            raise KeyError(ticker)
    kept = tuple(sorted(set(tickers)))
    returns = Returns(
        base_date=vol.returns.base_date,
        dates=vol.returns.dates,
        tickers=kept,
        values={t: vol.returns.values[t] for t in kept},
    )
    return VolSeries(
        returns=returns,
        window=vol.window,
        annualized=vol.annualized,
        values={t: vol.values[t] for t in kept},
    )

def test_rawrow_sample_has_eight_rows_of_two_tickers_one_blank(contract_sample):
    rows = [RawRow(**row) for row in contract_sample("mdata.RawRow")["rows"]]
    assert len(rows) == 8
    assert {row.ticker for row in rows} == {"AAA", "BBB"}
    assert [row.close for row in rows].count(None) == 1
    for row in rows:
        assert date.fromisoformat(row.date).isoformat() == row.date


def test_prices_sample_is_aligned(contract_fake, contract_sample):
    fake = contract_fake("mdata.load_prices")
    prices = fake.prices_from_json(contract_sample("mdata.Prices"))
    assert len(prices.dates) == 6
    assert list(prices.dates) == sorted(set(prices.dates))
    assert list(prices.tickers) == sorted(prices.tickers) == ["AAA", "BBB"]
    for ticker in prices.tickers:
        closes = prices.closes[ticker]
        assert len(closes) == len(prices.dates)
        assert all(close > 0 for close in closes)


def test_returns_sample_follows_the_base_date(contract_fake, contract_sample):
    fake = contract_fake("risk.vol_series")
    returns = fake.returns_from_json(contract_sample("risk.Returns"))
    prices = contract_fake("mdata.load_prices").prices_from_json(contract_sample("mdata.Prices"))

    assert returns.dates == prices.dates[1:]
