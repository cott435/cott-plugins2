"""Put every ticker on one calendar (DATA-03, DATA-04)."""

from __future__ import annotations

from collections.abc import Sequence

from mdata.types import Prices, RawRow


def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
    """Prices from the first date every ticker has a close; gaps take the previous close.

    Raises GapError naming the ticker when more than `max_gap` dates in a row are missing.
    Spine version: keeps only the dates every ticker has a close on, with no fill.
    """
    by_ticker: dict[str, dict[str, float]] = {}
    for row in rows:
        closes = by_ticker.setdefault(row.ticker, {})
        if row.close is not None:
            closes[row.date] = row.close
    tickers = tuple(sorted(by_ticker))
    shared = set.intersection(*(set(by_ticker[t]) for t in tickers)) if tickers else set()
    dates = tuple(sorted(shared))
    return Prices(
        dates=dates,
        tickers=tickers,
        closes={t: tuple(by_ticker[t][d] for d in dates) for t in tickers},
    )

from __future__ import annotations

def test_sample_rows_align_on_dates_every_ticker_has(contract_sample):
    rows = [RawRow(**row) for row in contract_sample("mdata.RawRow")["rows"]]

    prices = align_calendar(rows)

    assert prices.tickers == ("A", "B")
    assert prices.closes == {
        "A": (50.0, 51.0, 52.0047),
        "B": (120.0, 118.8, 119.97909),
    }


def test_unsorted_rows_give_ascending_dates_and_sorted_tickers():
    rows = [
        RawRow("2024-01-03", "Z", 3.0),
        RawRow("2024-01-02", "M", 20.0),
        RawRow("2024-01-03", "M", 30.0),
        RawRow("2024-01-02", "Z", 2.0),
    ]

    prices = align_calendar(rows)

    assert prices.dates == ("2024-01-02", "2024-01-03")
    assert prices.tickers == ("M", "Z")
    assert prices.closes == {"M": (20.0, 30.0), "Z": (2.0, 3.0)}


def test_date_one_ticker_lacks_is_dropped():
    rows = [
        RawRow("2024-01-02", "A", 1.0),
        RawRow("2024-01-02", "B", 10.0),
        RawRow("2024-01-03", "A", 1.5),
    ]

    prices = align_calendar(rows)

    assert prices.dates == ("2024-01-02",)
    assert prices.closes == {"A": (1.0,), "B": (10.0,)}


def test_no_rows_give_empty_prices():
    prices = align_calendar([])

    assert prices.dates == ()
    assert prices.tickers == ()
    assert prices.closes == {}
