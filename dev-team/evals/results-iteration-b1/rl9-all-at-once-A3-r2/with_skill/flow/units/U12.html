33	### AD-09: Every batch first runs the method's check world
34	**Status:** Accepted · **Date:** 2026-10-17 · **Scope:** system
35	- **Context:** TRAIN-06 needs every learning method to learn a hand-derived best behaviour before it is used in a world. Bugs in RL are silent (brief), and methods change while a world is built.
36	- **Decision:** `train.run_check` trains each team method in a config on its check world: for DQN a 6-state corridor with two exits, left of state 0 (reward 0.8) and right of state 5 (reward 1.0), γ = 0.9, so the best action is left in states 0–1 and right in 2–5 and a learner biased to one action can't pass; for PPO a 1-D point that is rewarded for reaching a target, where the best push is the sign of the offset. It needs ≥ 95% agreement over every check-world state. `train.run_batch` runs the same check steps with the batch's method settings before starting any run and refuses the batch on failure. `rllab check <config>` runs it alone.
37	- **Alternatives:** A recorded pass per method version: saves seconds but goes stale silently when code changes. A test only: passes in CI but not for the settings actually used.
38	- **Consequences:** Each batch spends under a minute on the check. A check-world settings override exists because a world's tuned settings (e.g. buffer size) may be too large for a 6-state world.

N_CELLS = 6
LEFT, RIGHT = 0, 1
LEFT_EXIT_REWARD = 0.8
RIGHT_EXIT_REWARD = 1.0
GAMMA = 0.9


def _onehot(cell: int | None) -> Array:
    """The observation of `cell`; all zeros once the agent has left through an exit."""
    obs = np.zeros(N_CELLS, np.float32)
    if cell is not None:
        obs[cell] = 1.0
    return obs


def _corridor_best_action(cell: int) -> int:
    """Hand-derived: compare the discounted exit rewards reachable from `cell` (AD-09)."""
    left = LEFT_EXIT_REWARD * GAMMA**cell
    right = RIGHT_EXIT_REWARD * GAMMA ** (N_CELLS - 1 - cell)
    return RIGHT if right > left else LEFT


class CheckDiscrete:
    """A 6-state corridor with exits left of 0 (0.8) and right of 5 (1.0), gamma 0.9."""

    spec: WorldSpec

    def __init__(self, **overrides: int | float) -> None:
        unknown = sorted(set(overrides) - {"max_steps"})
        if unknown:
            raise ValueError(f"unknown check-discrete overrides: {', '.join(unknown)}")
        self.spec = WorldSpec(
            world="check-discrete",
            variant="default",
            agents=("agent",),
            teams=(TeamSpec("agent", (0,), "return", "higher"),),
            obs_len=N_CELLS,
            action=Discrete(2),
            max_steps=int(overrides.get("max_steps", 20)),
            measures=(),
            sight=None,
        )
        self._seed = 0
        self._cell: int | None = None
        self._length = 0
        self._return = 0.0
        self._done = True

    def reset(self, seed: int) -> Step:
        rng = np.random.default_rng(seed)
        # A uniform start, so a learner sees every cell, not only those near one start.
        self._cell = int(rng.random() * N_CELLS)
        self._seed = seed
        self._length = 0
        self._return = 0.0
        self._done = False
        return self._step(0.0, truncated=False)

    def step(self, actions: Array) -> Step:
        if self._done or self._cell is None:
            raise RuntimeError("step after the episode is done; call reset first")
        action = self._checked(actions)
        cell = self._cell + (1 if action == RIGHT else -1)
        reward = 0.0
        if cell < 0:
            reward, self._cell = LEFT_EXIT_REWARD, None
        elif cell >= N_CELLS:
            reward, self._cell = RIGHT_EXIT_REWARD, None
        else:
            self._cell = cell
        self._length += 1
        self._return += reward
        truncated = self._cell is not None and self._length >= self.spec.max_steps
        self._done = self._cell is None or truncated
        return self._step(reward, truncated=truncated)

    def result(self) -> EpisodeResult:
        team = {"agent": self._return}
        return EpisodeResult(self._seed, self._length, team, dict(team), {})

    def draw(self) -> DrawState:
        exits = (

            Drawable("exit", float(N_CELLS), 0.0, 1.0, None, None),

            rewards=np.asarray([reward], np.float32),
            active=np.asarray([True]),
            done=self._done,
            truncated=truncated,
        )
