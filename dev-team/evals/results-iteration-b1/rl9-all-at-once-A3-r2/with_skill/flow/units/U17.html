import json
from collections.abc import Mapping
from datetime import UTC, datetime
from pathlib import Path

import torch

from rllab.core import EpisodeResult, RunConfig, WorldSpec
from rllab.runlog.types import (
    Checkpoint,
    EvalPoint,
    Failure,
    RunExistsError,
    RunRecord,
    RunStatus,
)

_CLOSING: dict[str, RunStatus] = {
    "finished": "finished",
    "incomplete": "incomplete",
    "failed": "failed",
}


class RunWriter:
    """Creates `runs/<world>/<variant>/<name>` under `root`; refuses an existing one."""

    def __init__(self, root: Path, config: RunConfig, spec: WorldSpec) -> None:
        self.path = root / config.world / config.variant / config.name
        if self.path.exists():
            raise RunExistsError(self.path)
        (self.path / "checkpoints").mkdir(parents=True)
        self.config = config
        self.spec = spec
        self._episodes: list[EpisodeResult] = []
        self._evals: list[EvalPoint] = []
        self._record = RunRecord(config, spec, datetime.now(UTC), None, "running", None, self.path)
        self._write_record()

    def episode(self, result: EpisodeResult) -> None:
        self._episodes.append(result)

    def eval(self, point: EvalPoint) -> None:
        self._evals.append(point)
        self._flush()

    def checkpoint(self, pct: int, step: int, state: Mapping[str, dict[str, object]]) -> Checkpoint:
        path = self.path / "checkpoints" / f"ckpt_{pct}.pt"
        ckpt = Checkpoint(self.config.name, pct, step, self.spec, self.config.methods, state, path)
        saved = {f.name: to_json(getattr(ckpt, f.name)) for f in dataclasses.fields(ckpt)}
        torch.save({**saved, "state": {team: dict(s) for team, s in state.items()}}, path)
        return ckpt

    def close(self, status: str, failure: Failure | None = None) -> RunRecord:
        if status not in _CLOSING:
            raise ValueError(f"a run closes as one of {sorted(_CLOSING)}, not {status!r}")
        self._flush()
        self._record = dataclasses.replace(
            self._record, ended=datetime.now(UTC), status=_CLOSING[status], failure=failure
        )
        self._write_record()
        return self._record

    def _flush(self) -> None:
        """Appends the buffered episodes and evaluation points to their JSON Lines logs."""
        for name, items in (("episodes", self._episodes), ("evals", self._evals)):
            with (self.path / f"{name}.jsonl").open("a") as log:
                log.writelines(json.dumps(to_json(item)) + "\n" for item in items)
            items.clear()

    def _write_record(self) -> None:
        """Replaces `run.json` atomically: write a temporary file, then rename it."""
        temporary = self.path / "run.json.tmp"
        temporary.write_text(json.dumps(to_json(self._record), indent=1))
        temporary.replace(self.path / "run.json")


def to_json(value: object) -> object:
    """A contract value as JSON data: dataclasses and mappings as objects, tuples as arrays."""
    if dataclasses.is_dataclass(value) and not isinstance(value, type):
        return {f.name: to_json(getattr(value, f.name)) for f in dataclasses.fields(value)}
    if isinstance(value, Mapping):
        return {str(key): to_json(item) for key, item in value.items()}
    if isinstance(value, (list, tuple)):
        return [to_json(item) for item in value]
    if isinstance(value, (Path, datetime)):
        return str(value) if isinstance(value, Path) else value.isoformat()
    return value

import json
from datetime import datetime
from pathlib import Path

from rllab.runlog.types import EvalPoint, Progress


def progress(path: Path) -> Progress:
    """One snapshot of the run at `path`: its record plus the last evaluation point written."""
    record_file = path / "run.json"
    if not record_file.is_file():
        raise FileNotFoundError(f"no run at {path}: {record_file} is missing")
    record = json.loads(record_file.read_text())
    latest = _latest_eval(path / "evals.jsonl")
    started = datetime.fromisoformat(record["started"])
    ended = record.get("ended")
    until = datetime.fromisoformat(ended) if ended else datetime.now(started.tzinfo)
    return Progress(
        step=latest.step if latest else 0,
        budget=int(record["config"]["budget"]),
        latest=latest,
        elapsed=until - started,
        status=str(record["status"]),
    )


def _latest_eval(log: Path) -> EvalPoint | None:
    """The last complete line of `evals.jsonl`, or None before the first evaluation point."""
    if not log.is_file():
        return None
    lines = [line for line in log.read_text().splitlines() if line.strip()]
    for line in reversed(lines):
        try:
            data = json.loads(line)
        except json.JSONDecodeError:
            continue
        return EvalPoint(int(data["step"]), dict(data["scores"]), dict(data["measures"]))
    return None
