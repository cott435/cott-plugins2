"""PL-03 step 3: maximum drawdown with its dates."""

from __future__ import annotations

import math

from risk.types import Drawdown, Returns


def max_drawdown(returns: Returns) -> tuple[Drawdown, ...]:
    """Per ticker, the deepest fall below the running peak of wealth from 1 (RISK-05).

    Spine version: depth only; both dates are `returns.base_date` until M1-T20 adds them.
    """
    return tuple(
        Drawdown(
            ticker=ticker,
            depth=_depth(returns.values[ticker]),
            peak_date=returns.base_date,
            trough_date=returns.base_date,
        )
        for ticker in returns.tickers
    )


def _depth(log_returns: tuple[float, ...]) -> float:
    """The lowest wealth / running peak - 1, with wealth starting at 1; 0.0 with no fall."""
    wealth = peak = 1.0
    depth = 0.0
    for r in log_returns:
        wealth *= math.exp(r)
        peak = max(peak, wealth)

"""PL-03 step 3: maximum drawdown with its dates."""

from __future__ import annotations

import math

from risk.types import Drawdown, Returns


def max_drawdown(returns: Returns) -> tuple[Drawdown, ...]:
    """Per ticker, the deepest fall below the running peak of wealth from 1 (RISK-05)."""
    return tuple(_drawdown(returns, ticker) for ticker in returns.tickers)


def _drawdown(returns: Returns, ticker: str) -> Drawdown:
    """Wealth starts at 1 on `base_date`; the first deepest fall wins, and the first peak.

    With no fall, both dates stay `base_date`.
    """
    wealth = peak = 1.0
    peak_date = returns.base_date
    best = Drawdown(ticker, 0.0, returns.base_date, returns.base_date)
    for date, r in zip(returns.dates, returns.values[ticker], strict=True):
        wealth *= math.exp(r)
        if wealth > peak:
            peak, peak_date = wealth, date
        depth = wealth / peak - 1.0
        if depth < best.depth:
            best = Drawdown(ticker, depth, peak_date, date)
    return best
