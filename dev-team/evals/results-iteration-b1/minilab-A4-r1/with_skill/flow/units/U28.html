+def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
+    """Prices from the first date every ticker has a close; a gap takes the previous close.
+
+    Raises GapError naming the ticker when more than `max_gap` dates in a row are missing.
+    """
+    by_ticker: dict[str, dict[str, float]] = defaultdict(dict)
+    for row in rows:
+        if row.close is not None:
+            by_ticker[row.ticker][row.date] = row.close
+    tickers = tuple(sorted({row.ticker for row in rows}))
+    if not tickers or not all(by_ticker[t] for t in tickers):
+        return Prices(dates=(), tickers=tickers, closes=dict.fromkeys(tickers, ()))
+    start = max(min(by_ticker[t]) for t in tickers)
+    dates = tuple(sorted({d for t in tickers for d in by_ticker[t] if d >= start}))
+    closes = {t: _forward_fill(t, by_ticker[t], dates, max_gap) for t in tickers}
+    return Prices(dates=dates, tickers=tickers, closes=closes)
+
+
+def _forward_fill(
+    ticker: str, known: Mapping[str, float], dates: Sequence[str], max_gap: int
+) -> tuple[float, ...]:
+    """The ticker's close on each date, a missing one taking the previous close.
+
+    A ticker missing the first date takes its last close before it.
+    """
+    previous = known[max(d for d in known if d <= dates[0])]
+    filled: list[float] = []
+    missing = 0
+    for date in dates:
+        if date in known:
+            previous, missing = known[date], 0
+        else:
+            missing += 1
+        if missing > max_gap:
+            raise GapError(f"{ticker} misses more than {max_gap} dates in a row by {date}")
+        filled.append(previous)
+    return tuple(filled)

+
+from __future__ import annotations
+
+import math
+
+from risk.types import Drawdown, Returns
+
+
+def max_drawdown(returns: Returns) -> tuple[Drawdown, ...]:
+    """Per ticker, the deepest fall of wealth (1 on `base_date`) below its running peak."""
+    wealth_dates = (returns.base_date, *returns.dates)
+    drawdowns = []
+    for ticker in returns.tickers:
+        depth, peak, trough = _deepest_fall(returns.values[ticker])
+        drawdowns.append(
+            Drawdown(
+                ticker=ticker,
+                depth=depth,
+                peak_date=wealth_dates[peak],
+                trough_date=wealth_dates[trough],
+            )
+        )
+    return tuple(drawdowns)
+
+
+def _deepest_fall(log_returns: tuple[float, ...]) -> tuple[float, int, int]:
+    """Depth (trough / peak - 1) and the wealth indices of its peak and trough.
+
+    Wealth index 0 is the base (wealth 1); index i + 1 follows return i. Ties keep
+    the first peak and the first trough; with no fall both indices are 0.
+    """
+    log_wealth = 0.0
+    log_peak = 0.0
+    peak = 0
+    worst = 0.0
+    worst_peak = 0
+    worst_trough = 0
+    for i, r in enumerate(log_returns, start=1):
+        log_wealth += r
+        if log_wealth > log_peak:
+            log_peak = log_wealth
+            peak = i
+        elif log_wealth - log_peak < worst:
+            worst = log_wealth - log_peak
+            worst_peak = peak
+            worst_trough = i
+    return math.expm1(worst), worst_peak, worst_trough
