"""PL-02 step 2: daily returns from aligned closes."""

from __future__ import annotations

import itertools
import math

from mdata import Prices
from risk.types import Returns


def log_returns(prices: Prices) -> Returns:
    """Return each ticker's log return r_t = ln(close_t / close_t-1), dated t."""
    if len(prices.dates) < 2:
        raise ValueError(f"need at least two dates for returns, got {len(prices.dates)}")
    values = {
        ticker: tuple(math.log(b / a) for a, b in itertools.pairwise(prices.closes[ticker]))
        for ticker in prices.tickers
    }
    return Returns(
        base_date=prices.dates[0],
        dates=prices.dates[1:],
        tickers=prices.tickers,
        values=values,
    )

from risk.returns import log_returns

LN_2 = 0.6931471805599453


def make_prices(closes: dict[str, tuple[float, ...]], dates: tuple[str, ...]) -> Prices:
    return Prices(dates=dates, tickers=tuple(sorted(closes)), closes=closes)


def test_dates_drop_the_first_which_becomes_the_base_date():
    prices = make_prices({"A": (100.0, 110.0, 99.0)}, ("2024-01-02", "2024-01-03", "2024-01-04"))
    returns = log_returns(prices)

    prices = make_prices(
