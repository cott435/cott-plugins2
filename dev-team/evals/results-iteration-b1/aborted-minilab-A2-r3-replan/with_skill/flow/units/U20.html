from __future__ import annotations

from collections.abc import Sequence
from pathlib import Path

from mdata import load_prices
from risk.annualize import annualize
from risk.returns import log_returns
from risk.types import VolSeries
from risk.vol import rolling_vol


def vol_series(
    path: str | Path, *, window: int = 20, tickers: Sequence[str] | None = None
) -> VolSeries:
    """Load the closes in `path`, take their returns, roll their volatility and annualize it."""
    prices = load_prices(path, tickers)
    returns = log_returns(prices)
    vol = rolling_vol(returns, window)
    return annualize(vol)

"""PL-03: the JSON risk report of a prices CSV."""

from __future__ import annotations

from collections.abc import Sequence
from pathlib import Path

from risk.drawdown import max_drawdown
from risk.report import write_report
from risk.summary import summary_stats
from risk.types import Report
from risk.vol_pipeline import vol_series


def risk_report(
    path: str | Path,
    out_path: str | Path,
    *,
    window: int = 20,
    tickers: Sequence[str] | None = None,
) -> Report:
    """Roll the volatility of `path`, summarize it, find each drawdown and write the report."""
    vol = vol_series(path, window=window, tickers=tickers)
    summary = summary_stats(vol)
    drawdowns = max_drawdown(vol.returns)
    return write_report(summary, drawdowns, out_path)

"""The risk contract types: returns, volatility, the summary, the drawdown and the report."""

from __future__ import annotations

from collections.abc import Mapping
from dataclasses import dataclass


@dataclass(frozen=True)
class Returns:
    """Daily returns on the prices' dates after the first, which is `base_date`."""

    base_date: str
    dates: tuple[str, ...]
    tickers: tuple[str, ...]
    values: Mapping[str, tuple[float, ...]]

    def __post_init__(self) -> None:
        if list(self.dates) != sorted(set(self.dates)) or not all(
            self.base_date < day for day in self.dates
        ):
            raise ValueError("dates must be unique, ascending and after base_date")
        if list(self.tickers) != sorted(set(self.tickers)):
            raise ValueError("tickers must be unique and sorted")
        if set(self.values) != set(self.tickers):
            raise ValueError("values must have exactly one series per ticker")
        for ticker, series in self.values.items():
            if len(series) != len(self.dates):
                raise ValueError(f"{ticker}: expected one return per date")


@dataclass(frozen=True)
class VolSeries:
    """Rolling volatility aligned to `returns.dates`; None until `window` earlier returns exist."""

    returns: Returns
    window: int
    annualized: bool
    values: Mapping[str, tuple[float | None, ...]]

    def __post_init__(self) -> None:
        if set(self.values) != set(self.returns.tickers):
            raise ValueError("values must have exactly one series per ticker")
        for ticker, series in self.values.items():
            if len(series) != len(self.returns.dates):
                raise ValueError(f"{ticker}: expected one value per return date")
            if any(value is not None for value in series[: self.window]):
                raise ValueError(f"{ticker}: no value until {self.window} earlier returns exist")


@dataclass(frozen=True)
class TickerSummary:
    """Whole-period statistics of one ticker."""

    ticker: str
    ann_return: float
    ann_vol: float
    sharpe: float
    last_vol: float | None

def write_report(
    summary: Sequence[TickerSummary], drawdowns: Sequence[Drawdown], out_path: str | Path
) -> Report:
    """Write `{"tickers": [...]}` to `out_path`, one row per ticker sorted by ticker."""
    by_ticker = {item.ticker: item for item in drawdowns}
    if {item.ticker for item in summary} != set(by_ticker):
        raise ValueError("summary and drawdowns cover different tickers")
    rows: tuple[Mapping[str, object], ...] = tuple(

            "peak_date": by_ticker[item.ticker].peak_date,
        }
        for item in sorted(summary, key=lambda s: s.ticker)
    )
    Path(out_path).write_text(json.dumps({"tickers": list(rows)}, indent=2))
    return Report(path=str(out_path), rows=rows)
