"""Uniform random play for any action format (EVAL-01)."""

import numpy as np

from rllab.core import ActionSpace, Array, Discrete, Policy


class RandomPolicy(Policy):
    """Draws each action uniformly from `action`, using only the `rng` passed to `act`."""

    def __init__(self, action: ActionSpace) -> None:
        self.action = action

    def act(self, obs: Array, rng: np.random.Generator) -> Array:
        """One uniform action per row of `obs`: an int in `[0, n)` or `dims` floats in range."""
        agents = len(obs)
        if isinstance(self.action, Discrete):
            return rng.integers(0, self.action.n, size=agents)
        return rng.uniform(self.action.low, self.action.high, size=(agents, self.action.dims))
