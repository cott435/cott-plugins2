from risk import log_returns
from risk.types import Returns

Golden = Callable[[str, str], tuple[dict, dict]]
Sample = Callable[[str], dict]


def prices_from_json(data: dict) -> Prices:
    return Prices(
        dates=tuple(data["dates"]),
        tickers=tuple(data["tickers"]),
        closes={t: tuple(float(c) for c in cs) for t, cs in data["closes"].items()},
    )


@pytest.mark.intent
def test_tc04_geometric_closes_match_the_closed_form(golden: Golden) -> None:
    given, expected = golden("risk", "TC-04")
    tol = expected["tol"]

    returns = log_returns(prices_from_json(given["prices"]))

    assert isinstance(returns, Returns)
    assert returns.base_date == expected["base_date"]
    assert returns.dates == tuple(expected["dates"])
    assert returns.tickers == tuple(expected["tickers"])
    assert set(returns.values) == set(expected["values"])
    for ticker, values in expected["values"].items():
        assert returns.values[ticker] == pytest.approx(values, abs=tol, rel=0)


@pytest.mark.intent
def test_returns_are_log_ratios_of_consecutive_closes_dated_t(contract_sample: Sample) -> None:
    data = contract_sample("mdata.Prices")
    prices = prices_from_json(data)

    returns = log_returns(prices)

    assert returns.base_date == prices.dates[0]
    assert returns.dates == prices.dates[1:]
    assert returns.tickers == prices.tickers
    for ticker in prices.tickers:
        closes = prices.closes[ticker]
        want = [math.log(closes[i] / closes[i - 1]) for i in range(1, len(closes))]
        assert returns.values[ticker] == pytest.approx(want, abs=1e-12, rel=0)
