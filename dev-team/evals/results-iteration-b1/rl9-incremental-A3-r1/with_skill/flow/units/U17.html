79	**Status:** Accepted · **Date:** 2026-10-17 · **Scope:** `runs`

Each run writes `runs/<name>-s<seed>-<timestamp>/` holding `settings.json` (the resolved settings), `curve.csv`, `checkpoint.pt` or `checkpoint.npz`, `games.npz` and `status` (`running`, `complete` or `incomplete`). class RunWriter:
    """Writes one run folder as the run goes; each curve point is flushed at once."""

    def __init__(self, path: Path, settings: RunSettings) -> None:
        self._path = path
        self._suffix = ".npz" if settings.agent == "tabular" else ".pt"

    def add_point(self, p: CurvePoint) -> None:
        with (self._path / _CURVE).open("a", newline="", encoding="utf-8") as f:
            csv.writer(f).writerow([getattr(p, name) for name in _CURVE_FIELDS])

    def save_checkpoint(self, agent: Agent, *, team: str = SOLO) -> None:
        stem = "checkpoint" if team == SOLO else f"checkpoint-{team}"
        agent.save(self._path / f"{stem}{self._suffix}")

    def save_generation(self, agent: Agent, *, team: str, index: int) -> Path:
        path = self._path / _GENERATIONS / team / str(index)
        path.parent.mkdir(parents=True, exist_ok=True)
        agent.save(path)
        return path

    def save_games(self, games: Sequence[GameRecording]) -> None:
        arrays: dict[str, np.ndarray] = {}
        for i, game in enumerate(games):
            arrays[f"{i}.seed"] = np.asarray(game.seed)
            arrays[f"{i}.board"] = np.asarray(game.board)
            arrays[f"{i}.frames"] = game.frames
            arrays[f"{i}.rewards"] = game.rewards
            for team, scores in game.scores.items():
                arrays[f"{i}.scores.{team}"] = scores
        np.savez_compressed(self._path / _GAMES, allow_pickle=False, **arrays)

    def finish(self, status: Literal["complete", "incomplete"]) -> RunRecord:
        (self._path / _STATUS).write_text(status, encoding="utf-8")
        return open_run(self._path)

def _load_kit() -> ModuleType:
    spec = importlib.util.spec_from_file_location("contract_kit", KIT_PATH)
    assert spec is not None

kit = _load_kit()


class SavingAgent:
    """An Agent whose save writes a marker file and records the path."""

    def __init__(self) -> None:
        self.saved: list[Path] = []

    def act(self, obs: np.ndarray, *, explore: bool) -> np.ndarray:
        return np.zeros(obs.shape[0], dtype=np.int64)

    def learn(self, t: Transition) -> None:
        self.saved.append(Path("learn"))

    def save(self, path: Path) -> None:
        path.write_bytes(b"weights")
        self.saved.append(path)

    def load(self, path: Path) -> None:
        self.saved.append(path)


def _same_games(a: tuple[GameRecording, ...], b: tuple[GameRecording, ...]) -> bool:
    return len(a) == len(b) and all(
        x.seed == y.seed
        and x.board == y.board
        and x.frames.dtype == y.frames.dtype
        and np.array_equal(x.frames, y.frames)
        and np.array_equal(x.rewards, y.rewards)
        and x.scores.keys() == y.scores.keys()
        and all(np.array_equal(x.scores[t], y.scores[t]) for t in x.scores)
        for x, y in zip(a, b, strict=True)
    )


def _same_record(a: RunRecord, b: RunRecord) -> bool:
    plain = ("path", "settings", "curve", "status", "checkpoints", "generations")
    return all(getattr(a, f) == getattr(b, f) for f in plain) and _same_games(a.games, b.games)


def _replay_writer_sample(root: Path) -> tuple[RunRecord, RunRecord]:
    recording: dict[str, Any] = kit.raw_sample("runs.RunWriter", "three_point_complete")
    settings = kit.decode(RunSettings, recording["settings"])
    writer = create_run(settings, root=root)
    finished: RunRecord | None = None
    for call in recording["calls"]:
        method = call["method"]
        if method == "add_point":
            writer.add_point(kit.decode(CurvePoint, call["point"]))
        elif method == "save_checkpoint":
            writer.save_checkpoint(SavingAgent(), team=call["team"])
        elif method == "save_games":
            writer.save_games(kit.decode(tuple[GameRecording, ...], call["games"]))
        else:
            finished = writer.finish(call["status"])
    assert finished is not None
    expected: RunRecord = kit.decode(RunRecord, recording["record"])
    return finished, expected


def test_the_writer_sample_reads_back_as_an_equal_record(tmp_path: Path) -> None:

I built the run store in `src/rllab/runs/store.py`: `create_run`, `RunWriter` and `open_run` now write a run folder and read it back. `scripts/check.sh gate` passes, and the 7 new tests in `tests/unit/rllab/test_store.py` pass.

**How it works:**
- **Folder name:** `<name>-s<seed>-<YYYYmmdd-HHMMSS>`. If that name is taken, it adds `-1`, `-2` and so on, so it is always unique.
- **On `create_run`:** writes `settings.json`, a `curve.csv` with only its header, and `status` set to `running`.
- **Curve points:** each `add_point` is appended to `curve.csv` at once, so the curve survives an interrupt.
- **Checkpoints:** `checkpoint.npz` for a tabular agent, `checkpoint.pt` otherwise. A multi-team run uses `checkpoint-<team>.<ext>`.
- **Generations:** saved under `generations/<team>/<index>`, and the path is returned. `open_run` lists them in number order, so 10 comes after 9.
- **Games:** all recordings go into one compressed `games.npz`, with no pickled data.
- **On `finish`:** writes the status, then reads the folder back through `open_run`.

**Errors from `open_run`:**
- A missing folder or missing `settings.json` raises `FileNotFoundError` naming the run.
- An unreadable `games.npz` raises `ValueError` naming the file. Unreadable settings, a bad curve or an unknown status also raise `ValueError`.

**Choices I made where the contract was silent:**
- **Constructor:** I gave `RunWriter` an `__init__(path, settings)`, because the frozen interface only lists its methods.
- **Checkpoint extension:** whether a checkpoint is `.npz` or `.pt` depends on whether the agent is tabular.
- **Status file:** a run folder with no `status` file raises `FileNotFoundError` rather than being read as some default status.
