from collections.abc import Mapping, Sequence
from typing import Literal

import numpy as np

from rllab.core import SOLO, ActionSpace, Env, Rewards, StepResult, TeamSpec
from rllab.envs.types import COMPACT_FIELDS

_DEFAULT_REWARDS = Rewards()

Cell = tuple[int, int]
Heading = Literal["up", "right", "down", "left"]

_HEADINGS: tuple[Heading, ...] = ("up", "right", "down", "left")
_MOVES: tuple[Cell, ...] = ((-1, 0), (0, 1), (1, 0), (0, -1))
_TURNS = (0, -1, 1)  # straight, turn left, turn right: the change in heading index

_HUNGER_PER_CELL = 100
_EMPTY, _HEAD, _BODY, _FOOD = 0, 1, 2, 3


class Snake(Env):
    """The Snake game with the compact or the full-grid view."""

    def __init__(
        self,
        *,
        view: Literal["compact", "grid"],
        rewards: Rewards = _DEFAULT_REWARDS,
        size: int = 10,
    ) -> None:
        if size < _START_LENGTH + 1:
            raise ValueError(f"size must be >= {_START_LENGTH + 1}, got {size}")
        self._view = view
        self._rewards = rewards
        self._size = size
        obs_shape = (len(COMPACT_FIELDS),) if view == "compact" else (3, size, size)
        self.teams: Mapping[str, TeamSpec] = {
            SOLO: TeamSpec(n_agents=1, obs_shape=obs_shape, actions=ActionSpace("discrete", n=3))
        }
        self.board: Literal["grid", "tank"] = "grid"
        self._rng = np.random.default_rng(0)
        self._body: list[Cell] = []
        self._heading = 1
        self._food: Cell | None = None
        self._steps = 0
        self._hunger = 0
        self._score = 0
        self._over = True

    def reset(self, seed: int) -> tuple[dict[str, np.ndarray], dict[str, float]]:
        """Start a length-3 snake in the middle heading right, with food on a seeded free cell."""
        self._rng = np.random.default_rng(seed)
        mid = self._size // 2
        self._start([(mid, mid - i) for i in range(_START_LENGTH)], 1)
        self._food = self._free_cell()
        return self._observe(), self._info()

    def reset_to(
        self,
        body: Sequence[tuple[int, int]],
        heading: Literal["up", "right", "down", "left"],
        food: tuple[int, int],
    ) -> tuple[dict[str, np.ndarray], dict[str, float]]:
        """Start a game from a given board: cells are (row, col), the body head first."""
        cells = [(int(r), int(c)) for r, c in body]
        self._check_board(cells, (int(food[0]), int(food[1])))
        if heading not in _HEADINGS:
            raise ValueError(f"heading must be one of {_HEADINGS}, got {heading!r}")
        self._start(cells, _HEADINGS.index(heading))
        self._food = (int(food[0]), int(food[1]))
        return self._observe(), self._info()

    def step(self, actions: Mapping[str, np.ndarray]) -> StepResult:
        if self._over:
            raise RuntimeError("the game is over: call reset first")
        turn = _TURNS[_action(actions)]
        self._heading = (self._heading + turn) % len(_HEADINGS)
        self._steps += 1
        head = _ahead(self._body[0], self._heading)
        terminated, truncated = False, False
        if self._collides(head):
            reward, terminated = self._rewards.death, True
        elif head == self._food:
            reward, terminated = self._rewards.food, self._eat(head)
        else:

            self._hunger += 1
            reward = self._rewards.step
            if self._hunger >= _HUNGER_PER_CELL * len(self._body):

            reward={SOLO: np.array([reward], dtype=np.float32)},
            done={SOLO: np.array([self._over], dtype=bool)},

            truncated=truncated,
            info=self._info(),

        grid = np.full((self._size, self._size), _EMPTY, dtype=np.int8)
        for r, c in self._body[1:]:
            grid[r, c] = _BODY
        if self._body:
            grid[self._body[0]] = _HEAD
        if self._food is not None:
            grid[self._food] = _FOOD
        return grid

    def _start(self, body: list[Cell], heading: int) -> None:
        self._body = body
        self._heading = heading
        self._steps = 0
        self._hunger = 0
        self._score = 0
        self._over = False

    def _check_board(self, body: list[Cell], food: Cell) -> None:
        if not body:
            raise ValueError("the body needs at least one cell")
        if not all(self._inside(cell) for cell in body):
            raise ValueError(f"the body leaves the {self._size}x{self._size} grid: {body}")

        if any(abs(a[0] - b[0]) + abs(a[1] - b[1]) != 1 for a, b in steps):
            raise ValueError(f"the body is not contiguous: {body}")
        if len(set(body)) != len(body):
            raise ValueError(f"the body overlaps itself: {body}")
        if not self._inside(food) or food in body:
            raise ValueError(f"the food {food} must be on a free grid cell, not on the snake")

    def _eat(self, head: Cell) -> bool:
        """Grow onto the food; True when the snake now fills the board."""
        self._body = [head, *self._body]
        self._score += 1
        self._hunger = 0
        if len(self._body) == self._size * self._size:
            self._food = None
            return True
        self._food = self._free_cell()
        return False

    def _free_cell(self) -> Cell:
        taken = set(self._body)
        free = [(r, c) for r in range(self._size) for c in range(self._size) if (r, c) not in taken]
        return free[int(self._rng.integers(len(free)))]

    def _inside(self, cell: Cell) -> bool:
        return 0 <= cell[0] < self._size and 0 <= cell[1] < self._size

    def _collides(self, cell: Cell) -> bool:
        """A wall or the body; the tail cell is free unless the snake eats and keeps it."""
        if not self._inside(cell):
            return True
        body = self._body if cell == self._food else self._body[:-1]
        return cell in body

    def _observe(self) -> dict[str, np.ndarray]:
        view = self._compact() if self._view == "compact" else self._grid()
        return {SOLO: view[np.newaxis]}

    def _compact(self) -> np.ndarray:
        head = self._body[0]
        danger = [self._collides(_ahead(head, self._heading + turn)) for turn in _TURNS]

        towards = [food[1] < head[1], food[1] > head[1], food[0] < head[0], food[0] > head[0]]
        return np.array([*danger, *facing, *towards], dtype=np.float32)

    def _grid(self) -> np.ndarray:
        frame = self.frame()
        return np.stack([frame == code for code in (_HEAD, _BODY, _FOOD)]).astype(np.float32)

    def _info(self) -> dict[str, float]:
        score = float(self._score)
        return {"steps": float(self._steps), f"score:{SOLO}": score, "score": score}


def _ahead(cell: Cell, heading: int) -> Cell:
    dr, dc = _MOVES[heading % len(_MOVES)]
    return cell[0] + dr, cell[1] + dc


def _action(actions: Mapping[str, np.ndarray]) -> int:
    """The single agent's action, checked: shape (1,), finite, an integer in 0..2."""
    if SOLO not in actions:
        raise ValueError(f"actions must name the team {SOLO!r}, got {sorted(actions)}")
    action = np.asarray(actions[SOLO])
    if action.shape != (1,):
        raise ValueError(f"the action must have shape (1,), got {action.shape}")
    value = float(action[0])
    if not np.isfinite(value) or value != int(value) or not 0 <= value < len(_TURNS):
