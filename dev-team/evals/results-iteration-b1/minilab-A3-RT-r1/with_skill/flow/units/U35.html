10	def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
11	    """One calendar from the first date every ticker has a close; a gap takes the previous close.
12	
13	    Raises GapError naming the ticker when more than `max_gap` dates in a row are missing.
14	    """
15	    by_ticker: dict[str, dict[str, float]] = {}
16	    for row in rows:
17	        if row.close is not None:
18	            by_ticker.setdefault(row.ticker, {})[row.date] = row.close
19	    tickers = tuple(sorted(by_ticker))
20	    date_sets = [set(closes) for closes in by_ticker.values()]
21	    dates = tuple(sorted(set.intersection(*date_sets))) if date_sets else ()
22	    return Prices(
23	        dates=dates,
24	        tickers=tickers,
25	        closes={ticker: tuple(by_ticker[ticker][date] for date in dates) for ticker in tickers},

"""PL-01 step 3: put every ticker on one calendar."""

from __future__ import annotations

from collections.abc import Mapping, Sequence

from mdata.types import GapError, Prices, RawRow


def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
    """One calendar from the first date every ticker has a close; a gap takes the previous close.

    The calendar is every date any ticker has a close on, from the first date on which every
    ticker has one. Raises GapError naming the ticker when more than `max_gap` calendar dates
    in a row are missing.
    """
    by_ticker: dict[str, dict[str, float]] = {}
    for row in rows:
        if row.close is not None:
            by_ticker.setdefault(row.ticker, {})[row.date] = row.close
    tickers = tuple(sorted(by_ticker))
    if not tickers:
        return Prices(dates=(), tickers=(), closes={})
    start = max(min(closes) for closes in by_ticker.values())
    dates = tuple(sorted({date for closes in by_ticker.values() for date in closes if date >= start}))
    return Prices(
        dates=dates,
        tickers=tickers,
        closes={ticker: _fill(ticker, by_ticker[ticker], dates, max_gap) for ticker in tickers},
    )


def _fill(
    ticker: str, closes: Mapping[str, float], dates: Sequence[str], max_gap: int
) -> tuple[float, ...]:
    """The ticker's close on each date, a missing date taking the previous close."""
    filled: list[float] = []
    missing = 0
    for date in dates:
        close = closes.get(date)
        if close is None:
            missing += 1
            if missing > max_gap:
                raise GapError(f"{ticker} is missing more than {max_gap} dates in a row")
            filled.append(filled[-1])
        else:
            missing = 0
            filled.append(close)
    return tuple(filled)
