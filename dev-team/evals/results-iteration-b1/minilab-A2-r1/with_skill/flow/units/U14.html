71	    """Rolling volatility over the last `window` days, today included."""
72	    return returns.rolling(window).apply(lambda x: np.std(x), raw=True)
73	
74	
75	def annualize(vol: pd.DataFrame) -> pd.DataFrame:
76	    """Annualize to a calendar year, like my brokerage statements: 365 days."""
77	    return vol * np.sqrt(365)

80	def summary(returns: pd.DataFrame) -> pd.DataFrame:
81	    """Yearly return and volatility per ticker."""
82	    return pd.DataFrame({"ann_return": returns.mean() * 365, "ann_vol": returns.std(ddof=0) * np.sqrt(365)})

"""PL-02 step 3: rolling volatility from the returns before each date."""

from __future__ import annotations

import statistics

from risk.types import Returns, VolSeries


def rolling_vol(returns: Returns, window: int) -> VolSeries:
    """Each date's sample stdev of the `window` returns before it: None until they exist."""
    if window < 2:
        raise ValueError(f"window must be at least 2, got {window}")
    values = {
        ticker: tuple(
            statistics.stdev(series[i - window : i]) if i >= window else None
            for i in range(len(series))
        )
        for ticker, series in returns.values.items()
    }
    return VolSeries(returns=returns, window=window, annualized=False, values=values)
