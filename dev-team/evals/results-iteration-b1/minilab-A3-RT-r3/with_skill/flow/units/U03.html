+def vol_series(
+    path: str | Path, *, window: int = 20, tickers: Sequence[str] | None = None
+) -> VolSeries:
+    """The fixture's VolSeries for `CSV`, narrowed to `tickers`.

+    ValueError if `window` < 2, as the real one; LookupError for a window with no fixture;

+        tickers=tuple(names),
