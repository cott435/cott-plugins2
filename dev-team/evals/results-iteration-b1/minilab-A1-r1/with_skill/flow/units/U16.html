"""Unit tests for risk.vol.rolling_vol (PL-02 step 3, spine: expanding stdev of earlier returns)."""

from __future__ import annotations

import json
from pathlib import Path

import pytest

from risk.types import Returns
from risk.vol import rolling_vol

SAMPLE = Path(__file__).parents[2] / "contracts" / "risk.Returns" / "sample.json"


def _sample_returns() -> Returns:
    data = json.loads(SAMPLE.read_text())
    return Returns(
        base_date=data["base_date"],
        dates=tuple(data["dates"]),
        tickers=tuple(data["tickers"]),
        values={ticker: tuple(values) for ticker, values in data["values"].items()},
    )


def _one_ticker(values: tuple[float, ...]) -> Returns:
    dates = tuple(f"2025-01-0{i + 2}" for i in range(len(values)))
    return Returns(base_date="2025-01-01", dates=dates, tickers=("X",), values={"X": values})


def test_sample_gives_daily_vol_aligned_to_the_returns() -> None:
    returns = _sample_returns()
    result = rolling_vol(returns, 2)
    assert result.returns is returns
    assert result.window == 2
    assert result.annualized is False
    assert set(result.values) == {"AAA", "BBB"}
    for ticker in ("AAA", "BBB"):
        assert len(result.values[ticker]) == len(returns.dates)
        assert result.values[ticker][:2] == (None, None)
    assert result.values["AAA"][2] == pytest.approx(0.0069666, abs=1e-6)
    assert result.values["BBB"][2] == pytest.approx(0.0069666, abs=1e-6)


def test_stdev_expands_over_every_earlier_return() -> None:
    result = rolling_vol(_one_ticker((0.01, 0.03, 0.05, 0.0)), 2)
    assert result.values["X"][:2] == (None, None)
    assert result.values["X"][2:] == pytest.approx((0.0141421356, 0.02))


def test_none_until_window_earlier_returns_exist() -> None:

    assert result.values["X"][:3] == (None, None, None)
    assert result.values["X"][3] == pytest.approx(0.02)


@pytest.mark.parametrize("window", [1, 0, -1])

"""PL-02 step 3: rolling volatility from earlier returns."""

from __future__ import annotations

from statistics import stdev

from risk.types import Returns, VolSeries


def rolling_vol(returns: Returns, window: int) -> VolSeries:
    """Sample stdev of the returns before each date, once `window` exist (RISK-02).

    Spine version: the stdev is expanding, over every earlier return.
    """
    if window < 2:
        raise ValueError(f"window must be at least 2, got {window}")
    values = {
        ticker: tuple(stdev(series[:i]) if i >= window else None for i in range(len(series)))
        for ticker, series in returns.values.items()
    }
