ISO_DATE = re.compile(r"^\d{4}-\d{2}-\d{2}$")


def _returns(data: dict) -> Returns:
    return Returns(
        base_date=data["base_date"],
        dates=tuple(data["dates"]),
        tickers=tuple(data["tickers"]),
        values={t: tuple(v) for t, v in data["values"].items()},
    )


def _check_returns(returns: Returns) -> None:
    assert ISO_DATE.match(returns.base_date)
    assert list(returns.dates) == sorted(returns.dates)
    assert returns.base_date < returns.dates[0]
    assert list(returns.tickers) == sorted(returns.tickers)
    assert set(returns.values) == set(returns.tickers)
    assert all(len(v) == len(returns.dates) for v in returns.values.values())


def _check_vol(vol: VolSeries) -> None:
    _check_returns(vol.returns)
    assert vol.window >= 2
    assert set(vol.values) == set(vol.returns.tickers)
    for series in vol.values.values():
        assert len(series) == len(vol.returns.dates)
        assert all(v is None for v in series[: vol.window])
        assert all(v is not None and v >= 0 for v in series[vol.window :])
