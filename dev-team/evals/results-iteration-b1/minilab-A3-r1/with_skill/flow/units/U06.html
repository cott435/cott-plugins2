    """TC-04: closes 100*1.01^t and 50*0.98^t give r = ln(1.01), ln(0.98) within 1e-12."""

    returns = log_returns(_prices(inputs["prices"]))
    assert isinstance(returns, Returns)

    assert list(returns.dates) == expected["dates"]
    assert list(returns.tickers) == expected["tickers"]
    assert set(returns.values) == set(expected["values"])
    for ticker, values in expected["values"].items():
        assert list(returns.values[ticker]) == pytest.approx(values, abs=expected["tol"])
