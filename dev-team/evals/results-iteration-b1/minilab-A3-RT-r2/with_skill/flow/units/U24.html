+"""Unit tests for risk.summary_stats (spine: daily mean and stdev)."""
+
+from __future__ import annotations
+
+import json
+import math
+from pathlib import Path
+
+import pytest
+
+from risk import Returns, TickerSummary, VolSeries, summary_stats
+
+SAMPLE = Path(__file__).parents[2] / "contracts" / "risk.VolSeries" / "sample.json"
+
+
+def _sample_vol() -> VolSeries:
+    data = json.loads(SAMPLE.read_text())
+    ret = data["returns"]
+    returns = Returns(
+        base_date=ret["base_date"],
+        dates=tuple(ret["dates"]),
+        tickers=tuple(ret["tickers"]),
+        values={t: tuple(v) for t, v in ret["values"].items()},
+    )
+    return VolSeries(
+        returns=returns,
+        window=data["window"],
+        annualized=data["annualized"],
+        values={t: tuple(v) for t, v in data["values"].items()},
+    )
+
+
+def _vol(values: dict[str, tuple[float, ...]]) -> VolSeries:
+    dates = tuple(f"2024-01-{day:02d}" for day in range(2, 2 + len(next(iter(values.values())))))
+    returns = Returns(base_date="2024-01-01", dates=dates, tickers=tuple(values), values=values)
+    return VolSeries(
+        returns=returns,
+        window=2,
+        annualized=False,
+        values={t: (None,) * len(dates) for t in values},
+    )
+
+
+def test_sample_gives_daily_mean_stdev_and_last_vol_per_ticker() -> None:
+    aaa, bbb = summary_stats(_sample_vol())
+
+    assert aaa.ticker == "AAA"
+    assert aaa.ann_return == pytest.approx(0.0088497765, rel=1e-6)
+    assert aaa.ann_vol == pytest.approx(0.016579, rel=1e-3)
+    assert aaa.sharpe == pytest.approx(0.5338, rel=1e-3)
+    assert aaa.last_vol == 0.3880623746755
+    assert bbb.ticker == "BBB"
+    assert bbb.ann_return == pytest.approx(0.0128986916, rel=1e-6)
+    assert bbb.ann_vol == pytest.approx(0.00662006, rel=1e-3)
+    assert bbb.sharpe == pytest.approx(1.9484, rel=1e-3)
+    assert bbb.last_vol == 0.0557071722029
+
+
+def test_two_returns_use_sample_stdev() -> None:
+    (summary,) = summary_stats(_vol({"AAA": (0.01, 0.03)}))
+
+    assert isinstance(summary, TickerSummary)
+    assert summary.ticker == "AAA"
+    assert summary.ann_return == pytest.approx(0.02)
+    assert summary.ann_vol == pytest.approx(math.sqrt(0.0002))
+    assert summary.sharpe == pytest.approx(math.sqrt(2))
+    assert summary.last_vol is None
+
+
+def test_flat_returns_give_zero_sharpe() -> None:

+

+    assert summary.sharpe == 0.0
