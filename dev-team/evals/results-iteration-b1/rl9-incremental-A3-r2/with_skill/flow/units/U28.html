"""viz.Replay: a windowless double stepping through a recording, and the provider check.

`check_replay(replay, game)` takes the provider, so the real Replay (on a headless backend)
runs the same check as the double.
"""

from rllab.runs import GameRecording


class ReplayDouble:
    """Frame indices only: `advance` moves on and stops at the last frame; `show` plays out."""

    def __init__(self, game: GameRecording, *, fps: float = 8.0) -> None:
        self.game = game
        self.fps = fps
        self.index = 0

    def advance(self) -> int:
        self.index = min(self.index + 1, len(self.game.frames) - 1)
        return self.index

    def show(self) -> None:
        while self.index < len(self.game.frames) - 1:
            self.advance()


def check_replay(replay: ReplayDouble, game: GameRecording) -> None:
    """`advance` returns each next frame index, then stays at the last."""
    last = len(game.frames) - 1
    assert [replay.advance() for _ in range(last + 3)] == [*range(1, last + 1), last, last, last]
