8	def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
9	    """Start on the first date every ticker has a close; gaps take the previous close.
10	
11	    GapError naming the ticker when more than `max_gap` dates in a row are missing.
12	    Spine version: keep only the dates every ticker has a close on, with no fill.
13	    """
14	    by_ticker: dict[str, dict[str, float]] = {}
15	    for row in rows:
16	        closes = by_ticker.setdefault(row.ticker, {})
17	        if row.close is not None:
18	            closes[row.date] = row.close
19	    tickers = tuple(sorted(by_ticker))
20	    shared = (
21	        set.intersection(*(set(closes) for closes in by_ticker.values())) if by_ticker else set()
22	    )
23	    dates = tuple(sorted(shared))
24	    return Prices(
25	        dates=dates,
26	        tickers=tickers,
27	        closes={ticker: tuple(by_ticker[ticker][date] for date in dates) for ticker in tickers},

"""PL-01 step 3: put every ticker on one calendar."""

from collections.abc import Mapping, Sequence

from mdata.types import GapError, Prices, RawRow


def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
    """Start on the first date every ticker has a close; gaps take the previous close.

    The calendar is every date on which any ticker has a close, from the first date on
    which every ticker has one. GapError naming the ticker when more than `max_gap`
    dates in a row are missing.
    """
    by_ticker: dict[str, dict[str, float]] = {}
    for row in rows:
        closes = by_ticker.setdefault(row.ticker, {})
        if row.close is not None:
            closes[row.date] = row.close
    tickers = tuple(sorted(by_ticker))
    dates = _calendar(by_ticker)
    return Prices(
        dates=dates,
        tickers=tickers,
        closes={
            ticker: _fill(ticker, by_ticker[ticker], dates, max_gap) for ticker in tickers
        },
    )


def _calendar(by_ticker: Mapping[str, Mapping[str, float]]) -> tuple[str, ...]:
    """Every date any ticker has a close on, from the first date every ticker has one."""
    if not by_ticker or not all(by_ticker.values()):
        return ()
    start = max(min(closes) for closes in by_ticker.values())
    every = set().union(*(set(closes) for closes in by_ticker.values()))
    return tuple(sorted(date for date in every if date >= start))


def _fill(
    ticker: str, closes: Mapping[str, float], dates: Sequence[str], max_gap: int
) -> tuple[float, ...]:
    """One close per date, a missing date taking the previous close; long gaps raise."""
    filled: list[float] = []
    missing = 0
    for date in dates:
        if date in closes:
            filled.append(closes[date])
            missing = 0
            continue
        missing += 1
        if missing > max_gap:
            msg = f"{ticker}: more than {max_gap} dates in a row missing, up to {date}"
            raise GapError(msg)
        filled.append(filled[-1])
    return tuple(filled)
