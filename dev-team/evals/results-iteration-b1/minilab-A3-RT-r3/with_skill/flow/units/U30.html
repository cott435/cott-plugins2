5	from collections.abc import Sequence

7	from mdata.types import Prices, RawRow

10	def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
11	    """Prices from the first date every ticker has a close; a gap takes the previous close.
12	
13	    Raises GapError naming the ticker when more than `max_gap` dates in a row are missing.
14	    Spine version: keeps only the dates every ticker has a close on, with no fill.
15	    """
16	    closes_by_ticker: dict[str, dict[str, float]] = {}
17	    for row in rows:
18	        ticker_closes = closes_by_ticker.setdefault(row.ticker, {})
19	        if row.close is not None:
20	            ticker_closes[row.date] = row.close
21	    tickers = tuple(sorted(closes_by_ticker))
22	    shared = set.intersection(*(set(c) for c in closes_by_ticker.values())) if tickers else set()
23	    dates = tuple(sorted(shared))
24	    closes = {t: tuple(closes_by_ticker[t][d] for d in dates) for t in tickers}
25	    return Prices(dates=dates, tickers=tickers, closes=closes)

from __future__ import annotations

from collections.abc import Sequence

from mdata.types import GapError, Prices, RawRow


def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
    """Prices from the first date every ticker has a close; a gap takes the previous close.

    The calendar is every date any ticker has a close on. Raises GapError naming the ticker
    when more than `max_gap` dates in a row are missing.
    """
    closes_by_ticker: dict[str, dict[str, float]] = {}
    for row in rows:
        ticker_closes = closes_by_ticker.setdefault(row.ticker, {})
        if row.close is not None:
            ticker_closes[row.date] = row.close
    tickers = tuple(sorted(closes_by_ticker))
    calendar = sorted({d for c in closes_by_ticker.values() for d in c})
    filled = {t: _forward_fill(t, closes_by_ticker[t], calendar, max_gap) for t in tickers}
    start = max((calendar.index(min(c)) for c in closes_by_ticker.values() if c), default=0)
    if any(not c for c in closes_by_ticker.values()):
        start = len(calendar)
    closes = {t: tuple(filled[t][start:]) for t in tickers}
    return Prices(dates=tuple(calendar[start:]), tickers=tickers, closes=closes)


def _forward_fill(
    ticker: str, closes: dict[str, float], calendar: list[str], max_gap: int
) -> list[float]:
    """The ticker's close on each calendar date, forward filled; 0.0 before its first close.

    Raises GapError when more than `max_gap` dates in a row after its first close are missing.
    """
    filled: list[float] = []
    previous = 0.0
    missing = 0
    for date in calendar:
        if date in closes:
            previous = closes[date]
            missing = 0
        elif filled and previous:
            missing += 1
            if missing > max_gap:
                raise GapError(f"{ticker} misses more than {max_gap} dates in a row")
        filled.append(previous)
    return filled
