def _rows_from_triples(triples: list[list[object]]) -> list[RawRow]:
    return [RawRow(date=d, ticker=t, close=c) for d, t, c in triples]  # type: ignore[arg-type]


def _rows_from_sample(sample: dict) -> list[RawRow]:
    return [RawRow(date=r["date"], ticker=r["ticker"], close=r["close"]) for r in sample["rows"]]


@pytest.mark.intent
def test_tc01_drops_unusable_rows_keeps_later_duplicate_and_sorts(golden) -> None:
    given, expected = golden("mdata", "TC-01")
    rows = _rows_from_triples(given["rows"])

    result = validate_rows(rows, given["tickers"])

    assert result == _rows_from_triples(expected["rows"])


@pytest.mark.intent
def test_tc01_returns_raw_rows_with_positive_closes_only(golden) -> None:
    given, _ = golden("mdata", "TC-01")

    result = validate_rows(_rows_from_triples(given["rows"]), given["tickers"])

    assert all(isinstance(row, RawRow) for row in result)
    assert all(row.close is not None and row.close > 0 for row in result)
