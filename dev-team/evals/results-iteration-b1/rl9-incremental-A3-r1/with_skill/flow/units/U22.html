def load_settings(path: Path) -> RunSettings:
    """Read a settings TOML into a fully resolved RunSettings."""
    with path.open("rb") as file:
        raw = tomllib.load(file)
    _check_top_level(raw)
    values = {key: raw[key] for key in _top_level_names()}
    values["opponents"] = _opponents(raw["opponents"])
    base_rewards = _TANK_REWARDS if raw["env"] == "tank" else Rewards()
    values["rewards"] = dataclasses.replace(base_rewards, **_table(raw, "rewards", Rewards))
    values["exploration"] = Exploration(**_table(raw, "exploration", Exploration))
    values["params"] = _params(_table(raw, "params", AgentParams))
    return RunSettings(**values)

    """The same settings with only the seed replaced."""
    return dataclasses.replace(settings, seed=seed)

    return tuple(f.name for f in dataclasses.fields(RunSettings) if f.name not in _SECTIONS)


def _check_top_level(raw: dict[str, Any]) -> None:
    known = {f.name for f in dataclasses.fields(RunSettings)}
    for key in raw:
        if key not in known:

def load_player(
    target: str, seed: int, *, env: str, team: str, vs: str
) -> tuple[dict[str, Agent], RunSettings]:
    """One agent per team: `team`'s from the run folder or baseline, the rest from `vs`.

    `seed` seeds a baseline; a run folder carries its own. Single-team environments ignore
    `team` and `vs`."""
    if target in {"random", "scripted"}:

    return _load_run(Path(target))


def score_players(targets: ReportTargets) -> dict[str, list[ScoreSet]]:
    """Each player's ScoreSet per training seed."""
    from_runs = {
        name: [_score(name, player, *_load_run(path)) for path in player.runs]
        for name, player in targets.players.items()
        if player.runs
    }
    seeds = sorted({s.train_seed for sets in from_runs.values() for s in sets})
    from_baselines = {
        name: [_score(name, player, *_load_baseline(player.baseline, seed, "snake")) for seed in seeds]
        for name, player in targets.players.items()
        if player.baseline is not None
    }
    scored = from_runs | from_baselines
    return {name: scored[name] for name in targets.players}


def open_curve_runs(targets: ReportTargets) -> CurveRuns:
    """The ladder and ablation runs; ValueError if an ablation's seeds or steps differ."""
    return CurveRuns(
        ladder={name: _open_runs(targets.players[name]) for name in targets.ladder},
        ablations={name: _open_runs(targets.players[name]) for name in targets.ablations},
    )


def _load_baseline(name: str, seed: int, env: str) -> tuple[dict[str, Agent], RunSettings]:
    """The named baseline on default Snake compact settings, seeded with `seed`."""
    if env != "snake":
        raise ValueError(f"no {name!r} baseline for env {env!r}: the baselines play Snake")
    settings = dataclasses.replace(_BASELINE_SETTINGS, name=name, agent=name, seed=seed)
    if name == "scripted":
        return {SOLO: ScriptedPlayer()}, settings
    n_actions = make_env(settings)().teams[SOLO].actions.n
    return {SOLO: RandomPlayer(n_actions, seed=seed)}, settings


def _load_run(path: Path) -> tuple[dict[str, Agent], RunSettings]:
    """The run's agents as PL-01 built them, each learning team loaded from its checkpoint."""
    record = open_run(path)
    if record.status != "complete":
        raise ValueError(f"run {path}: status {record.status!r}, so no final checkpoint")
    agents = make_agent(record.settings, make_env(record.settings)())
    for team, checkpoint in record.checkpoints.items():
        agents[team].load(checkpoint)
    return agents, record.settings


def _score(
    name: str, player: PlayerTarget, agents: dict[str, Agent], settings: RunSettings
) -> ScoreSet:
    """The held-out evaluation of one player's agents on its settings' training seed."""
    options = EvalOptions(team=player.team)
    return evaluate(
        make_env(settings), agents, player=name, train_seed=settings.seed, options=options
    )

    """A stand-in learner recording the checkpoints it loads."""

    def __init__(self) -> None:
        self.loaded: list[Path] = []

    def act(self, obs: np.ndarray, *, explore: bool) -> np.ndarray:

    def learn(self, t: Transition) -> None:
        raise AssertionError("players never trains")

    def save(self, path: Path) -> None:
        raise AssertionError("players never saves")

    def load(self, path: Path) -> None:
        self.loaded.append(path)
