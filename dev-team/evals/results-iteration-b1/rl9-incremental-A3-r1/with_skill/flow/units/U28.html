def _obs(*on: str) -> np.ndarray:
    """One compact observation with the named fields set to 1."""
    row = np.zeros(len(COMPACT_FIELDS), dtype=np.float32)
    for name in on:

    return row

    actions = ScriptedPlayer().act(np.stack(rows), explore=False)
