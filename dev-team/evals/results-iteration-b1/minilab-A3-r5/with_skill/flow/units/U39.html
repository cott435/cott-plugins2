8	def validate_rows(rows: Sequence[RawRow], tickers: Sequence[str] | None = None) -> list[RawRow]:
9	    """Rows with a positive close, one per (date, ticker), sorted by (date, ticker).
10	
11	    A later row in file order replaces an earlier one with the same (date, ticker).
12	    Raises `KeyError` naming a requested ticker that has no usable row.
13	    """
14	    latest = {
15	        (row.date, row.ticker): row for row in rows if row.close is not None and row.close > 0
16	    }
17	    present = {ticker for _, ticker in latest}
18	    for ticker in tickers or ():
19	        if ticker not in present:
20	            raise KeyError(ticker)
21	    return [latest[key] for key in sorted(latest)]
