+    """A Prices from its sample payload."""
+    return Prices(

+        closes={t: tuple(c) for t, c in data["closes"].items()},
+    )

+    return Prices(dates=prices.dates, tickers=keep, closes={t: prices.closes[t] for t in keep})

+    """A Returns from its sample payload."""
+    return Returns(
+        base_date=data["base_date"],

+        tickers=tuple(data["tickers"]),
+        values={t: tuple(v) for t, v in data["values"].items()},

+    """A VolSeries from its sample payload."""
+    return VolSeries(

+        values={t: tuple(v) for t, v in data["values"].items()},

+
+def vol_series(
+    path: str | Path, *, window: int = 20, tickers: Sequence[str] | None = None
+) -> VolSeries:
+    """The sample for any path; only the requested tickers, KeyError for one it doesn't have.
+
+    ValueError for a window below 2, as the contract says; LookupError for any window but the
+    sample's, which the fake does not hold.
+    """

+        raise ValueError(f"window must be at least 2, not {window}")

+    )
+    if tickers is None:
+        return vol
+    for ticker in tickers:
+        if ticker not in vol.values:
+            raise KeyError(ticker)
+    keep = tuple(sorted(set(tickers)))
+    returns = Returns(
+        base_date=vol.returns.base_date,
+        dates=vol.returns.dates,
+        tickers=keep,
+        values={t: vol.returns.values[t] for t in keep},
+    )
+    return VolSeries(
+        returns=returns,
+        window=vol.window,
+        annualized=vol.annualized,
+        values={t: vol.values[t] for t in keep},
