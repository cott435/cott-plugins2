"""PL-02 step 4: annualize the volatility."""

import math
from dataclasses import replace

from risk.types import VolSeries


def annualize(vol: VolSeries, periods_per_year: int = 252) -> VolSeries:
    """Volatility over a year of trading days; `ValueError` if `vol` is already annualized.

    Each daily value is scaled by the square root of `periods_per_year`; gaps (None) stay gaps.
    """
    if vol.annualized:
        raise ValueError("volatility is already annualized")
    scale = math.sqrt(periods_per_year)
    values = {
        ticker: tuple(None if v is None else v * scale for v in series)
        for ticker, series in vol.values.items()
    }
    return replace(vol, annualized=True, values=values)
