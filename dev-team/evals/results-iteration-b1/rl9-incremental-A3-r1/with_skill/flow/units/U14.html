def _recording(case: str, *, finish: bool = False) -> dict[str, Any]:
    recording: dict[str, Any] = copy.deepcopy(kit.raw_sample("core.Env", case))
    if finish:
        # The recording stops mid-game; end it at its last step (a step-cap truncation).
        recording["steps"][-1]["result"]["truncated"] = True
    return recording


def _greedy_agent(recording: dict[str, Any], team: str, n_games: int) -> Any:
    """An Agent replay double that expects, for each of `n_games` games, the recorded game's
    observations with `explore=False`, returns the recorded actions, and has no `learn` call."""
    observations = [recording["reset"]["obs"][team]]
    observations += [s["result"]["obs"][team] for s in recording["steps"][:-1]]
    actions = [s["actions"][team] for s in recording["steps"]]
    game = [
        {"method": "act", "obs": obs, "explore": False, "returns": act}
        for obs, act in zip(observations, actions, strict=True)
    ]
    return kit.AgentReplay(
        {"team": recording["teams"][team], "calls": game * n_games}
    )


class _EnvFactory:
    """`make_env`: each call builds a fresh replay double of the recording."""

    def __init__(self, recording: dict[str, Any]) -> None:
        self.recording = recording
        self.envs: list[Any] = []

    def __call__(self) -> Any:
        env = kit.EnvReplay(self.recording)
        self.envs.append(env)
        return env

    def reset_seeds(self) -> list[int]:
        return [seed for env in self.envs for seed in env.reset_seeds]


def _final_info(recording: dict[str, Any]) -> dict[str, float]:
    info: dict[str, float] = recording["steps"][-1]["result"]["info"]
    return info
