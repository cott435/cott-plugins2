9	def log_returns(prices: Prices) -> Returns:
10	    """Daily log returns of each ticker; `ValueError` with fewer than two dates."""
11	    if len(prices.dates) < 2:
12	        raise ValueError(f"need at least two dates, got {len(prices.dates)}")
