"""Maximum drawdown with its dates (RISK-05)."""

import math
from itertools import accumulate

from risk.types import Drawdown, Returns


def max_drawdown(returns: Returns) -> tuple[Drawdown, ...]:
    """Per ticker, the deepest fall of wealth (1 on `returns.base_date`) below its running peak.

    The fall's peak and trough dates come with it, the first if tied; with no fall,
    depth is 0 and both dates are `returns.base_date`.
    """
    dates = (returns.base_date, *returns.dates)
    drawdowns = []
    for ticker in returns.tickers:
        depth, peak, trough = _deepest_fall(returns.values[ticker])
        drawdowns.append(
            Drawdown(ticker=ticker, depth=depth, peak_date=dates[peak], trough_date=dates[trough])
        )
    return tuple(drawdowns)


def _deepest_fall(log_returns: tuple[float, ...]) -> tuple[float, int, int]:
    """The lowest wealth / running peak - 1, with its peak and trough indexes.

    Wealth is 1 at index 0, before the first return; the first peak and trough win ties.
    """
    wealth = [1.0, *(math.exp(total) for total in accumulate(log_returns))]
    depth, peak, trough = 0.0, 0, 0
    running_peak = 0
    for i, w in enumerate(wealth):
        if w > wealth[running_peak]:
            running_peak = i
        fall = w / wealth[running_peak] - 1.0
        if fall < depth:
            depth, peak, trough = fall, running_peak, i
    return depth, peak, trough
