def _load(path: Path) -> dict:
    return json.loads(path.read_text(encoding="utf-8"))


def _one_hot(cell: int) -> np.ndarray:
    obs = np.zeros((1, CORRIDOR_CELLS), dtype=np.float32)
    obs[0, cell] = 1.0
    return obs


def _transition(cell: int, action: int, nxt: int, reward: float, done: bool) -> Transition:
    return Transition(
        obs=_one_hot(cell),
        actions=np.array([action], dtype=np.int64),
        reward=np.array([reward], dtype=np.float32),
        next_obs=_one_hot(nxt),
        terminated=np.array([done], dtype=np.bool_),
        active=np.array([True]),
        truncated=False,
    )


def corridor_transitions(n: int, seed: int, goal_reward: float = 1.0) -> list[Transition]:
    """A random walk on the 10-cell corridor: action 1 moves right, 0 left; the last cell ends it."""
    rng = np.random.default_rng(seed)
    out = []
    cell = 0
    for _ in range(n):
        action = int(rng.integers(2))
        nxt = min(max(cell + (1 if action == 1 else -1), 0), CORRIDOR_CELLS - 1)
        done = nxt == CORRIDOR_CELLS - 1
        out.append(_transition(cell, action, nxt, goal_reward if done else 0.0, done))
        cell = 0 if done else nxt
    return out


def _agent(cfg: dict, ablation: str) -> DQNAgent:
    params = AgentParams(
        lr=cfg["lr"],
        gamma=cfg["gamma"],
        batch_size=cfg["batch_size"],
        memory_size=cfg["memory_size"],
        target_every=cfg["target_every"],
        learn_start=cfg["learn_start"],
        net=cfg["net"],
    )
    return DQNAgent(CORRIDOR_TEAM, params, Exploration(), seed=cfg["seed"], ablation=ablation)


def _same(a: dict[str, torch.Tensor], b: dict[str, torch.Tensor]) -> bool:
    return a.keys() == b.keys() and all(torch.equal(a[k], b[k]) for k in a)


def _learn_tracking_updates(agent: DQNAgent, transitions: list[Transition]):
    """Feed every transition; yield (update number, online, target) after each learn that updated."""
    updates = 0
    online = agent.online_state()
    for t in transitions:
        agent.learn(t)
        after = agent.online_state()
        if not _same(after, online):
            updates += 1
            yield updates, after, agent.target_state()
        online = after

    """Two same-seed agents: different early histories, then the same transitions from learn_start on."""
    cfg = data["agent"]

    a = _agent(cfg, ablation)

    for t in corridor_transitions(early, data["seed"]):
