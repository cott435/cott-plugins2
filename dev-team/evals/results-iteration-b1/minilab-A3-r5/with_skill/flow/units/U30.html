"""Whole-period summary per ticker (RISK-04)."""

import math
from statistics import fmean, stdev

from risk.types import TickerSummary, VolSeries


def summary_stats(vol: VolSeries, periods_per_year: int = 252) -> tuple[TickerSummary, ...]:
    """Annualized return, volatility, Sharpe ratio and last rolling volatility per ticker.

    The annualized return is `periods_per_year` times the mean, the annualized volatility
    the square root of `periods_per_year` times the sample standard deviation, and the
    Sharpe ratio their quotient (0 when the volatility is 0; no risk-free rate).
    """
    if len(vol.returns.dates) < 2:
        raise ValueError(f"summary_stats needs at least two returns, got {len(vol.returns.dates)}")
    return tuple(_ticker_summary(vol, ticker, periods_per_year) for ticker in vol.returns.tickers)


def _ticker_summary(vol: VolSeries, ticker: str, periods_per_year: int) -> TickerSummary:
    returns = vol.returns.values[ticker]
    ann_return = fmean(returns) * periods_per_year
    ann_vol = stdev(returns) * math.sqrt(periods_per_year)
    return TickerSummary(
        ticker=ticker,
        ann_return=ann_return,
        ann_vol=ann_vol,
        sharpe=ann_return / ann_vol if ann_vol else 0.0,
        last_vol=vol.values[ticker][-1],
    )
