"""PL-01 step 2: keep the usable rows, sorted."""

from __future__ import annotations

from collections.abc import Sequence

from mdata.types import RawRow


def validate_rows(rows: Sequence[RawRow], tickers: Sequence[str] | None = None) -> list[RawRow]:
    """Rows with a positive close, one per (date, ticker) (the later in file order), sorted by
    (date, ticker); only `tickers` when given.

    Raises KeyError for a requested ticker with no usable row.
    """
    wanted = None if tickers is None else set(tickers)
    latest: dict[tuple[str, str], RawRow] = {}
    for row in rows:
        if row.close is None or row.close <= 0:
            continue
        if wanted is not None and row.ticker not in wanted:
            continue
        latest[(row.date, row.ticker)] = row
    if tickers is not None:
        found = {ticker for _, ticker in latest}
        missing = [ticker for ticker in tickers if ticker not in found]
        if missing:
            raise KeyError(f"no usable row for ticker(s): {', '.join(missing)}")
    return [latest[key] for key in sorted(latest)]
