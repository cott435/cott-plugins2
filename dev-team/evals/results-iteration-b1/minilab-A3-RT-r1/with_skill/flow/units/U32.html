+
+from __future__ import annotations
+
+from pathlib import Path
+
+import mdata
+from mdata.pipeline import load_prices
+from mdata.types import Prices
+
+CSV = (
+    "date,ticker,close\n"
+    "2024-01-02,AAA,10\n"
+    "2024-01-02,BBB,20\n"
+    "2024-01-03,AAA,11\n"
+    "2024-01-03,BBB,21\n"
+    "2024-01-04,BBB,22\n"
+    "2024-01-04,AAA,12\n"
+    "2024-01-05,AAA,\n"
+    "2024-01-05,BBB,23\n"
+)
+
+
+def _write(tmp_path: Path) -> Path:
+    path = tmp_path / "prices.csv"
+    path.write_text(CSV)
+    return path
+
+
+def test_pl01_runs_and_returns_aligned_shape(tmp_path: Path) -> None:
+    prices = load_prices(_write(tmp_path))
+    assert isinstance(prices, Prices)
+    assert prices.tickers == ("AAA", "BBB")
+    assert set(prices.closes) == {"AAA", "BBB"}
+    assert len(prices.dates) > 0
+    for ticker in prices.tickers:
+        assert len(prices.closes[ticker]) == len(prices.dates)
+
+
+def test_pl01_runs_through_package_export_with_tickers(tmp_path: Path) -> None:
+    prices = mdata.load_prices(str(_write(tmp_path)), tickers=["BBB"])
+    assert isinstance(prices, Prices)
+    assert "BBB" in prices.tickers
+    assert len(prices.closes["BBB"]) == len(prices.dates)

"""PL-01 step 2: keep the usable rows, sorted."""

from __future__ import annotations

from collections.abc import Sequence

from mdata.types import RawRow


def validate_rows(rows: Sequence[RawRow], tickers: Sequence[str] | None = None) -> list[RawRow]:
    """Rows with a positive close, one per (date, ticker) (the later in file order), sorted by
    (date, ticker); only `tickers` when given.

    Raises KeyError for a requested ticker with no usable row.
    """
    kept = [row for row in rows if row.close is not None]
    return sorted(kept, key=lambda row: (row.date, row.ticker))

"""PL-01 step 3: put every ticker on one calendar."""

from __future__ import annotations

from collections.abc import Sequence

from mdata.types import Prices, RawRow


def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
    """One calendar from the first date every ticker has a close; a gap takes the previous close.

    Raises GapError naming the ticker when more than `max_gap` dates in a row are missing.
    """
    by_ticker: dict[str, dict[str, float]] = {}
    for row in rows:
        if row.close is not None:
            by_ticker.setdefault(row.ticker, {})[row.date] = row.close
    tickers = tuple(sorted(by_ticker))
    date_sets = [set(closes) for closes in by_ticker.values()]
    dates = tuple(sorted(set.intersection(*date_sets))) if date_sets else ()
    return Prices(
        dates=dates,
        tickers=tickers,
        closes={ticker: tuple(by_ticker[ticker][date] for date in dates) for ticker in tickers},
    )
