def new_figure(panels: int, title: str) -> Figure:
    """A figure of `panels` side-by-side axes on the surface colour, with a left-aligned title."""
    figure = Figure(figsize=(6.4 * panels, 4.6), facecolor=SURFACE, layout="constrained")

        ax.spines[side].set_color(AXIS)
        ax.spines[side].set_linewidth(HAIRLINE)
    ax.legend(frameon=False, labelcolor=SECONDARY, fontsize=8, loc="upper left")
