def load_run_configs(path: Path, seeds: Sequence[int]) -> list[RunConfig]:
    """One `RunConfig` per seed from the TOML at `path`, named `<stem>-s<seed>`."""
    with path.open("rb") as f:
        raw = tomllib.load(f)
    reject_unknown(raw, TOP_KEYS, "config")
    world = str(raw["world"])
    variant = str(raw.get("variant", "default"))
    teams = world_teams(world, variant)
    methods = team_methods(raw["methods"], teams)
    checks = {
        name: settings_for(name, given, defaults={})
        for name, given in raw.get("check_settings", {}).items()
    }
    return [
        RunConfig(
            name=f"{path.stem}-s{seed}",
            world=world,
            variant=variant,
            methods=methods,
            seed=seed,
            budget=int(raw["budget"]),
            check_settings=checks,
            device=str(raw.get("device", "cpu")),
        )
        for seed in seeds
    ]


def reject_unknown(table: Mapping[str, Any], known: frozenset[str], where: str) -> None:
    """Raise `ValueError` naming the first key of `table` not in `known`."""
    for key in table:
        if key not in known:
            raise ValueError(f"unknown key {key!r} in {where}")


def world_teams(world: str, variant: str) -> tuple[str, ...]:
    """The team names of `world` in `variant`; `ValueError` naming an unknown world or variant."""
    if world not in WORLD_TEAMS:
        raise ValueError(f"unknown world {world!r}")
    if variant not in WORLD_TEAMS[world]:
        raise ValueError(f"unknown variant {variant!r} of world {world!r}")
    return WORLD_TEAMS[world][variant]


def team_methods(raw: Mapping[str, Any], teams: tuple[str, ...]) -> dict[str, MethodConfig]:

    reject_unknown(raw, frozenset(teams), "methods")
