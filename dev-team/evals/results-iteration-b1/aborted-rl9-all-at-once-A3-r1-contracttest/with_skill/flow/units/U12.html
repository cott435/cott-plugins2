42	### AD-03: DQN for discrete actions, PPO for continuous ones, one learner per team
43	**Status:** Accepted · **Date:** 2026-10-17 · **Scope:** system
44	- **Context:** Snake has 3 discrete actions (SNAKE-02); fish and predator-prey push with two numbers in [−1, 1] (FISH-02, PREY-02). Many-fish has 8 agents (FISH-06), and predator-prey has two teams learning at once (TRAIN-07). The brief wants each world to teach one idea, and wants checking against a reference implementation (CHECK-02).
45	- **Decision:** `dqn` (replay buffer, target network, ε-greedy, Huber loss) for snake and the discrete check world. `ppo` (Gaussian policy, GAE, clipped objective) for fish, predator-prey and the continuous check world. All agents of a team share one learner (parameter sharing over a batch of agents). Teams have separate learners in the same run. Evaluation and replay act deterministically: greedy for DQN, the mean action for PPO.
46	- **Alternatives:** PPO everywhere: one method fewer, but snake would not teach value learning. SAC for fish: strong on continuous control, but a third method and harder to explain for many agents. Tabular Q-learning for snake: SNAKE-06 includes length, an unbounded number, so the table would need discretising.
47	- **Consequences:** Two methods to write, each with a check world (TRAIN-06) and a Stable-Baselines3 equivalent (AD-08). A prey that leaves the episode is masked out of its team's batch.

Spine version: online one-step Q-learning on a one-hidden-layer MLP with linearly decaying
epsilon-greedy exploration. Each `observe` takes one gradient step on that transition's active
rows; there is no replay buffer and no target network yet.

import copy
import math
from collections.abc import Mapping

import numpy as np
import torch
from torch import nn

from rllab.core.types import Array, Discrete, Policy, Settings, WorldSpec
from rllab.methods.types import MethodConfig, NonFiniteError, Transition

DEFAULTS: dict[str, int | float | str | bool] = {
    "lr": 0.001,
    "gamma": 0.99,
    "hidden": 64,
    "epsilon_start": 1.0,
    "epsilon_end": 0.05,
    "epsilon_decay_steps": 20000,
}

PREFIX = "q."  # the network's keys in `state_dict()`: `q.0.weight`, `q.0.bias`, `q.2.*`


def full_settings(settings: Settings) -> dict[str, int | float | str | bool]:
    """`settings` over the DQN defaults; `ValueError` naming any key DQN doesn't declare."""
    unknown = sorted(set(settings) - set(DEFAULTS))
    if unknown:
        raise ValueError(f"dqn has no settings {unknown}; it declares {sorted(DEFAULTS)}")
    return {**DEFAULTS, **settings}


def q_network(obs_len: int, hidden: int, n_actions: int) -> nn.Sequential:
    """Observation to one value per action: linear, ReLU, linear."""
    return nn.Sequential(nn.Linear(obs_len, hidden), nn.ReLU(), nn.Linear(hidden, n_actions))


def _init_from(net: nn.Sequential, rng: np.random.Generator) -> None:
    """Uniform weights in +-1/sqrt(fan_in) drawn from `rng`, zero biases (CR-13)."""
    with torch.no_grad():
        for layer in net:
            if isinstance(layer, nn.Linear):
                bound = 1.0 / math.sqrt(layer.in_features)
                values = rng.uniform(-bound, bound, size=tuple(layer.weight.shape))
                layer.weight.copy_(torch.as_tensor(values, dtype=torch.float32))
                layer.bias.zero_()


def _q_values(net: nn.Sequential, obs: Array) -> torch.Tensor:
    return net(torch.as_tensor(np.asarray(obs), dtype=torch.float32))


class GreedyPolicy:
    """A frozen Q-network acting greedily: the action with the highest value per row."""

    def __init__(self, net: nn.Sequential) -> None:
        self._net = net

    def act(self, obs: Array, rng: np.random.Generator) -> Array:
        with torch.no_grad():
            return _q_values(self._net, obs).argmax(dim=1).numpy().astype(np.int64)


def greedy_from_state(state: Mapping[str, object]) -> GreedyPolicy:
    """The greedy policy of a `DQN.state_dict()` (tensors, or nested lists from JSON)."""
    weights = {
        key.removeprefix(PREFIX): torch.as_tensor(value, dtype=torch.float32)
        for key, value in state.items()
        if key.startswith(PREFIX)
    }
    first, last = weights["0.weight"], weights["2.weight"]
    net = q_network(first.shape[1], first.shape[0], last.shape[0])
    net.load_state_dict(weights)
    return GreedyPolicy(net)


class DQN:
    """Online one-step Q-learning on an MLP with epsilon-greedy exploration (spine)."""

    def __init__(
        self, config: MethodConfig, spec: WorldSpec, team: str, rng: np.random.Generator
    ) -> None:
        if not isinstance(spec.action, Discrete):
            raise ValueError(f"dqn needs a Discrete action space, not {type(spec.action).__name__}")
        if team not in {t.name for t in spec.teams}:
            raise ValueError(f"{spec.world} has no team {team!r}")
        settings = full_settings(config.settings)
        self._n = spec.action.n
        self._gamma = float(settings["gamma"])
        self._eps_start = float(settings["epsilon_start"])
        self._eps_end = float(settings["epsilon_end"])
        self._eps_steps = int(settings["epsilon_decay_steps"])
        self._net = q_network(spec.obs_len, int(settings["hidden"]), self._n)
        _init_from(self._net, rng)
        self._optim = torch.optim.Adam(self._net.parameters(), lr=float(settings["lr"]))
        self._steps = 0

    def epsilon(self) -> float:
        """The exploration rate now: linear from start to end over the decay steps."""
        frac = min(1.0, self._steps / self._eps_steps) if self._eps_steps > 0 else 1.0
        return self._eps_start + frac * (self._eps_end - self._eps_start)

    def act(self, obs: Array, rng: np.random.Generator) -> Array:
        """Per row: a uniform random action with probability epsilon, else the greedy one."""
        with torch.no_grad():
            greedy = _q_values(self._net, obs).argmax(dim=1).numpy()
        explore = rng.random(len(greedy)) < self.epsilon()
        random = rng.integers(self._n, size=len(greedy))
        return np.where(explore, random, greedy).astype(np.int64)

    def observe(self, t: Transition) -> None:
        """One gradient step on `r + gamma * max Q(next)` (no bootstrap at a true end)."""
        self._steps += 1
        rows = np.asarray(t.active, dtype=bool)
        if not rows.any():
            return
        actions = torch.as_tensor(np.asarray(t.actions)[rows], dtype=torch.int64)
        rewards = torch.as_tensor(np.asarray(t.rewards)[rows], dtype=torch.float32)
        bootstrap = 0.0 if t.done and not t.truncated else self._gamma
        with torch.no_grad():
            future = _q_values(self._net, np.asarray(t.next_obs)[rows]).max(dim=1).values
        target = rewards + bootstrap * future
        chosen = _q_values(self._net, np.asarray(t.obs)[rows]).gather(1, actions[:, None])
        loss = nn.functional.mse_loss(chosen.squeeze(1), target)
        self._optim.zero_grad()
        loss.backward()
        _require_finite(loss, self._net)
        self._optim.step()

    def policy(self) -> Policy:
        """A greedy copy of the network as it is now; later training doesn't change it."""
        return GreedyPolicy(copy.deepcopy(self._net))

    def state_dict(self) -> dict[str, object]:
        """The network's tensors under `q.` and the number of transitions observed."""
        state: dict[str, object] = {
            PREFIX + key: value.detach().clone() for key, value in self._net.state_dict().items()
        }
        state["steps"] = self._steps
        return state


def _require_finite(loss: torch.Tensor, net: nn.Module) -> None:
    """`NonFiniteError` on a non-finite loss, parameter or gradient, before any update."""
    values = [loss.detach().reshape(1)]
    for param in net.parameters():
        values.append(param.detach().reshape(-1))
        if param.grad is not None:
            values.append(param.grad.reshape(-1))
    flat = torch.cat(values)
    bad = flat[~torch.isfinite(flat)]
    if bad.numel() > 0:
        raise NonFiniteError(float(bad[0]))

def dqn(seed: int = 0, **settings) -> DQN:
    config = MethodConfig("dqn", {**DEFAULTS.settings, **settings})
    return DQN(config, SNAKE, TEAM, np.random.default_rng(seed))


def q_values(learner: DQN, obs: np.ndarray) -> np.ndarray:
    """Q-values rebuilt from the learner's state dict, independent of its own forward pass."""
    state = learner.state_dict()
    weights = {k.removeprefix("q."): v for k, v in state.items() if k.startswith("q.")}
    first, last = weights["0.weight"], weights["2.weight"]
    net = q_network(first.shape[1], first.shape[0], last.shape[0])
    net.load_state_dict(weights)
    with torch.no_grad():
        return net(torch.as_tensor(obs, dtype=torch.float32)).numpy()


def loop(reward: float, *, done: bool, truncated: bool) -> Transition:
    """The snake sample's observation stepping to itself with `reward` under action 2."""
    return dataclasses.replace(
        STEP, rewards=np.array([reward], np.float32), done=done, truncated=truncated
    )

def test_act_explores_uniformly_at_epsilon_one():
    obs = np.repeat(STEP.obs, 3000, axis=0)
    actions = dqn(epsilon_start=1.0, epsilon_end=1.0).act(obs, np.random.default_rng(3))
    assert actions.dtype == np.int64
    assert actions.shape == (3000,)
    counts = np.bincount(actions, minlength=3) / 3000
    assert counts == pytest.approx([1 / 3] * 3, abs=0.04)


def test_act_is_greedy_once_epsilon_has_decayed():
    learner = dqn(epsilon_start=1.0, epsilon_end=0.0, epsilon_decay_steps=4)
    assert learner.epsilon() == pytest.approx(1.0)
    for _ in range(2):
        learner.observe(STEP)
    assert learner.epsilon() == pytest.approx(0.5)
    for _ in range(2):
        learner.observe(STEP)
    assert learner.epsilon() == 0.0
    obs = np.random.default_rng(5).normal(size=(50, 12)).astype(np.float32)

    assert np.array_equal(learner.act(obs, np.random.default_rng(6)), expected)

def test_load_policy_acts_greedily_on_the_checkpoint_weights():
    obs = np.random.default_rng(8).normal(size=(30, 12))
    state = CKPT.state[TEAM]
    w0, b0 = np.array(state["q.0.weight"]), np.array(state["q.0.bias"])
    w2, b2 = np.array(state["q.2.weight"]), np.array(state["q.2.bias"])
    expected = (np.maximum(obs @ w0.T + b0, 0) @ w2.T + b2).argmax(axis=1)

    assert np.array_equal(again.act(obs.astype(np.float32), np.random.default_rng(9)), expected)
