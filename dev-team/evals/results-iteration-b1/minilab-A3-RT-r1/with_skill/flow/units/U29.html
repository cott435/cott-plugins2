+"""Unit tests for the max_drawdown spine: depth only, dates at the base date."""
+
+from __future__ import annotations
+
+import json
+import math
+from pathlib import Path
+
+import pytest
+
+from risk.drawdown import max_drawdown
+from risk.types import Drawdown, Returns
+
+SAMPLE = Path(__file__).parents[2] / "contracts" / "risk.Returns" / "sample.json"
+
+
+def _sample_returns() -> Returns:
+    data = json.loads(SAMPLE.read_text())
+    return Returns(
+        base_date=data["base_date"],
+        dates=tuple(data["dates"]),
+        tickers=tuple(data["tickers"]),
+        values={t: tuple(v) for t, v in data["values"].items()},
+    )
+
+
+def test_sample_gives_one_drawdown_per_ticker_in_order() -> None:
+    out = max_drawdown(_sample_returns())
+    assert [d.ticker for d in out] == ["AAA", "BBB"]
+    assert all(isinstance(d, Drawdown) for d in out)
+
+
+def test_sample_depths() -> None:
+    aaa, bbb = max_drawdown(_sample_returns())
+    # AAA peaks after +0.01, then falls by log return -0.02: exp(-0.02) - 1
+    assert aaa.depth == pytest.approx(-0.019801326693244747)
+    # BBB only rises
+    assert bbb.depth == 0.0
+
+
+def test_spine_dates_are_the_base_date() -> None:
+    for d in max_drawdown(_sample_returns()):
+        assert d.peak_date == "2024-01-02"
+        assert d.trough_date == "2024-01-02"
+
+
+def test_fall_from_the_start_is_measured_from_base_wealth() -> None:
+    returns = Returns("2024-01-02", ("2024-01-03", "2024-01-04"), ("AAA",), {"AAA": (-0.1, -0.2)})
+    (d,) = max_drawdown(returns)
+    assert d.depth == pytest.approx(math.exp(-0.3) - 1)
+
+
+def test_deepest_of_two_falls_wins() -> None:
+    dates = ("d1", "d2", "d3", "d4")
+    returns = Returns("d0", dates, ("AAA",), {"AAA": (-0.05, 0.2, -0.15, 0.01)})
+    (d,) = max_drawdown(returns)
+    assert d.depth == pytest.approx(math.exp(-0.15) - 1)
