"""PL-03 step 3: maximum drawdown per ticker."""

from __future__ import annotations

import math
from itertools import accumulate

from risk.types import Drawdown, Returns


def max_drawdown(returns: Returns) -> tuple[Drawdown, ...]:
    """Deepest fall below the running peak of wealth starting at 1, with its dates (RISK-05).

    Spine version: depth only; both dates are the first date (`base_date`).
    """
    return tuple(
        Drawdown(
            ticker=ticker,
            depth=_depth(returns.values[ticker]),
            peak_date=returns.base_date,
            trough_date=returns.base_date,
        )
        for ticker in returns.tickers
    )


def _depth(log_returns: tuple[float, ...]) -> float:
    """Trough / peak - 1 at the deepest fall, with log wealth 0 on the first date."""
    log_wealth = [0.0, *accumulate(log_returns)]
    peaks = accumulate(log_wealth, max)
    return min(math.expm1(wealth - peak) for wealth, peak in zip(log_wealth, peaks, strict=True))

from __future__ import annotations

import math

import pytest

from risk.drawdown import max_drawdown
from risk.types import Returns


def _returns(values: dict[str, tuple[float, ...]], n: int) -> Returns:
    dates = tuple(f"2024-01-{day:02d}" for day in range(2, n + 2))
    return Returns(base_date="2024-01-01", dates=dates, tickers=tuple(values), values=values)


@pytest.fixture
def sample_returns(contract_sample) -> Returns:
    raw = contract_sample("risk.Returns")
    return Returns(
        base_date=raw["base_date"],
        dates=tuple(raw["dates"]),
        tickers=tuple(raw["tickers"]),
        values={ticker: tuple(v) for ticker, v in raw["values"].items()},
    )


def test_sample_gives_one_drawdown_per_ticker_with_its_depth(sample_returns):
    out = max_drawdown(sample_returns)
    assert [d.ticker for d in out] == ["A", "B"]
    # A: log wealth 0, .014, .007, .004, ... -> fall of .010 in log terms.
    assert out[0].depth == pytest.approx(math.exp(-0.010) - 1, abs=1e-12)
    # B: log wealth 0, .005, .010, .000, .000, .015 -> fall of .010 in log terms.
    assert out[1].depth == pytest.approx(math.exp(-0.010) - 1, abs=1e-12)


def test_spine_dates_are_the_first_date(sample_returns):
    out = max_drawdown(sample_returns)
    assert {(d.peak_date, d.trough_date) for d in out} == {("2024-03-04", "2024-03-04")}


def test_known_closes_fall_twenty_percent():
    closes = [100.0, 110.0, 99.0, 88.0, 95.0, 120.0]
    logs = tuple(math.log(b / a) for a, b in zip(closes, closes[1:], strict=False))
    (out,) = max_drawdown(_returns({"X": logs}, len(logs)))
    assert out.depth == pytest.approx(-0.2, abs=1e-12)


def test_rising_series_has_zero_depth():
    (out,) = max_drawdown(_returns({"Y": (0.01, 0.02, 0.0)}, 3))
    assert out.depth == 0.0


def test_first_day_fall_counts_from_wealth_one():
    (out,) = max_drawdown(_returns({"Z": (-0.05, 0.01)}, 2))
    assert out.depth == pytest.approx(math.exp(-0.05) - 1, abs=1e-12)
