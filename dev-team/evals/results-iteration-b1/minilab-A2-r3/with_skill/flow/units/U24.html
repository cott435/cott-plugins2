+    """Spine: only the dates every ticker has a close on; nothing is filled, so no gap applies."""
+    del max_gap
+    closes: dict[str, dict[str, float]] = {}
+    for row in rows:
+        if row.close is not None:
+            closes.setdefault(row.ticker, {})[row.date] = row.close

+    return Prices(
+        dates=dates,
+        tickers=tickers,
+        closes={ticker: tuple(closes[ticker][day] for day in dates) for ticker in tickers},
+    )

+"""PL-01 step 1: read the rows of a `date,ticker,close` CSV."""
+
+from __future__ import annotations
+
+import csv
+from datetime import date
+from pathlib import Path
+
+from mdata.types import RawRow
+
+COLUMNS = ("date", "ticker", "close")
+
+
+def read_csv(path: str | Path) -> list[RawRow]:
+    """Every row in file order; a blank close is None. ValueError for a missing column or bad date."""
+    with Path(path).open(newline="", encoding="utf-8") as handle:
+        reader = csv.DictReader(handle)
+        missing = [column for column in COLUMNS if column not in (reader.fieldnames or ())]
+        if missing:
+            msg = f"{path}: missing column(s) {', '.join(missing)}"
+            raise ValueError(msg)
+        return [_parse_row(record) for record in reader]
+
+
+def _parse_row(record: dict[str, str]) -> RawRow:
+    """One CSV record as a RawRow, checking the date is ISO yyyy-mm-dd."""
+    day = record["date"].strip()
+    if not _is_iso_date(day):
+        msg = f"malformed date {day!r}"
+        raise ValueError(msg)
+    close = record["close"].strip()
+    return RawRow(date=day, ticker=record["ticker"].strip(), close=float(close) if close else None)
+
+
+def _is_iso_date(text: str) -> bool:
+    """True when `text` is a real date written as yyyy-mm-dd."""
+    try:
+        return date.fromisoformat(text).isoformat() == text
+    except ValueError:
+        return False
