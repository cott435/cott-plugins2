"""PL-03 step 3: maximum drawdown per ticker."""

from __future__ import annotations

import math

from risk.types import Drawdown, Returns


def max_drawdown(returns: Returns) -> tuple[Drawdown, ...]:
    """The deepest fall below the running peak, with its dates, per ticker (RISK-05)."""
    dates = (returns.base_date, *returns.dates)
    return tuple(_ticker_drawdown(t, dates, returns.values[t]) for t in returns.tickers)


def _ticker_drawdown(ticker: str, dates: tuple[str, ...], values: tuple[float, ...]) -> Drawdown:
    """Walk the log-price path from the base date; keep the earliest deepest fall."""
    level = peak = 0.0
    peak_date = dates[0]
    worst = Drawdown(ticker=ticker, depth=0.0, peak_date=dates[0], trough_date=dates[0])
    for date, value in zip(dates[1:], values, strict=True):
        level += value
        if level >= peak:
            peak, peak_date = level, date
            continue
        depth = math.expm1(level - peak)
        if depth < worst.depth:
            worst = Drawdown(ticker=ticker, depth=depth, peak_date=peak_date, trough_date=date)
    return worst
