def check_plan(configs: Sequence[RunConfig]) -> dict[str, MethodConfig]:
    """Each distinct team method, keyed by method, with its `check_settings` applied."""
    plan: dict[str, MethodConfig] = {}
    for config in configs:
        for method_config in config.methods.values():
            method = method_config.method
            if method not in plan:
                overrides = config.check_settings.get(method, {})
                plan[method] = MethodConfig(method, {**method_config.settings, **overrides})
    return plan


def train_checks(plan: Mapping[str, MethodConfig], seed: int) -> dict[str, Learner]:
    """Each method's learner, trained for `CHECK_BUDGET` world steps on its check world."""
    return {method: _train_check(config, seed) for method, config in plan.items()}


def _train_check(config: MethodConfig, seed: int) -> Learner:
    world = _check_world(config.method)
    rng = np.random.default_rng(np.random.SeedSequence(seed))
    learner = make_learner(config, world.spec, world.spec.teams[0].name, rng)
    episode = 0
    step = world.reset(seed)
    for _ in range(CHECK_BUDGET):
        actions = learner.act(step.obs, rng)
        after = world.step(actions)
        learner.observe(
            Transition(
                step.obs, actions, after.rewards, after.obs, after.done, after.truncated, after.active
            )
        )
        if after.done or after.truncated:
            episode += 1
            after = world.reset(seed + episode)
        step = after
    return learner

) -> list[CheckResult]:
    """Each method's share of check-world states where its policy picks the best action."""
    return [_score_check(method, learners[method]) for method in plan]


def _score_check(method: str, learner: Learner) -> CheckResult:
    world = _check_world(method)
    states = world.states()
    acted = learner.policy().act(states, np.random.default_rng(0))
    best = world.best_actions()
    gap = np.abs(acted.astype(np.float64) - best.astype(np.float64)).reshape(len(states), -1)
    agreement = float(np.mean(np.all(gap <= BOX_TOLERANCE, axis=1)))
    return CheckResult(method, world.spec.world, agreement, agreement >= PASS_AGREEMENT)
