+"""Fake `risk.vol_series`: serves the risk.VolSeries sample (two tickers, window 2, annualized)."""
+
+from __future__ import annotations
+
+import json
+from collections.abc import Sequence
+from pathlib import Path
+
+from risk.types import Returns, VolSeries
+
+SAMPLE = Path(__file__).resolve().parent.parent / "risk.VolSeries" / "sample.json"
+
+
+def vol_series(
+    path: str | Path, *, window: int = 20, tickers: Sequence[str] | None = None
+) -> VolSeries:
+    """The sample's volatility, whatever `path` is.
+
+    `ValueError` if `window` < 2 (the contract) or is not the sample's window (the fake);
+    `KeyError` for a ticker the sample lacks.
+    """
+    sample = json.loads(SAMPLE.read_text())
+    if window < 2:
+        raise ValueError(f"window must be at least 2, got {window}")
+    if window != sample["window"]:
+        raise ValueError(f"the fake serves window {sample['window']} only, got {window}")
+    raw = sample["returns"]
+    chosen = tuple(raw["tickers"]) if tickers is None else tuple(sorted(set(tickers)))
+    for ticker in chosen:
+        if ticker not in raw["values"]:
+            raise KeyError(ticker)
+    returns = Returns(
+        base_date=raw["base_date"],
+        dates=tuple(raw["dates"]),
+        tickers=chosen,
+        values={ticker: tuple(raw["values"][ticker]) for ticker in chosen},
+    )
+    return VolSeries(
+        returns=returns,
+        window=sample["window"],
+        annualized=sample["annualized"],
+        values={ticker: tuple(sample["values"][ticker]) for ticker in chosen},
+    )
