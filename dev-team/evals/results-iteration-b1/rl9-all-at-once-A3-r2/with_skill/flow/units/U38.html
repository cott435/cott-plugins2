from collections.abc import Mapping

import numpy as np

def play_episode(
    world: World, policies: Mapping[str, Policy], seed: int, *, trace: bool = False
) -> Episode:
    """Reset `world` with `seed`, let each team's policy act until done or `max_steps`.

    The world gets `seed` itself; each team's policy a generator spawned from it (AD-05).
    """
    spec = world.spec
    for team in spec.teams:
        if team.name not in policies:
            raise KeyError(f"no policy for team {team.name!r}")
    rngs = policy_rngs(spec, seed)
    step = world.reset(seed)
    scores = {team.name: 0.0 for team in spec.teams}
    frames = [frame(world, 0, scores, np.zeros_like(step.rewards))] if trace else None
    length = 0
    while not step.done and length < spec.max_steps:
        step = world.step(team_actions(spec, policies, rngs, step))
        length += 1
        scores = add_rewards(spec, scores, step.rewards)
        if frames is not None:
            frames.append(frame(world, length, scores, step.rewards))
    return Episode(
        result=world.result(),
        frames=tuple(frames) if frames is not None else None,
        hit_step_limit=length == spec.max_steps,
    )


def policy_rngs(spec: WorldSpec, seed: int) -> dict[str, np.random.Generator]:
    """One generator per team, spawned from the episode seed."""
    children = np.random.SeedSequence(seed).spawn(len(spec.teams))
    return {
        team.name: np.random.default_rng(child)
        for team, child in zip(spec.teams, children, strict=True)
    }


def team_actions(
    spec: WorldSpec,
    policies: Mapping[str, Policy],
    rngs: Mapping[str, np.random.Generator],
    step: Step,
) -> Array:
    """Each team's policy acts on its agents' rows; one action row per agent, in agent order."""
    by_agent: dict[int, Array] = {}
    for team in spec.teams:
        agents = list(team.agents)
        acted = policies[team.name].act(step.obs[agents], rngs[team.name])
        by_agent.update(zip(agents, acted, strict=True))
    return np.stack([by_agent[agent] for agent in range(len(spec.agents))])


def add_rewards(spec: WorldSpec, scores: Mapping[str, float], rewards: Array) -> dict[str, float]:
    """Each team's running score after this step's rewards."""
    return {
        team.name: scores[team.name] + float(rewards[list(team.agents)].sum())
        for team in spec.teams
    }

    return Frame(step=at, draw=world.draw(), scores=dict(scores), rewards=rewards)

def check_actions(spec: WorldSpec, actions: Array, active: Array) -> None:
    """ActionError for the first active agent whose action is outside `spec.action`."""
    for i, row in enumerate(np.asarray(actions)):
        if active[i] and not inside(spec, row):
            raise ActionError(spec.agents[i], row.tolist())


def inside(spec: WorldSpec, action: Array) -> bool:
    if isinstance(spec.action, Box):
        low, high = spec.action.low, spec.action.high
        return action.shape == (spec.action.dims,) and bool(
            np.all((action >= low) & (action <= high))
        )
    return action.shape == () and float(action).is_integer() and 0 <= action < spec.action.n


@dataclass
class ReplayWorld:
    spec: WorldSpec
    steps: list[Step]
    draws: list[DrawState]
    recorded: EpisodeResult
    at: int | None = field(default=None, init=False)
    seed: int = field(default=0, init=False)

    def reset(self, seed: int) -> Step:
        self.at, self.seed = 0, seed
        return self.steps[0]

    def step(self, actions: Array) -> Step:
        if self.at is None or self.steps[self.at].done:
            raise RuntimeError("step before reset or after done")
        check_actions(self.spec, actions, self.steps[self.at].active)
        self.at += 1
        return self.steps[self.at]

    def result(self) -> EpisodeResult:
        return replace(self.recorded, seed=self.seed)

    def draw(self) -> DrawState:
        return self.draws[self.at or 0]

        kit.decode(EpisodeResult, recording["result"]),

def rule_for(world: str, variant: str, team: str) -> RulePolicy:
    """The rule baseline of `team` in `world`'s `variant`; `KeyError` naming all three if none."""
    match (world, variant, team):
        case ("snake", "default", "snake"):
            return SnakeRule()
    raise KeyError(f"no rule for team {team!r} of world {world!r}, variant {variant!r}")


def baseline_means(world: str, variant: str) -> BaselineMeans:
    """Random and rule mean score per team on `EVAL_SEEDS`, each against rule-based opponents."""
    try:
        played = make(world, variant)
    except ValueError as error:
        raise KeyError(f"no rule for world {world!r}, variant {variant!r}: {error}") from error
    spec = played.spec
    random: dict[str, float] = {}
    rule: dict[str, float] = {}
    for team in spec.teams:
        opponents: dict[str, Policy] = {
            other.name: rule_for(world, variant, other.name)
            for other in spec.teams
            if other.name != team.name
        }
        random[team.name] = mean_score(
            played, {**opponents, team.name: RandomPolicy(spec.action)}, team.name
        )
        rule[team.name] = mean_score(
            played, {**opponents, team.name: rule_for(world, variant, team.name)}, team.name
        )
    return BaselineMeans(world=world, variant=variant, random=random, rule=rule)

    return fmean(play_episode(world, policies, seed).result.scores[team] for seed in EVAL_SEEDS)
