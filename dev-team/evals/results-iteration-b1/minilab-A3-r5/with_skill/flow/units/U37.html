+"""`mdata.load_prices` re-frozen (M1-T22): with `tickers`, exactly those tickers, sorted, and the
+calendar start and gap check see only their rows; None keeps every ticker in the file.
+
+The cases run on `late_gap.csv`, where CRAG starts later than ACME and BOLT and then misses six
+dates in a row. They take the provider as the `load_prices` fixture: here the fake, which the real
+provider's answers match (the same CSV, the same answers).
+"""
+
+import csv
+import json
+from collections.abc import Callable, Sequence
+from pathlib import Path
+from typing import Any
+
+import pytest
+
+from mdata import GapError, Prices
+
+FOLDER = Path(__file__).resolve().parent
+LATE_GAP_CSV = FOLDER / "late_gap.csv"
+
+
+@pytest.fixture
+def load_prices(contract_fake) -> Callable[..., Prices]:
+    """The provider held to the cases below: the `mdata.load_prices` fake."""
+    return contract_fake("mdata.load_prices").load_prices
+
+
+def csv_closes(path: Path, tickers: Sequence[str]) -> dict[str, dict[str, float]]:
+    """Each requested ticker's close by date, read straight from the CSV."""
+    closes: dict[str, dict[str, float]] = {ticker: {} for ticker in tickers}
+    with path.open(newline="") as handle:
+        for row in csv.DictReader(handle):
+            if row["ticker"] in closes:
+                closes[row["ticker"]][row["date"]] = float(row["close"])
+    return closes
+
+
+def recorded() -> dict[str, Any]:
+    return json.loads((FOLDER / "late_gap.json").read_text())
+
+
+def test_late_gap_answers_match_the_prices_schema(contract_schema, check_schema):
+    schema = contract_schema("mdata.Prices")
+    assert check_schema(recorded()["prices"], schema) == []
+    assert check_schema(recorded()["gapped"]["CRAG"], schema) == []
+
+
+@pytest.mark.parametrize(
+    ("tickers", "answer"),
+    [(["ACME", "BOLT"], ("prices",)), (["CRAG"], ("gapped", "CRAG"))],
+)
+def test_late_gap_answers_hold_the_csv_closes(tickers, answer):
+    """Each recorded answer is its tickers' CSV closes, on every date they all trade."""
+    payload = recorded()
+    for key in answer:
+        payload = payload[key]
+    closes = csv_closes(LATE_GAP_CSV, tickers)
+    dates = sorted(set.intersection(*(set(by_date) for by_date in closes.values())))
+    assert payload["dates"] == dates
+    assert payload["tickers"] == tickers
+    assert payload["closes"] == {t: [by_date[d] for d in dates] for t, by_date in closes.items()}
+    assert list(recorded()["gapped"]) == ["CRAG"]
+
+
+@pytest.mark.parametrize("tickers", [["ACME", "BOLT"], ["BOLT", "ACME"], ["BOLT"]])
+def test_load_prices_aligns_only_the_requested_tickers(load_prices, tickers):
+    """CRAG, unrequested, neither moves the calendar's start nor raises GapError."""
+    prices = load_prices(LATE_GAP_CSV, tickers)
+    closes = csv_closes(LATE_GAP_CSV, tickers)
+    dates = sorted(set.intersection(*(set(by_date) for by_date in closes.values())))
+    assert prices.tickers == tuple(sorted(tickers))
+    assert prices.dates == tuple(dates)
+    assert prices.dates[0] == "2024-03-04"
+    assert len(prices.dates) == 10
+    assert dict(prices.closes) == {
+        ticker: tuple(by_date[d] for d in dates) for ticker, by_date in closes.items()
+    }
+
+
+def test_load_prices_checks_a_lone_ticker_on_its_own_dates(load_prices):
+    """CRAG alone: its calendar is its own two dates, so it has no gap to fill."""
+    prices = load_prices(LATE_GAP_CSV, ["CRAG"])
+    assert prices == Prices(
