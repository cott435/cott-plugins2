"""Rolling volatility from earlier returns (RISK-02)."""

from statistics import stdev

from risk.types import Returns, VolSeries


def rolling_vol(returns: Returns, window: int) -> VolSeries:
    """The sample stdev of the `window` returns before each date, never that date's own.

    Spine version: the stdev runs over every earlier return once `window` of them exist.
    """
    if window < 2:
        raise ValueError(f"window must be at least 2, got {window}")
    values = {
        ticker: tuple(stdev(series[:i]) if i >= window else None for i in range(len(series)))
        for ticker, series in returns.values.items()
    }
    return VolSeries(returns=returns, window=window, annualized=False, values=values)

import math

import pytest

from risk.types import Returns
from risk.vol import rolling_vol


def _returns(sample: dict) -> Returns:
    return Returns(
        base_date=sample["base_date"],
        dates=tuple(sample["dates"]),
        tickers=tuple(sample["tickers"]),
        values={t: tuple(v) for t, v in sample["values"].items()},

def test_sample_gives_none_until_window_earlier_returns(contract_sample) -> None:
    vol = rolling_vol(_returns(contract_sample("risk.Returns")), 2)
    for ticker in ("ACME", "BOLT"):
        assert vol.values[ticker][:2] == (None, None)
        assert all(v is not None for v in vol.values[ticker][2:])


def test_sample_values_are_expanding_stdev_of_earlier_returns(contract_sample) -> None:
    vol = rolling_vol(_returns(contract_sample("risk.Returns")), 2)

    # ACME returns: 0.012, -0.008, 0.021, 0.005, -0.015; each value excludes its own date.

"""Maximum drawdown with its dates (RISK-05)."""

import math
from itertools import accumulate

from risk.types import Drawdown, Returns


def max_drawdown(returns: Returns) -> tuple[Drawdown, ...]:
    """Each ticker's deepest fall below its running peak, wealth starting at 1 on `base_date`.

    Spine version: depth only; both dates are `base_date`.
    """
    return tuple(
        Drawdown(
            ticker=ticker,
            depth=_depth(returns.values[ticker]),
            peak_date=returns.base_date,
            trough_date=returns.base_date,
        )
        for ticker in returns.tickers
    )


def _depth(log_returns: tuple[float, ...]) -> float:
    """trough / peak - 1 for the worst fall of log wealth below its running peak."""
    log_wealth = list(accumulate(log_returns, initial=0.0))
    peaks = accumulate(log_wealth, max)
    worst = min(w - p for w, p in zip(log_wealth, peaks, strict=True))
    return math.expm1(worst)

def test_sample_depths_are_worst_fall_of_log_wealth(contract_sample) -> None:
    drawdowns = {d.ticker: d for d in max_drawdown(_returns(contract_sample("risk.Returns")))}
    # ACME log wealth: 0, .012, .004, .025, .030, .015 -> worst fall .030 to .015.
