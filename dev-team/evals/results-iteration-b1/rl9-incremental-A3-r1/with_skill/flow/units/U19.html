def train(
    make_env: Callable[[], Env],
    agents: Mapping[str, Agent],
    settings: RunSettings,
    *,
    root: Path = Path("runs"),
) -> RunRecord:
    """Train for `settings.steps` steps, evaluating every `eval_every`, into a run folder."""
    env = make_env()
    learners = learning_teams(env, agents, settings)
    writer = create_run(settings, root=root)
    loop = _Loop(make_env=make_env, agents=agents, settings=settings, writer=writer, learners=learners)
    try:
        games = loop.run(env)
    except KeyboardInterrupt:
        writer.finish("incomplete")
        raise
    for team in learners:
        writer.save_checkpoint(agents[team], team=team)
    writer.save_games(games)
    return writer.finish("complete")


def learning_teams(env: Env, agents: Mapping[str, Agent], settings: RunSettings) -> tuple[str, ...]:
    """The teams that learn, in the environment's order: those `opponents` marks `learn`, or
    the one team of a single-team run. ValueError unless `agents` has one agent per team."""
    if set(agents) != set(env.teams):
        raise ValueError(f"agents {sorted(agents)} must name the teams {sorted(env.teams)}")
    learners = tuple(t for t in env.teams if settings.opponents.get(t, "learn") == "learn")
    if not learners:
        raise ValueError("no team learns in this run")
    return learners

class RightAgent:
    """Always steps right (action 1); keeps every Transition and `explore` flag it gets."""

    def __init__(self, interrupt_at: int = -1) -> None:
        self.transitions: list[Transition] = []
        self.explores: list[bool] = []
        self.interrupt_at = interrupt_at
        self.loaded: Path | None = None

    def act(self, obs: np.ndarray, *, explore: bool) -> np.ndarray:
        self.explores.append(explore)
        return np.ones(len(obs), dtype=np.int64)

    def learn(self, t: Transition) -> None:
        if len(self.transitions) == self.interrupt_at:
            raise KeyboardInterrupt
        self.transitions.append(t)

    def save(self, path: Path) -> None:
        path.write_text("right", encoding="utf-8")

    def load(self, path: Path) -> None:
        self.loaded = path


class Clock:
