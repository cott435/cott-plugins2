def max_drawdown(returns: Returns) -> tuple[Drawdown, ...]:
    """Each ticker's worst fall of compounded returns from a running peak, as depth ≤ 0."""
    return tuple(
        Drawdown(ticker=ticker, depth=_depth(returns.values[ticker]), peak_date=returns.base_date)
        for ticker in returns.tickers
    )


def _depth(series: tuple[float, ...]) -> float:
    wealth = peak = 1.0
    depth = 0.0
    for value in series:
        wealth *= 1.0 + value
        peak = max(peak, wealth)
        depth = min(depth, wealth / peak - 1.0)
    return depth

    """Join summary and drawdown by ticker, write `{"tickers": rows}`; ValueError if they differ."""
    summary_tickers = sorted(stats.ticker for stats in summary)
