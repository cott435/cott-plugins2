"""Paired permutation tests against each baseline, with Holm's correction (EVAL-04).

Designed with the null-hypothesis-testing skill. H0: the learned agent's mean score is no
better than the baseline's; H1: it is better in the team's score direction (one-sided). The
statistic is the mean over seeds of the learned score (averaged over runs) minus the baseline
score, signed so positive means better. The null swaps the learned and baseline labels within
each seed, which flips the sign of that seed's difference.
"""

from collections.abc import Mapping, Sequence
from typing import Literal

import numpy as np

from rllab.core import Array, Direction
from rllab.evaluate.types import Comparison

ALPHA = 0.05


def compare(
    learned: Array,
    baseline: Array,
    direction: Direction,
    *,
    permutations: int = 10_000,
    rng: np.random.Generator,
) -> tuple[float, float]:
    """(difference, one-sided p) of learned (runs x seeds) against baseline (one per seed).

    The p-value counts the observed labelling as one of the permutations, so it is never 0.
    """
    per_seed = np.asarray(learned, dtype=np.float64).mean(axis=0)
    base = np.asarray(baseline, dtype=np.float64)
    if per_seed.shape != base.shape:
        raise ValueError(
            f"learned has {per_seed.shape[0]} seeds but baseline has {base.shape[0]} seeds"
        )
    sign = 1.0 if direction == "higher" else -1.0
    differences = sign * (per_seed - base)
    observed = float(differences.mean())
    swaps = rng.choice(np.array([-1.0, 1.0]), size=(permutations, differences.size))
    null = (swaps * differences).mean(axis=1)
    at_least = int(np.count_nonzero(null >= observed))
    return observed, (at_least + 1) / (permutations + 1)


def holm(p_values: Sequence[float]) -> tuple[float, ...]:
    """Holm's step-down adjusted p-values, in the input order."""
    m = len(p_values)
    adjusted = [0.0] * m
    running = 0.0
    for rank, i in enumerate(sorted(range(m), key=lambda j: p_values[j])):
        running = max(running, min(1.0, (m - rank) * p_values[i]))
        adjusted[i] = running
    return tuple(adjusted)


def compare_baselines(
    team: str,
    learned: Array,
    baselines: Mapping[Literal["random", "rule"], Array],
    direction: Direction,
    *,
    rng: np.random.Generator,
) -> tuple[Comparison, ...]:
    """One `Comparison` per baseline, with Holm's correction across them at `ALPHA`."""
    names = list(baselines)
    tests = [compare(learned, baselines[name], direction, rng=rng) for name in names]
    corrected = holm([p for _, p in tests])
    return tuple(
        Comparison(
            baseline=name,
            team=team,
            learned_mean=float(np.mean(learned)),
            baseline_mean=float(np.mean(baselines[name])),
            difference=difference,
            p_raw=p_raw,
            p_holm=p_holm,
            rejected=p_holm < ALPHA,
        )
        for name, (difference, p_raw), p_holm in zip(names, tests, corrected, strict=True)
    )
