def _cnn_params(**changes: Any) -> AgentParams:
    base = kit.sample("core.AgentParams", "dqn_compact", AgentParams)

def _tensors(state: Any) -> Iterator[torch.Tensor]:
    if isinstance(state, torch.Tensor):
        yield state
    elif isinstance(state, Mapping):
        for value in state.values():
            yield from _tensors(value)


def _weight_shapes(agent: DQNAgent, tmp_path: Path) -> list[tuple[int, ...]]:
    path = tmp_path / "checkpoint.pt"
    agent.save(path)
    state = torch.load(path, map_location="cpu", weights_only=True)
    return [tuple(t.shape) for t in _tensors(state)]


def _grid_transitions() -> list[Transition]:
    env = _grid_env()
    obs, _ = env.reset(env.seed)
    out = []
    for actions in env.recorded_actions:
        result = env.step(actions)
        done = np.asarray(result.done[SOLO], dtype=bool)
        out.append(
            Transition(
                obs=obs[SOLO],
                actions=actions[SOLO],
                reward=np.asarray(result.reward[SOLO], dtype=np.float32),
                next_obs=result.obs[SOLO],
                terminated=done if result.terminated else np.zeros_like(done),
                active=np.ones_like(done),
                truncated=bool(result.truncated),
            )
        )
        obs = result.obs
    return out

def _tensors(state: Any) -> Iterator[torch.Tensor]:
    if isinstance(state, torch.Tensor):
