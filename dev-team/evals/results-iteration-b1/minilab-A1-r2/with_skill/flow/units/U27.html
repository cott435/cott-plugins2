"""PL-02 step 2: log returns."""

from __future__ import annotations

import math

from mdata.types import Prices
from risk.types import Returns


def log_returns(prices: Prices) -> Returns:
    """Daily log returns `ln(close_t / close_t-1)` of each ticker, dated t.

    Raises `ValueError` with fewer than two dates.
    """
    if len(prices.dates) < 2:
        msg = f"log_returns needs at least two dates, got {len(prices.dates)}"
        raise ValueError(msg)
    values = {
        ticker: tuple(
            math.log(cur / prev)
            for prev, cur in zip(prices.closes[ticker], prices.closes[ticker][1:], strict=False)
        )
        for ticker in prices.tickers
    }
    return Returns(
        base_date=prices.dates[0],
        dates=prices.dates[1:],
        tickers=prices.tickers,
        values=values,
    )
