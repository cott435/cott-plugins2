        "statistic: difference in pooled mean score (one-sided); "
        f"null: permutation of player labels over pooled games ({PERMUTATIONS} shuffles); "
        f"correction: Holm over {len(specs)} claim(s)"
    )
    claims = [
        Claim(
            spec=spec,
            mean_a=float(pooled[spec.a].mean()),
            mean_b=float(pooled[spec.b].mean()),
            p_value=p,
            p_corrected=p_corr,
            passed=claim_passed(spec, pooled, p_corr, alpha),
            method=method,
        )
        for spec, p, p_corr in zip(specs, p_values, corrected, strict=True)
    ]
    write_report(out / "report.md", pooled, claims, charts)
    return claims

def tested_pair(spec: ClaimSpec, pooled: Mapping[str, np.ndarray]) -> tuple[np.ndarray, np.ndarray]:
    """The (x, y) whose H1 is mean(x) > mean(y): a over b, or b over a for "not worse"."""
    if spec.direction == "greater":
        return pooled[spec.a], pooled[spec.b]
    return pooled[spec.b], pooled[spec.a]


def permutation_p_value(x: np.ndarray, y: np.ndarray) -> float:
    """One-sided p-value of mean(x) - mean(y) under shuffled labels, (hits + 1) / (n + 1)."""
    rng = np.random.default_rng(PERMUTATION_SEED)
    both = np.concatenate([x, y])
    observed = x.mean() - y.mean()
    hits = 0
    for start in range(0, PERMUTATIONS, PERMUTATION_BATCH):
        n = min(PERMUTATION_BATCH, PERMUTATIONS - start)
        shuffled = rng.permuted(np.tile(both, (n, 1)), axis=1)
        diffs = shuffled[:, : len(x)].mean(axis=1) - shuffled[:, len(x) :].mean(axis=1)
        hits += int(np.count_nonzero(diffs >= observed))
    return (hits + 1) / (PERMUTATIONS + 1)


def holm(p_values: Sequence[float]) -> list[float]:
    """Holm step-down adjusted p-values, in input order."""
    m = len(p_values)
    adjusted = [0.0] * m
    running = 0.0
    for rank, i in enumerate(sorted(range(m), key=lambda k: p_values[k])):
        running = max(running, min(1.0, (m - rank) * p_values[i]))
        adjusted[i] = running
    return adjusted


def claim_passed(
    spec: ClaimSpec, pooled: Mapping[str, np.ndarray], p_corrected: float, alpha: float

    """`greater`: significant; `not_worse`: "a worse" not significant and the ratio bar holds."""
    significant = p_corrected <= alpha
    if spec.direction == "greater":
        return significant
    if spec.min_ratio is None:
        return not significant
    return not significant and ratio_holds(
        float(pooled[spec.a].mean()), float(pooled[spec.b].mean()), spec.min_ratio
    )


def ratio_holds(mean_a: float, mean_b: float, min_ratio: float) -> bool:
    """`mean_a >= min_ratio * mean_b`, inclusive up to float rounding."""
    bar = min_ratio * mean_b
    return mean_a >= bar or math.isclose(mean_a, bar)

def _step_and_learn(
    env: Env,
    agents: Mapping[str, Agent],
    learners: list[str],
    obs: Mapping[str, np.ndarray],
    *,
    active: Mapping[str, np.ndarray],
) -> StepResult:
    """Every team acts (exploring), the env steps, then each learner learns from its step."""
    actions = {team: agents[team].act(obs[team], explore=True) for team in env.teams}
    result = env.step(actions)
    for team in learners:
        agents[team].learn(_transition(team, obs, actions, result, active=active[team]))
    return result


def _all_active(obs: Mapping[str, np.ndarray]) -> dict[str, np.ndarray]:
    return {team: np.ones(len(batch), dtype=np.bool_) for team, batch in obs.items()}


def _transition(
    team: str,
    obs: Mapping[str, np.ndarray],
    actions: Mapping[str, np.ndarray],
    result: StepResult,
    *,
    active: np.ndarray,
) -> Transition:
    """One team's batch-first step; agents done only by truncation are not terminated."""
    return Transition(
        obs=obs[team],
        actions=actions[team],
        reward=result.reward[team],
        next_obs=result.obs[team],
        terminated=result.done[team] & active & (not result.truncated),
        active=active,
        truncated=result.truncated,
    )


def _mean_score(make_env: Callable[[], Env], agents: Mapping[str, Agent], spec: EvalSpec) -> float:
    """A greedy evaluation of the first learning team against the run's opposition."""
    return float(np.mean(evaluate(make_env, agents, spec).scores))

def open_curve_runs(targets: ReportTargets) -> CurveRuns:
    """The ladder's and the ablations' runs; `ValueError` if an ablation's seeds or steps differ."""
    players = {player.name: player for player in targets.players}
    ablation_names = [name for pair in targets.ablations.items() for name in pair]
    ablations = {name: _open_runs(players[name]) for name in dict.fromkeys(ablation_names)}
    for ablation, full in targets.ablations.items():
        _check_matches(ablation, ablations[ablation], full, ablations[full])
    return CurveRuns(
        ladder={name: _open_runs(players[name]) for name in targets.ladder},
        ablations=ablations,
    )


def _check_matches(
    ablation: str, runs: tuple[RunRecord, ...], full: str, full_runs: tuple[RunRecord, ...]
) -> None:
    """`ValueError` naming `seeds` or `steps` if the ablation's runs differ from its full agent's."""
    seeds = sorted(run.settings.seed for run in runs)
    full_seeds = sorted(run.settings.seed for run in full_runs)
    if seeds != full_seeds:
        raise ValueError(f"ablation {ablation}: seeds {seeds} differ from {full}'s {full_seeds}")
    steps = {run.settings.steps for run in runs}
    full_steps = {run.settings.steps for run in full_runs}
    if steps != full_steps:
        raise ValueError(
            f"ablation {ablation}: steps {sorted(steps)} differ from {full}'s {sorted(full_steps)}"
        )
