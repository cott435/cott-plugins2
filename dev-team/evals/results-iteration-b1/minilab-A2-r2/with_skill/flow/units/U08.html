from __future__ import annotations

import math

import pytest

from mdata import Prices
from risk import Returns, log_returns


def _prices(payload: dict) -> Prices:
    return Prices(
        dates=tuple(payload["dates"]),
        tickers=tuple(payload["tickers"]),
        closes={t: tuple(v) for t, v in payload["closes"].items()},
    )


@pytest.mark.intent
def test_tc04_geometric_closes_give_closed_form_log_returns(golden):
    given, expected = golden("risk", "TC-04")
    returns = log_returns(_prices(given["prices"]))
    tol = expected["tol"]
    assert isinstance(returns, Returns)
    assert returns.base_date == expected["base_date"]
    assert returns.dates == tuple(expected["dates"])
    assert returns.tickers == tuple(expected["tickers"])
    assert set(returns.values) == set(expected["values"])
    for ticker, want in expected["values"].items():
        got = returns.values[ticker]
        assert len(got) == len(want)
        assert got == pytest.approx(want, abs=tol, rel=0)


@pytest.mark.intent
def test_risk01_each_return_is_log_of_close_ratio_dated_t(contract_sample):
    sample = contract_sample("mdata.Prices")
    returns = log_returns(_prices(sample))
    assert returns.base_date == sample["dates"][0]
    assert returns.dates == tuple(sample["dates"][1:])
    assert returns.tickers == tuple(sample["tickers"])
    for ticker in sample["tickers"]:
        closes = sample["closes"][ticker]
        want = [math.log(closes[i] / closes[i - 1]) for i in range(1, len(closes))]
        assert returns.values[ticker] == pytest.approx(want, abs=1e-12, rel=0)


@pytest.mark.intent
def test_risk01_returns_are_log_not_simple(contract_sample):
    sample = contract_sample("mdata.Prices")
    returns = log_returns(_prices(sample))
    closes = sample["closes"]["AAPL"]
    simple = closes[1] / closes[0] - 1.0
    assert returns.values["AAPL"][0] != pytest.approx(simple, abs=1e-9, rel=0)
    assert returns.values["AAPL"][0] == pytest.approx(math.log(closes[1] / closes[0]), abs=1e-12, rel=0)


@pytest.mark.intent
def test_two_dates_give_one_return():
    prices = Prices(
        dates=("2024-01-02", "2024-01-03"),
        tickers=("X",),
        closes={"X": (100.0, 110.0)},
    )
    returns = log_returns(prices)
    assert returns.base_date == "2024-01-02"
    assert returns.dates == ("2024-01-03",)
    assert returns.values["X"] == pytest.approx((math.log(1.1),), abs=1e-12, rel=0)
