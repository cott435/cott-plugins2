+"""PL-01 step 3: align the rows to one calendar."""
+
+from __future__ import annotations
+
+from collections.abc import Mapping, Sequence
+
+from mdata.types import GapError, Prices, RawRow
+
+
+def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
+    """Prices on every trading date from the first date all tickers have a close.
+
+    A missing date takes the ticker's previous close; more than `max_gap` missing
+    dates in a row raise `GapError` naming the ticker.
+    """
+    closes = {(row.date, row.ticker): row.close for row in rows if row.close is not None}
+    tickers = tuple(sorted({ticker for _, ticker in closes}))
+    all_dates = sorted({date for date, _ in closes})
+    start = next(
+        (i for i, date in enumerate(all_dates) if all((date, t) in closes for t in tickers)),
+        len(all_dates),
+    )
+    dates = tuple(all_dates[start:])
+    return Prices(
+        dates=dates,
+        tickers=tickers,
+        closes={ticker: _fill(ticker, dates, closes, max_gap) for ticker in tickers},
+    )
+
+
+def _fill(
+    ticker: str,
+    dates: Sequence[str],
+    closes: Mapping[tuple[str, str], float],
+    max_gap: int,
+) -> tuple[float, ...]:
+    """One ticker's closes on `dates`, each missing date taking the previous close."""
+    filled: list[float] = []
+    missing = 0
+    for date in dates:
+        close = closes.get((date, ticker))
+        if close is None:
+            missing += 1
+            if missing > max_gap:
+                msg = f"{ticker} misses more than {max_gap} dates in a row (through {date})"
+                raise GapError(msg)
+            filled.append(filled[-1])
+        else:
+            missing = 0
+            filled.append(close)
+    return tuple(filled)
