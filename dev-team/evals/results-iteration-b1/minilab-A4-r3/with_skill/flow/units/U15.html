"""Fake `mdata.load_prices`: serves the mdata.Prices sample (the same closes as prices.csv)."""

from __future__ import annotations

import json
from collections.abc import Sequence
from pathlib import Path

from mdata import Prices

SAMPLE = Path(__file__).resolve().parent.parent / "mdata.Prices" / "sample.json"


def load_prices(path: str | Path, tickers: Sequence[str] | None = None) -> Prices:
    """The sample's prices, narrowed to `tickers`; `path` is not read.

    Raises KeyError for a requested ticker the sample does not have, as the real one does.
    """
    data = json.loads(SAMPLE.read_text(encoding="utf-8"))
    chosen = sorted(set(tickers)) if tickers is not None else data["tickers"]
    missing = [t for t in chosen if t not in data["closes"]]
    if missing:
        raise KeyError(missing[0])
    return Prices(
        dates=tuple(data["dates"]),
        tickers=tuple(chosen),
        closes={t: tuple(data["closes"][t]) for t in chosen},
    )

"""rolling_vol spine: expanding sample stdev of the returns before each date."""

from __future__ import annotations

import math

import pytest

from risk import Returns
from risk.vol import rolling_vol


def _returns(values: dict[str, tuple[float, ...]], n: int) -> Returns:
    dates = tuple(f"2024-01-{d:02d}" for d in range(2, 2 + n))
    return Returns(base_date="2024-01-01", dates=dates, tickers=tuple(values), values=values)


def test_sample_returns_give_none_until_window_then_earlier_stdev(contract_sample):
    data = contract_sample("risk.Returns")
    returns = Returns(
        base_date=data["base_date"],
        dates=tuple(data["dates"]),
        tickers=tuple(data["tickers"]),
        values={t: tuple(v) for t, v in data["values"].items()},
    )
    vol = rolling_vol(returns, 2)
    assert vol.returns is returns
    assert vol.window == 2
    assert vol.annualized is False
    assert set(vol.values) == {"A", "B"}
    assert all(len(v) == len(returns.dates) for v in vol.values.values())
    assert vol.values["A"][:2] == (None, None)
    # two returns 0.014 and -0.007: sample stdev |a - b| / sqrt(2)
    assert vol.values["A"][2] == pytest.approx(0.021 / math.sqrt(2), abs=1e-12)
    # B's first two returns are equal
    assert vol.values["B"][2] == pytest.approx(0.0, abs=1e-12)


def test_uses_every_earlier_return_never_its_own():
    returns = _returns({"X": (0.0, 2.0, 4.0, 100.0)}, 4)
    vol = rolling_vol(returns, 2)
    assert vol.values["X"][0] is None
    assert vol.values["X"][1] is None
    assert vol.values["X"][2] == pytest.approx(math.sqrt(2.0))
    # mean 2 of (0, 2, 4): squares 4 + 0 + 4 over 2 -> 4; the 100 is not used
    assert vol.values["X"][3] == pytest.approx(2.0)


def test_window_three_waits_for_three_earlier_returns():
    returns = _returns({"X": (1.0, 2.0, 3.0, 4.0)}, 4)
    assert rolling_vol(returns, 3).values["X"] == (None, None, None, pytest.approx(1.0))

from __future__ import annotations

from collections.abc import Sequence
from pathlib import Path

from mdata import load_prices

from risk.annualize import annualize
from risk.returns import log_returns
from risk.types import VolSeries
from risk.vol import rolling_vol


def vol_series(
    path: str | Path, *, window: int = 20, tickers: Sequence[str] | None = None
) -> VolSeries:
    """Annualized rolling volatility of a prices CSV (RISK-01 to RISK-03).

    Raises as `mdata.load_prices`, and ValueError if `window` < 2.
    """
    prices = load_prices(path, tickers)
    returns = log_returns(prices)
    vol = rolling_vol(returns, window)
    return annualize(vol)
