def poisson_scores(
    players: Mapping[str, float], *, seeds: int, games: int, seed: int
) -> dict[str, list[ScoreSet]]:
    """The `poisson_scores` generator: per player, `seeds` ScoreSets of `games` Poisson scores."""
    rng = np.random.default_rng(seed)
    return {
        name: [
            ScoreSet(
                player=name,
                train_seed=s,
                scores=rng.poisson(lam, games).astype(np.float64),
                games=(),
            )
            for s in range(seeds)
        ]
        for name, lam in players.items()
    }


def _fixed_scores(values: Sequence[float], name: str, seeds: int = 5) -> list[ScoreSet]:
    """A fixed score list split evenly over `seeds` training seeds (pooled, it is the list)."""
    chunks = np.array_split(np.asarray(values, dtype=np.float64), seeds)
    return [
        ScoreSet(player=name, train_seed=s, scores=chunk, games=())
        for s, chunk in enumerate(chunks)
    ]

def run_claims(
    scores: Mapping[str, Sequence[ScoreSet]],
    specs: Sequence[ClaimSpec],

    charts: Sequence[Path] = (),
    out: Path,

def _check_players(scores: Mapping[str, Sequence[ScoreSet]], specs: Sequence[ClaimSpec]) -> None:
    missing = sorted({name for s in specs for name in (s.a, s.b)} - set(scores))
    if missing:
        raise ValueError(f"claims name players missing from the scores: {missing}")


def pool(sets: Sequence[ScoreSet]) -> np.ndarray:
    """One player's scores pooled across its training seeds."""
    return np.concatenate([s.scores for s in sets]).astype(np.float64)


def permutation_p(a: np.ndarray, b: np.ndarray, direction: str) -> float:
    """One-sided p-value of mean(a) - mean(b) under random relabelling of the pooled games.

    `greater` tests H1 "a > b"; `not_worse` tests H1 "a < b" (H0 "a >= b").
    """
    both = np.concatenate([a, b])
    n_a, n_b = a.size, b.size
    total = float(both.sum())

    def diff(sum_a: np.ndarray | float) -> np.ndarray:
        return np.asarray(sum_a) / n_a - (total - np.asarray(sum_a)) / n_b

    observed = diff(float(a.sum()))
    sign = 1.0 if direction == "greater" else -1.0
    rng = np.random.default_rng(_SEED)
    extreme = 0
    for start in range(0, PERMUTATIONS, _BATCH):
        count = min(_BATCH, PERMUTATIONS - start)
        shuffled = rng.permuted(np.tile(both, (count, 1)), axis=1)
        null = diff(shuffled[:, :n_a].sum(axis=1))
        extreme += int(np.count_nonzero(sign * null >= sign * observed - 1e-12))
    return (extreme + 1) / (PERMUTATIONS + 1)


def holm(p_values: Sequence[float]) -> list[float]:
    """Holm step-down adjusted p-values, in the input order."""
    m = len(p_values)
    adjusted = [0.0] * m
    running = 0.0
    for rank, i in enumerate(sorted(range(m), key=lambda k: p_values[k])):
        running = max(running, min(1.0, (m - rank) * p_values[i]))
        adjusted[i] = running
    return adjusted


def ratio_holds(mean_a: float, mean_b: float, min_ratio: float | None) -> bool:
    """REF-02: `mean_a >= min_ratio * mean_b`, inclusive; true when no ratio is set."""
    if min_ratio is None:
        return True
    bar = min_ratio * mean_b
    return mean_a >= bar or math.isclose(mean_a, bar)


def _claim(
    spec: ClaimSpec,
    pooled: Mapping[str, np.ndarray],
    p: float,
    p_adj: float,
    alpha: float,
    method: str,
) -> Claim:
    mean_a, mean_b = float(pooled[spec.a].mean()), float(pooled[spec.b].mean())

    return Claim(spec, mean_a, mean_b, p, p_adj, passed, method)
