from collections.abc import Sequence

from mdata.types import RawRow


def validate_rows(rows: Sequence[RawRow], tickers: Sequence[str] | None = None) -> list[RawRow]:
    """Rows with a positive close, one per (date, ticker), sorted by (date, ticker).

    A later row in file order replaces an earlier one with the same (date, ticker).
    Raises `KeyError` naming a requested ticker that has no usable row.
    """
    latest = {
        (row.date, row.ticker): row for row in rows if row.close is not None and row.close > 0
    }
    present = {ticker for _, ticker in latest}
    for ticker in tickers or ():
        if ticker not in present:
            raise KeyError(ticker)
    return [latest[key] for key in sorted(latest)]
