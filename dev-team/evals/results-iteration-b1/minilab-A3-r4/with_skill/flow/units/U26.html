8	def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
9	    """One calendar from the first date every ticker has a close. GapError past `max_gap` missing dates."""
10	    closes_by_ticker: dict[str, dict[str, float]] = {}
11	    for row in rows:
12	        ticker_closes = closes_by_ticker.setdefault(row.ticker, {})
13	        if row.close is not None:
14	            ticker_closes[row.date] = row.close
15	    tickers = tuple(sorted(closes_by_ticker))
16	    date_sets = [set(closes_by_ticker[ticker]) for ticker in tickers]
17	    dates = tuple(sorted(set.intersection(*date_sets))) if date_sets else ()
18	    closes = {ticker: tuple(closes_by_ticker[ticker][date] for date in dates) for ticker in tickers}
19	    return Prices(dates=dates, tickers=tickers, closes=closes)

from collections.abc import Mapping, Sequence

from mdata.types import GapError, Prices, RawRow


def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
    """One calendar from the first date every ticker has a close. GapError past `max_gap` missing dates."""
    closes_by_ticker = _closes_by_ticker(rows)
    tickers = tuple(sorted(closes_by_ticker))
    dates = _calendar(closes_by_ticker)
    closes = {
        ticker: _forward_fill(ticker, closes_by_ticker[ticker], dates, max_gap)
        for ticker in tickers
    }
    return Prices(dates=dates, tickers=tickers, closes=closes)


def _closes_by_ticker(rows: Sequence[RawRow]) -> dict[str, dict[str, float]]:
    """Each ticker's closes by date; rows without a close are left out."""
    closes_by_ticker: dict[str, dict[str, float]] = {}
    for row in rows:
        ticker_closes = closes_by_ticker.setdefault(row.ticker, {})
        if row.close is not None:
            ticker_closes[row.date] = row.close
    return closes_by_ticker


def _calendar(closes_by_ticker: Mapping[str, Mapping[str, float]]) -> tuple[str, ...]:
    """Every date any ticker has a close, from the first date every ticker has one."""
    if not closes_by_ticker or not all(closes_by_ticker.values()):
        return ()
    start = max(min(ticker_closes) for ticker_closes in closes_by_ticker.values())
    all_dates = set().union(*closes_by_ticker.values())
    return tuple(sorted(date for date in all_dates if date >= start))


def _forward_fill(
    ticker: str, ticker_closes: Mapping[str, float], dates: Sequence[str], max_gap: int
) -> tuple[float, ...]:
    """The ticker's close on each date, a missing date taking the previous close."""
    filled: list[float] = []
    missing_in_row = 0
    for date in dates:
        if date in ticker_closes:
            filled.append(ticker_closes[date])
            missing_in_row = 0
            continue
        missing_in_row += 1
        if missing_in_row > max_gap:
            msg = f"{ticker}: more than {max_gap} dates in a row missing (through {date})"
            raise GapError(msg)
        filled.append(filled[-1])
    return tuple(filled)
