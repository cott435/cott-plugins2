7	def log_returns(prices: Prices) -> Returns:

"""Daily log returns (RISK-01)."""

import math

from mdata.types import Prices
from risk.types import Returns


def log_returns(prices: Prices) -> Returns:
    """r_t = ln(close_t / close_t-1), dated t. ValueError with fewer than two dates."""
    if len(prices.dates) < 2:
        raise ValueError(f"log_returns needs at least two dates, got {len(prices.dates)}")
    values = {
        ticker: tuple(
            math.log(now / before)
            for before, now in zip(prices.closes[ticker], prices.closes[ticker][1:], strict=False)
        )
        for ticker in prices.tickers
    }
    return Returns(
        base_date=prices.dates[0],
        dates=prices.dates[1:],
        tickers=prices.tickers,
        values=values,
    )

from __future__ import annotations

import math
from collections.abc import Callable

import pytest

from mdata.types import Prices
from risk.returns import log_returns


def prices_from_sample(data: dict) -> Prices:
    return Prices(
        dates=tuple(data["dates"]),
        tickers=tuple(data["tickers"]),
        closes={ticker: tuple(closes) for ticker, closes in data["closes"].items()},
    )


def test_sample_gives_log_returns_dated_after_the_first_date(
    contract_sample: Callable[[str], dict],
) -> None:
    returns = log_returns(prices_from_sample(contract_sample("mdata.Prices")))
    assert returns.base_date == "2024-03-04"
    assert returns.dates == ("2024-03-05", "2024-03-06", "2024-03-07", "2024-03-08", "2024-03-11")
    assert returns.tickers == ("ACME", "BOLT")
    assert set(returns.values) == {"ACME", "BOLT"}
    # ACME closes 50.0, 51.0, 49.5, 48.0, 50.5, 52.0
    assert returns.values["ACME"] == pytest.approx(
        (
            math.log(51.0 / 50.0),
            math.log(49.5 / 51.0),
            math.log(48.0 / 49.5),
            math.log(50.5 / 48.0),
            math.log(52.0 / 50.5),
        ),
        abs=1e-12,
    )


def test_doubling_close_gives_ln_two() -> None:
    prices = Prices(dates=("2024-01-02", "2024-01-03"), tickers=("X",), closes={"X": (10.0, 20.0)})
    returns = log_returns(prices)
    assert returns.base_date == "2024-01-02"
    assert returns.dates == ("2024-01-03",)
    assert returns.values["X"] == pytest.approx((0.6931471805599453,), abs=1e-15)


def test_round_trip_returns_sum_to_zero() -> None:
    prices = Prices(
        dates=("2024-01-02", "2024-01-03", "2024-01-04"),
        tickers=("X",),
        closes={"X": (40.0, 50.0, 40.0)},
    )
    first, second = log_returns(prices).values["X"]
    assert first == pytest.approx(-second, abs=1e-15)
    assert first > 0.0


def test_unchanged_close_gives_zero() -> None:
    prices = Prices(dates=("2024-01-02", "2024-01-03"), tickers=("X",), closes={"X": (7.5, 7.5)})
    assert log_returns(prices).values["X"] == (0.0,)

    prices = Prices(dates=("2024-01-02",), tickers=("X",), closes={"X": (10.0,)})
    with pytest.raises(ValueError, match="two dates"):
        log_returns(prices)
