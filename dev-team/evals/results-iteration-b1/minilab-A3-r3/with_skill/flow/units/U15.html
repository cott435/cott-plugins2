"""Rolling volatility from earlier returns (RISK-02)."""

from __future__ import annotations

from statistics import stdev

from risk.types import Returns, VolSeries


def rolling_vol(returns: Returns, window: int) -> VolSeries:
    """Sample stdev of all returns before each date; None until `window` of them exist.

    Spine version: an expanding window over every earlier return.
    Raises ValueError if `window` < 2.
    """
    if window < 2:
        raise ValueError(f"window must be at least 2, got {window}")
    values = {
        ticker: tuple(
            stdev(series[:i]) if i >= window else None for i in range(len(series))
        )
        for ticker, series in returns.values.items()
    }
    return VolSeries(returns=returns, window=window, annualized=False, values=values)

"""rolling_vol spine: expanding sample stdev of earlier returns."""

from __future__ import annotations

import pytest

from risk import Returns
from risk.vol import rolling_vol


def _sample_returns(contract_sample) -> Returns:
    data = contract_sample("risk.Returns")
    return Returns(
        base_date=data["base_date"],
        dates=tuple(data["dates"]),
        tickers=tuple(data["tickers"]),
        values={t: tuple(v) for t, v in data["values"].items()},
    )


def test_sample_gives_none_then_expanding_stdev(contract_sample):
    returns = _sample_returns(contract_sample)
    vol = rolling_vol(returns, 2)
    assert vol.returns is returns
    assert vol.window == 2
    assert vol.annualized is False
    assert set(vol.values) == {"A", "B"}
    a = vol.values["A"]
    assert len(a) == len(returns.dates)
    assert a[:2] == (None, None)
    # stdev(0.0198026272962, -0.0100503358535), by hand: |diff| / sqrt(2)
    assert a[2] == pytest.approx(0.0211093, rel=1e-5)
    assert all(v is not None and v > 0 for v in vol.values["B"][2:])


def test_uses_only_earlier_returns():
    returns = Returns(
        base_date="d0",
        dates=("d1", "d2", "d3", "d4"),
        tickers=("X",),
        values={"X": (1.0, 3.0, 100.0, -50.0)},
    )
    vol = rolling_vol(returns, 2)
    # d3 sees (1, 3): stdev sqrt(2); d4 sees (1, 3, 100): stdev sqrt(3201)
    assert vol.values["X"][2] == pytest.approx(2**0.5)
    assert vol.values["X"][3] == pytest.approx(3201**0.5)


def test_larger_window_delays_first_value():
    returns = Returns(
        base_date="d0",
        dates=("d1", "d2", "d3", "d4"),
        tickers=("X",),
        values={"X": (1.0, 2.0, 3.0, 4.0)},
    )
    vol = rolling_vol(returns, 3)
    assert vol.values["X"][:3] == (None, None, None)
    assert vol.values["X"][3] == pytest.approx(1.0)


@pytest.mark.parametrize("window", [1, 0, -3])
def test_window_below_two_raises(contract_sample, window):
    with pytest.raises(ValueError, match="window"):
        rolling_vol(_sample_returns(contract_sample), window)
