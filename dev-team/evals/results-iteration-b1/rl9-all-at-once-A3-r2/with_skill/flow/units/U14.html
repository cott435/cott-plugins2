The tests keep their own model of the board (read from `draw()` at the start, then advanced by
the rules in spec.md) and check every step of the real world against it. Which action index is
which relative turn is not part of the contract, so the mapping is read off the world once and
then held to for every later step and seed.

from collections import deque
from collections.abc import Callable
from dataclasses import dataclass
from pathlib import Path
from typing import Any

import numpy as np
import pytest

from rllab.core import ActionError, Discrete, Step
from rllab.worlds.snake import Snake

FIXTURES = Path(__file__).resolve().parents[2] / "fixtures" / "rllab"
GRID = 10
HUNGER = 100
Cell = tuple[int, int]


def load(case: str, name: str) -> Any:
    return json.loads((FIXTURES / case / f"{name}.json").read_text())


TC06 = load("TC-06", "input")
TC06_EXPECTED = load("TC-06", "expected")

# ----------------------------------------------------------------------------- board model


def add(a: Cell, b: Cell) -> Cell:
    return (a[0] + b[0], a[1] + b[1])


def sub(a: Cell, b: Cell) -> Cell:
    return (a[0] - b[0], a[1] - b[1])


def quarter(v: Cell) -> Cell:
    """A quarter turn that takes the table's `straight` (up) to its `right`."""
    up, right = TABLE["straight"], TABLE["right"]
    turned = (-v[1], v[0])
    return turned if (-up[1], up[0]) == right else (v[1], -v[0])


def turns(heading: Cell) -> dict[str, Cell]:
    """The relative-turn table rotated from heading up to `heading`."""
    table = dict(TABLE)
    for _ in range(4):
        if table["straight"] == heading:
            return table
        table = {k: quarter(v) for k, v in table.items()}
    raise AssertionError(f"heading {heading} is not one grid cell")


def inside(c: Cell) -> bool:
    return 0 <= c[0] < GRID and 0 <= c[1] < GRID


def cell(obj: Any) -> Cell:
    assert obj.size == pytest.approx(1.0)
    x, y = round(obj.x), round(obj.y)
    assert (x, y) == (obj.x, obj.y)
    return (x, y)


def board(world: Any) -> tuple[Cell, list[Cell], list[Cell]]:
    """Head, body cells and food cells as `draw()` shows them."""
    state = world.draw()
    assert (state.width, state.height) == (GRID, GRID)
    snake = [o for o in state.objects if o.agent == 0]
    heads = [o for o in snake if "head" in o.kind]
    assert len(heads) == 1
    body = [cell(o) for o in snake if o is not heads[0]]
    food = [cell(o) for o in state.objects if o.agent is None]
    return cell(heads[0]), body, food


@dataclass
class Model:
    cells: list[Cell]  # head first
    food: Cell
    hungry: int = 0
    eaten: int = 0
    steps: int = 0

    @property
    def heading(self) -> Cell:
        return sub(self.cells[0], self.cells[1])

    def moves(self) -> dict[str, Cell]:
        return {k: add(self.cells[0], d) for k, d in turns(self.heading).items()}


def start(world: Any) -> Model:
    """Read the start board; check SNAKE-01 on it."""
    head, body, food = board(world)
    assert len(body) + 1 == START_LENGTH
    assert len(food) == 1
    cells = [head]
    rest = list(body)
    while rest:
        nxt = [c for c in rest if sum(map(abs, sub(c, cells[-1]))) == 1]
        assert len(nxt) == 1, "the start snake is one straight-ish chain"
        cells.append(nxt[0])
        rest.remove(nxt[0])
    assert all(2 <= x <= GRID - 3 and 2 <= y <= GRID - 3 for x, y in cells), "near the centre"
    assert inside(food[0])
    assert food[0] not in cells
    return Model(cells, food[0])


def act(world: Any, action: int) -> Step:
    return world.step(np.asarray([action]))


def mapping() -> dict[str, int]:
    """Which action index is left, right and straight, read from the seed-0 start."""
    found: dict[str, int] = {}
    for action in range(3):
        world = Snake()
        world.reset(SEEDS[0])
        model = start(world)
        act(world, action)
        head, _, _ = board(world)
        labels = [k for k, c in model.moves().items() if c == head]
        assert len(labels) == 1, f"action {action} moved the head off the turn table"
        found[labels[0]] = action
    assert set(found) == set(TABLE)
    return found


def advance(world: Any, model: Model, label: str, actions: dict[str, int]) -> tuple[Step, str]:
    """Step the world and the model; check they agree. Returns the step and why it ended."""
    new = model.moves()[label]
    tail = model.cells[-1]
    step = act(world, actions[label])
    model.steps += 1
    if not inside(new):
        assert step.done, f"head left the grid at {new} but the episode went on"
        return step, "wall"
    if new in model.cells[:-1]:
        assert step.done, f"head hit the body at {new} but the episode went on"
        return step, "body"
    if new == tail and step.done:
        return step, "body"  # moving into the tail's cell: either rule is allowed
    eats = new == model.food
    model.cells = [new, *model.cells] if eats else [new, *model.cells[:-1]]
    model.eaten += eats
    model.hungry = 0 if eats else model.hungry + 1
    hungry, out_of_time = model.hungry >= HUNGER, model.steps >= world.spec.max_steps
    assert step.done == (hungry or out_of_time), (model.steps, model.hungry)
    assert step.truncated == (out_of_time and not hungry)
    if step.done:
        return step, "hunger" if hungry else "max_steps"
    head, body, food = board(world)
    assert head == new
    assert sorted([head, *body]) == sorted(model.cells)
    assert len(food) == 1
    assert inside(food[0])
    assert food[0] not in model.cells, "food is on a free cell"
    if not eats:
        assert food[0] == model.food, "food moves only when eaten"
    model.food = food[0]
    return step, ""


# ------------------------------------------------------------------------------- policies

Policy = Callable[[Model, np.random.Generator], str]


def wander(model: Model, rng: np.random.Generator) -> str:
    return str(rng.choice(sorted(TABLE)))


def safe(model: Model, rng: np.random.Generator) -> str:
    """A move that neither dies nor eats, when there is one."""
    moves = model.moves()
    ok = [k for k in sorted(moves) if inside(moves[k]) and moves[k] not in model.cells]
    hungry = [k for k in ok if moves[k] != model.food]
    return str(rng.choice(hungry or ok or sorted(moves)))


def seek(model: Model, rng: np.random.Generator) -> str:
    """The first move of a shortest path to the food around the body."""
    moves = model.moves()
    first = {c: k for k, c in moves.items() if inside(c) and c not in model.cells}
    queue = deque((c, k) for c, k in first.items())
    seen = set(first) | set(model.cells)
    while queue:
        here, label = queue.popleft()
        if here == model.food:
            return label
        for d in TABLE.values():
            for n in (add(here, d), sub(here, d)):
                if inside(n) and n not in seen:
                    seen.add(n)
                    queue.append((n, label))
    return safe(model, rng)


def play(seed: int, policy: Policy, actions: dict[str, int]) -> tuple[Any, Model, str]:
    world = Snake()
    world.reset(seed)
    model = start(world)
    rng = np.random.default_rng(seed)
    for _ in range(world.spec.max_steps):
        step, why = advance(world, model, policy(model, rng), actions)
        if step.done:
            return world, model, why
    raise AssertionError("the episode outlived max_steps")


def eat_then(target: int, after: Policy) -> Policy:
    def policy(model: Model, rng: np.random.Generator) -> str:
        return seek(model, rng) if model.eaten < target else after(model, rng)

    return policy


def spin(model: Model, rng: np.random.Generator) -> str:
    return "left"

        and np.array_equal(a.active, b.active)
        and (a.done, a.truncated) == (b.done, b.truncated)
    )

"""Replay double of `core.World`: serves one recorded episode.

`reset(seed)` returns the recorded start for any seed (and `result()` reports that seed), so a
training loop can reset it as often as it likes. `step` refuses an action outside `spec.action`
with `ActionError` and a step after `done` with `RuntimeError`, as every World must; any action
inside the action space advances the recording by one step.
"""

from dataclasses import dataclass, field, replace
from typing import Any

import numpy as np

from rllab.core import ActionError, Array, Box, DrawState, EpisodeResult, Step, WorldSpec


def check_actions(spec: WorldSpec, actions: Array, active: Array) -> None:
    """ActionError for the first active agent whose action is outside `spec.action`."""
    for i, row in enumerate(np.asarray(actions)):
        if active[i] and not inside(spec, row):
            raise ActionError(spec.agents[i], row.tolist())


def inside(spec: WorldSpec, action: Array) -> bool:
    if isinstance(spec.action, Box):
        low, high = spec.action.low, spec.action.high
        return action.shape == (spec.action.dims,) and bool(
            np.all((action >= low) & (action <= high))
        )
    return action.shape == () and float(action).is_integer() and 0 <= action < spec.action.n


@dataclass
class ReplayWorld:
    spec: WorldSpec
    steps: list[Step]
    draws: list[DrawState]
    recorded: EpisodeResult
    at: int | None = field(default=None, init=False)
    seed: int = field(default=0, init=False)

    def reset(self, seed: int) -> Step:
        self.at, self.seed = 0, seed
        return self.steps[0]

    def step(self, actions: Array) -> Step:
        if self.at is None or self.steps[self.at].done:
            raise RuntimeError("step before reset or after done")
        check_actions(self.spec, actions, self.steps[self.at].active)
        self.at += 1
        return self.steps[self.at]

    def result(self) -> EpisodeResult:
        return replace(self.recorded, seed=self.seed)

    def draw(self) -> DrawState:
        return self.draws[self.at or 0]

    return ReplayWorld(

        kit.decode(list[Step], recording["steps"]),
        kit.decode(list[DrawState], recording["draws"]),
        kit.decode(EpisodeResult, recording["result"]),
    )

import numpy as np

from rllab.core import (
    ActionError,
    Array,
    Discrete,
    DrawState,
    Drawable,
    EpisodeResult,
    Step,
    TeamSpec,
    WorldSpec,
)

Cell = tuple[int, int]

TEAM = "snake"
START_LENGTH = 3
STRAIGHT, LEFT, RIGHT = 0, 1, 2
# Headings in observation order; y grows downwards, so up is (0, -1).
HEADINGS: tuple[Cell, ...] = ((0, -1), (1, 0), (0, 1), (-1, 0))
DEFAULTS: dict[str, int] = {"size": 10, "max_steps": 1000, "hunger": 100}
# Danger straight, left and right; heading (4); food up, right, down, left (4); length.
OBS_LEN = 3 + len(HEADINGS) + 4 + 1


def turn(heading: Cell, action: int) -> Cell:
    """The heading after a relative turn: left of up is left, right of up is right."""
    dx, dy = heading
    if action == LEFT:
        return (dy, -dx)
    if action == RIGHT:
        return (-dy, dx)
    return heading


class Snake:
    """The snake world; `overrides` change only the spec's starting values."""

    spec: WorldSpec

    def __init__(self, **overrides: int | float) -> None:
        unknown = sorted(set(overrides) - set(DEFAULTS))
        if unknown:
            raise ValueError(f"unknown snake overrides: {', '.join(unknown)}")
        settings = {**DEFAULTS, **{k: int(v) for k, v in overrides.items()}}
        self.size = settings["size"]
        self.hunger = settings["hunger"]
        self.spec = WorldSpec(
            world="snake",
            variant="default",
            agents=(TEAM,),
            teams=(TeamSpec(TEAM, (0,), "food eaten", "higher"),),
            obs_len=OBS_LEN,
            action=Discrete(3),
            max_steps=settings["max_steps"],
            measures=(),
            sight=None,
        )
        self.started = False
        self.reset(0)
        self.started = False

    def reset(self, seed: int) -> Step:
        self.rng = np.random.default_rng(np.random.SeedSequence(seed))
        self.seed = seed
        self.heading = HEADINGS[int(self.rng.integers(len(HEADINGS)))]
        centre = (self.size // 2, self.size // 2)
        dx, dy = self.heading
        self.cells: list[Cell] = [
            (centre[0] - i * dx, centre[1] - i * dy) for i in range(START_LENGTH)
        ]
        self.food = self.place_food()
        self.steps = self.hungry = self.eaten = 0
        self.done = False
        self.started = True
        return self.observe(0.0, truncated=False)

    def step(self, actions: Array) -> Step:
        if not self.started or self.done:
            raise RuntimeError("snake: step before reset or after the episode ended")
        action = self.checked(actions)
        self.heading = turn(self.heading, action)
        head = (self.cells[0][0] + self.heading[0], self.cells[0][1] + self.heading[1])
        self.steps += 1
        if self.fatal(head):
            self.done = True
            return self.observe(0.0, truncated=False)
        eats = head == self.food
        self.cells = [head, *self.cells] if eats else [head, *self.cells[:-1]]
        self.eaten += eats
        self.hungry = 0 if eats else self.hungry + 1
        if eats:
            self.food = self.place_food()
        starved = self.hungry >= self.hunger or self.food is None
        out_of_time = self.steps >= self.spec.max_steps
        self.done = starved or out_of_time
        return self.observe(float(eats), truncated=out_of_time and not starved)

    def result(self) -> EpisodeResult:
        return EpisodeResult(
            seed=self.seed,
            length=self.steps,
            scores={TEAM: float(self.eaten)},
            team_rewards={TEAM: float(self.eaten)},
            measures={},
        )

    def draw(self) -> DrawState:
        snake = tuple(
            Drawable("snake-head" if i == 0 else "snake-body", float(x), float(y), 1.0, TEAM, 0)
            for i, (x, y) in enumerate(self.cells)
        )
        food = () if self.food is None else (self.drawn_food(self.food),)
        return DrawState(float(self.size), float(self.size), snake + food)

    def drawn_food(self, cell: Cell) -> Drawable:
        return Drawable("food", float(cell[0]), float(cell[1]), 1.0, None, None)

    def checked(self, actions: Array) -> int:
        """The one action as an int; `ActionError` for anything outside `Discrete(3)`."""
        rows = np.asarray(actions)
        value = rows.tolist()[0] if rows.shape == (1,) else rows.tolist()
        if not isinstance(value, int | float) or isinstance(value, bool):
            raise ActionError(TEAM, value)
        if not float(value).is_integer() or not 0 <= value < 3:
            raise ActionError(TEAM, value)
        return int(value)

    def fatal(self, cell: Cell) -> bool:
        """A wall or the body; the tail cell is free, since the tail moves on."""
        x, y = cell
        off_grid = not (0 <= x < self.size and 0 <= y < self.size)
        return off_grid or cell in self.cells[:-1]

    def place_food(self) -> Cell | None:
        """A random cell the snake does not cover; none when the snake fills the grid."""
        taken = set(self.cells)
        free = [(x, y) for y in range(self.size) for x in range(self.size) if (x, y) not in taken]
        if not free:
            return None
        return free[int(self.rng.integers(len(free)))]

            done=self.done,
            truncated=truncated,
        )
