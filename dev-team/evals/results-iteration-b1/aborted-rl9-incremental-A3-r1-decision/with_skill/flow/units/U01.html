135	**Status:** Accepted · **Date:** 2026-10-17 · **Scope:** `core`

137	- **Decision:** Every environment is batch-first over its agents. Rewards and per-agent `done` are `(n_agents,)`, and the game-level `terminated`/`truncated` stay booleans. Snake and the corridor are the `n_agents == 1` case. `core.Agent.act` takes and returns batches, and `core.Transition` carries per-agent arrays with an `active` mask. 130	def _diff_path(raw: str) -> str:
131	    raw = raw.rstrip("\t")
132	    return raw[2:] if raw[:2] in {"a/", "b/"} else raw

135	def parse_diff(text: str) -> Diff:
136	    """Added lines (with new line numbers), removed lines and deleted files, per file."""
137	    diff = Diff()
138	    old = new = ""
139	    line_no = 0
140	    in_header = False
141	    for line in text.splitlines():
142	        if line.startswith("diff --git "):
143	            old, new, in_header = "", "", True
144	        elif in_header and line.startswith("--- "):
145	            old = _diff_path(line[4:])
146	        elif in_header and line.startswith("+++ "):
147	            new = _diff_path(line[4:])
148	            if new == "/dev/null":
149	                diff.deleted.add(old)
150	        elif match := HUNK.match(line):
151	            in_header, line_no = False, int(match[1])
152	        elif in_header:
153	            continue
154	        elif line.startswith("+"):
155	            diff.added.setdefault(new, []).append((line_no, line[1:]))
156	            line_no += 1
157	        elif line.startswith("-"):
158	            diff.removed.setdefault(old, []).append(line[1:])
159	    return diff

245	    """Only pass, `...` or `raise NotImplementedError`, after an optional docstring."""

250	def _task_stub(body: list[ast.stmt]) -> str | None:
251	    """The task an interface stub names: a body that is only `raise NotImplementedError("M1-T05")`."""
252	    stmts = body[1:] if body and _is_docstring(body[0]) else body
253	    if len(stmts) != 1 or not isinstance(stmts[0], ast.Raise):
254	        return None
255	    exc = stmts[0].exc
256	    if not (isinstance(exc, ast.Call) and _name(exc.func) == "NotImplementedError"):
257	        return None
258	    arg = exc.args[0] if len(exc.args) == 1 and not exc.keywords else None
259	    if (
260	        isinstance(arg, ast.Constant)
261	        and isinstance(arg.value, str)
262	        and TASK_ID.fullmatch(arg.value)
263	    ):
264	        return arg.value
265	    return None

277	    if isinstance(stmt, ast.Pass):
278	        return True

298	    """Abstract methods, overloads and Protocol members may have placeholder bodies."""
299	    exempt: set[ast.AST] = set()
300	    for node in ast.walk(tree):

303	                exempt.add(node)

305	            exempt.update(
306	                n for n in node.body if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef))
307	            )

345	    """sys.path.insert/append/extend, site.addsitedir, or an assignment to sys.path."""
346	    if isinstance(node, ast.Call) and isinstance(node.func, ast.Attribute):

370	    for node in ast.walk(func):

372	            return True

381	def ast_findings(path: str, lines: set[int]) -> list[Finding]:
382	    try:
383	        tree = ast.parse(Path(path).read_text(encoding="utf-8"))
384	    except (SyntaxError, UnicodeDecodeError, OSError):
385	        return []  # ruff reports syntax errors; a deleted file has nothing to check
386	    exempt, test = _exempt_functions(tree), is_test_file(path)
387	    found: list[Finding] = []
388	    for node in ast.walk(tree):
389	        if not _touches(node, lines):
390	            continue
391	        if outside := _outside_finding(path, node):
392	            found.append(outside)
393	        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
394	            if (
395	                node not in exempt
396	                and not path.endswith(".pyi")
397	                and _is_trivial(node.body)
398	                and not _task_stub(node.body)
399	            ):
400	                found.append(
401	                    Finding("stub", path, node.lineno, f"`{node.name}` has a placeholder body")
402	                )
403	            if test and node.name.startswith("test") and not _asserts(node):
404	                found.append(
405	                    Finding(
406	                        "test-without-assertion",
407	                        path,
408	                        node.lineno,
409	                        f"`{node.name}` asserts nothing",

414	                found.append(
415	                    Finding("swallowed-exception", path, node.lineno, "exception swallowed")
416	                )

503	def _limit_findings(old: dict[str, Any], new: dict[str, Any]) -> list[str]:
504	    pylint = _get(old, "tool.ruff.lint.pylint") or {}
505	    maxes = [*MAX_KEYS, *(f"tool.ruff.lint.pylint.{k}" for k in pylint)]
506	    out = [
507	        f"{k}: {_get(old, k)} -> {_get(new, k)}"
508	        for k in maxes
509	        if _loosened(_get(old, k), _get(new, k), +1)
510	    ]
511	    out += [
512	        f"{k}: {_get(old, k)} -> {_get(new, k)}"
513	        for k in MIN_KEYS
514	        if _loosened(_get(old, k), _get(new, k), -1)
515	    ]
516	    out += [
517	        f"{k} turned off" for k in TRUE_KEYS if _get(old, k) is True and _get(new, k) is not True
518	    ]
519	    out += [f"{k} turned on" for k in OFF_KEYS if _get(new, k) is True and _get(old, k) is not True]
520	    return out

523	def _loosened(was: Any, now: Any, direction: int) -> bool:
524	    if not isinstance(was, (int, float)):
525	        return False
526	    if not isinstance(now, (int, float)):
527	        return True  # a limit that disappeared
528	    return (now - was) * direction > 0
