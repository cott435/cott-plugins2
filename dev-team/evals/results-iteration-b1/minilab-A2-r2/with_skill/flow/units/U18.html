def annualize(vol: VolSeries, periods_per_year: int = 252) -> VolSeries:
    """Scale each daily volatility by sqrt(periods_per_year) trading days; gaps stay None."""
    if vol.annualized:
        raise ValueError("vol is already annualized")

def rolling_vol(returns: Returns, window: int) -> VolSeries:
    """Return each date's sample stdev of the `window` returns before it, None until they exist.

    A date's own return never enters its value (RISK-02).
    """
    if window < 2:
        raise ValueError(f"window must be at least 2, got {window}")
    values = {
        ticker: tuple(
            None if i < window else statistics.stdev(series[i - window : i])
            for i in range(len(series))
        )
        for ticker, series in returns.values.items()
    }

def risk_report(
    path: str | Path,
    out_path: str | Path,
    *,
    window: int = 20,
    tickers: Sequence[str] | None = None,
) -> Report:
    """Take the rolling vol of the closes at `path`, summarize it and its drawdowns, and write them."""
