def write_report(
    summary: Sequence[TickerSummary], drawdowns: Sequence[Drawdown], out_path: str | Path
) -> Report:
    """Write {"tickers": [...]}, one row per ticker, sorted. ValueError if the two cover different tickers."""
    by_ticker = {d.ticker: d for d in drawdowns}
    if {s.ticker for s in summary} != set(by_ticker) or len(summary) != len(drawdowns):
        msg = "summary and drawdowns cover different tickers"
        raise ValueError(msg)
    rows = tuple(_row(s, by_ticker[s.ticker]) for s in sorted(summary, key=lambda s: s.ticker))
    Path(out_path).write_text(json.dumps({"tickers": rows}, indent=2) + "\n", encoding="utf-8")
    return Report(path=str(out_path), rows=rows)

        **dataclasses.asdict(summary),
        "max_drawdown": drawdown.depth,
        "peak_date": drawdown.peak_date,
        "trough_date": drawdown.trough_date,
