"""PL-01 step 2: keep the usable rows, sorted."""

from collections.abc import Sequence

from mdata.types import RawRow


def validate_rows(rows: Sequence[RawRow], tickers: Sequence[str] | None = None) -> list[RawRow]:
    """Positive closes only, the later of duplicate (date, ticker) rows, sorted by (date, ticker).

    KeyError for a requested ticker with no usable row.
    """
    latest = {
        (row.date, row.ticker): row for row in rows if row.close is not None and row.close > 0
    }
    usable_tickers = {ticker for _, ticker in latest}
    for ticker in tickers or ():
        if ticker not in usable_tickers:
            raise KeyError(ticker)
    return [latest[key] for key in sorted(latest)]
