`tickers` selects from the answer as the real provider does: exactly those tickers, sorted, on the

import json

from mdata import GapError, Prices

HERE = Path(__file__).resolve().parent
SAMPLE = HERE.parent / "mdata.Prices" / "sample.json"
LATE_GAP = HERE / "late_gap.json"


def _answer(path: str | Path) -> tuple[dict[str, list[object]], list[str]]:
    """The recorded Prices payload for the fixture CSV at `path`, and its gapped tickers."""
    if Path(path).name == "late_gap.csv":
        recorded = json.loads(LATE_GAP.read_text())
        return recorded["prices"], recorded["gapped"]
    return json.loads(SAMPLE.read_text()), []


def load_prices(path: str | Path, tickers: Sequence[str] | None = None) -> Prices:
    payload, gapped = _answer(path)
    in_file = [*payload["closes"], *gapped]
    chosen = sorted(set(in_file if tickers is None else tickers))
    for ticker in chosen:
        if ticker not in in_file:
            raise KeyError(ticker)
    for ticker in chosen:
        if ticker in gapped:
            msg = f"{ticker} misses more than 5 dates in a row"
            raise GapError(msg)
    return Prices(
        dates=tuple(payload["dates"]),
        tickers=tuple(chosen),
        closes={ticker: tuple(payload["closes"][ticker]) for ticker in chosen},
    )

def _recorded(path: str | Path) -> tuple[dict[str, Any], dict[str, Any]]:
    """The recorded Prices payload for the fixture CSV at `path`, and its gapped tickers'."""
    if Path(path).name == "late_gap.csv":

        return recorded["prices"], recorded["gapped"]

def _prices(payload: dict[str, Any], tickers: Sequence[str]) -> Prices:
    return Prices(
        dates=tuple(payload["dates"]),
        tickers=tuple(tickers),
        closes={ticker: tuple(payload["closes"][ticker]) for ticker in tickers},
    )


def load_prices(path: str | Path, tickers: Sequence[str] | None = None) -> Prices:
    payload, gapped = _recorded(path)
    in_file = [*payload["closes"], *gapped]

            msg = f"{ticker} misses more than 5 dates in a row"
            raise GapError(msg)
    return _prices(payload, chosen)
