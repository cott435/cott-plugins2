+"""Fake `mdata.load_prices`: serves the `mdata.Prices` sample and nothing else.
+
+Any path gives the sample; `tickers` picks a subset of it, and a ticker the sample lacks is a
+`KeyError`, as the real function raises for a requested ticker with no usable row.
+"""
+
+from __future__ import annotations
+
+import json
+from collections.abc import Sequence
+from pathlib import Path
+
+from mdata import Prices
+
+SAMPLE = Path(__file__).resolve().parent.parent / "mdata.Prices" / "sample.json"
+
+
+def load_prices(path: str | Path, tickers: Sequence[str] | None = None) -> Prices:
+    """The sample as `Prices`, narrowed to `tickers` when given; `path` is ignored."""
+    del path
+    payload = json.loads(SAMPLE.read_text())
+    closes = {ticker: tuple(values) for ticker, values in payload["closes"].items()}
+    chosen = sorted(closes) if tickers is None else sorted(set(tickers))
+    for ticker in chosen:
+        if ticker not in closes:
+            raise KeyError(ticker)
+    return Prices(
+        dates=tuple(payload["dates"]),
+        tickers=tuple(chosen),
+        closes={ticker: closes[ticker] for ticker in chosen},
+    )

+def test_prices_sample_builds_prices_and_holds_its_invariants(contract_sample) -> None:
+    payload = contract_sample("mdata.Prices")
+    prices = Prices(
+        dates=tuple(payload["dates"]),
+        tickers=tuple(payload["tickers"]),
+        closes={t: tuple(v) for t, v in payload["closes"].items()},
+    )
+    assert prices.tickers == ("AAA", "BBB")
