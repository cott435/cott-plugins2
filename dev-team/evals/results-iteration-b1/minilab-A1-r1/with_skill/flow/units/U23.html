def log_returns(prices: Prices) -> Returns:
    """Spine: simple returns r_t = close_t / close_t-1 - 1, dated t (RISK-01)."""
    if len(prices.dates) < 2:
        raise ValueError(f"need at least two dates for returns, got {len(prices.dates)}")

        for ticker, closes in prices.closes.items()
    }

        dates=prices.dates[1:],
        tickers=prices.tickers,
        values=values,
    )

from __future__ import annotations

from statistics import stdev

def rolling_vol(returns: Returns, window: int) -> VolSeries:
    """Sample stdev of the returns before each date, once `window` exist (RISK-02).

    Spine version: the stdev is expanding, over every earlier return.
    """
    if window < 2:
        raise ValueError(f"window must be at least 2, got {window}")
    values = {
        ticker: tuple(stdev(series[:i]) if i >= window else None for i in range(len(series)))
        for ticker, series in returns.values.items()
    }
    return VolSeries(returns=returns, window=window, annualized=False, values=values)

"""Smoke test for PL-02 (risk.vol_series) on the mdata.load_prices fake: shape only."""

from __future__ import annotations

from pathlib import Path

import pytest

from risk import VolSeries, vol_series
from risk import vol_pipeline

CSV = Path(__file__).parents[2] / "contracts" / "mdata.load_prices" / "prices.csv"


@pytest.fixture(autouse=True)
def _fake_load_prices(monkeypatch, contract_fake) -> None:
    fake = contract_fake("mdata.load_prices")
    monkeypatch.setattr(vol_pipeline, "load_prices", fake.load_prices)


def test_pipeline_returns_a_vol_series_per_return_date() -> None:
    vol = vol_series(CSV, window=2)
    assert isinstance(vol, VolSeries)
    assert vol.window == 2
    assert vol.returns.tickers == ("AAA", "BBB")
    assert set(vol.values) == {"AAA", "BBB"}
    assert len(vol.returns.dates) == 5
    assert all(len(vol.values[ticker]) == 5 for ticker in vol.returns.tickers)


def test_pipeline_leaves_dates_before_the_window_empty() -> None:
    vol = vol_series(str(CSV), window=3)
    for ticker in vol.returns.tickers:
        assert vol.values[ticker][:3] == (None, None, None)
        assert all(value is not None for value in vol.values[ticker][3:])


def test_pipeline_narrows_to_the_requested_tickers() -> None:
    vol = vol_series(CSV, window=2, tickers=["BBB"])
    assert vol.returns.tickers == ("BBB",)
    assert set(vol.values) == {"BBB"}


def test_pipeline_rejects_a_window_below_two() -> None:
    with pytest.raises(ValueError, match="window"):
        vol_series(CSV, window=1)
