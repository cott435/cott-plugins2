+"""PL-01 step 3: put every ticker on one calendar."""
+
+from __future__ import annotations
+
+from collections.abc import Sequence
+
+from mdata.types import GapError, Prices, RawRow
+
+
+def align_calendar(rows: Sequence[RawRow], *, max_gap: int = 5) -> Prices:
+    """Prices on one calendar from the first date every ticker has a close.
+
+    The calendar is every date any ticker has a close on, from the date the last ticker
+    starts; a missing date takes the ticker's previous close. More than `max_gap` missing
+    dates in a row raises `GapError` naming the ticker.
+    """
+    by_ticker: dict[str, dict[str, float]] = {}
+    for row in rows:
+        series = by_ticker.setdefault(row.ticker, {})
+        if row.close is not None:
+            series[row.date] = row.close
+    tickers = tuple(sorted(by_ticker))
+    if not tickers or not all(by_ticker.values()):
+        return Prices(dates=(), tickers=tickers, closes=dict.fromkeys(tickers, ()))
+    every_date = sorted({date for series in by_ticker.values() for date in series})
+    dates = tuple(every_date[every_date.index(max(min(s) for s in by_ticker.values())) :])
+    closes = {
+        ticker: _forward_fill(ticker, by_ticker[ticker], every_date, max_gap)[-len(dates) :]
+        for ticker in tickers
+    }
+    return Prices(dates=dates, tickers=tickers, closes=closes)
+
+
+def _forward_fill(
+    ticker: str, series: dict[str, float], dates: list[str], max_gap: int
+) -> tuple[float, ...]:
+    """The ticker's close on each date from its first, a missing date taking the previous.
+
+    Dates before the ticker's first close have no value and are left out, so the result
+    lines up with the tail of `dates`.
+    """
+    filled: list[float] = []
+    missing_run = 0
+    for date in dates:
+        if date in series:
+            filled.append(series[date])
+            missing_run = 0
+        elif filled:
+            missing_run += 1
+            if missing_run > max_gap:
+                msg = f"{ticker}: more than {max_gap} dates in a row are missing (to {date})"
+                raise GapError(msg)
+            filled.append(filled[-1])
+    return tuple(filled)
