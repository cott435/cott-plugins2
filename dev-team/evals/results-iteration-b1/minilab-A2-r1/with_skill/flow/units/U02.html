+
+from __future__ import annotations
+
+import json
+from collections.abc import Sequence
+from pathlib import Path
+
+from mdata import Prices
+
+SAMPLE = Path(__file__).resolve().parent.parent / "mdata.Prices" / "sample.json"
+
+
+def load_prices(path: str | Path, tickers: Sequence[str] | None = None) -> Prices:
+    """The sample's prices, narrowed to `tickers`; KeyError for a ticker the sample lacks."""
+    data = json.loads(SAMPLE.read_text())
+    closes = {ticker: tuple(values) for ticker, values in data["closes"].items()}
+    chosen = sorted(closes) if tickers is None else sorted(tickers)
+    return Prices(
+        dates=tuple(data["dates"]),
+        tickers=tuple(chosen),
+        closes={ticker: closes[ticker] for ticker in chosen},
+    )

+"""The mdata contract: samples build the frozen types; the load_prices fake serves the sample."""
+
+from __future__ import annotations
+
+import dataclasses
+
+import pytest
+
+from mdata import Prices, RawRow
+
+
+def test_prices_sample_builds_frozen_prices(contract_sample):
+    data = contract_sample("mdata.Prices")
+    prices = Prices(
+        dates=tuple(data["dates"]),
+        tickers=tuple(data["tickers"]),
+        closes={ticker: tuple(values) for ticker, values in data["closes"].items()},
+    )
+    assert prices.tickers == ("AAPL", "MSFT")
+    assert len(prices.dates) == 6
+    assert list(prices.dates) == sorted(prices.dates)
+    assert all(len(prices.closes[t]) == 6 for t in prices.tickers)
+    assert all(close > 0 for t in prices.tickers for close in prices.closes[t])
+    with pytest.raises(dataclasses.FrozenInstanceError):
+        prices.dates = ()
+
+
+def test_rawrow_sample_builds_rows(contract_sample):
+    rows = [RawRow(**row) for row in contract_sample("mdata.RawRow")["rows"]]
+    assert len(rows) == 8
+    assert {row.ticker for row in rows} == {"AAPL", "MSFT"}
+    assert sum(row.close is None for row in rows) == 1
+
+
+def test_fake_load_prices_serves_the_sample(contract_fake, contract_sample):
+    fake = contract_fake("mdata.load_prices")
+    prices = fake.load_prices("any.csv")
+    data = contract_sample("mdata.Prices")
+    assert isinstance(prices, Prices)
+    assert prices.dates == tuple(data["dates"])
+    assert prices.tickers == ("AAPL", "MSFT")
+    assert prices.closes["MSFT"] == tuple(data["closes"]["MSFT"])
+
+

+    fake = contract_fake("mdata.load_prices")
+    assert fake.load_prices("any.csv", ["MSFT"]).tickers == ("MSFT",)
+    with pytest.raises(KeyError):
+        fake.load_prices("any.csv", ["GOOG"])
