        dataclasses.replace(c, run=name, path=path / "checkpoints" / c.path.name)
        for c in run.checkpoints()
    ]

def check_actions(spec: WorldSpec, actions: Array, active: Array) -> None:
    """`ActionError` for the first active agent whose action is outside `spec.action`."""
    actions = np.asarray(actions)
    space = spec.action
    for i, agent in enumerate(spec.agents):
        if not active[i]:
            continue
        a = actions[i]
        if isinstance(space, Box):
            inside = np.shape(a) == (space.dims,) and bool(
                np.all((a >= space.low) & (a <= space.high))
            )
        else:
            inside = np.ndim(a) == 0 and float(a).is_integer() and 0 <= a < space.n
        if not inside:
            raise ActionError(agent, a.tolist())


class ReplayWorld:
    """Serves a recorded episode: `recording` holds spec, reset, steps, draws and result."""

    def __init__(self, fx, recording) -> None:
        self.spec = fx.decode(WorldSpec, recording["spec"])
        self._reset = fx.decode(Step, recording["reset"])
        self._steps = [fx.decode(Step, s["step"]) for s in recording["steps"]]
        self.recorded_actions = [fx.decode(Array, s["actions"]) for s in recording["steps"]]
        self._draws = [fx.decode(DrawState, d) for d in recording["draws"]]
        self._result = fx.decode(EpisodeResult, recording["result"])
        self.seeds: list[int] = []
        self.actions: list[Array] = []
        self._t = -1

    def reset(self, seed: int) -> Step:
        self.seeds.append(seed)
        self._t = 0
        return self._reset

    def step(self, actions: Array) -> Step:
        if self._t < 0:
            raise RuntimeError("step before reset")
        if self._t >= len(self._steps):
            raise RuntimeError("step after the episode ended")
        last = self._reset if self._t == 0 else self._steps[self._t - 1]
        check_actions(self.spec, actions, last.active)
        self.actions.append(np.array(actions, copy=True))
        step = self._steps[self._t]
        self._t += 1
        return step

    def result(self) -> EpisodeResult:
        return self._result

    def draw(self) -> DrawState:
        return self._draws[max(self._t, 0)]

def summarize(spec: WorldSpec, team: TeamSpec, scores: TeamScores, runs: Sequence[Run]) -> Summary:
    """The team's EVAL-06 summary: means, range, tests (5+ runs only), verdict and collapse."""
    run_means = {name: float(np.mean(s)) for name, s in scores.learned.items()}
    random_mean = float(np.mean(scores.random))
    rule_mean = float(np.mean(scores.rule))
    comparisons = _comparisons(team, scores) if len(run_means) >= MIN_RUNS else ()
    return Summary(
        world=spec.world,
        variant=spec.variant,
        team=team.name,
        runs=tuple(run_means),
        learned_mean=float(np.mean(list(run_means.values()))),
        learned_range=(min(run_means.values()), max(run_means.values())),

        random_mean=random_mean,
        rule_mean=rule_mean,
        comparisons=comparisons,

    )


def _comparisons(team: TeamSpec, scores: TeamScores) -> tuple[Comparison, ...]:
    """The learned runs against each baseline, Holm-adjusted across the two."""
    learned = np.stack(list(scores.learned.values()))
    rng = np.random.default_rng(np.random.SeedSequence(PERMUTATION_SEED))
    baselines: tuple[tuple[Literal["random", "rule"], Array], ...] = (
        ("random", scores.random),
        ("rule", scores.rule),
    )
    tests = [compare(learned, baseline, team.direction, rng=rng) for _, baseline in baselines]
    adjusted = holm([p for _, p in tests])
    return tuple(
        Comparison(
            baseline=name,
            team=team.name,
            learned_mean=float(np.mean(learned)),
            baseline_mean=float(np.mean(baseline)),
            difference=difference,
            p_raw=p_raw,
            p_holm=p_holm,
            rejected=p_holm <= ALPHA,
        )
        for (name, baseline), (difference, p_raw), p_holm in zip(

        )
    )


def holm(p_values: Sequence[float]) -> list[float]:
    """Holm step-down adjusted p-values, in the order given."""
    m = len(p_values)
    adjusted = [0.0] * m
    running = 0.0
    for rank, i in enumerate(sorted(range(m), key=lambda j: p_values[j])):
        running = max(running, min(1.0, (m - rank) * p_values[i]))
        adjusted[i] = running
    return adjusted


def collapse_step(runs: Sequence[Run], team: TeamSpec, random_mean: float) -> int | None:
    """The step of the first of the last 4+ eval points where the team's in-training score,
    averaged over runs, is no better than `random_mean`; None when there is no such streak."""
    by_step: defaultdict[int, list[float]] = defaultdict(list)
    for run in runs:
        for point in run.evals():
            by_step[point.step].append(point.scores[team.name])

    def no_better(step: int) -> bool:
        mean = float(np.mean(by_step[step]))
        return mean <= random_mean if team.direction == "higher" else mean >= random_mean

    streak = list(itertools.takewhile(no_better, sorted(by_step, reverse=True)))
    return streak[-1] if len(streak) >= COLLAPSE_POINTS else None

    out.mkdir(parents=True, exist_ok=True)
    path = out / "summary.json"
    data = [dataclasses.asdict(s) for s in summaries]
    path.write_text(json.dumps(data, indent=1), encoding="utf-8")
    return path

def final_evaluation(
    runs: Sequence[Run], out: Path, *, workers: int | None = None
) -> list[Summary]:
    """Play every final checkpoint and both baselines on `FINAL_SEEDS`, test, judge, and

    world = make(spec.world, spec.variant)

    write_summaries(summaries, out)

def shared_spec(runs: Sequence[Run]) -> WorldSpec:
    """The one world spec every run was trained on; `ValueError` for none or several."""
    specs = {run.record.spec for run in runs}
    if len(specs) != 1:
        raise ValueError(f"final evaluation needs runs of exactly one world spec, got {len(specs)}")
    return specs.pop()


def final_policy(run: Run, team: str, spec: WorldSpec) -> Policy:
    """`team`'s policy at the run's last checkpoint."""
    return load_policy(max(run.checkpoints(), key=lambda c: c.pct), team, spec)


def play_seeds(world: World, policies: Mapping[str, Policy]) -> dict[str, Array]:
    """Each team's score on each of `FINAL_SEEDS`, in seed order."""
    results = [play_episode(world, policies, seed).result for seed in FINAL_SEEDS]
    return {
        t.name: np.array([r.scores[t.name] for r in results], dtype=float)
        for t in world.spec.teams
    }

@dataclass(frozen=True)
class Tag:
    """A stand-in policy naming what it is: `learned` (run, pct), `random` or `rule`."""

    kind: str
    run: str = ""
    pct: int = -1


def score(tag: Tag, seed: int) -> float:
    if tag.kind == "learned":
        return float(tag.run.rsplit("s", 1)[1]) + seed % 2  # run sN averages N + 0.5
    return RULE_SCORE if tag.kind == "rule" else 0.0


@pytest.fixture

    episodes: list[tuple[dict, int]] = []

        episodes.append((dict(policies), seed))

        result = EpisodeResult(seed, 10, scores, scores, {})
