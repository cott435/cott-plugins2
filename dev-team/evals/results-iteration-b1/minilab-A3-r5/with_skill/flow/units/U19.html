from collections.abc import Mapping, Sequence
from pathlib import Path

from risk.types import Drawdown, Report, TickerSummary


def write_report(
    summary: Sequence[TickerSummary], drawdowns: Sequence[Drawdown], out_path: str | Path
) -> Report:
    """`{"tickers": [...]}` with one row per ticker, sorted, written to `out_path`."""
    by_ticker = {d.ticker: d for d in drawdowns}
    stats = {s.ticker: s for s in summary}
    if set(stats) != set(by_ticker):
        msg = f"summary tickers {sorted(stats)} differ from drawdown tickers {sorted(by_ticker)}"
        raise ValueError(msg)
    rows = tuple(_row(stats[t], by_ticker[t]) for t in sorted(stats))
    Path(out_path).write_text(json.dumps({"tickers": list(rows)}, indent=2) + "\n")
    return Report(path=str(out_path), rows=rows)
